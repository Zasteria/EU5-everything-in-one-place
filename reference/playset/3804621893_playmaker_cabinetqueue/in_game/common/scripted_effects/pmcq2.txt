# Country preference persists in saves. Old saves default to three months.
pmcq2_init_months = {
 if = { limit = { NOT = { has_variable = pmcq2_months } } set_variable = { name = pmcq2_months value = 3 } }
 clamp_variable = { name = pmcq2_months min = 1 max = 6 }
}
# Refresh optional participants from the current assignments, not saved stale slots.
pmcq2_sync_members = {
 save_scope_as = pmcq2_owner
 remove_variable = pmcq2_control
 remove_variable = pmcq2_nobles
 if = {
  limit = { any_cabinet = { has_cabinet_action = yes cabinet_action = cabinet_action:pmcq2_control_action exists = cabinet_member } }
  random_cabinet = {
   limit = { has_cabinet_action = yes cabinet_action = cabinet_action:pmcq2_control_action exists = cabinet_member }
   scope:pmcq2_owner = { set_variable = { name = pmcq2_control value = prev } }
  }
 }
 if = {
  limit = {
   has_reform = government_reform:noblesse_oblige_reform
   any_cabinet = { has_cabinet_action = yes cabinet_action = cabinet_action:pmcq2_nobles_action exists = cabinet_member }
  }
  random_cabinet = {
   limit = { has_cabinet_action = yes cabinet_action = cabinet_action:pmcq2_nobles_action exists = cabinet_member }
   scope:pmcq2_owner = { set_variable = { name = pmcq2_nobles value = prev } }
  }
 }
}
# Deactivation removes its slot before checking the remaining participant.
# Do not rediscover the deactivating native action within its own callback.
pmcq2_membership_changed = {
 remove_variable = pmcq2_application_reported
 if = {
  limit = { has_variable = pmcq2_running }
  if = {
   limit = { pmcq2_ready = yes }
   if = { limit = { exists = var:pmcq2_target } pmcq2_apply = yes }
  }
  else = { pmcq2_pause = yes }
 }
}
# All these effects run in country scope.
pmcq2_clear_bonuses = {
 every_in_list = {
  variable = pmcq2_locations
  remove_location_modifier = pmcq2_control_bonus
  remove_location_modifier = pmcq2_nobles_bonus
 }
 clear_variable_list = pmcq2_locations
}
pmcq2_pause = {
 pmcq2_clear_bonuses = yes
 if = { limit = { NOT = { has_variable = pmcq2_generation } } set_variable = { name = pmcq2_generation value = 0 } }
 change_variable = { name = pmcq2_generation add = 1 }
 remove_variable = pmcq2_running
 remove_variable = pmcq2_target
 # Migrate/disable the 0.1.x picker state on existing saves.
 remove_variable = pmcq_running
 remove_variable = pmcq_pending
 remove_variable = pmcq_wait
 remove_variable = pmcq_phase
 remove_variable = pmcq_click_armed
 remove_variable = pmcq_confirm_armed
 clear_variable_list = pmcq_open
 if = {
  limit = { exists = var:pmcq_target }
  var:pmcq_target = { clear_variable_list = pmcq_click clear_variable_list = pmcq_confirm }
 }
 remove_variable = pmcq_target
}
pmcq2_apply = {
 save_scope_as = pmcq2_owner
 # Retain bonuses on the current target; clear only locations that left it.
 every_in_list = {
  variable = pmcq2_locations
  if = {
   limit = { NOT = { owner = scope:pmcq2_owner province = scope:pmcq2_owner.var:pmcq2_target } }
   remove_location_modifier = pmcq2_control_bonus
   remove_location_modifier = pmcq2_nobles_bonus
  }
 }
 clear_variable_list = pmcq2_locations
 set_variable = { name = pmcq2_checked_locations value = 0 }
 set_variable = { name = pmcq2_buffed_locations value = 0 }
 # Use the native scripted-effect convention, 1 + cabinet.effective_skill,
 # seen in send_people_to_the_colonies.txt. Vanilla parity needs measurement.
 set_variable = { name = pmcq2_control_scale value = 0 }
 if = {
  limit = { pmcq2_control_valid = yes }
  set_variable = { name = pmcq2_control_scale value = { value = 1 add = var:pmcq2_control.effective_skill min = 0 } }
 }
 set_variable = { name = pmcq2_nobles_scale value = 0 }
 if = {
  limit = { pmcq2_nobles_valid = yes }
  set_variable = { name = pmcq2_nobles_scale value = { value = 1 add = var:pmcq2_nobles.effective_skill min = 0 } }
 }
 var:pmcq2_target = {
  every_location_in_province = {
   limit = { owner = scope:pmcq2_owner }
   if = {
    limit = { scope:pmcq2_owner = { pmcq2_control_valid = yes } }
    if = {
     limit = { OR = {
      NOT = { has_location_modifier = pmcq2_control_bonus }
      scope:pmcq2_owner = {
       OR = {
        NOT = { has_variable = pmcq2_control_applied_scale }
        AND = { has_variable = pmcq2_control_applied_scale NOT = { var:pmcq2_control_applied_scale = var:pmcq2_control_scale } }
       }
      }
     } }
     add_location_modifier = { modifier = pmcq2_control_bonus days = -1 mode = replace size = scope:pmcq2_owner.var:pmcq2_control_scale }
    }
   }
   else = { remove_location_modifier = pmcq2_control_bonus }
   if = {
    limit = { scope:pmcq2_owner = { pmcq2_nobles_valid = yes } }
    if = {
     limit = { OR = {
      NOT = { has_location_modifier = pmcq2_nobles_bonus }
      scope:pmcq2_owner = {
       OR = {
        NOT = { has_variable = pmcq2_nobles_applied_scale }
        AND = { has_variable = pmcq2_nobles_applied_scale NOT = { var:pmcq2_nobles_applied_scale = var:pmcq2_nobles_scale } }
       }
      }
     } }
     add_location_modifier = { modifier = pmcq2_nobles_bonus days = -1 mode = replace size = scope:pmcq2_owner.var:pmcq2_nobles_scale }
    }
   }
   else = { remove_location_modifier = pmcq2_nobles_bonus }
   scope:pmcq2_owner = {
    add_to_variable_list = { name = pmcq2_locations target = prev }
    change_variable = { name = pmcq2_checked_locations add = 1 }
   }
   if = {
    limit = {
     OR = {
      AND = { scope:pmcq2_owner = { pmcq2_control_valid = yes } has_location_modifier = pmcq2_control_bonus }
      AND = { scope:pmcq2_owner = { NOT = { pmcq2_control_valid = yes } } NOT = { has_location_modifier = pmcq2_control_bonus } }
     }
     OR = {
      AND = { scope:pmcq2_owner = { pmcq2_nobles_valid = yes } has_location_modifier = pmcq2_nobles_bonus }
      AND = { scope:pmcq2_owner = { NOT = { pmcq2_nobles_valid = yes } } NOT = { has_location_modifier = pmcq2_nobles_bonus } }
     }
    }
    scope:pmcq2_owner = { change_variable = { name = pmcq2_buffed_locations add = 1 } }
   }
  }
 }
 set_variable = { name = pmcq2_control_applied_scale value = var:pmcq2_control_scale }
 set_variable = { name = pmcq2_nobles_applied_scale value = var:pmcq2_nobles_scale }
 if = {
  limit = { NOT = { has_variable = pmcq2_application_reported } }
  set_variable = pmcq2_application_reported
  if = {
   limit = { var:pmcq2_checked_locations > 0 var:pmcq2_buffed_locations = var:pmcq2_checked_locations }
   debug_log = "PMCQ2 0.2.4: location modifiers match active queue participants"
  }
  else = { debug_log = "PMCQ2 0.2.4: participant modifier verification failed or no owned locations found" }
  save_scope_value_as = { name = pmcq2_checked value = var:pmcq2_checked_locations }
  save_scope_value_as = { name = pmcq2_buffed value = var:pmcq2_buffed_locations }
  save_scope_value_as = { name = pmcq2_control_strength value = var:pmcq2_control_scale }
  save_scope_value_as = { name = pmcq2_nobles_strength value = var:pmcq2_nobles_scale }
  debug_log_scopes = yes
 }
}
pmcq2_choose_next = {
 pmcq2_init_months = yes
 remove_variable = pmcq2_application_reported
 save_scope_as = pmcq2_owner
 pmcq2_clear_bonuses = yes
 remove_variable = pmcq2_target
 if = {
  limit = { has_variable = pmcq2_culture_mode }
  pmcq2_choose_culture = yes
 }
 else = {
 ordered_in_list = {
  variable = pmcq_queue
  limit = { owner = scope:pmcq2_owner province_average_control < 0.95 var:pmcq_order > scope:pmcq2_owner.var:pmcq_cursor }
  order_by = { value = 0 subtract = var:pmcq_order }
  max = 1 check_range_bounds = no
  scope:pmcq2_owner = { set_variable = { name = pmcq2_target value = prev } }
 }
 if = {
  limit = { NOT = { exists = var:pmcq2_target } }
  ordered_in_list = {
   variable = pmcq_queue
   limit = { owner = scope:pmcq2_owner province_average_control < 0.95 }
   order_by = { value = 0 subtract = var:pmcq_order }
   max = 1 check_range_bounds = no
   scope:pmcq2_owner = { set_variable = { name = pmcq2_target value = prev } }
  }
 }
 }
 if = {
  limit = { exists = var:pmcq2_target }
  if = {
   limit = { NOT = { has_variable = pmcq2_culture_mode } }
   set_variable = { name = pmcq_cursor value = var:pmcq2_target.var:pmcq_order }
  }
  pmcq2_apply = yes
  debug_log = "PMCQ2 0.2.5: target selected; configured stay started"
  debug_log_date = yes
  save_scope_value_as = { name = pmcq2_generation value = var:pmcq2_generation }
  if = { limit = { NOT = { has_variable = pmcq2_stay_token } } set_variable = { name = pmcq2_stay_token value = 0 } }
  change_variable = { name = pmcq2_stay_token add = 1 }
  save_scope_value_as = { name = pmcq2_stay_token value = var:pmcq2_stay_token }
  set_variable = { name = pmcq2_active_months value = var:pmcq2_months }
  if = {
   limit = { var:pmcq2_active_months = 1 }
   trigger_event_silently = { on_action = pmcq2_due months = 1 }
  }
  else_if = {
   limit = { var:pmcq2_active_months = 2 }
   trigger_event_silently = { on_action = pmcq2_due months = 2 }
  }
  else_if = {
   limit = { var:pmcq2_active_months = 3 }
   trigger_event_silently = { on_action = pmcq2_due months = 3 }
  }
  else_if = {
   limit = { var:pmcq2_active_months = 4 }
   trigger_event_silently = { on_action = pmcq2_due months = 4 }
  }
  else_if = {
   limit = { var:pmcq2_active_months = 5 }
   trigger_event_silently = { on_action = pmcq2_due months = 5 }
  }
  else_if = {
   limit = { var:pmcq2_active_months = 6 }
   trigger_event_silently = { on_action = pmcq2_due months = 6 }
  }
 }
 else_if = {
  limit = { NOT = { has_variable = pmcq2_culture_mode } }
  debug_log = "PMCQ2: no eligible queued province; paused"
  pmcq2_pause = yes
 }
}
pmcq2_daily = {
 pmcq2_sync_members = yes
 save_scope_as = pmcq2_owner
 if = {
  limit = { NOT = { pmcq2_ready = yes } }
  pmcq2_pause = yes
 }
 else = {
  if = {
   limit = { has_variable = pmcq2_culture_mode }
   if = {
    limit = { exists = var:pmcq2_target }
    if = {
     limit = { var:pmcq2_target = { NOT = { pmcq2_culture_eligible = yes } } }
     pmcq2_clear_bonuses = yes
     remove_variable = pmcq2_target
     # Invalidate the abandoned stay, including callbacks delivered while idle.
     change_variable = { name = pmcq2_stay_token add = 1 }
    }
   }
   if = { limit = { NOT = { exists = var:pmcq2_target } } pmcq2_choose_next = yes }
   else = { pmcq2_apply = yes }
  }
  else_if = {
   limit = { exists = var:pmcq2_target var:pmcq2_target = { owner = scope:pmcq2_owner } }
   pmcq2_apply = yes
  }
  else = { pmcq2_pause = yes }
  if = {
   limit = { has_variable = pmcq2_running }
   save_scope_value_as = { name = pmcq2_generation value = var:pmcq2_generation }
   trigger_event_silently = { on_action = pmcq2_daily days = 1 }
  }
 }
}
# The visited list belongs to the country; manual queue order is untouched.
pmcq2_choose_culture = {
 ordered_province = {
  limit = {
   pmcq2_culture_eligible = yes
   NOT = { scope:pmcq2_owner = { is_target_in_variable_list = { name = pmcq2_culture_visited target = prev } } }
  }
  order_by = population
  max = 1 check_range_bounds = no
  scope:pmcq2_owner = { set_variable = { name = pmcq2_target value = prev } }
 }
 if = {
  limit = { NOT = { exists = var:pmcq2_target } }
  clear_variable_list = pmcq2_culture_visited
  ordered_province = {
   limit = { pmcq2_culture_eligible = yes }
   order_by = population
   max = 1 check_range_bounds = no
   scope:pmcq2_owner = { set_variable = { name = pmcq2_target value = prev } }
  }
 }
 if = {
  limit = { exists = var:pmcq2_target }
  add_to_variable_list = { name = pmcq2_culture_visited target = var:pmcq2_target }
 }
}
pmcq2_start = {
 if = {
  limit = { pmcq2_has_source = yes pmcq2_can_start = yes }
  pmcq2_pause = yes
  pmcq2_sync_members = yes
  set_variable = { name = pmcq_cursor value = 0 }
  clear_variable_list = pmcq2_culture_visited
  set_variable = pmcq2_running
  pmcq2_choose_next = yes
  if = {
   limit = { has_variable = pmcq2_running }
   save_scope_value_as = { name = pmcq2_generation value = var:pmcq2_generation }
   trigger_event_silently = { on_action = pmcq2_daily days = 1 }
  }
 }
}
