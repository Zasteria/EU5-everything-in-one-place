# Reserve provinces, not the global area, so different countries can work independently.
pmcqie_release_provinces = {
 every_in_list = { variable = pmcqie_provinces
  if = { limit = { exists = var:pmcqi_reservation var:pmcqi_reservation = scope:pmcqi_slot }
   remove_province_modifier = pmcqi_bonus
   remove_variable = pmcqi_reservation
  }
 }
 clear_variable_list = pmcqie_provinces
}
pmcqie_clear = {
 pmcqie_release_provinces = yes
 remove_variable = pmcqie_area
}
pmcqie_choose = {
 scope:pmcqi_country = {
  ordered_province = {
   limit = { pmcqi_eligible = yes area = { pmcqie_available = yes } }
   order_by = population max = 1 check_range_bounds = no
   area = { scope:pmcqi_slot = { set_variable = { name = pmcqie_area value = prev } } }
  }
 }
}
pmcqie_apply = {
 # Drop bonuses on lost provinces before renewing the current area's owned provinces.
 pmcqie_release_provinces = yes
 set_variable = { name = pmcqi_slot_scale value = { value = 1 add = effective_skill min = 0 } }
 var:pmcqie_area = {
  every_province_in_area = {
   limit = { owner = scope:pmcqi_country NOT = { has_variable = pmcqi_reservation } }
   set_variable = { name = pmcqi_reservation value = scope:pmcqi_slot }
   add_province_modifier = { modifier = pmcqi_bonus days = 2 mode = replace size = scope:pmcqi_slot.var:pmcqi_slot_scale }
   scope:pmcqi_slot = { add_to_variable_list = { name = pmcqie_provinces target = prev } }
  }
 }
}
pmcqie_tick = {
 if = { limit = { exists = var:pmcqie_area }
  if = { limit = { var:pmcqie_area = { OR = {
   pmcqie_finished = yes
   any_province_in_area = { owner = scope:pmcqi_country pmcqi_native_integration_reserved = yes }
  } } } pmcqie_clear = yes }
 }
 if = { limit = { NOT = { exists = var:pmcqie_area } } pmcqie_choose = yes }
 if = { limit = { exists = var:pmcqie_area }
  pmcqie_apply = yes
  scope:pmcqi_country = { change_variable = { name = pmcqi_multi_count add = 1 } }
 }
 else = {
  pmcqie_clear = yes
  remove_variable = pmcqi_slot_running
  stop_cabinet_action = yes
 }
}
