# Country scope. Each user command invalidates all older pending workers.
pmcqpp_rearm = {
 if = { limit = { NOT = { has_variable = pmcqpp_generation } } set_variable = { name = pmcqpp_generation value = 0 } }
 change_variable = { name = pmcqpp_generation add = 1 }
 remove_variable = pmcqpp_running
 if = { limit = { any_navy = { has_variable = pmcqpp_v3_enabled } }
  save_scope_as = pmcqpp_country
  if = { limit = { has_variable = pmcqpp_generation }
   save_scope_value_as = { name = pmcqpp_generation value = var:pmcqpp_generation }
   trigger_event_silently = { on_action = pmcqpp_v3_daily days = 1 }
  }
 }
}
