# Country scope; the caller supplies the empty cabinet slot and required skill.
pmcqcm_fill = {
 ordered_character = {
  limit = {
   is_alive = yes
   in_cabinet = no
   is_general = no
   is_admiral = no
   can_serve_in_cabinet_of = scope:pmcqcm_country
  }
  order_by = $SKILL$
  max = 1
  check_range_bounds = no
  save_scope_as = pmcqcm_candidate
  scope:pmcqcm_slot = {
   if = { limit = {
    NOT = { exists = cabinet_member }
    has_cabinet_action = yes
    scope:pmcqcm_country = { can_pay_price = price:set_cabinet_member }
   }
    add_to_cabinet = scope:pmcqcm_candidate
    # Charge only after the appointment is observed to have succeeded.
    if = { limit = { cabinet_member = scope:pmcqcm_candidate }
     scope:pmcqcm_country = { pay_price = price:set_cabinet_member }
    }
   }
  }
 }
}
pmcqcm_scan = {
 save_scope_as = pmcqcm_country
 # Avoid competing with the native member selector, including older saves.
 set_automated_system = { system = cabinetmembers activate = no }
 every_cabinet = {
  limit = { has_cabinet_action = yes NOT = { exists = cabinet_member } }
  save_scope_as = pmcqcm_slot
  if = { limit = { cabinet_action = { cabinet_action_type = adm } }
   scope:pmcqcm_country = { pmcqcm_fill = { SKILL = adm } }
  }
  else_if = { limit = { cabinet_action = { cabinet_action_type = dip } }
   scope:pmcqcm_country = { pmcqcm_fill = { SKILL = dip } }
  }
  else_if = { limit = { cabinet_action = { cabinet_action_type = mil } }
   scope:pmcqcm_country = { pmcqcm_fill = { SKILL = mil } }
  }
 }
}
pmcqcm_tick = {
 pmcqcm_scan = yes
 save_scope_value_as = { name = pmcqcm_generation value = var:pmcqcm_generation }
 trigger_event_silently = { on_action = pmcqcm_daily days = 1 }
}
