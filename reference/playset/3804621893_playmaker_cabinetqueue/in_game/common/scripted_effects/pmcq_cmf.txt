pmcq_handle_cmf_callback = {
 switch = { trigger = var:cmf_callback
  flag:pmcq__scripted_gifts = { set_variable = pmcqsg_open_requested }
  flag:pmcq__favorites_system = { cmm_sync_bool_alias = { setting = pmcq__favorites_system alias = pmcq_favorites_system_enabled } }
  flag:pmcq__swap_war_clicks = { cmm_sync_bool_alias_inverted = { setting = pmcq__swap_war_clicks alias = pmcq_native_war_clicks } }
  flag:pmcq__reset_shortcuts = { pmcq_reset_diplo_shortcuts = yes }
  flag:pmcq__recommended_setup = { set_variable = pmcq_cmf_request_recommended_setup }
  flag:pmcq__remove_forts = { set_variable = pmcq_cmf_request_remove_forts }
  flag:pmcq__block_circles = { cmm_sync_bool_alias_inverted = { setting = pmcq__block_circles alias = pmcq_circle_requests_allowed } }
  flag:pmcq__mute_notifications = { cmm_sync_bool_alias = { setting = pmcq__mute_notifications alias = pmcq_mute_notifications } }
  flag:pmcq__remember_events = { cmm_sync_bool_alias = { setting = pmcq__remember_events alias = pmcq_event_memory_enabled }
 cmm_sync_bool_alias_inverted = { setting = pmcq__remember_events alias = pmcq_event_memory_disabled } }
  flag:pmcq__event_catalog = { set_variable = pmcq_cmf_request_event_catalog }
  flag:pmcq__load_preset = { set_variable = pmcq_cmf_request_load_preset }
  flag:pmcq__add_hre = { save_scope_as = pmcq_cmf_actor
   if = { limit = { can_do_generic_action = { generic_action = generic_action:pmcqp_add_hre_locations recipient = scope:pmcq_cmf_actor } }
    fire_generic_action = { type = pmcqp_add_hre_locations actor = scope:pmcq_cmf_actor recipient = scope:pmcq_cmf_actor }
   } }
  flag:pmcq__dismantle_hre = { set_variable = pmcq_cmf_request_dismantle_hre }
  flag:pmcq__diplo_slot_1 = { set_variable = { name = pmcq_diplo_edit_slot value = 1 } set_variable = pmcq_cmf_request_shortcut_picker }
  flag:pmcq__diplo_slot_2 = { set_variable = { name = pmcq_diplo_edit_slot value = 2 } set_variable = pmcq_cmf_request_shortcut_picker }
  flag:pmcq__diplo_slot_3 = { set_variable = { name = pmcq_diplo_edit_slot value = 3 } set_variable = pmcq_cmf_request_shortcut_picker }
  flag:pmcq__diplo_slot_4 = { set_variable = { name = pmcq_diplo_edit_slot value = 4 } set_variable = pmcq_cmf_request_shortcut_picker }
  flag:pmcq__diplo_slot_5 = { set_variable = { name = pmcq_diplo_edit_slot value = 5 } set_variable = pmcq_cmf_request_shortcut_picker }
  flag:pmcq__diplo_slot_6 = { set_variable = { name = pmcq_diplo_edit_slot value = 6 } set_variable = pmcq_cmf_request_shortcut_picker }
 }
}
