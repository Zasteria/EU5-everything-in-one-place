# Native Improve Relations queue. No opinion, diplomat or treasury bonuses.
# All persistent ownership, cooldowns and order are stored on the player country.
pmcqr_init = {
 if = { limit = { NOT = { has_variable = pmcqx_limit } } set_variable = { name = pmcqx_limit value = 1 } }
 if = { limit = { NOT = { has_variable = pmcqr_scans } } set_variable = { name = pmcqr_scans value = 0 } }
 if = { limit = { NOT = { has_variable = pmcqr_antagonism_threshold } } set_variable = { name = pmcqr_antagonism_threshold value = 30 } }
 if = { limit = { NOT = { has_variable = pmcqr_limit } } set_variable = { name = pmcqr_limit value = 1 } }
 if = { limit = { NOT = { has_variable = pmcqr_serial } } set_variable = { name = pmcqr_serial value = 0 } }
}
pmcqr_recount = {
 set_variable = { name = pmcqr_count value = 0 }
 set_variable = { name = pmcqr_queued value = 0 }
 save_scope_as = pmcqr_owner
 every_in_list = { variable = pmcqr_owned scope:pmcqr_owner = { change_variable = { name = pmcqr_count add = 1 } } }
 every_in_list = { variable = pmcqr_queue scope:pmcqr_owner = { change_variable = { name = pmcqr_queued add = 1 } } }
}
pmcqr_begin_scan = {
 pmcqr_auto_add_subjects = yes
 if = {
  limit = { NOT = { has_variable = pmcqr_scan } }
  clear_variable_list = pmcqr_sampled
  clear_variable_list = pmcqr_seen
  clear_variable_list = pmcqr_pending
  clear_variable_list = pmcqr_ready
  clear_variable_list = pmcqr_capped
  clear_variable_list = pmcqx_ready
  clear_variable_list = pmcqx_pending
  clear_variable_list = pmcqx_subjects
  if = {
   limit = { has_variable = pmcqx_running }
   save_scope_as = pmcqx_owner
   every_subject = {
    limit = { num_locations > 0 }
    save_scope_as = pmcqx_target
    scope:pmcqx_owner = { add_to_variable_list = { name = pmcqx_subjects target = scope:pmcqx_target } }
   }
  }
  set_variable = pmcqr_scan
 }
}
pmcqr_wake = {
 pmcqr_init = yes
 remove_variable = pmcqr_scheduler
 remove_variable = pmcqr_scan
 set_variable = pmcqr_refresh
}
# pmcqr_target is a country; called only for this queue's tracked assignments.
# Keep the slot reserved until a later native observation confirms cancellation.
pmcqr_cancel_native = {
 if = {
  limit = {
   is_target_in_variable_list = { name = pmcqr_seen target = scope:pmcqr_target }
   NOT = { exists = var:pmcqr_cancel_target }
  }
  # This action is NOT exposed to perform_diplomatic_action. The GUI executes
  # the native Improve Opinion outliner action, one tracked target per scan.
  set_variable = { name = pmcqr_cancel_target value = scope:pmcqr_target }
  # Only an intentional interruption may return to waiting after recall.
  if = {
   limit = { scope:pmcqr_target = { NOT = { pmcqr_finished = yes } } NOT = { is_key_in_variable_map = { name = pmcqr_done target = scope:pmcqr_target } } }
   add_to_variable_map = { name = pmcqr_interrupted key = scope:pmcqr_target value = 1 }
  }
 }
}
pmcqr_release = {
 remove_list_variable = { name = pmcqr_owned target = scope:pmcqr_target }
 remove_from_variable_map = { name = pmcqr_grace key = scope:pmcqr_target }
 remove_from_variable_map = { name = pmcqr_cancel_retry key = scope:pmcqr_target }
 # Native auto-recall can happen between observations, before 100% is seen.
 # Never automatically retry an ended dispatch unless this queue interrupted it.
 if = {
  limit = { is_key_in_variable_map = { name = pmcqr_interrupted target = scope:pmcqr_target } }
  add_to_variable_map = { name = pmcqr_retry key = scope:pmcqr_target value = 1 days = 30 }
 }
 else = { add_to_variable_map = { name = pmcqr_retired key = scope:pmcqr_target value = 1 } }
 remove_from_variable_map = { name = pmcqr_interrupted key = scope:pmcqr_target }
}
pmcqr_process = {
 change_variable = { name = pmcqr_scans add = 1 }
 save_scope_as = pmcqr_owner
 remove_variable = pmcqr_cancel_target
 pmcqr_recount = yes
 clear_variable_list = pmcqr_drop
 set_variable = { name = pmcqr_kept value = 0 }
 # Iterate queue, not owned: releasing owned entries cannot invalidate iteration.
 ordered_in_list = {
  variable = pmcqr_queue
  order_by = { value = 0 scope:pmcqr_owner = { subtract = "variable_map(pmcqr_order|prev)" } }
  max = scope:pmcqr_owner.var:pmcqr_queued
  check_range_bounds = no
  save_scope_as = pmcqr_target
  if = {
   limit = { scope:pmcqr_owner = { is_target_in_variable_list = { name = pmcqr_sampled target = scope:pmcqr_target } } }
   if = {
    limit = { scope:pmcqr_owner = { is_target_in_variable_list = { name = pmcqr_owned target = scope:pmcqr_target } } }
    scope:pmcqr_owner = {
     if = {
      limit = { is_target_in_variable_list = { name = pmcqr_seen target = scope:pmcqr_target } }
      remove_from_variable_map = { name = pmcqr_grace key = scope:pmcqr_target }
      if = {
       limit = { scope:pmcqr_target = { pmcqr_finished = yes } }
       add_to_variable_map = { name = pmcqr_done key = scope:pmcqr_target value = 1 }
      }
      if = {
       limit = {
        OR = {
         NOT = { has_variable = pmcqr_running }
         var:pmcqr_kept >= var:pmcqr_limit
         is_key_in_variable_map = { name = pmcqr_done target = scope:pmcqr_target }
         scope:pmcqr_target = { NOT = { pmcqr_eligible = yes } }
        }
       }
       pmcqr_cancel_native = yes
      }
      else = { change_variable = { name = pmcqr_kept add = 1 } }
     }
     else_if = {
      limit = {
       NOT = { is_target_in_variable_list = { name = pmcqr_pending target = scope:pmcqr_target } }
       NOT = { is_key_in_variable_map = { name = pmcqr_grace target = scope:pmcqr_target } }
      }
      pmcqr_release = yes
     }
     else = { change_variable = { name = pmcqr_kept add = 1 } }
    }
   }
   # Only remove a completed entry after its native assignment has ended.
   if = {
    limit = {
     scope:pmcqr_owner = { NOT = { is_target_in_variable_list = { name = pmcqr_owned target = scope:pmcqr_target } } }
     OR = {
      num_locations = 0
      pmcqr_finished = yes
      scope:pmcqr_owner = { is_key_in_variable_map = { name = pmcqr_done target = scope:pmcqr_target } }
     }
    }
    scope:pmcqr_owner = { add_to_variable_list = { name = pmcqr_drop target = scope:pmcqr_target } }
   }
  }
 }
 every_in_list = {
  variable = pmcqr_drop
  save_scope_as = pmcqr_target
  scope:pmcqr_owner = {
   add_to_variable_map = { name = pmcqr_retired key = scope:pmcqr_target value = 1 }
   remove_list_variable = { name = pmcqr_queue target = scope:pmcqr_target }
   remove_from_variable_map = { name = pmcqr_order key = scope:pmcqr_target }
   remove_from_variable_map = { name = pmcqr_done key = scope:pmcqr_target }
   remove_from_variable_map = { name = pmcqr_cap_done key = scope:pmcqr_target }
  }
 }
 pmcqr_recount = yes
 pmcqx_process = yes
 pmcqb_process = yes
 set_variable = { name = pmcqr_budget value = num_of_diplomats }
 # Reserve the native annexation request's diplomat until effects propagate.
 if = { limit = { has_variable = pmcqx_sent } change_variable = { name = pmcqr_budget subtract = 1 } }
 if = {
  limit = { has_variable = pmcqr_running }
  ordered_in_list = {
   variable = pmcqr_queue
   order_by = { value = 0 scope:pmcqr_owner = { subtract = "variable_map(pmcqr_order|prev)" } }
   max = scope:pmcqr_owner.var:pmcqr_queued
   check_range_bounds = no
   save_scope_as = pmcqr_target
   if = {
    limit = {
     pmcqr_eligible = yes
     scope:pmcqr_owner = {
      var:pmcqr_count < var:pmcqr_limit
      num_of_diplomats >= 1
      var:pmcqr_budget >= 1
      is_target_in_variable_list = { name = pmcqr_ready target = scope:pmcqr_target }
      NOT = { is_target_in_variable_list = { name = pmcqr_seen target = scope:pmcqr_target } }
      NOT = { is_target_in_variable_list = { name = pmcqr_pending target = scope:pmcqr_target } }
      NOT = { is_target_in_variable_list = { name = pmcqr_owned target = scope:pmcqr_target } }
      NOT = { is_key_in_variable_map = { name = pmcqr_retry target = scope:pmcqr_target } }
     }
    }
    scope:pmcqr_owner = {
     perform_diplomatic_action = { type = improve_relation actor = scope:pmcqr_owner recipient = scope:pmcqr_target }
     add_to_variable_list = { name = pmcqr_owned target = scope:pmcqr_target }
     # Covers action propagation/travel before native pending state is visible.
     add_to_variable_map = { name = pmcqr_grace key = scope:pmcqr_target value = 1 days = 7 }
     change_variable = { name = pmcqr_count add = 1 }
     change_variable = { name = pmcqr_budget subtract = 1 }
    }
   }
  }
 }
 pmcqr_recount = yes
 if = { limit = { var:pmcqr_queued = 0 } remove_variable = pmcqr_running }
 if = {
  limit = { NOT = { has_variable = pmcqr_running } var:pmcqr_count = 0 }
  remove_variable = pmcqr_scheduler
 }
 remove_variable = pmcqr_scan
 remove_variable = pmcqr_refresh
}

# Independent opt-in watcher. Native availability checks subject-specific rules.
# Start at most one annexation per pass; never force acceptance or instant annex.
pmcqx_process = {
 remove_variable = pmcqx_sent
 remove_variable = pmcqx_start_target
 save_scope_as = pmcqx_owner
 set_variable = { name = pmcqx_count value = 0 }
 every_country_we_are_annexing = {
  scope:pmcqx_owner = { change_variable = { name = pmcqx_count add = 1 } }
 }
 clear_variable_list = pmcqx_drop
 every_in_list = {
  variable = pmcqx_requested
  save_scope_as = pmcqx_target
  scope:pmcqx_owner = {
   if = {
    limit = {
     OR = {
      is_annexing = scope:pmcqx_target
      scope:pmcqx_target = { NOT = { is_subject_of = scope:pmcqx_owner } }
      AND = {
       NOT = { is_key_in_variable_map = { name = pmcqx_grace target = scope:pmcqx_target } }
       NOT = { is_target_in_variable_list = { name = pmcqx_pending target = scope:pmcqx_target } }
      }
     }
    }
    add_to_variable_list = { name = pmcqx_drop target = scope:pmcqx_target }
   }
   else = { change_variable = { name = pmcqx_count add = 1 } }
  }
 }
 every_in_list = {
  variable = pmcqx_drop
  save_scope_as = pmcqx_target
  scope:pmcqx_owner = { remove_list_variable = { name = pmcqx_requested target = scope:pmcqx_target } }
 }
 if = {
  limit = { has_variable = pmcqx_running }
  save_scope_as = pmcqx_owner
  every_in_list = {
   variable = pmcqx_subjects
   save_scope_as = pmcqx_target
   if = {
    limit = {
     num_locations > 0
     is_subject_of = scope:pmcqx_owner
     scope:pmcqx_owner = {
      num_of_diplomats >= 1
      var:pmcqx_count < var:pmcqx_limit
      NOT = { has_variable = pmcqx_sent }
      NOT = { is_annexing = scope:pmcqx_target }
      NOT = { is_key_in_variable_map = { name = pmcqx_retry target = scope:pmcqx_target } }
      is_target_in_variable_list = { name = pmcqx_ready target = scope:pmcqx_target }
      NOT = { is_target_in_variable_list = { name = pmcqx_pending target = scope:pmcqx_target } }
     }
    }
    scope:pmcqx_owner = {
     set_variable = { name = pmcqx_start_target value = scope:pmcqx_target }
     add_to_variable_map = { name = pmcqx_retry key = scope:pmcqx_target value = 1 days = 30 }
     add_to_variable_map = { name = pmcqx_grace key = scope:pmcqx_target value = 1 days = 7 }
     add_to_variable_list = { name = pmcqx_requested target = scope:pmcqx_target }
     change_variable = { name = pmcqx_count add = 1 }
     set_variable = pmcqx_sent
    }
   }
  }
 }
}
# Use the existing estate action so its cost, cooldown and timer stay native.
pmcqb_process = {
 if = {
  limit = {
   has_variable = pmcqb_running
   stability <= 80
   estate_satisfaction = { estate_type = estate_type:peasants_estate value >= 0.45 }
   NOT = { is_key_in_variable_map = { name = pmcqb_retry target = this } }
  }
  save_scope_as = pmcqb_owner
  ordered_estate = {
   limit = {
    OR = { estate_type = estate_type:peasants_estate estate_type = estate_type:dhimmi_estate }
    scope:pmcqb_owner = {
     can_do_generic_action = { generic_action = generic_action:ask_commoners_for_stability recipient = scope:pmcqb_owner target = prev }
    }
   }
   order_by = satisfaction
   max = 1
   check_range_bounds = no
   save_scope_as = pmcqb_estate
   scope:pmcqb_owner = {
    add_to_variable_map = { name = pmcqb_retry key = this value = 1 days = 1 }
    fire_generic_action = { type = ask_commoners_for_stability actor = scope:pmcqb_owner recipient = scope:pmcqb_owner target = scope:pmcqb_estate }
   }
  }
 }
}

# Add each subject once while enabled. Retain seen entries so completed/removed targets stay retired.
pmcqr_auto_add_subjects = {
 if = { limit = { is_ai = no has_variable = pmcqr_auto_subjects }
  pmcqr_init = yes
  save_scope_as = pmcqr_auto_owner
  every_subject = {
   limit = { num_locations > 0 }
   save_scope_as = pmcqr_auto_target
   scope:pmcqr_auto_owner = {
    if = { limit = { NOT = { is_target_in_variable_list = { name = pmcqr_auto_subjects_seen target = scope:pmcqr_auto_target } } }
     add_to_variable_list = { name = pmcqr_auto_subjects_seen target = scope:pmcqr_auto_target }
     if = { limit = { NOT = { is_target_in_variable_list = { name = pmcqr_queue target = scope:pmcqr_auto_target } } }
      remove_from_variable_map = { name = pmcqr_retired key = scope:pmcqr_auto_target }
      remove_from_variable_map = { name = pmcqr_done key = scope:pmcqr_auto_target }
      remove_from_variable_map = { name = pmcqr_cap_done key = scope:pmcqr_auto_target }
      remove_from_variable_map = { name = pmcqr_retry key = scope:pmcqr_auto_target }
      change_variable = { name = pmcqr_serial add = 1 }
      add_to_variable_list = { name = pmcqr_queue target = scope:pmcqr_auto_target }
      add_to_variable_map = { name = pmcqr_order key = scope:pmcqr_auto_target value = var:pmcqr_serial }
     }
    }
   }
  }
  pmcqr_recount = yes
 }
}
