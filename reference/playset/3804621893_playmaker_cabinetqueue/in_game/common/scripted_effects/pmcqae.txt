pmcqae_release = {
 every_in_list = { variable = pmcqae_locations remove_location_modifier = pmcqae_bonus }
 clear_variable_list = pmcqae_locations
 every_in_list = { variable = pmcqae_provinces
  if = { limit = { exists = var:pmcqa_reservation var:pmcqa_reservation = scope:pmcqa_slot }
   if = { limit = { exists = var:promoting_culture var:promoting_culture = var:pmcqa_reserved_culture }
    remove_variable = promoting_culture
   }
   remove_variable = pmcqa_reservation
   remove_variable = pmcqa_reserved_culture
  }
 }
 clear_variable_list = pmcqae_provinces
}
pmcqae_clear = { pmcqae_release = yes remove_variable = pmcqae_area }
pmcqae_choose = {
 scope:pmcqa_country = {
  ordered_province = {
   limit = { pmcqa_eligible = yes area = { pmcqae_available = yes } }
   order_by = population max = 1 check_range_bounds = no
   area = { scope:pmcqa_slot = { set_variable = { name = pmcqae_area value = prev } } }
  }
 }
}
pmcqae_apply = {
 set_variable = { name = pmcqa_slot_scale value = { value = 1 add = effective_skill min = 0 } }
 var:pmcqae_area = {
  every_province_in_area = {
   limit = { OR = {
    pmcqae_retained = yes
    AND = { pmcqa_eligible = yes NOT = { has_variable = pmcqa_reservation } }
   } }
   set_variable = { name = pmcqa_reservation value = scope:pmcqa_slot }
   set_variable = { name = promoting_culture value = scope:pmcqa_slot.var:pmcqa_slot_culture }
   set_variable = { name = pmcqa_reserved_culture value = scope:pmcqa_slot.var:pmcqa_slot_culture }
   scope:pmcqa_slot = { add_to_variable_list = { name = pmcqae_provinces target = prev } }
   every_location_in_province = {
    limit = { owner = scope:pmcqa_country OR = { integration_level = integrated integration_level = core } }
    if = { limit = { OR = { NOT = { has_location_modifier = pmcqae_bonus } NOT = { exists = scope:pmcqa_slot.var:pmcqa_slot_applied_scale } AND = { exists = scope:pmcqa_slot.var:pmcqa_slot_applied_scale NOT = { scope:pmcqa_slot.var:pmcqa_slot_applied_scale = scope:pmcqa_slot.var:pmcqa_slot_scale } } } }
     add_location_modifier = { modifier = pmcqae_bonus days = -1 mode = replace size = scope:pmcqa_slot.var:pmcqa_slot_scale }
    }
    scope:pmcqa_slot = { add_to_variable_list = { name = pmcqae_locations target = prev } }
   }
  }
 }
 set_variable = { name = pmcqa_slot_applied_scale value = var:pmcqa_slot_scale }
}
pmcqae_tick = {
 # Keep valid culture assignments and bonuses continuously active.
 every_in_list = {
  variable = pmcqae_locations
  if = { limit = { NOT = {
   owner = scope:pmcqa_country
   OR = { integration_level = integrated integration_level = core }
   province = { pmcqae_retained = yes }
  } } remove_location_modifier = pmcqae_bonus }
 }
 every_in_list = {
  variable = pmcqae_provinces
  if = { limit = { NOT = { pmcqae_retained = yes } }
   if = { limit = { exists = var:pmcqa_reservation var:pmcqa_reservation = scope:pmcqa_slot }
    if = { limit = { exists = var:promoting_culture exists = var:pmcqa_reserved_culture var:promoting_culture = var:pmcqa_reserved_culture }
     remove_variable = promoting_culture
    }
    remove_variable = pmcqa_reservation
    remove_variable = pmcqa_reserved_culture
   }
  }
 }
 if = { limit = { NOT = { var:pmcqa_slot_culture = scope:pmcqa_country.culture } }
  pmcqae_clear = yes
  set_variable = { name = pmcqa_slot_culture value = scope:pmcqa_country.culture }
 }
 if = { limit = { exists = var:pmcqae_area }
  if = { limit = { var:pmcqae_area = { NOT = { any_province_in_area = { OR = {
   pmcqae_retained = yes
   AND = { pmcqa_eligible = yes NOT = { has_variable = pmcqa_reservation } }
  } } } } }
   pmcqae_clear = yes
  }
 }
 if = { limit = { NOT = { exists = var:pmcqae_area } } pmcqae_choose = yes }
 if = { limit = { exists = var:pmcqae_area }
  clear_variable_list = pmcqae_locations
  clear_variable_list = pmcqae_provinces
  pmcqae_apply = yes
  scope:pmcqa_country = { change_variable = { name = pmcqa_multi_count add = 1 } }
 }
 else = {
  pmcqae_clear = yes
  remove_variable = pmcqa_slot_running
  stop_cabinet_action = yes
 }
}
