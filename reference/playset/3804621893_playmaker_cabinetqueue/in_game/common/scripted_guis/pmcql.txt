pmcql_toggle = {
 scope = country
 effect = { root = {
  if = { limit = { is_ai = no }
   if = { limit = { NOT = { has_variable = pmcql_generation } }
    set_variable = { name = pmcql_generation value = 0 }
   }
   change_variable = { name = pmcql_generation add = 1 }
   remove_variable = pmcql_army_pending
   remove_variable = pmcql_open_pending
   remove_variable = pmcql_navy_pending
   if = { limit = { has_variable = pmcql_enabled }
    remove_variable = pmcql_enabled
   }
   else = {
    set_variable = pmcql_enabled
    pmcql_tick = yes
   }
  }
 } }
}
pmcql_army_ready = {
 scope = country
 is_shown = { is_ai = no has_variable = pmcql_enabled at_war = no has_raised_army_levies = yes has_variable = pmcql_army_pending }
}
