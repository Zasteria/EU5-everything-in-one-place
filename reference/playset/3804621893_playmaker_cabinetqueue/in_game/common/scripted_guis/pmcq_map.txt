pmcq_map_integrate = {
 scope = province
 saved_scopes = { pmcq_map_owner }
 is_shown = { root = {
  owner = scope:pmcq_map_owner
  exists = var:pmcqi_reservation
  var:pmcqi_reservation = { has_cabinet_action = yes exists = cabinet_member has_variable = pmcqi_slot_running }
 } }
}
pmcq_map_assimilate = {
 scope = province
 saved_scopes = { pmcq_map_owner }
 is_shown = { root = {
  owner = scope:pmcq_map_owner
  exists = var:pmcqa_reservation
  var:pmcqa_reservation = {
   has_cabinet_action = yes
   exists = cabinet_member
   has_variable = pmcqa_slot_running
   OR = {
    cabinet_action = cabinet_action:pmcqa_action
    cabinet_action = cabinet_action:pmcqae_action
   }
  }
 } }
}
pmcq_assimilation_waiting = {
 scope = cabinet
 is_shown = { root = {
  has_cabinet_action = yes
  cabinet_action = cabinet_action:pmcqa_action
  NOT = { has_variable = pmcqa_slot_target }
 } }
}

pmcq_manual_setup_visible = {
 scope = cabinet
 is_shown = { root = {
  has_cabinet_action = yes
  OR = {
   cabinet_action = cabinet_action:pmcq2_control_action
   cabinet_action = cabinet_action:pmcq2_nobles_action
  }
 } }
}
