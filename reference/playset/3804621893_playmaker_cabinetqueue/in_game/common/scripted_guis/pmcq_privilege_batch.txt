pmcqpriv_begin = { scope = country effect = { root = {
 clear_variable_list = pmcqpriv_pending
 clear_variable_list = pmcqpriv_processing
 set_variable = { name = pmcqpriv_granted value = 0 }
} } }


pmcqpriv_has_selection = { scope = country is_shown = { has_variable_list = pmcqpriv_pending } }

pmcqpriv_confirm = { scope = country effect = { root = {
 save_scope_as = pmcqpriv_owner
 set_variable = { name = pmcqpriv_granted value = 0 }
 clear_variable_list = pmcqpriv_processing
 every_in_list = { variable = pmcqpriv_pending
  save_scope_as = pmcqpriv_copy
  root = { add_to_variable_list = { name = pmcqpriv_processing target = scope:pmcqpriv_copy } }
 }
 every_in_list = { variable = pmcqpriv_processing
  save_scope_as = pmcqpriv_target
  scope:pmcqpriv_owner = {
   if = { limit = {
    NOT = { has_estate_privilege = scope:pmcqpriv_target }
    country_has_estate = scope:pmcqpriv_target.estate_type
    scope:pmcqpriv_target = { is_visible_for = root is_available_for = root }
    can_pay_price = price:grant_privilege
    any_estate = { any_possible_privilege = { this = scope:pmcqpriv_target } }
   } pmcqpriv_grant = yes }
  }
 }
 clear_variable_list = pmcqpriv_processing
} } }

pmcqpriv_multi_enabled = { scope = country is_shown = { has_variable = pmcqpriv_multi } }
pmcqpriv_multi_toggle = { scope = country effect = { root = {
 if = { limit = { has_variable = pmcqpriv_multi } remove_variable = pmcqpriv_multi }
 else = { set_variable = pmcqpriv_multi }
} } }
