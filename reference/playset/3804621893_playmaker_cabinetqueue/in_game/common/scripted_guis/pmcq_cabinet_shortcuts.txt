pmcqcs_prepare = {
 scope = country
 saved_scopes = { pmcqcs_slot pmcqcs_action_id }
 effect = { root = { set_variable = { name = pmcqcs_slot value = scope:pmcqcs_slot } set_variable = { name = pmcqcs_expected_action value = scope:pmcqcs_action_id } } }
}

pmcqcs_has_noblesse = {
 scope = country
 is_shown = { has_reform = government_reform:noblesse_oblige_reform }
}

pmcqcs_values_available = {
 scope = country
 is_shown = {
  NOT = { any_cabinet = {
   has_cabinet_action = yes
   cabinet_action = cabinet_action:change_societal_values
  } }
 }
}

# Match Auto Integrate's conquered-land eligibility, including work in progress.
pmcqcs_integration_available = {
 scope = country
 is_shown = {
  any_province = {
   owner = root
   any_location_in_province = {
    owner = root
    integration_level = conquered
   }
  }
 }
}
