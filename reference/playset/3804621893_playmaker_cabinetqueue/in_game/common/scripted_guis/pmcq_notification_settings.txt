# Absent flag preserves the existing default: block routine Circle requests.
pmcq_toggle_circle_requests = { scope = country effect = { root = {
 if = { limit = { has_variable = pmcq_circle_requests_allowed } remove_variable = pmcq_circle_requests_allowed }
 else = { set_variable = pmcq_circle_requests_allowed }
} } }
pmcq_toggle_notification_mute = { scope = country effect = { root = {
 if = { limit = { has_variable = pmcq_mute_notifications } remove_variable = pmcq_mute_notifications }
 else = { set_variable = pmcq_mute_notifications }
} } }

# Flags passed by the event UI are stored on the country and serialized with the save.
pmcq_event_remember = { scope = country effect = { root = {
 add_to_variable_map = { name = pmcq_event_choices key = scope:event_key value = scope:option_key }
} } }
pmcq_event_forget = { scope = country effect = { root = {
 remove_from_variable_map = { name = pmcq_event_choices key = scope:event_key }
} } }
pmcq_event_memory_toggle = { scope = country effect = { root = {
 if = { limit = { has_variable = pmcq_event_memory_enabled }
  remove_variable = pmcq_event_memory_enabled
  set_variable = pmcq_event_memory_disabled
  add_to_variable_map = { name = cmm key = flag:pmcq__remember_events value = 0 }
 }
 else = {
  set_variable = pmcq_event_memory_enabled
  remove_variable = pmcq_event_memory_disabled
  add_to_variable_map = { name = cmm key = flag:pmcq__remember_events value = 1 }
 }
} } }

} } }
