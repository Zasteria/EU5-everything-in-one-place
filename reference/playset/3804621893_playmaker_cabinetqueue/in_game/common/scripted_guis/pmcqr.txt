pmcqr_prepare_scan = {
 effect = { root = { 
  pmcqr_init = yes
  remove_variable = pmcqr_confirm_armed
  remove_variable = pmcqr_cancel_target
  remove_variable = pmcqx_start_target
  remove_variable = pmcqx_confirm_armed
  pmcqr_begin_scan = yes
  } }
}
pmcqr_arm_recall = {
 saved_scopes = { pmcqr_target }
 effect = { root = { 
  if = {
   limit = {
    exists = var:pmcqr_cancel_target
    var:pmcqr_cancel_target = scope:pmcqr_target
    is_target_in_variable_list = { name = pmcqr_owned target = scope:pmcqr_target }
   }
   set_variable = pmcqr_confirm_armed
  }
  } }
}
pmcqr_clear_recall = {
 effect = { root = {  remove_variable = pmcqr_confirm_armed remove_variable = pmcqr_cancel_target remove_variable = pmcqx_confirm_armed remove_variable = pmcqx_start_target  } }
}
pmcqx_start_requested = {
 saved_scopes = { pmcqx_target }
 is_shown = { root = { 
  has_variable = pmcqx_running
  exists = var:pmcqx_start_target
  var:pmcqx_start_target = scope:pmcqx_target
  NOT = { is_annexing = scope:pmcqx_target }
  } }
}
pmcqx_arm_start = {
 saved_scopes = { pmcqx_target }
 effect = { root = {  set_variable = pmcqx_confirm_armed  } }
}
pmcqr_recall_requested = {
 saved_scopes = { pmcqr_target }
 is_shown = { root = { 
  exists = var:pmcqr_cancel_target
  var:pmcqr_cancel_target = scope:pmcqr_target
  is_target_in_variable_list = { name = pmcqr_owned target = scope:pmcqr_target }
  } }
}
pmcqr_is_owned = {
 saved_scopes = { pmcqr_target }
 is_shown = { root = {  is_target_in_variable_list = { name = pmcqr_owned target = scope:pmcqr_target }  } }
}
pmcqr_finish_scan = {
 is_shown = { root = {  has_variable = pmcqr_scan  } }
 effect = { root = {  if = { limit = { has_variable = pmcqr_scan is_ai = no } pmcqr_process = yes }  } }
}

   limit = { has_variable = pmcqr_scan exists = scope:pmcqr_target is_target_in_variable_list = { name = pmcqr_queue target = scope:pmcqr_target } NOT = { is_target_in_variable_list = { name = pmcqr_sampled target = scope:pmcqr_target } } }
   add_to_variable_list = { name = pmcqr_sampled target = scope:pmcqr_target }
   
  }
  } }
}
