pmcqcm_initialize = {
 scope = country
 effect = { root = {
  # Generation also records initialization; a saved Off choice must stay Off.
  if = { limit = { is_ai = no NOT = { has_variable = pmcqcm_generation } }
   set_variable = { name = pmcqcm_generation value = 1 }
   set_variable = pmcqcm_enabled
   pmcqcm_tick = yes
  }
 } }
}
pmcqcm_toggle = {
 scope = country
 effect = { root = {
  if = { limit = { NOT = { has_variable = pmcqcm_generation } }
   set_variable = { name = pmcqcm_generation value = 0 }
  }
  change_variable = { name = pmcqcm_generation add = 1 }
  if = { limit = { has_variable = pmcqcm_enabled }
   remove_variable = pmcqcm_enabled
  }
  else = {
   set_variable = pmcqcm_enabled
   pmcqcm_tick = yes
  }
 } }
}
