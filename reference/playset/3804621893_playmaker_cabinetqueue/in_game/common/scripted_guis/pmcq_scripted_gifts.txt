pmcqsg_open = { scope = country effect = { root = {
 pmcqsg_refresh = yes remove_variable = pmcqsg_open_requested
} } }
pmcqsg_toggle = { scope = country effect = { root = {
 pmcqsg_init = yes
 if = { limit = { has_variable = pmcqsg_enabled } remove_variable = pmcqsg_enabled }
 else = { set_variable = pmcqsg_enabled pmcqsg_process = yes }
 pmcqsg_refresh = yes
} } }
pmcqsg_run = { scope = country effect = { root = {
 pmcqsg_process = yes pmcqsg_refresh = yes
} } }
pmcqsg_less = { scope = country effect = { root = {
 pmcqsg_init = yes change_variable = { name = pmcqsg_reserve subtract = 100 }
 clamp_variable = { name = pmcqsg_reserve min = 0 max = 1000000 }
 pmcqsg_refresh = yes
} } }
pmcqsg_more = { scope = country effect = { root = {
 pmcqsg_init = yes change_variable = { name = pmcqsg_reserve add = 100 }
 clamp_variable = { name = pmcqsg_reserve min = 0 max = 1000000 }
 pmcqsg_refresh = yes
} } }
pmcqsg_remove = { scope = country saved_scopes = { pmcqsg_target } effect = { root = {
 remove_list_variable = { name = pmcqsg_targets target = scope:pmcqsg_target }
 # Deliberately keep cooldown entries until their expiration.
 pmcqsg_refresh = yes
} } }

pmcqsg_begin_selection = { scope = country effect = { root = {
 clear_variable_list = pmcqsg_pending
 remove_variable = pmcqsg_reopen_picker
} } }
pmcqsg_selected = { scope = country saved_scopes = { pmcqsg_target } is_shown = {
 is_target_in_variable_list = { name = pmcqsg_pending target = scope:pmcqsg_target }
} }
pmcqsg_has_selection = { scope = country is_shown = { has_variable_list = pmcqsg_pending } }
pmcqsg_select = { scope = country saved_scopes = { pmcqsg_target } effect = { root = {
 if = { limit = { exists = scope:pmcqsg_target }
  if = { limit = { is_target_in_variable_list = { name = pmcqsg_pending target = scope:pmcqsg_target } }
   remove_list_variable = { name = pmcqsg_pending target = scope:pmcqsg_target }
  }
  else_if = { limit = { NOT = { is_target_in_variable_list = { name = pmcqsg_targets target = scope:pmcqsg_target } } }
   add_to_variable_list = { name = pmcqsg_pending target = scope:pmcqsg_target }
  }
 }
} } }
pmcqsg_confirm_selection = { scope = country effect = { root = {
 save_scope_as = pmcqsg_sender
 every_in_list = { variable = pmcqsg_pending
  limit = { num_locations > 0 NOT = { this = scope:pmcqsg_sender } is_discovered_by = scope:pmcqsg_sender within_diplomatic_range = scope:pmcqsg_sender }
  save_scope_as = pmcqsg_target
  scope:pmcqsg_sender = {
   if = { limit = { NOT = { is_target_in_variable_list = { name = pmcqsg_targets target = scope:pmcqsg_target } } }
    add_to_variable_list = { name = pmcqsg_targets target = scope:pmcqsg_target }
   }
  }
 }
 clear_variable_list = pmcqsg_pending
 remove_variable = pmcqsg_reopen_picker
 pmcqsg_refresh = yes
} } }
pmcqsg_manual_cooldown = { scope = country saved_scopes = { pmcqsg_target } is_shown = {
 exists = scope:pmcqsg_target
 is_key_in_variable_map = { name = pmcqsg_cooldowns target = scope:pmcqsg_target }
} }
# Clear only removes unconfirmed selections.
pmcqsg_clear_selection = { scope = country effect = { root = {
 clear_variable_list = pmcqsg_pending
} } }
pmcqsg_reopen_consumed = { scope = country effect = { root = {
 remove_variable = pmcqsg_reopen_picker
} } }
