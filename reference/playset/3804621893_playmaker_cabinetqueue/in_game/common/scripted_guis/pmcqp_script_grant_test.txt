pmcqp_script_grant_test = { scope = country
 is_valid = { root = { NOT = { has_estate_privilege = estate_privilege:clergy_strengthen_faith }
  NOT = { has_variable = pmcqp_script_grant_attempted }
  can_pay_price = price:grant_privilege
  any_estate = { any_possible_privilege = { this = estate_privilege:clergy_strengthen_faith } } } }
 effect = { root = {
  if = { limit = { NOT = { has_estate_privilege = estate_privilege:clergy_strengthen_faith }
  NOT = { has_variable = pmcqp_script_grant_attempted }
  can_pay_price = price:grant_privilege
  any_estate = { any_possible_privilege = { this = estate_privilege:clergy_strengthen_faith } } }
   set_variable = pmcqp_script_grant_attempted
   if = { limit = { country_has_estate = estate_type:nobles_estate }
    set_variable = { name = pmcqp_test_penalty_nobles value = {
     value = "estate_power(estate_type:nobles_estate)"
     multiply = 0.20
     multiply = "define:NCountry|GRANT_PRIVILEGE_PENALTY_FOR_OTHERS_SCALE"
    } }
   }
   if = { limit = { country_has_estate = estate_type:burghers_estate }
    set_variable = { name = pmcqp_test_penalty_burghers value = {
     value = "estate_power(estate_type:burghers_estate)"
     multiply = 0.20
     multiply = "define:NCountry|GRANT_PRIVILEGE_PENALTY_FOR_OTHERS_SCALE"
    } }
   }
   if = { limit = { country_has_estate = estate_type:peasants_estate }
    set_variable = { name = pmcqp_test_penalty_peasants value = {
     value = "estate_power(estate_type:peasants_estate)"
     multiply = 0.20
     multiply = "define:NCountry|GRANT_PRIVILEGE_PENALTY_FOR_OTHERS_SCALE"
    } }
   }
   if = { limit = { country_has_estate = estate_type:dhimmi_estate }
    set_variable = { name = pmcqp_test_penalty_dhimmi value = {
     value = "estate_power(estate_type:dhimmi_estate)"
     multiply = 0.20
     multiply = "define:NCountry|GRANT_PRIVILEGE_PENALTY_FOR_OTHERS_SCALE"
    } }
   }
   if = { limit = { country_has_estate = estate_type:tribes_estate }
    set_variable = { name = pmcqp_test_penalty_tribes value = {
     value = "estate_power(estate_type:tribes_estate)"
     multiply = 0.20
     multiply = "define:NCountry|GRANT_PRIVILEGE_PENALTY_FOR_OTHERS_SCALE"
    } }
   }
   if = { limit = { country_has_estate = estate_type:cossacks_estate }
    set_variable = { name = pmcqp_test_penalty_cossacks value = {
     value = "estate_power(estate_type:cossacks_estate)"
     multiply = 0.20
     multiply = "define:NCountry|GRANT_PRIVILEGE_PENALTY_FOR_OTHERS_SCALE"
    } }
   }
   set_variable = { name = pmcqp_test_power_before value = government_power }
   set_variable = { name = pmcqp_test_clergy_before value = "estate_satisfaction(estate_type:clergy_estate)" }
   set_variable = { name = pmcqp_test_nobles_before value = "estate_satisfaction(estate_type:nobles_estate)" }
   set_variable = { name = pmcqp_test_burghers_before value = "estate_satisfaction(estate_type:burghers_estate)" }
   set_variable = { name = pmcqp_test_peasants_before value = "estate_satisfaction(estate_type:peasants_estate)" }
   debug_log = "PMCQ42|script_grant_attempted"
   grant_estate_privilege = estate_privilege:clergy_strengthen_faith
   if = { limit = { has_estate_privilege = estate_privilege:clergy_strengthen_faith }
    pay_price = price:grant_privilege
    add_estate_satisfaction = { type = estate_type:clergy_estate value = {
     value = medium_privilege_target_satisfaction
     multiply = "define:NCountry|GRANT_PRIVILEGE_SATISFACTION_IMPACT"
    } }
    if = { limit = { country_has_estate = estate_type:nobles_estate has_variable = pmcqp_test_penalty_nobles }
     add_estate_satisfaction = { type = estate_type:nobles_estate value = var:pmcqp_test_penalty_nobles }
    }
    if = { limit = { country_has_estate = estate_type:burghers_estate has_variable = pmcqp_test_penalty_burghers }
     add_estate_satisfaction = { type = estate_type:burghers_estate value = var:pmcqp_test_penalty_burghers }
    }
    if = { limit = { country_has_estate = estate_type:peasants_estate has_variable = pmcqp_test_penalty_peasants }
     add_estate_satisfaction = { type = estate_type:peasants_estate value = var:pmcqp_test_penalty_peasants }
    }
    if = { limit = { country_has_estate = estate_type:dhimmi_estate has_variable = pmcqp_test_penalty_dhimmi }
     add_estate_satisfaction = { type = estate_type:dhimmi_estate value = var:pmcqp_test_penalty_dhimmi }
    }
    if = { limit = { country_has_estate = estate_type:tribes_estate has_variable = pmcqp_test_penalty_tribes }
     add_estate_satisfaction = { type = estate_type:tribes_estate value = var:pmcqp_test_penalty_tribes }
    }
    if = { limit = { country_has_estate = estate_type:cossacks_estate has_variable = pmcqp_test_penalty_cossacks }
     add_estate_satisfaction = { type = estate_type:cossacks_estate value = var:pmcqp_test_penalty_cossacks }
    }

    debug_log = "PMCQ42|script_grant_owned_and_price_paid"
   }
   else = { debug_log = "PMCQ42|script_grant_not_owned_no_price_paid" }
   remove_variable = pmcqp_test_penalty_nobles
   remove_variable = pmcqp_test_penalty_burghers
   remove_variable = pmcqp_test_penalty_peasants
   remove_variable = pmcqp_test_penalty_dhimmi
   remove_variable = pmcqp_test_penalty_tribes
   remove_variable = pmcqp_test_penalty_cossacks
   set_variable = { name = pmcqp_test_power_after value = government_power }
   set_variable = { name = pmcqp_test_clergy_after value = "estate_satisfaction(estate_type:clergy_estate)" }
   set_variable = { name = pmcqp_test_nobles_after value = "estate_satisfaction(estate_type:nobles_estate)" }
   set_variable = { name = pmcqp_test_burghers_after value = "estate_satisfaction(estate_type:burghers_estate)" }
   set_variable = { name = pmcqp_test_peasants_after value = "estate_satisfaction(estate_type:peasants_estate)" }
  }
 } }
}
