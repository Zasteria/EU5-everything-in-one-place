pmcqp_economy = {
 scope = country
 effect = { root = {
  set_variable = { name = nobles_estate_target value = 0.01 }
  set_variable = { name = burghers_estate_target value = 0.01 }
  set_variable = { name = clergy_estate_target value = 1 }
  set_variable = { name = peasants_estate_target value = 0.45 }
  set_variable = { name = tribes_estate_target value = 1 }
  set_variable = { name = dhimmi_estate_target value = 0.25 }
  set_automated_system = { system = finances activate = no }
  set_automated_system = { system = replacegenerals activate = yes }
  set_automated_system = { system = replaceadmirals activate = yes }
  # Enable only once; repeated Economy clicks must not restart daily loops.
  if = { limit = { is_ai = no NOT = { has_variable = pmcqcm_enabled } }
   if = { limit = { NOT = { has_variable = pmcqcm_generation } }
    set_variable = { name = pmcqcm_generation value = 0 }
   }
   change_variable = { name = pmcqcm_generation add = 1 }
   set_variable = pmcqcm_enabled
   pmcqcm_tick = yes
  }
  if = { limit = { is_ai = no NOT = { has_variable = pmcql_enabled } }
   if = { limit = { NOT = { has_variable = pmcql_generation } }
    set_variable = { name = pmcql_generation value = 0 }
   }
   change_variable = { name = pmcql_generation add = 1 }
   remove_variable = pmcql_army_pending
   remove_variable = pmcql_open_pending
   remove_variable = pmcql_navy_pending
   set_variable = pmcql_enabled
   # First scan tomorrow so opening Military cannot interrupt budget setup.
   save_scope_value_as = { name = pmcql_generation value = var:pmcql_generation }
   trigger_event_silently = { on_action = pmcql_daily days = 1 }
  }
  if = { limit = { is_ai = no NOT = { has_variable = pmcqb_running } }
   set_variable = pmcqb_running
   pmcqr_wake = yes
  }
 } }
}
