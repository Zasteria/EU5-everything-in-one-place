pmcq_add = {
type = owncountry
ai_tick = never
automation_tick = never
show_message = no

potential = { scope:actor = { is_ai = no } }
allow = { scope:actor = { NOT = { has_variable = pmcq2_running } } }
select_trigger = {
 looking_for_a = province source = actor target_flag = pmcq_province
 move_to_next_section_on_click = yes
 enabled = { scope:actor = { NOT = { is_target_in_variable_list = { name = pmcqpm_pending target = prev } } } }
 selected = { scope:actor = { is_target_in_variable_list = { name = pmcqpm_pending target = prev } } }
 interaction_source_list = { scope:actor = { every_province = { add_to_list = source } } }
 name = pmcqpm_pick column = { data = name } column = { data = control }
 visible = { NOT = { scope:actor = { is_target_in_variable_list = { name = pmcq_queue target = prev } } } }
}
effect = {
 if = { limit = { exists = scope:pmcq_province scope:pmcq_province = { owner = scope:actor } scope:actor = { NOT = { has_variable = pmcq2_running } } }
  scope:actor = {
   if = { limit = {
    NOT = { is_target_in_variable_list = { name = pmcqpm_pending target = scope:pmcq_province } }
    NOT = { is_target_in_variable_list = { name = pmcq_queue target = scope:pmcq_province } }
   } add_to_variable_list = { name = pmcqpm_pending target = scope:pmcq_province } }
   set_variable = pmcqpm_resume
   set_variable = pmcqpm_reopen
  }
 }
}
}

  }
 }
}

}

pmcq_toggle_culture = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { always = yes }
 effect = { scope:actor = {
  if = { limit = { has_variable = pmcq2_running } set_variable = pmcq2_resume_mode }
  pmcq2_pause = yes
  if = { limit = { has_variable = pmcq2_culture_mode } remove_variable = pmcq2_culture_mode }
  else = { set_variable = pmcq2_culture_mode }
  if = { limit = { has_variable = pmcq2_resume_mode } remove_variable = pmcq2_resume_mode pmcq2_start = yes }
 } }
}

pmcq_add_all = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { scope:actor = { NOT = { has_variable = pmcq2_running } } }
 effect = { scope:actor = {
  if = { limit = { NOT = { has_variable = pmcq2_running } }
   save_scope_as = pmcq_all_owner
   if = { limit = { NOT = { has_variable = pmcq_serial } } set_variable = { name = pmcq_serial value = 0 } }
   every_province = {
    limit = { owner = scope:pmcq_all_owner }
    save_scope_as = pmcq_all_province
    scope:pmcq_all_owner = {
     if = { limit = { NOT = { is_target_in_variable_list = { name = pmcq_queue target = scope:pmcq_all_province } } }
      change_variable = { name = pmcq_serial add = 1 }
      scope:pmcq_all_province = { set_variable = { name = pmcq_order value = scope:pmcq_all_owner.var:pmcq_serial } }
      add_to_variable_list = { name = pmcq_queue target = scope:pmcq_all_province }
     }
    }
   }
  }
 } }
