pmcqr_add = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { scope:actor = { always = yes } }
 select_trigger = {
 looking_for_a = country source = actor target_flag = pmcqr_target
 name = pmcqr_pick
 move_to_next_section_on_click = yes
 enabled = { scope:actor = { NOT = { is_target_in_variable_list = { name = pmcqrm_pending target = prev } } } }
 selected = { scope:actor = { is_target_in_variable_list = { name = pmcqrm_pending target = prev } } }
 interaction_source_list = { scope:actor = { every_country_in_diplomatic_range = { add_to_list = source } } }
 column = { data = name }
 visible = {
  NOT = { this = scope:actor }
  num_locations > 0
  is_discovered_by = scope:actor
  scope:actor = { NOT = { is_target_in_variable_list = { name = pmcqr_queue target = prev } } }
 }
}
 effect = { scope:actor = {
  if = { limit = {
   exists = scope:pmcqr_target
   NOT = { is_target_in_variable_list = { name = pmcqrm_pending target = scope:pmcqr_target } }
   NOT = { is_target_in_variable_list = { name = pmcqr_queue target = scope:pmcqr_target } }
  }
   add_to_variable_list = { name = pmcqrm_pending target = scope:pmcqr_target }
  }
  set_variable = pmcqrm_resume
  set_variable = pmcqrm_reopen
 } }
}

 } }

pmcqr_subjects = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { scope:actor = { always = yes } }
 
 effect = { scope:actor = { 
 pmcqr_init = yes
 save_scope_as = pmcqr_owner
 every_subject = {
  limit = { num_locations > 0 }
  save_scope_as = pmcqr_target
  scope:pmcqr_owner = {
   if = {
    limit = { NOT = { is_target_in_variable_list = { name = pmcqr_queue target = scope:pmcqr_target } } }
    # Explicit bulk re-add starts a fresh attempt, like manual selection.
    # Do not reset active/already queued assignments or their queue positions.
    remove_from_variable_map = { name = pmcqr_retired key = scope:pmcqr_target }
    remove_from_variable_map = { name = pmcqr_done key = scope:pmcqr_target }
    remove_from_variable_map = { name = pmcqr_cap_done key = scope:pmcqr_target }
    remove_from_variable_map = { name = pmcqr_retry key = scope:pmcqr_target }
    change_variable = { name = pmcqr_serial add = 1 }
    add_to_variable_list = { name = pmcqr_queue target = scope:pmcqr_target }
    add_to_variable_map = { name = pmcqr_order key = scope:pmcqr_target value = var:pmcqr_serial }
   }
  }
 }
 pmcqr_recount = yes
 pmcqr_wake = yes
 } }
}
