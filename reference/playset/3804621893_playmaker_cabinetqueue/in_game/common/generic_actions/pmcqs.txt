pmcqs_subjects = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { scope:actor = {
  trigger_if = {
   limit = { is_subject = yes }
   subject_type = { allow_subject_creation = yes }
  }
  trigger_if = {
   limit = { exists = union }
   union = { OR = { NOT = { has_active_senior_partner = yes } country_is_senior_partner = { country = scope:actor } } }
  }
  save_temporary_scope_as = pmcqs_country
  any_province = { pmcqs_subject_eligible = yes }
 } }
 effect = { scope:actor = { pmcqs_create_subjects = yes set_variable = pmcqs_close_relations } }
}
