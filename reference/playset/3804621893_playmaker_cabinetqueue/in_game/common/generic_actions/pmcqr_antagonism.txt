# Explicit bulk add: same completion reset as confirming a manual selection.
pmcqr_antagonism = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { always = yes }
 effect = { scope:actor = {
  pmcqr_init = yes
  save_scope_as = pmcqr_owner
  every_country_in_diplomatic_range = {
   limit = {
    NOT = { this = scope:pmcqr_owner }
    num_locations > 0
    is_discovered_by = scope:pmcqr_owner
    # Candidate's antagonism toward the player, not the reverse.
    antagonism = { target = scope:pmcqr_owner value > scope:pmcqr_owner.var:pmcqr_antagonism_threshold }
   }
   save_scope_as = pmcqr_target
   scope:pmcqr_owner = {
    if = {
     limit = { NOT = { is_target_in_variable_list = { name = pmcqr_queue target = scope:pmcqr_target } } }
     remove_from_variable_map = { name = pmcqr_retired key = scope:pmcqr_target }
     remove_from_variable_map = { name = pmcqr_done key = scope:pmcqr_target }
     remove_from_variable_map = { name = pmcqr_cap_done key = scope:pmcqr_target }
     remove_from_variable_map = { name = pmcqr_retry key = scope:pmcqr_target }
     change_variable = { name = pmcqr_serial add = 1 }
     add_to_variable_list = { name = pmcqr_queue target = scope:pmcqr_target }
     add_to_variable_map = { name = pmcqr_order key = scope:pmcqr_target value = var:pmcqr_serial }
    }
   }
  }
  pmcqr_recount = yes
  pmcqr_wake = yes
 } }
}
pmcqr_antagonism_less = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { scope:actor = { OR = { NOT = { has_variable = pmcqr_antagonism_threshold } var:pmcqr_antagonism_threshold > 10 } } }
 effect = { scope:actor = {
  pmcqr_init = yes
  if = { limit = { var:pmcqr_antagonism_threshold > 10 }
   change_variable = { name = pmcqr_antagonism_threshold subtract = 10 }
  }
 } }
}
pmcqr_antagonism_more = {
 type = owncountry
 ai_tick = never
 automation_tick = never
 show_message = no
 potential = { scope:actor = { is_ai = no } }
 allow = { scope:actor = { OR = { NOT = { has_variable = pmcqr_antagonism_threshold } var:pmcqr_antagonism_threshold < 50 } } }
 effect = { scope:actor = {
  pmcqr_init = yes
  if = { limit = { var:pmcqr_antagonism_threshold < 50 }
   change_variable = { name = pmcqr_antagonism_threshold add = 10 }
  }
 } }
}
