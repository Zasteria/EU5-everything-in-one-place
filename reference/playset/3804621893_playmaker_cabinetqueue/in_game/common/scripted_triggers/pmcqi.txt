# Province selection starts with conquered land, matching integrate_province.
pmcqi_eligible = {
 owner = scope:pmcqi_country
 NOT = { pmcqi_native_integration_reserved = yes }
 any_location_in_province = { owner = scope:pmcqi_country integration_level = conquered }
}
# Completion is integrated OR core for every owned location, not full cores.
pmcqi_finished = {
 NOT = {
  any_location_in_province = {
   owner = scope:pmcqi_country
   NOT = { OR = { integration_level = integrated integration_level = core } }
  }
 }
}
pmcqi_ready = {
 is_ai = no
 exists = var:pmcqi_cabinet
 var:pmcqi_cabinet = {
  has_cabinet_action = yes
  cabinet_action = cabinet_action:pmcqi_action
  exists = cabinet_member
 }
}

pmcqil_unlocked = { tag = MLO has_variable = lombard_ambitions_variable }
pmcqil_region = { area = { OR = {
 this = area:lombardy_area this = area:emilia_area this = area:piemonte_area
 this = area:liguria_area this = area:veneto_area this = area:tuscany_area
} } }
pmcqi_supported_action = {
 has_cabinet_action = yes
 OR = {
  cabinet_action = cabinet_action:pmcqi_action
  AND = { cabinet_action = cabinet_action:pmcqie_action scope:pmcqi_country = { has_advance = integrate_area_advance } }
  AND = { cabinet_action = cabinet_action:pmcqil_action scope:pmcqi_country = { pmcqil_unlocked = yes } }
 }
}

# Province scope: inspect live native cabinet targets, including already-running saves.
pmcqi_native_integration_reserved = {
 save_temporary_scope_as = pmcqi_checked_province
 owner = {
  any_cabinet = {
   has_cabinet_action = yes
   OR = {
    AND = {
     OR = {
      cabinet_action = cabinet_action:integrate_province
      cabinet_action = cabinet_action:lat_reintegrate_province
      cabinet_action = cabinet_action:mlo_restore_kingdom_lombardy
      cabinet_action = cabinet_action:oma_zanzibar_integration
     }
     interaction_target:target ?= scope:pmcqi_checked_province
    }
    AND = {
     cabinet_action = cabinet_action:integrate_area
     interaction_target:target ?= scope:pmcqi_checked_province.area
    }
   }
  }
 }
}
# Ignore stale reservations whose slot has stopped or changed to a different action.
pmcqi_auto_integration_reserved = {
 exists = var:pmcqi_reservation
 var:pmcqi_reservation = {
  has_cabinet_action = yes
  OR = {
   cabinet_action = cabinet_action:pmcqi_action
   cabinet_action = cabinet_action:pmcqie_action
   cabinet_action = cabinet_action:pmcqil_action
  }
 }
}
