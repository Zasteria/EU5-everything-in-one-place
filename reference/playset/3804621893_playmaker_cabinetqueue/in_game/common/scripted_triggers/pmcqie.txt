# Area scope, with the acting country saved by the shared scheduler.
pmcqie_available = {
 any_location_in_area = { owner ?= scope:pmcqi_country integration_level = conquered }
 NOT = { any_province_in_area = { owner = scope:pmcqi_country has_variable = pmcqi_reservation } }
 NOT = { any_province_in_area = { owner = scope:pmcqi_country pmcqi_native_integration_reserved = yes } }
}
pmcqie_finished = {
 NOT = { any_location_in_area = {
  owner ?= scope:pmcqi_country
  OR = { integration_level = conquered integration_level = colonized }
 } }
}
