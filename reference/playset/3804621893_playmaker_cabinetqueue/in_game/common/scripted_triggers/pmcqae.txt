pmcqa_supported_action = {
 has_cabinet_action = yes
 OR = {
  cabinet_action = cabinet_action:pmcqa_action
  AND = { cabinet_action = cabinet_action:pmcqae_action scope:pmcqa_country = { modifier:allow_cabinet_assimilate_area = yes } }
 }
}
# Province scope; keep this slot's valid culture assignment across daily refreshes.
pmcqae_retained = {
 owner = scope:pmcqa_country
 exists = var:pmcqa_reservation
 var:pmcqa_reservation = scope:pmcqa_slot
 exists = var:promoting_culture
 var:promoting_culture = scope:pmcqa_slot.var:pmcqa_slot_culture
 pmcqa_needs_assimilation = yes
 NOT = { any_location_in_province = {
  owner = scope:pmcqa_country
  NOT = { OR = { integration_level = integrated integration_level = core } }
 } }
}
# Area scope; avoid interfering with native or automatic culture assignments.
pmcqae_available = {
 any_province_in_area = { pmcqa_eligible = yes }
 NOT = { any_province_in_area = { owner = scope:pmcqa_country OR = { has_variable = promoting_culture has_variable = pmcqa_reservation } } }
}
