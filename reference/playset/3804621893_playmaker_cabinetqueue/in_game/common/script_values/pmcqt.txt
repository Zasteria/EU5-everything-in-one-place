pmcqt_month_index = {
 value = current_year
 multiply = 12
 add = current_month
}
pmcqt_stability_months = {
 value = 0
 if = {
  limit = { has_variable = pmcqt_stability_month has_variable = pmcqt_stability_day has_cooldown = ask_commoners_for_stability_cooldown }
  add = var:pmcqt_stability_month
  subtract = pmcqt_month_index
  if = { limit = { current_day < var:pmcqt_stability_day } add = 1 }
 }
}
