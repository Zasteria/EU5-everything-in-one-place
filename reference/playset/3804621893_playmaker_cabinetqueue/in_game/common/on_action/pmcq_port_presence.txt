# Retire already-scheduled jobs from the old implementation without applying a boost.
pmcqpp_daily = {
 effect = { scope:pmcqpp_country ?= {
  remove_variable = pmcqpp_running
  every_navy = {
   remove_variable = pmcqpp_enabled
   if = { limit = { NOT = { has_variable = pmcqpp_v3_enabled } }
    set_variable = { name = pmcqpp_last_power value = 0 }
    set_variable = { name = pmcqpp_last_added value = 0 }
   }
  }
 } }
}
# Explicit country scope: ability callbacks begin in unit scope.
pmcqpp_v3_daily = {
 trigger = {
  exists = scope:pmcqpp_country
  scope:pmcqpp_country = { var:pmcqpp_generation = scope:pmcqpp_generation }
 }
 effect = { scope:pmcqpp_country = {
  if = {
   limit = { is_ai = no any_navy = { has_variable = pmcqpp_v3_enabled } }
   # A moving/exiled/undocked fleet stops permanently; war merely suspends gain.
   every_navy = {
    limit = { has_variable = pmcqpp_v3_enabled }
    # Existing enabled fleets from v3 did not yet store their starting port.
    if = { limit = { NOT = { exists = var:pmcqpp_home_port } }
     set_variable = { name = pmcqpp_home_port value = unit_location }
    }
    if = { limit = { OR = { NOT = { unit_location = var:pmcqpp_home_port } is_moving = yes in_combat = yes is_exiled = yes NOT = { unit_location = { is_land = yes is_port = yes exists = sea_zone } } } }
     remove_variable = pmcqpp_v3_enabled remove_variable = pmcqpp_credit remove_variable = pmcqpp_home_port clear_variable_list = pmcqpp_seas_served
    }
   }
   # Daily worker only handles movement shutdown; payouts use the native monthly pulse.

   save_scope_as = pmcqpp_country
   save_scope_value_as = { name = pmcqpp_generation value = var:pmcqpp_generation }
   trigger_event_silently = { on_action = pmcqpp_v3_daily days = 1 }
  }
  else = { remove_variable = pmcqpp_running remove_variable = pmcqpp_month_days }
 } }
}

# Add our callback to the native country pulse without replacing its other actions.
monthly_country_pulse = {
 on_actions = { pmcqpp_monthly_payout }
}
pmcqpp_monthly_payout = {
 trigger = {
  is_ai = no
  at_war = no
  any_navy = { has_variable = pmcqpp_v3_enabled }
  NOT = { has_variable = pmcqpp_monthly_lock }
 }
 effect = {
  save_scope_as = pmcqpp_country
  set_variable = { name = pmcqpp_monthly_lock days = 1 }
  remove_variable = pmcqpp_month_days
    # Monthly combined contribution ceiling, replicated on every covered sea tile.
    set_variable = { name = pmcqpp_daily_budget value = 0.9 }
    every_navy = {
     limit = {
      has_variable = pmcqpp_v3_enabled is_moving = no in_combat = no is_exiled = no
      exists = var:pmcqpp_home_port
      unit_location = var:pmcqpp_home_port
      unit_location = { is_land = yes is_port = yes exists = sea_zone }
     }
     save_scope_as = pmcqpp_fleet
     set_variable = { name = pmcqpp_last_added value = 0 }
     set_variable = { name = pmcqpp_last_power value = 0 }
     every_sub_unit = {
      save_scope_value_as = { name = pmcqpp_ship_value value = pmcqpp_ship_power }
      scope:pmcqpp_fleet = { change_variable = { name = pmcqpp_last_power add = scope:pmcqpp_ship_value } }
     }
     # Script-value min is a FLOOR, max is a CEILING (not math min/max functions).
     change_variable = { name = pmcqpp_last_power multiply = { value = 1 add = modifier:global_maritime_presence_modifier min = 0 } }
     # Thirty days of the tripled daily rate, paid together.
     change_variable = { name = pmcqpp_last_power multiply = 30 }
     # Carry fractional gains between payouts.
     if = { limit = { NOT = { has_variable = pmcqpp_credit } } set_variable = { name = pmcqpp_credit value = 0 } }
     change_variable = { name = pmcqpp_credit add = var:pmcqpp_last_power }
     # Limit saved credit to one monthly budget; never build up a large backlog.
     clamp_variable = { name = pmcqpp_credit min = 0 max = 144 }
     set_variable = { name = pmcqpp_last_power value = { value = var:pmcqpp_credit divide = 160 max = scope:pmcqpp_country.var:pmcqpp_daily_budget } }
     if = { limit = { NOT = { has_variable = pmcqpp_pulses } } set_variable = { name = pmcqpp_pulses value = 0 } }
     change_variable = { name = pmcqpp_pulses add = 1 }
     # Replicate the full monthly contribution on every sea tile.
     set_variable = { name = pmcqpp_used_power value = 0 }
     clear_variable_list = pmcqpp_seas_served
     if = { limit = { var:pmcqpp_last_power > 0 }
      unit_location.sea_zone.province_definition = {
       every_location_in_province_definition = {
        limit = {
         is_land = no
         location_maritime_presence_power = { country = scope:pmcqpp_country value < 100 }
        }
        save_scope_value_as = { name = pmcqpp_before value = "location_maritime_presence_power(scope:pmcqpp_country)" }
        change_maritime_presence_power = {
         country = scope:pmcqpp_country
         value = { value = scope:pmcqpp_fleet.var:pmcqpp_last_power max = { value = 100 subtract = "location_maritime_presence_power(scope:pmcqpp_country)" min = 0 } }
        }
        save_scope_value_as = { name = pmcqpp_actual value = { value = "location_maritime_presence_power(scope:pmcqpp_country)" subtract = scope:pmcqpp_before } }
        scope:pmcqpp_fleet = {
         change_variable = { name = pmcqpp_last_added add = scope:pmcqpp_actual }
         # min is a floor: retain the largest actual per-tile gain, not the sum.
         set_variable = { name = pmcqpp_used_power value = { value = var:pmcqpp_used_power min = scope:pmcqpp_actual } }
        }
       }
      }
     }
     # Charge the replicated contribution once, independent of the number of tiles.
     change_variable = { name = pmcqpp_credit subtract = { value = var:pmcqpp_used_power multiply = 160 } }
     clamp_variable = { name = pmcqpp_credit min = 0 max = 144 }
     save_scope_value_as = { name = pmcqpp_used_power value = var:pmcqpp_used_power }
     scope:pmcqpp_country = { change_variable = { name = pmcqpp_daily_budget subtract = scope:pmcqpp_used_power } clamp_variable = { name = pmcqpp_daily_budget min = 0 max = 0.9 } }
    }
 }
}
