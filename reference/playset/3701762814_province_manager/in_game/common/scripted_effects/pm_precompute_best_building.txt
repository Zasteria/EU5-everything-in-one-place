pm_precompute_best_building = {
	# Scope: location
	# Populates pm_new_buildings list with candidate buildings NOT YET BUILT in this location.
	# Building-upgrade candidates (existing buildings with remaining levels) are no longer
	# collected here — vanilla 1.3's native Auto-Expand Buildings system handles upgrades.
	# GUI bridge then compares income via GetBuildingTypeIncomeToOwnerInLocation
	# (no script trigger for income exists, so all comparison happens in GUI)

	save_scope_as = pm_loc
	owner = {
		save_scope_as = pm_owner

		scope:pm_loc = {
			set_variable = { name = pm_best_profit value = 0 }
			remove_variable = pm_best_bt
			clear_variable_list = pm_new_buildings
		}

		# Collect NEW (not yet built) buildings into pm_new_buildings
		every_buildable_building_type = {
			limit = {
				is_special_building = no
				NOT = {
					scope:pm_owner = {
						building_type_is_obsolete = prev
					}
				}
				NOT = { building_category = estate_category }
				scope:pm_owner = {
					can_build_building = prev
				}
				scope:pm_loc = {
					can_build_building = prev
					NOT = { has_building = prev }
				}
				NOT = {
					any_construction_material_for_building_type = {
						NOT = {
							scope:pm_loc = {
								is_produced_in_location_market = prev
							}
						}
					}
				}
			}
			scope:pm_loc = {
				add_to_variable_list = {
					name = pm_new_buildings
					target = prev
				}
			}
		}
	}
}
