# GUI bridge: called from state animation for each candidate building in a location.
# Compares building income (from GUI's GetBuildingTypeIncomeToOwnerInLocation)
# with current best. Updates if better.
# Scopes: ROOT = location, bt = building_type, profit = CFixedPoint via MakeScopeValue
pm_sgui_compare_new_profit = {
	scope = location
	saved_scopes = { bt profit cost }
	effect = {
		if = {
			limit = {
				scope:profit > var:pm_best_profit
				can_build_building = scope:bt
				# Profit threshold filter
				OR = {
					NOT = { owner = { has_variable = pm_profit_threshold } }
					scope:profit >= owner.var:pm_profit_threshold
				}
				# For existing buildings: check remaining capacity including constructions
				OR = {
					NOT = { has_building = scope:bt }
					"building(scope:bt|owner)" = {
						pm_remaining_build_capacity > 0
					}
				}
			}
			set_variable = { name = pm_best_profit value = scope:profit }
			set_variable = { name = pm_best_bt value = scope:bt }
			set_variable = { name = pm_best_cost value = scope:cost }
		}
	}
}

pm_sgui_toggle_all_auto_build = {
	scope = country
	effect = {
		if = {
			limit = {
				any_owned_location = {
					has_variable = pm_auto_build
				}
			}
			# Some locations enabled → disable all
			every_owned_location = {
				limit = { has_variable = pm_auto_build }
				remove_variable = pm_auto_build
			}
		}
		else = {
			# None enabled → enable all
			every_owned_location = {
				set_variable = { name = pm_auto_build value = 1 }
			}
		}
	}
}

pm_sgui_toggle_core_auto_build = {
	scope = country
	effect = {
		if = {
			limit = {
				any_owned_location = {
					has_variable = pm_auto_build
					integration_level = core
				}
			}
			# Some core locations enabled → disable core only
			every_owned_location = {
				limit = {
					has_variable = pm_auto_build
					integration_level = core
				}
				remove_variable = pm_auto_build
			}
		}
		else = {
			# None enabled → enable core only
			every_owned_location = {
				limit = {
					integration_level = core
				}
				set_variable = { name = pm_auto_build value = 1 }
			}
		}
	}
}

	scope = location
	saved_scopes = { cost }

	}
	effect = {
		set_variable = { name = pm_best_cost value = scope:cost }
	}
}

pm_sgui_toggle_auto_build = {
	scope = location
	effect = {
		if = {
			limit = { has_variable = pm_auto_build }
			remove_variable = pm_auto_build
		}
		else = {
			set_variable = { name = pm_auto_build value = 1 }
			# Precompute immediately so a newly-toggled location doesn't sit idle
			# until the next construction-completion pass. Purely script-side —
			# appending to pm_ab_locations/pm_ab_trigger reactively triggers the
			# existing GUI bridge in pm_auto_build_hidden.gui, no GUI onclick needed
			# (chaining onclick handlers on the checkbox itself broke its toggle visual).
			pm_precompute_best_building = yes
			owner = {
				add_to_variable_list = {
					name = pm_ab_locations
					target = prev
				}
				add_to_variable_list = {
					name = pm_ab_trigger
					target = this
				}
			}
		}
	}
}

# Estate filter: mark a location as having estate construction
# Called from hidden GUI widget per-construction via state animation
# scope:is_estate = 1 if construction is estate, 0 otherwise (passed via MakeScopeValue)
pm_sgui_mark_estate_loc = {
	scope = location
	saved_scopes = { is_estate }
	effect = {
		if = {
			limit = { scope:is_estate >= 1 }
			set_variable = { name = pm_has_estate_build value = 1 }
		}
	}
}
