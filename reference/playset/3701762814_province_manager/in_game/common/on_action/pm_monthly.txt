monthly_country_pulse = {
	on_actions = {
		pm_auto_build_monthly
	}
}

pm_auto_build_monthly = {
	trigger = { is_ai = no }
	effect = {
		save_scope_as = pm_owner
		if = {
			limit = { NOT = { has_variable = pm_profit_threshold } }
			set_variable = { name = pm_profit_threshold value = 0 }
		}
		# === Phase 1: BUILD from last month's precomputed pm_best_bt ===
		# Track spending this tick (gold trigger doesn't update mid-iteration)
		set_variable = { name = pm_ab_spent value = 0 }
		every_owned_location = {
			limit = {
				has_variable = pm_auto_build
				has_variable = pm_best_bt
				has_variable = pm_best_cost
				can_build_building = var:pm_best_bt
				num_civil_constructions = 0
				OR = {
					NOT = { owner = { has_variable = pm_profit_threshold } }
					var:pm_best_profit >= owner.var:pm_profit_threshold
				}
			}
			# Check gold - already_spent >= cost + gold_limit
			if = {
				limit = {
					owner = { gold >= prev.pm_ab_min_gold_with_spent }
				}
				construct_building = {
					building_type = var:pm_best_bt
				}
				owner = {
					change_variable = { name = pm_ab_spent add = prev.var:pm_best_cost }
				}
			}
		}
		remove_variable = pm_ab_spent

		# === Phase 2: PRECOMPUTE only locations whose construction just finished ===
		# (Locations newly opted into auto-build get an immediate recompute from the
		# checkbox's onclick chain instead of waiting for this monthly pass — see
		# pm_sgui_toggle_auto_build usage in locations_list_view.gui.)
		clear_variable_list = pm_ab_locations
		clear_variable_list = pm_ab_trigger

		every_owned_location = {
			limit = {
				has_variable = pm_auto_build
				num_civil_constructions = 0
				has_variable = pm_ab_was_building
			}
			remove_variable = pm_ab_was_building
			pm_precompute_best_building = yes
			ROOT = {
				add_to_variable_list = {
					name = pm_ab_locations
					target = prev
				}
			}
		}

		# Track which auto-build locations are currently under construction, so next
		# month's pass above can detect the 1 -> 0 transition (construction finished).
		every_owned_location = {
			limit = {
				has_variable = pm_auto_build
				num_civil_constructions > 0
				NOT = { has_variable = pm_ab_was_building }
			}
			set_variable = { name = pm_ab_was_building value = 1 }
		}

		# Signal hidden GUI for new building profit comparison
		if = {
			limit = {
				any_in_list = {
					variable = pm_ab_locations
					always = yes
				}
			}
			add_to_variable_list = {
				name = pm_ab_trigger
				target = this
			}
		}
	}
}
