# Responsive Universalis - performance / multiplayer-responsiveness define overrides
# Target: EU5 1.3.x (built against 1.3.8). Every override lists the vanilla value.
# Philosophy: cut tick count and AI busywork; keep gameplay-visible behaviour close to vanilla.

NGame = {
	# Vanilla: 2. Hours of game time advanced per simulation tick.
	# 6 = 4 sim ticks per day instead of 12 -> ~3x less per-tick overhead.
	# At bounded speeds (1-6) calendar pacing is unchanged (seconds-per-hour is fixed);
	# what improves is the game's ability to keep up = less tick lag, smoother MP.
	# On speed 7 (unbounded) days simply pass faster.
	# 6 divides 24 evenly and stays close to the 5h combat phase length; if you want
	# more aggression, 8 or 12 also work (Faster Universalis ships 47 with rebalancing).
	HOUR_TICK = 6

	# Vanilla: 3. Reverted to vanilla in v1.0.2 after field testing: raising this
	# to 5 correlated with desyncs at war declaration on speed 4+ in MP sessions
	# (scaling with war participants and moving units). Best reading: the engine
	# limits war speed in multiplayer games because war load breaks sync above 3.
	# The earlier "never saw a forced speed 3" observation was made with the mod
	# already at 5, so vanilla's limit was never actually observed.
	GAME_SPEED_FOR_MULTI_PLAYERS_WARS = 3
}

NUnit = {
	# Army and navy movement accrues PER TICK, not per game hour (proven
	# 2026-08-16: Paris to Corbeil, 1 regiment, vanilla 3-6 days = 36-72
	# ticks at 12/day; modded 9 days = 36 ticks at 4/day, same tick count).
	# 6 hour ticks therefore made all movement 3x slower per calendar day
	# since release. Compensate by the tick ratio (12/4 = 3x vanilla).
	# Found by user Горничная on the Workshop page.
	ARMY_MOVEMENT_SPEED = 0.39	# vanilla 0.13
	NAVY_MOVEMENT_SPEED = 1.5	# vanilla 0.5
}

# Combat compensation (added v1.0.4). Combat advances a FIXED 2 hours per
# tick regardless of HOUR_TICK (proven by dice cadence measurement
# 2026-08-19: vanilla dice every 5 visible hours, modded every ~15 = 3x;
# independently confirmed by Faster Universalis' duration formula). At 4
# ticks/day combat gets 8 processing hours per calendar day instead of
# vanilla 24, so battles ran ~3x longer in calendar days. Scale the
# per-processing-hour rates x3 so battles pace, bleed and end at vanilla
# calendar rates. Total losses and winners are unchanged (morale budgets
# and dice are untouched). Every key RTTI-verified in the exe; changed
# values of all six keys run live in Faster Universalis (23k subs), so
# engine acceptance is proven. The aggressive-ticks addon overrides these
# six keys again with its own x6 values (last filename wins per key).
NCombat = {
	# Dice phase every 2 processing hours instead of 5: 4 phases per
	# calendar day vs vanilla 4.8 (closest integer; exact would be the
	# float 5/3 but float acceptance here is unproven)
	HOURS_PER_PHASE = 2	# vanilla 5
	# Passive morale drain per processing hour x3 = vanilla drain per day
	COMBAT_HOURLY_MORALE_TICK = 0.03	# vanilla 0.01
	# Strength casualties per processing hour x3 = vanilla losses per day
	# (battles now end at vanilla pace, so without this total casualties
	# would drop 3x)
	COMBAT_DAMAGE_MULT = 0.03	# vanilla 0.01
	# Minimums are processing hours: 8/day now, so 8 = 1 day and 24 = 3
	# days, the same real-time minimums as vanilla (24 and 72 at 24/day)
	MINIMUM_COMBAT_DURATION = 8	# vanilla 24
	MINIMUM_NAVAL_COMBAT_DURATION = 24	# vanilla 72
	# Naval bombard cadence scaled with the phase cadence
	BOMBARD_HOURS = 2	# vanilla 5
}
