ism_block_columbian_exchange = {
	type = subject
	category = CATEGORY_SUBJECT_ACTIONS
	show_message = no
	show_message_to_target = no

	potential = {
		scope:actor = {
			NOT = { has_variable = ism__block_columbian_exchange }
		}
	}

	allow = {
	}

	ai_tick = never

	use_enroute = no

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_subject = {
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			situation:columbian_exchange = {
				situation_is_active = yes
			}
			NOR = {
				country_type = building
				country_type = pop
			}
			NOT = { has_country_modifier = ism_overlord_blocking_columbian_exchange }
		}
		enabled = {
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:recipient
			}
			scope:recipient = {
				add_country_modifier = {
					modifier = ism_overlord_blocking_columbian_exchange
					days = -1
					mode = replace
				}
			}
		}
	}
}
