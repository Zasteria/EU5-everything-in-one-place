# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
ism_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# QoL (qol) Tab
		## Automations (automations) Group
		flag:ism__auto_take_over_subject_wars = {
			cmm_sync_bool_alias = {
				setting = ism__auto_take_over_subject_wars
				alias = ism__auto_take_over_subject_wars
			}
			ism_take_over_all_subject_wars = yes
		}
		flag:ism__auto_subject_map_sharing = {
			cmm_sync_bool_alias = {
				setting = ism__auto_subject_map_sharing
				alias = ism__auto_subject_map_sharing
			}
		}
		## Subject Rights (subject_rights) Group
		flag:ism__block_colonization = {
			cmm_sync_bool_alias = {
				setting = ism__block_colonization
				alias = ism__block_colonization
			}
			ism_block_subject_colonization = yes
		}
		flag:ism__block_urbanization = {
			cmm_sync_bool_alias = {
				setting = ism__block_urbanization
				alias = ism__block_urbanization
			}
			ism_block_subject_urbanization = yes
		}
		flag:ism__block_columbian_exchange = {
			cmm_sync_bool_alias = {
				setting = ism__block_columbian_exchange
				alias = ism__block_columbian_exchange
			}
			ism_block_subject_columbian_exchange = yes
		}
		flag:ism__block_declaring_war = {
			cmm_sync_bool_alias = {
				setting = ism__block_declaring_war
				alias = ism__block_declaring_war
			}
			ism_block_subject_war_declarations = yes
		}
	}
}
