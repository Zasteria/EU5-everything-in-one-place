# Disable columbian exchange when blocked
INJECT:move_nw_good_to_new_location = {
	select_trigger = {
		looking_for_a = province_definition
		interaction_source_list = { }
		target_flag = verification
		name = "choose_country"
		show_if = {
			scope:actor = {
				AND = {
					is_subject = yes
					OR = {
						has_country_modifier = ism_overlord_blocking_columbian_exchange
						overlord = { has_variable = ism__block_columbian_exchange }
					}
				}
			}
		}
		visible = {
			always = no
		}
	}
}
