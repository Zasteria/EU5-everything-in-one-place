# Responsive Universalis - Faster Speeds (the add-on formerly called Aggressive Ticks)
# Target: EU5 1.4.x. Every override lists the vanilla value.
#
# On game 1.3 this add-on set 12 hour simulation ticks. Game 1.4 removed the
# tick-length define (NGame.HOUR_TICK): ticks are daily and fixed in the
# engine, so that job is gone. What can still be tuned is how long the game
# WAITS per daily tick at each speed. This add-on halves that wait for
# speeds 1 to 4. Speed 5 has no wait in vanilla (the vanilla comment on the
# key says "last entry is uncapped"), it already runs as fast as the
# computer allows and stays the same.
#
# Measured in game 2026-10-02 at speed 2, same start, same machine: a day
# took 1.02 s in vanilla (table value 1.0) and 0.55 s with this file (table
# value 0.5). That is 1.84 times faster, a little under double because
# computing each day takes time too.
#
# The 1.3 version of this file is in git tag all-mods-for-game-1.3.11.

NGame = {
	# wall-clock seconds per daily tick, one entry per game speed
	GAME_SPEED_TICKS = {
		1.0		# speed 1, vanilla 2.0
		0.5		# speed 2, vanilla 1.0
		0.25	# speed 3, vanilla 0.5
		0.1		# speed 4, vanilla 0.2
		0.0		# speed 5, vanilla 0.0 (uncapped)
	}

	# Vanilla: 2. The speed step the game sets by itself when a war starts
	# in multiplayer. Whether the engine counts this value from 0 or from 1
	# is not known (its own log counts speeds from 0). It does not matter
	# here: for the first steps our table is the vanilla table moved one
	# step (our step N waits as long as vanilla step N+1), so one step lower
	# keeps multiplayer wars at the vanilla real-time pace under either
	# reading. Raising the real pace at war declaration was the desync
	# trigger in July 2026. Not tested in multiplayer.
	GAME_SPEED_FOR_MULTI_PLAYERS_WARS = 1
}
