INJECT:policy_vote = {
	# The patriarchate leader's vote is decisive for patriarchate laws.
	votes = {
		if = {
			limit = {
				exists = scope:recipient
				scope:recipient = {
					international_organization_type = international_organization_type:autocephalous_patriarchate
				}
				scope:actor = {
					is_leader_of_international_organization_of_type = {
						type = autocephalous_patriarchate
					}
				}
			}
			value = 100
		}
		else_if = {
			limit = {
				exists = scope:recipient
				scope:recipient = international_organization:catholic_church
			}
			add = {
				desc = "[cardinals|e]"
				value = scope:actor.num_cardinals
			}
			multiply = {
				desc = "[country_rank|e]"
				value = scope:actor.country_rank_level
				min = 1
			}
		}
		else_if = {
			limit = {
				exists = scope:recipient
				scope:recipient = {
					uses_parliament_for_law_votes_trigger = yes
				}
			}
			value = {
				desc = "[special_status_power|e]"
				value = {
					save_temporary_value_as = {
						name = special_status_power_value
						value = "scope:actor.country_combined_special_status_power(scope:recipient)"
					}
					if = {
						limit = { exists = scope:special_status_power_value }
						value = scope:special_status_power_value
					}
					else = {
						value = 1
					}
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:recipient
				scope:recipient = {
					international_organization_type = international_organization_type:union
				}
			}
			add = {
				desc = "[tax_base|e]"
				value = scope:actor.country_tax_base
			}
		}
		else = {
			value = 1 # 1 vote per country
		}
		min = 0
	}
}
