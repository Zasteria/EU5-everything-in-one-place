famine = {
	monthly_spawn_chance = monthly_spawn_chance_unique

	can_start = {
		current_year >= 1338
		current_year < 1342
	}

	can_end = {
		current_year >= 1342
	}

	visible = {
		capital ?= {
			OR = {
				sub_continent = sub_continent:south_asia
				region = region:north_china_region
				region = region:east_china_region
				region = region:south_china_region
				region = region:west_china_region
			}
		}
	}

	on_start = {
		set_variable = { name = ludi_famine_total_deaths value = 0 }
		sub_continent:south_asia = {
			every_location_in_sub_continent = {
				limit = { is_ownable = yes }
				add_location_modifier = {
					modifier = ludi_famine_modifier
					mode = add_and_extend
					years = -1
				}
				ludi_famine_initialize_location_population_counter_effect = yes
			}
		}
		region:north_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				add_location_modifier = {
					modifier = ludi_famine_modifier
					mode = add_and_extend
					years = -1
				}
				ludi_famine_initialize_location_population_counter_effect = yes
			}
		}
		region:east_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				add_location_modifier = {
					modifier = ludi_famine_modifier
					mode = add_and_extend
					years = -1
				}
				ludi_famine_initialize_location_population_counter_effect = yes
			}
		}
		region:south_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				add_location_modifier = {
					modifier = ludi_famine_modifier
					mode = add_and_extend
					years = -1
				}
				ludi_famine_initialize_location_population_counter_effect = yes
			}
		}
		region:west_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				add_location_modifier = {
					modifier = ludi_famine_modifier
					mode = add_and_extend
					years = -1
				}
				ludi_famine_initialize_location_population_counter_effect = yes
			}
		}
	}

	on_monthly = {
		if = {
			limit = {
				NOT = { has_variable = ludi_famine_total_deaths }
			}
			set_variable = { name = ludi_famine_total_deaths value = 0 }
		}
		sub_continent:south_asia = {
			every_location_in_sub_continent = {
				limit = { is_ownable = yes }
				ludi_famine_count_location_deaths_effect = yes
			}
		}
		region:north_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				ludi_famine_count_location_deaths_effect = yes
			}
		}
		region:east_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				ludi_famine_count_location_deaths_effect = yes
			}
		}
		region:south_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				ludi_famine_count_location_deaths_effect = yes
			}
		}
		region:west_china_region = {
			every_location_in_region = {
				limit = { is_ownable = yes }
				ludi_famine_count_location_deaths_effect = yes
			}
		}
	}

	tooltip = {
		custom_tooltip = ludi_great_asian_famine_active_years_tt
		if = {
			limit = {
				OR = {
					sub_continent = sub_continent:south_asia
					region = region:north_china_region
					region = region:east_china_region
					region = region:south_china_region
					region = region:west_china_region
				}
			}
			custom_tooltip = famine_affected_location_tt
		}
	}

	on_ended = {
		sub_continent:south_asia = {
			every_location_in_sub_continent = {
				remove_location_modifier = ludi_famine_modifier
				remove_location_modifier = ludi_famine_relief_modifier
				ludi_famine_clear_location_population_counter_effect = yes
			}
		}
		region:north_china_region = {
			every_location_in_region = {
				remove_location_modifier = ludi_famine_modifier
				remove_location_modifier = ludi_famine_relief_modifier
				ludi_famine_clear_location_population_counter_effect = yes
			}
		}
		region:east_china_region = {
			every_location_in_region = {
				remove_location_modifier = ludi_famine_modifier
				remove_location_modifier = ludi_famine_relief_modifier
				ludi_famine_clear_location_population_counter_effect = yes
			}
		}
		region:south_china_region = {
			every_location_in_region = {
				remove_location_modifier = ludi_famine_modifier
				remove_location_modifier = ludi_famine_relief_modifier
				ludi_famine_clear_location_population_counter_effect = yes
			}
		}
		region:west_china_region = {
			every_location_in_region = {
				remove_location_modifier = ludi_famine_modifier
				remove_location_modifier = ludi_famine_relief_modifier
				ludi_famine_clear_location_population_counter_effect = yes
			}
		}
		every_country = {
			limit = {
				OR = {
					has_country_modifier = ludi_famine_goods_drive_modifier
					has_country_modifier = ludi_famine_goods_drive_cost_modifier
				}
			}
			remove_country_modifier = ludi_famine_goods_drive_modifier
			remove_country_modifier = ludi_famine_goods_drive_cost_modifier
		}
	}

	map_color = {
		if = {
			limit = {
				OR = {
					sub_continent = sub_continent:south_asia
					region = region:north_china_region
					region = region:east_china_region
					region = region:south_china_region
					region = region:west_china_region
				}
			}
			value = rgb { 105 62 28 }
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "famine_affected_location_tt"
		color = rgb { 105 62 28 }
	}
}
