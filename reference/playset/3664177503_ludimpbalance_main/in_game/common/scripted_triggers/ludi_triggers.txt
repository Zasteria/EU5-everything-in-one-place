TRY_REPLACE:can_upgrade_country_rank_empire = {
	is_great_power = yes
	trigger_if = {
		limit = { 
			country_exists = c:PAP
			NOT = { tag = PAP }
			religion = religion:catholic 
		}
		c:PAP = {
			opinion = { target = root value > 100 }
		}
	}
	OR = {
		total_population > 10000
		total_true_faith_population > 7500
		total_accepted_culture_population > 5000
		
	}
	num_locations >= 200
	prestige > 70
	is_subject = no
}

TRY_REPLACE:is_cossacks = {
	culture = culture:cossack_culture
	owner ?= {
		OR = {
			culture = { has_culture_group = culture_group:russian_group }
			culture = { has_culture_group = culture_group:lithuanian_group }
			culture = { has_culture_group = culture_group:polish_group }
			culture = { has_culture_group = culture_group:romanian_group }
		}
	}
}
