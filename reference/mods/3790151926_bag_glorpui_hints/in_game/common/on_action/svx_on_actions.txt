# CMF's registration hook. It merges with whatever else declares it, so nothing
# of Glorp UI's or Construction Manager's own wiring is displaced.
#
# New game, save load and country transfer, once per human country.
# Scope: country
cmf_on_mod_registration = {
	on_actions = {
		svx_on_registration
	}
}

# Scope: country
svx_on_registration = {
	effect = {
		svx_register = yes
	}
}
