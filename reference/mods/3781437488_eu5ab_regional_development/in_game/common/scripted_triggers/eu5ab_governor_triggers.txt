# Structural filter used for relocation as well as new slots. It mirrors the
# vanilla location potential and non-slot allow conditions without invoking the
# construction-UI can-build trigger, whose vanilla allow block requires actor.
eu5ab_local_governor_candidate = {
	has_owner = yes
	owner ?= root
	owner.capital != this
	OR = {
		location_rank = location_rank:city
		location_rank = location_rank:megalopolis
	}
	has_road_to_capital = yes
	NOT = { has_building = building_type:local_governor }
	NOT = { has_building = building_type:itinerant_court_seat }
	NOT = { has_building = building_type:kings_manor }
	NOT = { has_building = building_type:seljuk_mint }
	NOT = { any_buildings_in_location = { building_type = building_type:local_governor building_levels_under_construction > 0 } }
	NOT = { any_buildings_in_location = { OR = { building_type = building_type:itinerant_court_seat building_type = building_type:kings_manor } building_levels_under_construction > 0 } }
}

eu5ab_naval_governor_candidate = {
	has_owner = yes
	owner ?= root
	OR = {
		location_rank = location_rank:city
		location_rank = location_rank:megalopolis
	}
	is_port = yes
	NOT = { is_connected_to_through_realm = owner.capital }
	is_overseas_for_owner = no
	NOT = { has_building = building_type:naval_governor }
	NOT = { any_buildings_in_location = { building_type = building_type:naval_governor building_levels_under_construction > 0 } }
}


eu5ab_fresh_naval_governor_slot_available = {

}
