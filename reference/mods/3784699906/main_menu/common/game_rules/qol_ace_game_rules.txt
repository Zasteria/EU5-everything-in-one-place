# Auto child education when Community Mod Framework (CMF) is absent.
# The Community Mod Menu overrides this rule mid-game when CMF is present.
# When enabled, every child is auto-assigned the education matching their
# highest aptitude; exclusions and expensive education are only configurable
# from the CMF menu and are NOT applied by this rule.
qol_ace_auto_educate_rule = {
	default = qol_ace_auto_educate_enabled

	qol_ace_auto_educate_off = {
		flag = general_rule
	}

	qol_ace_auto_educate_enabled = {
		flag = general_rule
	}
}
