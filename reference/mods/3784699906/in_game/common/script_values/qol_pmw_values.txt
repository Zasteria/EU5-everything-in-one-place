# Post-war maintenance - script values.
#
# The maintenance sliders run on a 0..1 scale and the lowest value the engine
# accepts is MaintenanceSetting.GetMin, currently 0.50: qol_pmw_target is that
# floor. The slider is additionally clamped by mincap, so a change of the
# engine minimum could only stop the run, never push maintenance below it.
#
# qol_pmw_arm is the arming value. economy_slider raises onvaluechanged only
# when the number under an already displayed slider changes, so the driver
# shows the widget holding this value first and qol_pmw_set_*_target writes the
# real target a beat later - that change is what reaches the engine. One
# hundredth above the target stays inside [mincap, maxcap].
# Scope: country
qol_pmw_target = {
	value = 0.5
}

# Scope: country
qol_pmw_arm = {
	value = 0.51
}
