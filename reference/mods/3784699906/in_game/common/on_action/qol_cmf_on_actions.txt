# Hook Calidad de vida EU5 into the Community Mod Framework on_actions.
# Scope: country

# Settings registration and alias sync on game start, load and yearly pulse.
cmf_on_mod_registration = {
	on_actions = {
		qol_on_register_cmf_mod
	}
}

qol_on_register_cmf_mod = {
	effect = {
		qol_register_cmf_mod = yes
	}
}

# Unified callback: setting toggles, alert clicks and action bar clicks.
cmf_on_callback = {
	on_actions = {
		qol_on_cmf_callback
	}
}

qol_on_cmf_callback = {
	effect = {
		qol_handle_cmf_callback = yes
	}
}
