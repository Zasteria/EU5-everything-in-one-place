imd_auto_accept_is_off = {
	type = internationalorganization

	show_message = no

	ai_tick = never
	automation_tick = never

	select_trigger = {
		looking_for_a = international_organization
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:hre
			scope:actor = {
				NOT = { has_variable = imd_auto_accept }
			}
		}
	}

	potential = {
		exists = international_organization:hre
		scope:actor = {
			NOT = { has_variable = imd_auto_accept }
		}
	}

	effect = {
		scope:actor = {
			set_variable = {
				name = imd_auto_accept
				value = yes
			}
			set_variable = {
				name = request_circle_infrastructure_expansion_timeout
				value = yes
			}
			set_variable = {
				name = request_new_circle_titles_timeout
				value = yes
			}
		}
	}
}

imd_auto_accept_is_on = {
	type = internationalorganization

	show_message = no

	ai_tick = never
	automation_tick = never

	select_trigger = {
		looking_for_a = international_organization
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:hre
			scope:actor = {
				has_variable = imd_auto_accept
			}
		}
	}

	potential = {
		exists = international_organization:hre
		scope:actor = {
			has_variable = imd_auto_accept
		}
	}

	effect = {
		scope:actor = {
			remove_variable = imd_auto_accept
			remove_variable = request_circle_infrastructure_expansion_timeout
			remove_variable = request_new_circle_titles_timeout
			remove_variable = imd_auto_infrastructure_expansion_timeout
			remove_variable = imd_auto_elevate_in_circle_timeout
			remove_variable = imd_auto_county_privileges_timeout
		}
	}
}
