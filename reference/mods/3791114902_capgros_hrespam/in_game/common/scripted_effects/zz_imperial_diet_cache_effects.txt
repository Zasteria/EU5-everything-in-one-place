# Vote projection cache for the petition popup, stored on the HRE.
#
#   imd_votes_<policy>   weight projected to back <policy> over the one in force
#   imd_votes_eligible   weight of everyone whose vote is projected at all
#   imd_cache_group      rotation cursor, 0-3

#weight projected to back one policy over the policy in force for its law
imd_cache_policy_votes = {
	international_organization:hre ?= {
		set_variable = { name = imd_votes_$POLICY$ value = 0 }
		if = {
			limit = {
				international_organization_has_law = law:$LAW$
				policy_enabled_to_international_organization = policy:$POLICY$
				NOT = { international_organization_has_policy = policy:$POLICY$ }
			}
			#a macro inside a quoted data function is mangled, so both operands go through variables
			set_variable = { name = imd_cache_law value = law:$LAW$ }
			set_variable = { name = imd_cache_policy value = policy:$POLICY$ }
			every_international_organization_member = {
				limit = {
					is_ai = yes
					can_vote_in_parliament = international_organization:hre
				}
				save_temporary_scope_as = imd_voter
				save_temporary_scope_value_as = {
					name = imd_bias_proposed
					value = "international_organization:hre.var:imd_cache_policy.ai_policy_resolution_vote_bias(scope:imd_voter|international_organization:hre)"
				}
				save_temporary_scope_value_as = {
					name = imd_bias_in_force
					value = "international_organization:hre.law_policy(international_organization:hre.var:imd_cache_law).ai_policy_resolution_vote_bias(scope:imd_voter|international_organization:hre)"
				}
				if = {
					limit = { scope:imd_bias_proposed >= scope:imd_bias_in_force }
					international_organization:hre = {
						change_variable = {
							name = imd_votes_$POLICY$
							add = "scope:imd_voter.country_combined_special_status_power(international_organization:hre)"
						}
					}
				}
			}
			remove_variable = imd_cache_law
			remove_variable = imd_cache_policy
		}
	}
}

#everyone whose vote is projected at all, refreshed in full every month
imd_cache_eligible_votes = {
	international_organization:hre ?= {
		set_variable = { name = imd_votes_eligible value = 0 }
		every_international_organization_member = {
			limit = {
				is_ai = yes
				can_vote_in_parliament = international_organization:hre
			}
			save_temporary_scope_as = imd_voter
			international_organization:hre = {
				change_variable = {
					name = imd_votes_eligible
					add = "scope:imd_voter.country_combined_special_status_power(international_organization:hre)"
				}
			}
		}
	}
}
