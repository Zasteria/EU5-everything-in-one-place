INJECT:request_county_privileges = {
	ai_prerequisite = {
		NOT = { has_cooldown = ai_spam_protection_request_county_privileges }
		NOT = {
			international_organization:hre ?= {
				leader_country ?= {
					has_variable = imd_auto_accept
				}
			}
		}
	}

	reject_effect = {
		hidden_effect = {
			scope:actor = {
				add_cooldown = {
					type = ai_spam_protection_request_county_privileges
					years = 5
				}
			}
		}
	}

	ai_will_do = {
		scope:actor = { add = ai_spam_protection }
		add = {
			desc = "USEFULNESS_OF_GOVERNMENT_REFORM"
			value = "scope:target.modifier_utility(scope:actor)"
			multiply = hre_county_privileges_utility_weight
		}
		add = {
			desc = "[gold|e]"
			value = hre_county_privileges_gold_score
		}
		add = {
			desc = "TARGET_ACCEPTANCE"
			value = "scope:actor.country_interaction_acceptance(country_interaction:request_county_privileges|scope:recipient)"
		}
		add = {
			desc = "HRE_COUNTY_PRIVILEGES_REFORM_FIT"
			value = hre_county_privileges_drift_score
		}
		subtract = {
			desc = "HRE_COUNTY_PRIVILEGES_LOW_GOVERNMENT_POWER"
			value = hre_county_privileges_government_power_shortfall
		}
		subtract = {
			desc = "HRE_COUNTY_PRIVILEGES_LOW_STABILITY"
			value = hre_county_privileges_stability_shortfall
		}
	}
}

imd_auto_county_privileges = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS

	show_message = no
	show_in_gui_list = no

	ai_limit_per_check = 1

	ai_prerequisite = {
		NOT = { has_cooldown = ai_spam_protection_request_county_privileges }
	}

	potential = {
		exists = international_organization:hre
		international_organization:hre = {
			leader_country ?= {
				has_variable = imd_auto_accept
				NOT = { has_variable = imd_auto_county_privileges_timeout }
			}
		}
		scope:actor = {
			is_member_of_international_organization = international_organization:hre
			is_emperor = no
			NOT = { has_reform = government_reform:margraviate }
			NOT = { has_reform = government_reform:landgraviate }
			government_type = government_type:monarchy
			num_open_reform_slots > 0
		}
	}

	allow = {
		scope:actor = {
			gold >= {
				value = grant_county_privileges_cost
				multiply = 4
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			international_organization:hre = {
				leader_country ?= {
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_emperor"
		column = {
			data = name
		}
		visible = {
			is_emperor = yes
		}
		enabled = {
			international_organization:hre = { international_organization_has_leader = yes }
		}
	}

	select_trigger = {
		looking_for_a = government_reform
		interaction_source_list = {
			government_reform:margraviate = {
				add_to_list = source
			}
			government_reform:landgraviate = {
				add_to_list = source
			}
		}
		cache_interaction_source_list = yes
		target_flag = target
		name = "choose_government_reform"
		column = {
			data = name
		}
		visible = {
		}
		enabled = {
			is_allowed_for = scope:actor
		}
	}

	effect = {
		scope:actor = {
			transfer_gold_to_org_country_effect = {
				international_organization = international_organization:hre
				country = scope:recipient
				value = grant_county_privileges_cost
			}
			add_opinion = {
				target = scope:recipient
				modifier = opinion_granted_country_privileges
			}
			if = {
				limit = { exists = scope:target }
				add_reform = scope:target
			}
		}
		scope:recipient = {
			set_variable = {
				name = imd_auto_county_privileges_timeout
				years = 2
			}
		}
	}

	ai_will_do = {
		scope:actor = { add = ai_spam_protection }
		add = {
			desc = "USEFULNESS_OF_GOVERNMENT_REFORM"
			value = "scope:target.modifier_utility(scope:actor)"
			multiply = hre_county_privileges_utility_weight
		}
		add = {
			desc = "[gold|e]"
			value = hre_county_privileges_gold_score
		}
		add = {
			desc = "TARGET_ACCEPTANCE"
			value = "scope:actor.country_interaction_acceptance(country_interaction:request_county_privileges|scope:recipient)"
		}
		add = {
			desc = "HRE_COUNTY_PRIVILEGES_REFORM_FIT"
			value = hre_county_privileges_drift_score
		}
		subtract = {
			desc = "HRE_COUNTY_PRIVILEGES_LOW_GOVERNMENT_POWER"
			value = hre_county_privileges_government_power_shortfall
		}
		subtract = {
			desc = "HRE_COUNTY_PRIVILEGES_LOW_STABILITY"
			value = hre_county_privileges_stability_shortfall
		}
	}
}
