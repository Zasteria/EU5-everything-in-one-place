INJECT:request_circle_infrastructure_expansion = {
	ai_limit_per_check = 1

	cooldown = {
		type = request_circle_infrastructure_expansion
		months = 120
	}

	ai_prerequisite = {
		NOT = { has_country_modifier = granted_circle_infrastructure_expansion }
	}

	reject_effect = {
		scope:recipient = {
			every_imperial_circle_country_is_member_of = {
				add_circle_satisfaction = -2
			}
		}
		hidden_effect = {
			scope:actor = {
				add_cooldown = {
					type = request_circle_infrastructure_expansion
					years = 5
				}
			}
		}
	}
}

imd_auto_infrastructure_expansion = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS
	icon = circle_leader_action

	show_message = no
	show_in_gui_list = no

	ai_limit_per_check = 1

	cooldown = {
		type = imd_auto_infrastructure_expansion
		months = 120
	}

	potential = {
		scope:actor = {
			is_member_of_international_organization = international_organization:hre
			is_imperial_circle_leader = no
			is_emperor = no
			NOT = { has_country_modifier = granted_circle_infrastructure_expansion }
			any_imperial_circle_country_is_member_of = {
				has_circle_leader = yes
				circle_leader ?= {
					has_variable = imd_auto_accept
					NOT = { has_variable = imd_auto_infrastructure_expansion_timeout }
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			international_organization:hre = {
				every_international_organization_member = {
					limit = { is_imperial_circle_leader = yes }
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_circle_leader"
		column = {
			data = name
		}
		visible = {
			is_imperial_circle_leader = yes
			in_same_imperial_circle = scope:actor
		}
	}

	effect = {
		scope:actor = {
			add_country_modifier = {
				modifier = granted_circle_infrastructure_expansion
				years = 10
			}
		}
		scope:recipient = {
			every_imperial_circle_country_is_member_of = {
				add_circle_satisfaction = 1
			}
			set_variable = {
				name = imd_auto_infrastructure_expansion_timeout
				years = 2
			}
		}
	}

	ai_will_do = {
		add = {
			desc = "ACTOR_ACCEPTANCE"
			value = "scope:actor.country_interaction_acceptance(country_interaction:request_circle_infrastructure_expansion|scope:recipient)"
		}
	}
}
