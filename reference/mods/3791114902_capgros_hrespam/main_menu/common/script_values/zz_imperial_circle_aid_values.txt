imperial_circle_aid_yearly_share = 0.2

imperial_circle_aid_ceiling_months = 1

imperial_circle_aid_max_reserve_months = 6

imperial_circle_aid_fatigue_per_grant = 1

imperial_circle_aid_fatigue_decay_monthly = 0.0278

imperial_circle_aid_request = {
	if = {
		limit = { exists = scope:actor }
		value = scope:actor.monthly_income_trade_and_tax
	}
	else = {
		value = monthly_income_trade_and_tax
	}
	multiply = 12
	multiply = imperial_circle_aid_yearly_share
	min = 0
}

imperial_circle_aid_max_reserve = {
	if = {
		limit = { exists = scope:actor }
		value = scope:actor.monthly_income_trade_and_tax
	}
	else = {
		value = monthly_income_trade_and_tax
	}
	multiply = imperial_circle_aid_max_reserve_months
	min = 0
}

imperial_circle_aid_ceiling = {
	if = {
		limit = { exists = scope:recipient }
		value = scope:recipient.monthly_income_trade_and_tax
	}
	else = {
		value = monthly_income_trade_and_tax
	}
	multiply = imperial_circle_aid_ceiling_months
	min = 0
}

imperial_circle_aid_amount = {
	value = imperial_circle_aid_request
	max = imperial_circle_aid_ceiling
}

imperial_circle_aid_fatigue = {
	value = 0
	if = {
		limit = {
			exists = scope:recipient
			scope:recipient = { has_variable = imd_circle_aid_fatigue }
		}
		add = scope:recipient.var:imd_circle_aid_fatigue
	}
	min = 0
}

imperial_circle_aid_fatigue_penalty = {
	value = imperial_circle_aid_fatigue
	multiply = imperial_circle_aid_fatigue
	multiply = 5
	add = {
		value = imperial_circle_aid_fatigue
		multiply = 15
	}
}

imperial_circle_aid_need_from_loans = {
	value = "scope:actor.num_loans"
	multiply = 8
	max = 32
}
