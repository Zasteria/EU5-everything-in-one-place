#the petitioner's votes in the Diet: its combined special status power
imperial_diet_petition_votes = {
	if = {
		limit = { exists = scope:actor }
		value = "scope:actor.country_combined_special_status_power(international_organization:hre)"
	}
	else = {
		value = "country_combined_special_status_power(international_organization:hre)"
	}
	min = 1
}

imperial_diet_petition_authority_base = 0.5
imperial_diet_petition_authority_span = 2
imperial_diet_petition_authority_half_weight = 50

#base + span * votes / (votes + half_weight)
imperial_diet_petition_accept_authority = {
	save_temporary_value_as = {
		name = imperial_diet_petition_vote_denominator
		value = {
			value = imperial_diet_petition_votes
			add = imperial_diet_petition_authority_half_weight
		}
	}

	value = imperial_diet_petition_votes
	divide = scope:imperial_diet_petition_vote_denominator
	multiply = imperial_diet_petition_authority_span
	add = imperial_diet_petition_authority_base
}
