# Post-war maintenance - the calls the driver GUI makes back into script. The
# widget cannot write country variables itself, so the two-step arm -> target
# dance runs through these. Scope: country

# Second step: the real number instead of the one the slider was armed with.
# The change under the displayed slider is what reaches the engine.
qol_pmw_set_army_target = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = { set_variable = { name = qol_pmw_army_target value = qol_pmw_target } }
}

qol_pmw_set_navy_target = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = { set_variable = { name = qol_pmw_navy_target value = qol_pmw_target } }
}

qol_pmw_set_fort_target = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = { set_variable = { name = qol_pmw_fort_target value = qol_pmw_target } }
}

# The drive is over for this branch: drop its flag and let the busy flag
# follow, so the driver can go back to sleep. Scope: country
qol_pmw_clear_army_go = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		cmf_log = { action = qol_pmw_log_done }
		remove_variable = qol_pmw_army_go
		qol_pmw_update_busy = yes
	}
}
