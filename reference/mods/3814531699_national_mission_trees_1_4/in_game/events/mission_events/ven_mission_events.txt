ven_events.18 = {
	type = country_event
	category = mission_event
	title = ven_events.18.t
	desc  = ven_events.18.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = ven_events.18.a
		add_country_modifier = { modifier = ven_latin_empire_unlock years = -1 mode = add }
		add_prestige = 150
		add_legitimacy = legitimacy_major_bonus
		ruler = {
			save_scope_as = ven_latin_empire_ruler
		}
		province:constantinople_province = {
			create_location_country_from_province = {
				ruler = scope:ven_latin_empire_ruler
				subject_type = subject_type:vassal

				capital = {
					province = {
						save_scope_as = ven_latin_empire_capital
					}
				}

				save_scope_as = ven_latin_empire_state
			}
		}
		every_owned_location = {
			limit = {
				OR = {
					area = area:morea_area
					area = area:northern_greece_area
					area = area:aegean_archipelago_area
					area = area:thrace_area
					area = area:macedonia_area
				}
				NOT = { owner = scope:ven_latin_empire_state }
			}
			change_location_owner = scope:ven_latin_empire_state
		}
		scope:ven_latin_empire_state = {
			change_tag_cosmetic = { tag = LAT }
		}
	}
}
