teo_events.1 = {
	type = country_event
	category = country_event
	title = teo_events.1.t
	desc  = teo_events.1.d
	fire_only_once = yes
	immediate = {}
	option = {
		name = teo_events.1.a
		ai_chance = { base = 100 }
	}
	option = {
		name = teo_events.1.b
		ai_chance = { base = 0 }
		change_culture = culture:gothic_culture
		add_country_modifier = { modifier = teo_gothic_penalty_mod years = 100 mode = add }
	}
}

	type = country_event
	category = mission_event
	title = teo_events.5.t

	option = {
		name = teo_events.5.a

			limit = { country_exists = c:GAZ }

			}
		}
	}
	option = {
		name = teo_events.5.b
		ai_chance = { base = 100 }
	}
}
