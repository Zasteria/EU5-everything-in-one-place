# Event 1: ブルサの揺り籠
# =============================================================================

som_ottoman_events.1 = {
	type = country_event
	category = mission_event

	title = som_ottoman_events.1.t
	desc  = som_ottoman_events.1.d

	fire_only_once = yes

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:clergy_estate
			background = estate_type:clergy_estate
		}
	}

	option = {
		name = som_ottoman_events.1.a
		custom_tooltip = som_ottoman_events.1.a.tt

		location:bursa = {
			# 宗教変換: 全POPをスンニ派へ
			every_pop = {
				limit = {
					NOT = { religion = religion:sunni }
				}
				change_pop_religion = religion:sunni
			}
			# 文化変換: 全POPをトルコ文化へ
			every_pop = {
				limit = {
					NOT = { culture = culture:turkish_culture }
				}
				change_pop_culture = culture:turkish_culture
			}
			change_prosperity = prosperity_weak_bonus
		}

		add_legitimacy = legitimacy_weak_bonus
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_weak_bonus
		}
	}
}

# =============================================================================
# Event 61: マルジュ・ダービクの戦い — マムルーク制圧後
# =============================================================================

som_ottoman_events.61 = {
	type = country_event
	category = mission_event
	title = som_ottoman_events.61.t
	desc  = som_ottoman_events.61.d
	fire_only_once = yes
	illustration_tags = { 10 = exterior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
		# MAMとの戦争をスコープに保存
		any_current_war = {
			limit = {
				OR = {
					attacker_leader = root
					defender_leader = root
				}
				OR = {
					attacker_leader = c:MAM
					defender_leader = c:MAM
				}
			}
			save_scope_as = mam_war_scope
		}
	}
	option = {
		name = som_ottoman_events.61.a
		# 白紙講和
		if = {
			limit = { exists = scope:mam_war_scope }
			white_peace = scope:mam_war_scope
		}
		# エヤレットとして属国化
		if = {
			limit = {
				country_exists = c:MAM
				NOT = { c:MAM = { is_subject = yes } }
			}
			c:MAM = {
				make_subject_of = {
					target = root
					type = subject_type:eyalet
				}
			}
		}
	}
	option = {
		name = som_ottoman_events.61.b
		# 和平のみ
		if = {
			limit = { exists = scope:mam_war_scope }
			white_peace = scope:mam_war_scope
		}
	}
}
