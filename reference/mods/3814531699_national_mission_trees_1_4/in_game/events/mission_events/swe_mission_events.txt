namespace = swe_events

# =============================================================================
# Event 30: スウェーデン帝国の誕生——ストールマクツティデン
# =============================================================================

swe_events.30 = {
	type = country_event
	category = mission_event
	title = swe_events.30.t
	desc  = swe_events.30.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = swe_events.30.a
		add_country_modifier = { modifier = swe_scandinavia_united_mod years = -1 mode = add }
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
	}
}

# =============================================================================
# Event 40: 信仰の選択——カトリック・ルター派・ノルド
# =============================================================================

swe_events.40 = {
	type = country_event
	category = mission_event
	title = swe_events.40.t
	desc  = swe_events.40.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:clergy_estate
			background = estate_type:clergy_estate
		}
	}
	option = {
		name = swe_events.40.a
		# カトリック堅守
		add_country_modifier = { modifier = swe_faith_steadfast_mod years = 50 mode = add }
	}
	option = {
		name = swe_events.40.b
		# ルター派改革
		add_country_modifier = { modifier = swe_reform_winds_mod years = 50 mode = add }
	}
	option = {
		name = swe_events.40.c
		# ノルド改宗
		change_religion = religion:norse
		add_country_modifier = { modifier = swe_old_gods_calling_mod years = 50 mode = add }
	}
}
