	type = country_event
	category = mission_event
	title = byz_events.2.t

	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:burghers_estate

		}

		name = byz_events.2.a

			}
			change_prosperity = prosperity_mild_bonus
		}
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = 12
		}
	}
}

# =============================================================================
# Event 22: イスラムの聖地
# =============================================================================
byz_events.22 = {
	type = country_event
	category = mission_event
	title = byz_events.22.t
	desc  = byz_events.22.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:clergy_estate
			background = estate_type:clergy_estate
		}
	}
	# 選択肢A: 聖地を破壊する
	option = {
		name = byz_events.22.a
		location:mecca = {
			every_pop = {
				limit = { religion = religion:sunni }
				split_pop = { fraction = 0.10 religion = root.religion }
			}
		}
		location:medina = {
			every_pop = {
				limit = { religion = religion:sunni }
				split_pop = { fraction = 0.10 religion = root.religion }
			}
		}
		add_country_modifier = {
			modifier = byz_islamic_sites_destroyed
			years = -1
			mode = add
		}
	}
	# 選択肢B: 聖地を受容する
	option = {
		name = byz_events.22.b
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = 36
		}
		add_country_modifier = {
			modifier = byz_islamic_sites_accepted
			years = -1
			mode = add
		}
	}
}
