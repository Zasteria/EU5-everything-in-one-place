# =============================================================================
# Event 6: アヴィニョン問題の解決
# =============================================================================

pro_events.6 = {
	type = country_event
	category = mission_event
	title = pro_events.6.t
	desc  = pro_events.6.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:clergy_estate
			background = estate_type:clergy_estate
		}
	}
	option = {
		name = pro_events.6.a
		# PAPがavignonを割譲し、プロヴァンスが中核化
		if = {
			limit = {
				country_exists = c:PAP
				c:PAP = { owns = location:avignon }
			}
			c:PAP = {
				location:avignon = {
					remove_core = c:PAP
					add_core = root
					change_location_owner = root
				}
			}
		}
		if = {
			limit = { owns = location:avignon }
			location:avignon = {
				add_core = root
			}
		}
		add_country_modifier = { modifier = pro_avignon_question_mod years = -1 mode = add }
	}
}
