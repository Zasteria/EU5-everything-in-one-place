hab_events.30 = {
	type = country_event
	category = mission_event
	title = hab_events.30.t
	desc  = hab_events.30.d
	fire_only_once = yes

	immediate = {}

	option = {
		name = hab_events.30.a
		add_prestige = 35

		# ① イタリア地方の自国所有ロケーションをリスト化
		every_owned_location = {
			limit = { region = region:italy_region }
			add_to_list = list_of_ita_locations
		}

		# ② 起点ロケーション選択（ローマ優先・なければ人口最大）
		if = {
			limit = {
				owns = location:rome
				location:rome = { region = region:italy_region }
			}
			location:rome = {
				save_scope_as = ita_capital_location
			}
		}
		else = {
			ordered_in_list = {
				list = list_of_ita_locations
				order_by = pop_size
				descending = yes
				max = 1
				save_scope_as = ita_capital_location
			}
		}

		# ③ 起点から新国家ITA作成
		scope:ita_capital_location = {
			create_country_from_location = {
				define_unique_country_tag = ITA
				hidden_effect = {
					change_country_name = ITA
					change_country_adjective = ITA_ADJ
					change_country_color = map_ITA
					change_country_flag = ITA
				}
				save_scope_as = ita_new_state
			}
		}

		# ④ 王国ランクに設定
		scope:ita_new_state = {
			set_country_rank = country_rank:rank_kingdom
		}

		# ⑤ イタリア地方の全自国領をITAへ移譲・コア化
		every_in_list = {
			list = list_of_ita_locations
			limit = { NOT = { owner = scope:ita_new_state } }
			change_location_owner = scope:ita_new_state
			add_core = scope:ita_new_state
			change_integration_level = core
		}

		# ⑥ 統合された王冠領として属国化
		scope:ita_new_state = {
			make_subject_of = {
				target = root
				type = subject_type:hab_integrated_crown
			}
		}
	}

	option = {
		name = hab_events.30.b
		add_prestige = 5
	}
}
