namespace = nor_events

# =============================================================================
# Event 30: スカンジナビア統一
# =============================================================================

nor_events.30 = {
	type = country_event
	category = mission_event
	title = nor_events.30.t
	desc  = nor_events.30.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = nor_events.30.a
		add_country_modifier = { modifier = nor_scandinavia_united_mod years = -1 mode = add }
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
	}
}
