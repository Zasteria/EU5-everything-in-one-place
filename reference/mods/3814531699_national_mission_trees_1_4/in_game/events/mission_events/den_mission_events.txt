namespace = den_events

# =============================================================================
# Event 10: ノルウェー同君連合
# =============================================================================

den_events.10 = {
	type = country_event
	category = mission_event
	title = den_events.10.t
	desc  = den_events.10.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = den_events.10.a
		add_country_modifier = { modifier = den_norway_union_bonus_mod years = -1 mode = add }
		add_prestige = 10
	}
}

# =============================================================================
# Event 40: 宗教の問い——カトリック維持・ルター派改革・ノルド復興
# =============================================================================

den_events.40 = {
	type = country_event
	category = mission_event
	title = den_events.40.t
	desc  = den_events.40.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:clergy_estate
			background = estate_type:clergy_estate
		}
	}
	option = {
		name = den_events.40.a
		# カトリック維持
		if = {
			limit = { NOT = { dominant_religion = religion:catholic } }
			custom_tooltip = { text = den_events.40.a.tt }
		}
		add_country_modifier = { modifier = den_faith_steadfast_mod years = 50 mode = add }
	}
	option = {
		name = den_events.40.b
		# ルター派への道
		custom_tooltip = { text = den_events.40.b.tt }
		add_country_modifier = { modifier = den_reform_winds_mod years = 50 mode = add }
	}
	option = {
		name = den_events.40.c
		# ノルド復興への道——norsに改宗
		custom_tooltip = { text = den_events.40.c.tt }
		change_religion = religion:norse
		add_country_modifier = { modifier = den_old_gods_calling_mod years = 50 mode = add }
	}
}
