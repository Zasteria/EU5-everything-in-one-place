# =============================================================================
# Event 99: 聖ステファンの王冠の栄光
# =============================================================================

hun_events.99 = {
	type = country_event
	category = mission_event
	title = hun_events.99.t
	desc  = hun_events.99.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:burghers_estate
			background = estate_type:burghers_estate
		}
	}
	option = {
		name = hun_events.99.a
		add_country_modifier = { modifier = hun_eternal_crown_mod years = -1 mode = add }
		set_country_rank_effect = { rank = country_rank:rank_empire }
	}
}

		name = hun_events.20.b

		location:brasov = {
			change_location_rank = location_rank:city
		}
		add_gold = { value = -1000 }
	}
}

# =============================================================================
# Event 30: クロアチアとの歴史的合意
# =============================================================================

hun_events.30 = {
	type = country_event
	category = mission_event
	title = hun_events.30.t
	desc  = hun_events.30.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = hun_events.30.a
		if = {
			limit = { country_exists = c:CRO }
			c:CRO = {
				make_subject_of = {
					target = root
					type = subject_type:fiefdom
				}
			}
		}
	}
	option = {
		name = hun_events.30.b
		add_prestige = 10
	}
}
