	type = country_event

	}
}

# ブルゴーニュへの野心
brb_events.2 = {
	type = country_event
	category = mission_event
	title = brb_events.2.t
	desc  = brb_events.2.d
	fire_only_once = yes
	immediate = {}
	option = {
		name = brb_events.2.a
		trigger = { country_exists = c:BUR }
		change_gold_effect = { scale = -60 }
		c:BUR = {
			make_subject_of = { target = root type = subject_type:vassal }
		}
	}
	option = {
		name = brb_events.2.b
		add_casus_belli = {
			type = casus_belli:cb_conquer_province
			target = c:BUR
		}
	}
	option = {
		name = brb_events.2.c
	}
}
