# =============================================================================
# Event 20: ハプスブルクの打倒——ボヘミアの覇権
# =============================================================================

boh_events.20 = {
	type = country_event
	category = mission_event
	title = boh_events.20.t
	desc  = boh_events.20.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = boh_events.20.a
		add_country_modifier = { modifier = boh_habsburg_crushed_mod years = -1 mode = add }
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
		add_prestige = 25
	}
}
