# Gothic Invasion CB - Theodoro (FEO) unique CB
# Unlocked by setting teo_gothic_invasion_unlocked variable via Gothic Empire final mission
# Targets countries with German culture group

cb_gothic_invasion = {
	create_visible = {
		has_variable = teo_gothic_invasion_unlocked
		scope:target = {
			culture ?= { has_culture_group = culture_group:german_group }
		}
	}
	create_enabled = {
		has_variable = teo_gothic_invasion_unlocked
		NOT = { has_truce_with = scope:target }
		scope:target = {
			culture ?= { has_culture_group = culture_group:german_group }
		}
	}
	declare_enabled = {
		has_variable = teo_gothic_invasion_unlocked
		scope:target = {
			culture ?= { has_culture_group = culture_group:german_group }
		}
	}
	war_goal_type = gothic_invasion_wargoal
	additional_war_enthusiasm_attacker = 0.50
}
