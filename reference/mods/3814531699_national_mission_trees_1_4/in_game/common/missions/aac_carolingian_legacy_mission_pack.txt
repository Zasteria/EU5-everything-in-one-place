aac_carolingian_legacy_mission_pack = {
	icon = nmt_bohemia
	repeatable = no
	player_playstyle = military
	chance = 10000000

	visible = {
		game_has_missions_enabled = yes
		has_or_had_tag = AAC
		NOR = { country_type = pop country_type = building }
	}

	enabled = { always = yes }

	on_start = { set_variable = aac_mission_started }
	on_completion = { remove_variable = aac_mission_started }
	on_abort = { remove_variable = aac_mission_started }

	# ==================== ROOT ====================

	aac_throne_of_charlemagne = {
		icon = frankokratia_frankokratia_advance
		requires = { }
		enabled = {
			always = yes
		}
		on_completion = {
			trigger_event_non_silently = aac_events.10
			area:rhineland_area = {
				every_location_in_area = {
					limit = { owner ?= { NOT = { this = root } } }
					save_scope_as = target_location
					root = {
						add_casus_belli = {
							target = scope:target_location.owner
							type = casus_belli:cb_conquer_province
							province = scope:target_location.province
							years = 50
						}
					}
				}
			}
		}
	}

	# ==================== BRANCH A: ライン川・ドイツ ===================

	aac_control_rhineland = {
		icon = legacy_of_great_empires
		requires = { aac_throne_of_charlemagne }
		enabled = {
			owns_or_non_sovereign_subject_owns = location:cologne
			owns_or_non_sovereign_subject_owns = location:trier
			any_owned_location = {
				area = area:rhineland_area
				count >= 10
			}
		}
		on_completion = {
			add_country_modifier = { modifier = aac_rhineland_mod years = -1 mode = add }
		}
	}

	aac_integrate_lorraine = {
		icon = legacy_of_great_empires

		}
	}

	aac_conquer_hre = {
		icon = legacy_of_great_empires
		requires = { aac_integrate_lorraine aac_control_flanders }
		enabled = {
			OR = {
				is_leader_of_international_organization = international_organization:hre
				AND = {
					any_owned_location = {
						region = region:north_german_region
						count >= 150
					}
					any_owned_location = {
						region = region:south_german_region
						count >= 150
					}
				}
			}
		}
		on_completion = {
			add_country_modifier = { modifier = aac_hre_master_mod years = -1 mode = add }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
	}
