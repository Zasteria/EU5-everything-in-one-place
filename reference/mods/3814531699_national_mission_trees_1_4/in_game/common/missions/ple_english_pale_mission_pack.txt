ple_english_pale_mission_pack = {
	icon = nmt_pale
	repeatable = no
	player_playstyle = military
	chance = 10000000

	visible = {
		game_has_missions_enabled = yes
		has_or_had_tag = PLE
		NOR = { country_type = pop country_type = building }
	}

	enabled = { always = yes }

	on_start = { set_variable = ple_mission_started }
	on_completion = { remove_variable = ple_mission_started }
	on_abort = { remove_variable = ple_mission_started }

	# ===== ROOT =====

	ple_english_pale = {
		icon = a_fortified_border
		requires = { }
		enabled = { legitimacy >= 100 }
		on_completion = {
			add_country_modifier = { modifier = ple_pale_mod years = 10 mode = add }
		}
	}

	# ===== A: ペイルの防衛と統治 =====

	ple_pale_consolidated = {
		icon = bureaucracy_law_advance
		requires = { ple_english_pale }
		enabled = {
			location:fingal = {
				is_core_of = root
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_consolidated_mod years = 30 mode = add }
		}
	}

	ple_pale_defense = {
		icon = new_frontiers
		requires = { ple_pale_consolidated }
		enabled = {
			any_owned_location = {
				OR = {
					has_building_with_at_least_one_level = castle
					has_building_with_at_least_one_level = bastion
				}
				count >= 4
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_defense_mod years = 20 mode = add }
		}
	}

	ple_lord_deputy = {
		icon = consolidate_the_royal_domain
		requires = { ple_anglo_irish_parliament }
		enabled = {
			OR = {
				NOT = { is_subject = yes }
				c:ENG = { opinion = { target = root value >= 200 } }
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_lord_deputy_mod years = 20 mode = add }
		}
	}

	ple_anglo_irish_parliament = {
		icon = estates_general
		requires = { ple_pale_consolidated }
		enabled = {
			estate_satisfaction:burghers_estate >= 0.75
		}
		on_completion = {
			add_country_modifier = { modifier = ple_parliament_mod years = 20 mode = add }
		}
	}

	ple_dublin_castle = {
		icon = city_building_advance
		requires = { ple_pale_consolidated }
		enabled = {
			location:dublin = {
				owner = root
				total_building_levels >= 30
			}
		}
		on_completion = {
			location:dublin = {
				add_location_modifier = { modifier = ple_dublin_mod years = -1 mode = add }
			}
		}
	}

	# ===== B: アングロ＝ノルマン諸侯 =====

	ple_hiberno_norman_lords = {
		icon = noble_society_cologne
		requires = { ple_pale_consolidated }
		enabled = {
			OR = {
				AND = {
					"estate_power(estate_type:nobles_estate)" >= 0.25
					estate_satisfaction:nobles_estate >= 0.60
				}
				"estate_power(estate_type:crown_estate)" >= 0.15
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_norman_lords_mod years = 20 mode = add }
		}
	}

	ple_fitzgerald_kildare = {
		icon = noble_officers
		requires = { ple_hiberno_norman_lords }
		enabled = {
			OR = {
				AND = {
					c:KID = { is_subject_of = root }
					c:KID = { opinion = { target = root value >= 180 } }
				}
				NOT = { country_exists = c:KID }
			}
		}
		on_completion = {
			if = {
				limit = { country_exists = c:KID }
				annex_country = { country = c:KID }
			}
			else = {
				add_prestige = 20
			}
		}
	}

	ple_butler_ormond = {
		icon = noble_resilience
		requires = { ple_hiberno_norman_lords }
		enabled = {
			OR = {
				AND = {
					c:ORD = { is_subject_of = root }
					c:ORD = { opinion = { target = root value >= 180 } }
				}
				NOT = { country_exists = c:ORD }
			}
		}
		on_completion = {
			if = {
				limit = { country_exists = c:ORD }
				annex_country = { country = c:ORD }
			}
			else = {
				add_prestige = 20
			}
		}
	}

	ple_statutes_of_kilkenny = {
		icon = development_of_maritime_law
		requires = { ple_fitzgerald_kildare ple_butler_ormond }
		enabled = {
			tolerance_own >= 1
		}
		on_completion = {
			add_country_modifier = { modifier = ple_kilkenny_mod years = 30 mode = add }
		}
	}

	# ===== C: ゲール人領域への進出 =====

	ple_gaelic_frontier_wars = {
		icon = sme_frontier_warriors
		requires = { ple_pale_consolidated }
		enabled = {
			any_current_war = {
				OR = { attacker_leader = root defender_leader = root }
				any_war_participant = {
					NOT = { this = root }
					culture = culture:irish
				}
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_frontier_wars_mod years = 20 mode = add }
		}
	}

	ple_conquer_leinster = {
		icon = settle_the_frontier_advance
		requires = { ple_gaelic_frontier_wars }
		enabled = {
			any_owned_location = {
				area = area:leinster_area
				OR = { owner = root owner = { is_subject_of = root } }
				count >= 20
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_leinster_mod years = 20 mode = add }
		}
	}

	ple_conquer_munster = {
		icon = geo_georgian_highlands
		requires = { ple_gaelic_frontier_wars }
		enabled = {
			any_owned_location = {
				area = area:munster_area
				OR = { owner = root owner = { is_subject_of = root } }
				count >= 22
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_munster_mod years = 20 mode = add }
		}
	}

	ple_conquer_connacht = {
		icon = chg_border_wars
		requires = { ple_gaelic_frontier_wars }
		enabled = {
			any_owned_location = {
				area = area:connaught_area
				OR = { owner = root owner = { is_subject_of = root } }
				count >= 15
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_connacht_mod years = 20 mode = add }
		}
	}

	ple_conquer_ulster = {
		icon = eternal_crusade
		requires = { ple_gaelic_frontier_wars }
		enabled = {
			any_owned_location = {
				area = area:ulster_area
				OR = { owner = root owner = { is_subject_of = root } }
				count >= 22
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_ulster_mod years = 20 mode = add }
		}
	}

	ple_master_of_ireland = {
		icon = imperial_ambitions
		requires = { ple_conquer_leinster ple_conquer_munster ple_conquer_connacht ple_conquer_ulster }
		enabled = {
			num_locations >= 80
		}
		on_completion = {
			add_country_modifier = { modifier = ple_master_ireland_mod years = 30 mode = add }
		}
	}

	# ===== D: 宗教改革と植民 =====

	ple_reformation_choice = {
		icon = hun_hungarian_reformation
		requires = { ple_pale_consolidated }
		enabled = {
			OR = {
				situation:reformation = { situation_is_active = yes }
				situation:reformation = { situation_has_ended = yes }
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_reformation_mod years = 20 mode = add }
		}
	}

	ple_church_of_ireland = {
		icon = reform_the_ecclesiastical_authorities
		requires = { ple_reformation_choice }
		enabled = {
			OR = {
				religion = religion:anglican
				religion = religion:lutheran
				religion = religion:calvinist
			}
			NOT = { mission_task_completed = mission_task:ple_recusant_catholics }
		}
		on_completion = {
			add_country_modifier = { modifier = ple_church_ireland_mod years = -1 mode = add }
		}
	}

	ple_recusant_catholics = {
		icon = ties_with_the_holy_see
		requires = { ple_reformation_choice }
		enabled = {
			religion = religion:catholic
			NOT = { mission_task_completed = mission_task:ple_church_of_ireland }
		}
		on_completion = {
			add_country_modifier = { modifier = ple_recusant_mod years = -1 mode = add }
		}
	}

	ple_dublin_university = {
		icon = library_advance
		requires = { ple_reformation_choice }
		enabled = {
			location:dublin = {
				total_building_levels >= 60
			}
			OR = {
				has_country_modifier = ple_church_ireland_mod
				has_country_modifier = ple_recusant_mod
			}
		}
		on_completion = {
			location:dublin = {
				add_location_modifier = { modifier = ple_university_mod years = -1 mode = add }
			}
		}
	}

	# ===== E: アイルランド王国の樹立 =====

	ple_english_crown_relations = {
		icon = imperial_school_of_naval_engineering
		requires = { ple_statutes_of_kilkenny ple_master_of_ireland ple_dublin_university }
		enabled = {
			diplomatic_reputation >= 1
		}
		on_completion = {
			add_country_modifier = { modifier = ple_crown_relations_mod years = 20 mode = add }
		}
	}

	ple_kingdom_of_ireland = {
		icon = legacy_of_great_empires
		requires = { ple_english_crown_relations }
		enabled = {
			tag = c:IRE
		}
		on_completion = {
			trigger_event_non_silently = ple_events.99
		}
	}

	# ===== F: 反抗か、従属か =====

	ple_defy_the_crown = {
		icon = liberty_egalite_fraternity
		requires = { ple_kingdom_of_ireland }
		enabled = {
			NOT = { mission_task_completed = mission_task:ple_loyal_crown_partnership }
		}
		on_completion = {
			add_country_modifier = { modifier = ple_defy_crown_mod years = 20 mode = add }
		}
	}

	ple_war_of_irish_independence = {
		icon = a_neverending_crusade
		requires = { ple_defy_the_crown }
		enabled = {
			any_current_war = {
				OR = { attacker_leader = root defender_leader = root }
				any_war_participant = { NOT = { this = root } this = c:ENG }
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ple_irish_independence_war_mod years = 10 mode = add }
		}
	}

	ple_sovereign_kingdom_of_ireland = {
		icon = golden_fleece
		requires = { ple_war_of_irish_independence }
		final = yes
		enabled = {
			NOT = { is_subject = yes }
			country_rank_level >= 3
		}
		on_completion = {
			add_country_modifier = { modifier = ple_sovereign_ireland_mod years = -1 mode = add }
		}
	}

	ple_loyal_crown_partnership = {
		icon = crown_power_advance_renaissance
		requires = { ple_kingdom_of_ireland }
		enabled = {
			NOT = { mission_task_completed = mission_task:ple_defy_the_crown }
		}
		on_completion = {
			add_country_modifier = { modifier = ple_loyal_partnership_mod years = 20 mode = add }
		}
	}

	ple_anglo_irish_union = {
		icon = the_pentarchy
		requires = { ple_loyal_crown_partnership }
		enabled = {
			c:ENG = { is_subject_of = root }
		}
		on_completion = {
			add_country_modifier = { modifier = ple_anglo_irish_union_mod years = -1 mode = add }
		}
	}

	ple_united_kingdoms = {
		icon = imperial_creed
		requires = { ple_anglo_irish_union }
		final = yes
		enabled = {
			country_rank_level >= 3
			monthly_income_trade_and_tax >= 300
			is_subject_of = c:ENG
		}
		on_completion = {
			if = {
				limit = { is_subject_of = c:ENG }
				change_subject_type = subject_type:eng_welsh_principality
			}
		}
	}
}
