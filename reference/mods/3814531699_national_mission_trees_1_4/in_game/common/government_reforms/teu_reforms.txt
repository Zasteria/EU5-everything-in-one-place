# 北方十字軍の道（crusade_path / ev.11）
teu_crusade_path_reform = {
	unique = yes
	icon = nmt_teu_crusade_path_reform

	potential = {
		OR = {
			has_unlocked_government_reform_trigger = { type = teu_crusade_path_reform }
			has_reform = government_reform:teu_crusade_path_reform
		}
	}

	allow = {
		government_type = government_type:theocracy
	}

	country_modifier = {
		monthly_devotion = -0.20
		government_reform_slots = 1
	}

	country_modifier = {
		scale = {
			value = legitimacy
			multiply = 0.01
			max = 1.0
		}
		discipline = 0.10
		global_estate_target_satisfaction = 0.10
		monthly_rebel_growth = -0.001
	}

	on_deactivate = {}

	years = 2
}

# 北方十字軍の完成（northern_crusade_final / ev.50）
teu_northern_crusade_reform = {
	unique = yes
	icon = nmt_teu_northern_crusade_final

	potential = {
		OR = {
			has_unlocked_government_reform_trigger = { type = teu_northern_crusade_reform }
			has_reform = government_reform:teu_northern_crusade_reform
		}
	}

	allow = {
		government_type = government_type:theocracy
	}

	country_modifier = {
		potential_trigger = {
			custom_tooltip = {
				text = teu_northern_crusade_reform_loot_tt
			}
		}
		monthly_devotion = -0.30
		government_reform_slots = 1
		amount_looted_modifier = 0.30
		allow_rgo_slave_demand = yes
		auto_slave_raid_different_religion = yes
		slave_raid_efficiency = 0.04
	}

	# 改革取得時にCBをアンロック
	on_activate = {
		unlock_casus_belli_effect = { type = cb_crusade }
		unlock_casus_belli_effect = { type = cb_heretic }
	}


	on_deactivate = {}

	years = 2
}
