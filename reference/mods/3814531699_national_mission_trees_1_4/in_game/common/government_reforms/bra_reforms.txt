# 質の軍制（プロイセン軍制改革・質寄り）
bra_quality_doctrine_reform = {
	unique = yes
	icon = modernization_of_the_military

	potential = {
		OR = {
			has_unlocked_government_reform_trigger = { type = bra_quality_doctrine_reform }
			has_reform = government_reform:bra_quality_doctrine_reform
		}
	}

	allow = {
		always = yes
	}

	country_modifier = {
		army_light_infantry_power = 0.10
		army_heavy_infantry_power = 0.10
		levy_combat_efficiency_modifier = 0.15
		army_maintenance_efficiency = -0.10
	}

	on_deactivate = {}

	years = 2
}
