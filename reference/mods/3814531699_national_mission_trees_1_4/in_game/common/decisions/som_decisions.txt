# Ottoman Empire (SOM): decisions unlocked by the mission som_tur_new_capital
som_develop_kostantiniyye_decision = {
	decision_category = country_specific_decisions

	only_once = yes

	potential = {
		has_variable = som_tur_constantinople_development_unlocked
	}

	allow = {
		gold >= 500
	}

	ai_will_do = {
		value = -1
	}

	option = {
		name = som_develop_kostantiniyye_decision.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -500
			location:constantinople = {
				add_location_modifier = { modifier = som_tur_constantinople_dev_mod years = -1 mode = add }
			}
			set_variable = {
				name = som_tur_constantinople_dev_1_done
				value = 1
			}
		}
	}
}
