# Muscovy / Russia (MOS): decisions unlocked by the mission mos_dev_agriculture
mos_agriculture_decision = {
	decision_category = country_specific_decisions

	only_once = yes

	potential = {
		has_variable = mos_agriculture_unlocked
	}

	allow = {
		gold >= 3000
		political_influence >= 50
	}

	ai_will_do = {
		value = -1
	}

	option = {
		name = mos_agriculture_decision.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -3000
			add_political_influence = -50
			add_country_modifier = { modifier = mos_agriculture_mod years = -1 mode = add }
			set_variable = {
				name = mos_agriculture_1_done
				value = 1
			}
		}
	}
}

mos_agriculture_upgrade_decision = {
	decision_category = country_specific_decisions

	only_once = yes

	potential = {

	}

	allow = {
		gold >= 20000
		political_influence >= 75
	}

	ai_will_do = {
		value = -1
	}

	option = {
		name = mos_agriculture_upgrade_decision.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -20000
			add_political_influence = -75
			remove_country_modifier = mos_agriculture_mod
			add_country_modifier = { modifier = mos_black_earth_mod years = -1 mode = add }
		}
	}
}
