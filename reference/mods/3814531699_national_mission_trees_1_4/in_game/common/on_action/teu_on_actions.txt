# root = country, scope:target = location

# 要塞なしのロケーション占領時
on_location_occupied = {
	effect = {
		if = {
			limit = {
				has_reform = government_reform:teu_northern_crusade_reform
				scope:target = {
					NOT = { owner = root }
				}
			}
			scope:target = {
				add_location_modifier = {
					modifier = teu_crusade_loot_mod
					years = 10
					mode = add
				}
			}
		}
	}
}

				}
			}
			scope:target = {
				add_location_modifier = {
					modifier = teu_crusade_loot_mod
					years = 10
					mode = add
				}
			}
		}
	}
}
