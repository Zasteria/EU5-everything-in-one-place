			}
		}
		duration = 0
		on_completion = {
			add_country_modifier = { modifier = mlo_curb_della_scala_mod years = 20 mode = add }

				limit = {
					country_exists = c:VER

				}
				add_casus_belli = {
					target = location:verona.owner
					type = casus_belli:cb_conquer_province
