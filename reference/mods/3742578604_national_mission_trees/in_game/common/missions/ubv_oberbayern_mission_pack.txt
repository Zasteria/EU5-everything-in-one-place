ubv_oberbayern_mission_pack = {
	icon = nmt_bavaria
	repeatable = no
	player_playstyle = military
	chance = 10000000

	visible = {
		game_has_missions_enabled = yes
		has_or_had_tag = UBV
		NOR = { country_type = pop country_type = building }
	}

	enabled = { always = yes }

	on_start = { set_variable = ubv_mission_started }
	on_completion = { remove_variable = ubv_mission_started }
	on_abort = { remove_variable = ubv_mission_started }

	# ===== ROOT =====

	ubv_wittelsbach_legacy = {
		icon = crown_power_advance_renaissance
		requires = { }
		enabled = { always = yes }
		on_completion = {
			add_country_modifier = { modifier = ubv_legacy_mod years = 20 mode = add }
		}
	}

	# ===== A: ミュンヘン家の礎 =====

	ubv_secure_munich = {
		icon = city_building_advance
		requires = { ubv_wittelsbach_legacy }
		enabled = {
			location:munich = {
				owner = root
				total_building_levels >= 25
				OR = {
					location_rank = location_rank:city
					location_rank = location_rank:megalopolis
				}
			}
		}
		on_completion = {
			location:munich = {
				add_location_modifier = { modifier = ubv_munich_mod years = -1 mode = add }
			}
		}
	}

	ubv_salt_trade = {
		icon = generic_trade
		requires = { ubv_wittelsbach_legacy }
		enabled = {
			produced_in_country:salt >= 3
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_salt_mod years = 20 mode = add }
		}
	}

	ubv_imperial_relations = {
		icon = imperial_ambitions
		requires = { ubv_wittelsbach_legacy }
		enabled = {
			OR = {
				is_allied_with = { target = c:HAB }
				NOT = { country_exists = c:HAB }
				has_royal_marriage_with = c:HAB
				c:HAB = { is_rival_of = root }
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_imperial_mod years = 20 mode = add }
		}
	}

	ubv_bavarian_ambition = {
		icon = imperial_creed
		requires = { ubv_secure_munich ubv_salt_trade ubv_imperial_relations }
		enabled = {
			OR = {
				c:LBV = { monthly_income_trade_and_tax <= root.monthly_income_trade_and_tax }
				c:LBV = { is_rival_of = root }
				location:landshut = { owner = root }
				NOT = { country_exists = c:LBV }
			}
		}
		on_completion = {
			add_casus_belli = {
				type = casus_belli:cb_annex
				target = c:LBV
				years = 20
			}
		}
	}

	ubv_reunite_bavaria = {
		icon = imperial_ambitions
		requires = { ubv_bavarian_ambition }
		enabled = {
			OR = {
				NOT = { country_exists = c:LBV }
				c:LBV = { is_subject_of = root }
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_reunite_mod years = -1 mode = add }
		}
	}

	ubv_duke_of_all_bavaria = {
		icon = legacy_of_great_empires
		requires = { ubv_reunite_bavaria }
		enabled = {
			any_owned_location = {
				area = area:bavaria_area
				owner = root
				count >= 38
			}
		}
		on_completion = {
			trigger_event_non_silently = ubv_events.1
			set_variable = ubv_bavaria_unified_flag
		}
	}

	# ===== ROOT =====

	ubv_unified_duchy = {
		icon = legacy_of_great_empires
		requires = { }
		enabled = { has_variable = ubv_bavaria_unified_flag }
		on_completion = {
			add_country_modifier = { modifier = ubv_unified_mod years = 20 mode = add }
		}
	}

	# ===== B: 統一公国の再編 =====

	ubv_landrecht = {
		icon = crown_rights_laws_advance
		requires = { ubv_unified_duchy }
		enabled = {
			OR = {
				stability >= 60
				"estate_power(estate_type:crown_estate)" >= 0.25
			}
		}
		on_completion = {
			unlock_government_reform_effect = { type = ubv_landrecht_reform }
			add_reform = government_reform:ubv_landrecht_reform
		}
	}


	ubv_tyrol_ambition = {
		icon = settle_the_frontier_advance
		requires = { ubv_unified_duchy }
		enabled = {
			OR = {
				NOT = { country_exists = c:TYR }
				c:TYR = { is_rival_of = root }
				has_royal_marriage_with = c:TYR
				has_insulted = c:TYR
			}
		}
		on_completion = {
			area:tirol_area = {
				every_location_in_area = {
					limit = { NOT = { owner = root } }
					save_scope_as = temp_location
					root = {
						add_casus_belli = {
							target = scope:temp_location.owner
							type = casus_belli:cb_conquer_province
							province = scope:temp_location.province
							years = 20
						}
					}
				}
			}
		}
	}

	ubv_swabian_expansion = {
		icon = expansionism
		requires = { ubv_unified_duchy }
		enabled = {
			any_owned_location = {
				area = area:swabia_area
				OR = { owner = root owner = { is_subject_of = root } }
				count >= 2
			}
		}
		on_completion = {
			area:swabia_area = {
				every_location_in_area = {
					limit = { NOT = { owner = root } }
					save_scope_as = temp_location
					root = {
						add_casus_belli = {
							target = scope:temp_location.owner
							type = casus_belli:cb_conquer_province
							province = scope:temp_location.province
							years = 20
						}
					}
				}
			}
		}
	}

	ubv_greater_bavaria = {
		icon = imperial_ambitions
		requires = { ubv_landrecht ubv_tyrol_ambition ubv_swabian_expansion }
		enabled = {
			num_locations >= 60
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_greater_mod years = 30 mode = add }
		}
	}

	# ===== C: 反宗教改革とカトリック同盟 =====


	ubv_counter_reformation_leader = {
		icon = church_councils
		requires = { ubv_greater_bavaria }
		enabled = {
			OR = {
				situation:reformation = { situation_is_active = yes }
				situation:reformation = { situation_has_ended = yes }
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_counter_reformation_mod years = 20 mode = add }
		}
	}



	ubv_electoral_dignity = {
		icon = crown_power_advance_renaissance
		requires = { ubv_counter_reformation_leader }
		enabled = {
			OR = {
				country_rank_level >= 3
				any_owned_location = {
					owner = root
					count >= 90
				}
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_electoral_dignity_mod years = 20 mode = add }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}

	# ===== D: オーストリアとの対抗、帝冠 =====

	ubv_austrian_rivalry = {
		icon = austrian_military_flexibility
		requires = { ubv_electoral_dignity }
		enabled = {
			OR = {
				c:HAB = { is_rival_of = root }
				NOT = { country_exists = c:HAB }
				has_royal_marriage_with = c:HAB
				has_casus_belli = { target = c:HAB type = casus_belli:cb_claim_throne }
			}
		}
		on_completion = {
			add_casus_belli = {
				type = casus_belli:cb_claim_throne
				target = c:HAB
				target_character = root.ruler
				years = 20
			}
		}
	}

	ubv_war_of_austrian_succession = {
		icon = a_fortified_border
		requires = { ubv_austrian_rivalry }
		enabled = {
			OR = {
				any_current_war = {
					OR = { attacker_leader = root defender_leader = root }
					any_war_participant = { NOT = { this = root } this = c:HAB }
				}
				NOT = { country_exists = c:HAB }
			}
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_succession_war_mod years = 20 mode = add }
		}
	}

	ubv_imperial_crown = {
		icon = imperial_creed
		requires = { ubv_war_of_austrian_succession }
		enabled = {
			OR = {
				is_leader_of_international_organization = international_organization:hre
				country_rank_level >= 4
			}
		}
		on_completion = {
			trigger_event_non_silently = ubv_events.3
		}
	}

	ubv_enlightened_absolutism = {
		icon = crown_rights_laws_advance
		requires = { ubv_electoral_dignity }
		enabled = {
			average_country_literacy >= 30
		}
		on_completion = {
			add_country_modifier = { modifier = ubv_enlightenment_mod years = 20 mode = add }
		}
	}

	# ===== E: 王国の樹立 =====

	ubv_napoleonic_alliance = {
		icon = liberty_egalite_fraternity
		requires = { ubv_imperial_crown ubv_enlightened_absolutism }
		enabled = {
			c:FRA = { monthly_income_trade_and_tax <= root.monthly_income_trade_and_tax }
		}
		on_completion = {
			add_casus_belli = {
				type = casus_belli:cb_claim_throne
				target = c:FRA
				target_character = root.ruler
				years = 20
			}
		}
	}

	ubv_kingdom_of_bavaria = {
		icon = legacy_of_great_empires
		requires = { ubv_napoleonic_alliance }
		final = yes
		enabled = {
			country_rank_level = 4
			any_owned_location = {
				owner = root
				count >= 500
			}
		}
		on_completion = {
			trigger_event_non_silently = ubv_events.99
		}
	}
}
