# ソフィアの歴史
bul_events.4 = {
	type = country_event
	category = mission_event
	title = bul_events.4.t
	desc  = bul_events.4.d
	fire_only_once = yes
	illustration = "gfx/interface/illustrations/missions/nmt_bulgaria.dds"
	immediate = {}
	option = {
		name = bul_events.4.a
		location:sredets = {
			rename_location = bul_sofia_location_name
			create_art = {
				quality = 85
				type = work_of_art_type:monument
				key = bul_hagia_sophia_sofia
			}
		}
	}
	option = {
		name = bul_events.4.b
		location:sredets = {
			rename_location = bul_sofia_location_name
			create_art = {
				quality = 85
				type = work_of_art_type:monument
				key = bul_hagia_sophia_sofia
			}
		}
		set_capital = location:sredets
	}
}
