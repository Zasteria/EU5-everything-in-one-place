# Catalonia (CAT): timed modifiers
# National Destinies mod
#
# Level 2 formable: formation choices last 20 years. Values are taken from
# what vanilla's own timed modifiers grant for each key on 1.4
# (main_menu/common/static_modifiers/country.txt), with named values where
# vanilla writes a name.

# Formation cost, applied in form_effect to a country that was Aragon (10
# years). Catalonia chosen as the state's identity turns Aragon and Valencia
# from peers of a union into possessions, and their estates know it.
# global_separatism is a key where a positive value harms.
nd_cat_crown_unmade = {
	game_data = {
		category = country
	}
	global_separatism = 0.05
	nobles_estate_target_satisfaction = medium_permanent_target_satisfaction_penalty
}

# Constitutions before the Corts acknowledge him; he gains the estates and

		category = country

# ===== DESTINY TIMED MODIFIERS =====
# Eight blocks: two choices each for the stage 2 (15 years) and stage 3 (20
# years) events of both destiny paths. No key here appears in a heritage
# advance, in a formation choice, in an advance event modifier above or
# anywhere else in the destiny tree. Values sit at or under what vanilla's
# own timed modifiers grant for each key on 1.4 (static_modifiers/country.txt),
# with names where vanilla writes a name. Four of the eight carry a cost, each
# from outside what its path is about, and each written as a number so the
# sign can be read.

# Event nd_cat.9 (the oath at Valencia), choice A: the prince swears the Furs
# and the kingdom keeps its own law. The Moors of the kingdom, a third of its
# people, keep their own judges under it, and a charter that holds draws
# settlers. monthly_rebel_growth is a key where a negative value is the
# benefit.
nd_cat_dest_regne_per_si = {
	game_data = {
		category = country
	}
	tolerance_heathen = 1
	monthly_rebel_growth = -0.001
	global_migration_attraction = 0.1
}

# Event nd_cat.9, choice B: the Usatges and the Constitutions are carried
# south. One law makes one people sooner and puts the prince's officers on
# familiar ground; those who farmed under the older franchises lose them.
nd_cat_dest_una_sola_llei = {
	game_data = {
		category = country
	}
	global_pop_assimilation_speed_modifier = 0.15
	global_monthly_control = 0.002
	peasants_estate_target_satisfaction = -0.025
}
