# Mallorca (MLL): timed modifiers
# National Destinies mod
#
# Level 2 formable: formation choices last 20 years. Values are taken from
# what vanilla's own timed modifiers grant for each key on 1.4
# (main_menu/common/static_modifiers/country.txt), with named values where
# vanilla writes a name.

# Formation choice A: the City. The jurats and merchants of Ciutat de
# Mallorca govern the kingdom from inside the walls, as they did in fact:
# ships are laid down, the sea lanes lengthen, and the villages pay for a
# council in which they are outvoted.
nd_mll_ciutat_e_regne = {
	game_data = {
		category = country
	}
	trade_sea_efficiency = medium_trade_sea_efficiency_bonus
	global_merchant_power = 0.1
	burghers_estate_target_satisfaction = medium_permanent_target_satisfaction
	ship_build_speed = 0.1
	peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
}

# Formation choice B: the Part Forana. The villages get what King Sancho's
# ruling of 1315 promised them: their share of seats and a look at the
# accounts. The countryside builds, feeds and musters willingly; the
# merchants of the City find their own council slower to do their bidding.
nd_mll_part_forana = {
	game_data = {
		category = country
	}
	peasants_estate_target_satisfaction = medium_permanent_target_satisfaction
	global_monthly_food_modifier = monthly_food_productivity_mild_bonus
	global_rural_build_buildings_efficiency = 0.1
	global_levy_size_modifier = 0.1
	burghers_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
}

# Advance event nd_mll.2 (the Contracte Sant), choice A: the contract is
# kept. The creditors' treasurer collects the taxes and pays the interest
# first, so lenders trust the kingdom and lend to it cheaply; what is left
# for the kingdom itself is little. tax_income_efficiency is negative on
# purpose.
nd_mll_universal_consignacio = {
	game_data = {
		category = country
	}
	bank_interest = -0.01
	total_loan_capacity_modifier = 0.25
	tax_income_efficiency = small_tax_income_efficiency_penalty
}

# Advance event nd_mll.2, choice B: the interest is suspended, as the
# jurats did in 1425. The taxes stay on the island and the villages
# breathe; no banker in Barcelona will forget it. bank_interest is positive
# on purpose (a penalty).
nd_mll_pensions_suspeses = {
	game_data = {
		category = country
	}
	tax_income_efficiency = small_tax_income_efficiency_bonus
	peasants_estate_target_satisfaction = small_permanent_target_satisfaction
	bank_interest = 0.01
}

# to the last day. Every bastion is manned and the island learns what it

	game_data = {
		category = country
	}

# Advance event nd_mll.5, choice B: terms are asked for and given. Lives,
# goods and harvests are spared and trade resumes within the season; the
# name of the place is not spoken of with the same respect.
# monthly_war_exhaustion is a key where a negative value is the benefit, and
# monthly_prestige is negative on purpose.

	game_data = {
		category = country
	}
	monthly_war_exhaustion = -0.03

}
