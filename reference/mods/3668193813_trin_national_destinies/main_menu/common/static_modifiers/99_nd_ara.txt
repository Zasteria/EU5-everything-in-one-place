# Aragon (ARA): timed modifiers
# National Destinies mod
#
# Level 2 formable: formation choices last 20 years. Values are taken from
# what vanilla's own timed modifiers grant for each key on 1.4
# (main_menu/common/static_modifiers/country.txt), with named values where
# vanilla writes a name.

# Formation choice A: the kingdom in its own land. The king keeps his court
# at Zaragoza and governs with the four arms; the barons and the frontier
# towns turn out for a kingdom that is theirs, and the crown's own hand is
# the weaker for it. global_crown_estate_power is negative on purpose.
nd_ara_reino_en_su_tierra = {
	game_data = {
		category = country
	}
	nobles_estate_target_satisfaction = medium_permanent_target_satisfaction
	global_levy_size_modifier = 0.1
	global_defensive = 0.15
	stability_cost_efficiency = 0.2
	global_crown_estate_power = -0.1
}

# Formation choice B: the head of the Crown. Aragon gives its name and its

		category = country
	}
