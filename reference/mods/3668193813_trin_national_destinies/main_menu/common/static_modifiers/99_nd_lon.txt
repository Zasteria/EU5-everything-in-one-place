# Leon (LON): timed modifiers
# National Destinies mod

# grant for each key on 1.4, with named values where vanilla writes a name.

# Formation choice A: Imperator totius Hispaniae. The kings of Leon held
# themselves heirs of Toledo and overlords of every Christian prince in the
# peninsula, and Alfonso VII was crowned emperor in Leon cathedral in 1135
# with neighbouring Christian princes acknowledging him as overlord. Claims are
# found quickly, there is room for more vassals and they are kept loyal; the
# estates at home pay for the pageant and say so.
nd_lon_imperium_totius_hispaniae = {
	game_data = {
		category = country
	}
	casus_belli_creation_speed_modifier = 0.25
	diplomatic_capacity_modifier = 0.2
	subject_loyalty = 10
	monthly_prestige = 0.1
	global_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
}
