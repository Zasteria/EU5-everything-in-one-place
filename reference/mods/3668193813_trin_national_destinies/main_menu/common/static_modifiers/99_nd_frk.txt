# ===== Francia (FRK) Static Modifiers =====
# Level 4: 40 year duration on formation choices, at most 8 lines each.
# The formation question is the one the Carolingian realm itself never
# settled: the capitularies were territorial law binding every subject,
# while the leges nationum left each people under its own law. Francia is
# proclaimed over French, German, Dutch, Frisian and Lombard subjects all at
# once, so the question arrives on the first day.

# ===== FORMATION CHOICES =====

# Path A: One law over all. The capitularies bind every people of the realm
# and the missi carry one law into every county. Integration and control
# come fast; the peoples who kept their own customs lose standing, which is
# why cultures capacity falls while assimilation quickens.
nd_frk_one_law_over_all = {
	game_data = {
		category = country
	}
	global_integration_speed_modifier = 0.25
	global_monthly_control = 0.004
	global_pop_assimilation_speed_modifier = 0.25
	tax_income_efficiency = 0.1
	global_crown_estate_power = 0.15
	cultures_capacity = -1
}

# Path B: Each people under its own law. The assembly at Aachen in 802 had
# the laws of the Frisians, Saxons, Thuringians and Chamavian Franks written
# down so that each nation would be judged by its own custom. The realm
# holds its peoples by leaving them their law, and the crown gives up the
# jurisdiction that would have made it the only judge.
nd_frk_law_of_the_peoples = {
	game_data = {
		category = country
	}
	cultures_capacity = 1
	add_accepted_culture_cost_modifier = -0.33
	global_estate_target_satisfaction = 0.05
	subject_loyalty = 10
	diplomatic_annexation_efficiency = 0.25
	global_crown_estate_power = -0.1
}
