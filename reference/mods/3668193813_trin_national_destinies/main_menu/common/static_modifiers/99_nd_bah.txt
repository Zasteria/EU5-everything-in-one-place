# ==========================================================================
# FORMATION EVENT (nd_bah.10) choices. Added 2026-08-14 when BAH became a
# real formable; before that this file's header was accurate in saying the
# tag had no formation event.
#
# The two choices are the sultanate's own founding quarrel, not a generic
# military/economic split. The Deccanis were the amirs already settled in
# the plateau who raised Hasan Gangu at Gulbarga in 1347 and took the tarafs
# as their own; the Afaqis were the Persians, Turks and Arabs who came in by
# sea afterwards for gunnery, medicine and administration, and whose
# champion Mahmud Gawan was executed in 1481 on a forged charge. Each choice
# therefore carries the other faction's resentment as its downside.
# ==========================================================================

# Choice A: the tarafs. Power sits with the amirs and their countries.
nd_bah_taraf_amirs = {
	game_data = {
		category = country
	}
	# The amirs bring their own men, and so do the Hindu chiefs under them:
	# Maratha and Telugu troops served Bahmani armies throughout, which is
	# what a Muslim sultanate over a Hindu plateau actually looked like.
	nobles_estate_levy_size = 0.25
	dhimmi_estate_levy_size = 0.1
	# Armies quartered on their own tarafs rather than supplied from a
	# capital.
	global_supply_limit_modifier = 0.1
	# The cost, and the reason the state fragmented in 1490: a court that
	# governs through provincial magnates does not govern much.
	country_cabinet_efficiency = -0.1
}

# Choice B: the court of Bidar. Power sits with the men brought in for it.
nd_bah_court_of_bidar = {
	game_data = {
		category = country
	}
	# Foreign professionals on contract: Ottoman and Persian gunners, the
	# Afaqi captains, the physicians and secretaries of the Bidar court.
	mercenary_maintenance_efficiency = 0.2
	# The madrasa Mahmud Gawan raised at Bidar in 1472, and everything else
	# that arrived with the men who staffed it.
	global_institution_growth_modifier = 0.15
	# Dabhol: the port the horses, the immigrants and the money came
	# through.
	merchant_maintenance_efficiency = 0.15
	# The cost: the Deccani amirs watch foreigners take the offices and the
	# revenue, and they do not forget it.
	nobles_estate_target_satisfaction = -0.1
}
