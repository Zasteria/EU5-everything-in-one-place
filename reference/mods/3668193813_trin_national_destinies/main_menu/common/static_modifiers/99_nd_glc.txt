# Galicia (GLC): timed modifiers
# National Destinies mod
#
# Level 2 formable: formation choices last 20 years, advance event choices
# 15. Values are taken from what vanilla's own timed modifiers grant for
# each key on 1.4 (main_menu/common/static_modifiers/country.txt), with
# named values where vanilla writes a name.

# Formation choice A: the patrimony of the Apostle. The kingdom is governed
# from Compostela, where the archbishop is the first lord of the land, the
# Voto comes in from half of Spain and every road ends at the shrine. The
# church is content, the kingdom is known wherever pilgrims come from, and
# the townsmen who rose against their archbishops in 1117 and 1318 are
# ruled by one again.
nd_glc_patrimonio_do_apostolo = {
	game_data = {
		category = country
	}
	clergy_estate_target_satisfaction = medium_permanent_target_satisfaction
	tax_income_efficiency = medium_tax_income_efficiency_bonus
	cultural_influence_modifier = 0.15
	tolerance_own = 1
	burghers_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
}
