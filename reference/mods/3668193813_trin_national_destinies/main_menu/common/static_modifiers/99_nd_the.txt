# ===== DESTINY TIMED MODIFIERS =====
#
# Four pairs, one per branching destiny stage. No key here appears in the
# heritage advances, in the three formation modifiers, or anywhere else in
# the destiny tree. The negatives are deliberately orthogonal: a war of
# conquest against Orthodox Epirus costs the crown its Greek clergy, an
# assembly of barons costs it the ability to legislate, and a harbour run
# by its own merchants costs it the freedom to start wars and to be liked
# while taxing foreigners.

# ---------- The reckoning with Epirus ----------

# Beat the Doukai in open field and dictate the terms. The Greek clergy of
# the realm watch a Latin king campaign against an Orthodox despotate and
# draw the obvious conclusion.

	game_data = {
		category = country
	}
	global_war_score_efficiency = 0.1
	regiment_recruit_speed = 0.15

}
