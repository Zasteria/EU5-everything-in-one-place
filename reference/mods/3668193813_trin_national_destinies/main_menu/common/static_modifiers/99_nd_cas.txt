# Castile (CAS): timed modifiers
# National Destinies mod
#
# Level 2 formable: formation choices last 20 years. Values are taken from
# what vanilla's own timed modifiers grant for each key on 1.4
# (main_menu/common/static_modifiers/country.txt), with named values where
# vanilla writes a name. No government power key is used: the formable fixes
# no government type, and a key of one pool pays nothing to the others.

# Formation choice A: the king's justice. The Crown rules through its
# judges, its accountants and the councils of the towns, as Alfonso XI did
# after 1325 and Isabella after 1480. The great houses are kept from the
# rents and the offices and resent it.
nd_cas_justicia_del_rey = {
	game_data = {
		category = country
	}
	global_crown_estate_power = 0.1
	tax_income_efficiency = large_tax_income_efficiency_bonus
	global_monthly_control = 0.002
	burghers_estate_target_satisfaction = medium_permanent_target_satisfaction
	monthly_towards_centralization = societal_value_monthly_move
	nobles_estate_target_satisfaction = medium_permanent_target_satisfaction_penalty
}

# ===== DESTINY TIMED MODIFIERS =====
# Eight choices, two for each stage 2 and stage 3 event: 15 years at stage 2,
# 20 at stage 3, three keys each. No key repeats a key of the destiny
# advances, of the heritage advances or of the formation choices. Three of
# the eight carry a small cost from outside the path's own theme.

# Path A stage 2 (nd_cas.11), choice A: corregidores de toga. Graduates on
# every bench: new learning finds readers, and the roads and bridges a
# corregidor must inspect are kept up. The men who lead the host think
# little of a kingdom run by clerks.
nd_cas_dest_toga = {
	game_data = {
		category = country
	}
	embrace_institution_cost_modifier = -0.1
	global_road_building_time = -0.1
	land_morale_modifier = -0.05
}
