# Regnum Gothorum (GOT): Timed modifiers
# National Destinies mod
#
# Formation choices (Level 3: 30 year duration). The fork is the name of the
# restored kingdom, and the name is the claim. Latin continuity makes the
# state a juridical and ecclesiastical person; Gothic revivalism makes it a
# people with a tongue and a sworn king. Each carries a real cost drawn from
# an orthogonal domain: the Latin crown binds itself to the councils, and the
# Gothic-named crown is read by Latin Christendom as a barbarian archaism.

# ===== DESTINY TIMED MODIFIERS =====
# Stage 2 and stage 3 of each path offer two readings of the same milestone.
# Each pair splits the path's own logic rather than offering a generic good
# and a generic better. Four of the eight carry a small orthogonal cost, drawn
# from a domain the choice would genuinely neglect.

# --- Path A stage 2: the fisc and the patrimony (nd_got.12) ---

# Option A: what the crown takes belongs to the kingdom. The treasury is
# audited by the council and cannot be drawn on as a household purse.
nd_got_dest_fiscus_regni = {
	game_data = {
		category = country
	}
	trade_income = 0.1
	global_raw_material_output = 0.1
	# The cost: an audited treasury pays the host late, and a host paid late
	# recovers its nerve slowly.
	land_morale_recovery = -0.01
}

# Option B: the king keeps his patrimony and the realm is paid from it. A
# royal domain worked hard rather than a public revenue accounted for.
nd_got_dest_patrimonium_regis = {
	game_data = {
		category = country
	}
	global_construction_speed = 0.1
	global_devastation_recovery = 0.0025
	rural_migration_attraction = 0.25
}

# --- Path A stage 3: the king under the code (nd_got.13) ---

# Option A: the king swears to the code before the crown touches him. A bound
# crown is a reliable counterparty, and every court in Christendom knows it.
nd_got_dest_rex_eris = {
	game_data = {
		category = country
	}
	diplomatic_capacity_modifier = 0.15
	monthly_diplomats = 0.1
	# The cost: a crown that answers to lawyers is not a crown the veterans
	# gather round, and the army's reputation fades between wars.
	army_tradition_decay = 0.0005
}

# Option B: the code binds the realm and the crown is its judge. Legalist
# absolutism: the king enforces the law downward instead of standing under it.
nd_got_dest_lex_ministrorum = {
	game_data = {
		category = country
	}
	spy_network_construction = 0.15
	global_max_literacy = 5
	# The cost: a realm administered by judges is not administered by
	# merchants, and the merchants notice what it costs them.
	merchant_maintenance_efficiency = -0.1
}
