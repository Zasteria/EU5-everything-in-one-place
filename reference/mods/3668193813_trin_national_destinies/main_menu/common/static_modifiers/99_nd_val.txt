# Valencia (VAL): timed modifiers
# National Destinies mod
#
# Level 2 formable: formation choices last 20 years. Values are taken from
# what vanilla's own timed modifiers grant for each key on 1.4
# (main_menu/common/static_modifiers/country.txt), with named values where
# vanilla writes a name.

# Formation choice A: the king's towns under the Furs. The kingdom James I
# meant: royal towns peopled by free settlers under one written law, as a
# counterweight to the barons. Strong towns and quick lawmaking; the lords
# who wanted the custom of Aragon are not reconciled.
nd_val_furs_i_viles_reials = {
	game_data = {
		category = country
	}
	burghers_estate_target_satisfaction = medium_permanent_target_satisfaction
	legislative_efficiency = 0.1
	town_upgrade_cost_modifier = -0.1
	global_merchant_capacity_modifier = 0.1
	nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
}

# Formation choice B: the Alfonsine jurisdiction (1329). The lords accept the
# Furs and are paid for it with the lower justice over their own villages.
# The barons are content, their Muslim tenants are left in peace because
# they are the barons' income, and the lords' horse turns out for the king.
# The Crown has given away part of its own hand: global_crown_estate_power
# is negative on purpose.
nd_val_jurisdiccio_alfonsina = {
	game_data = {
		category = country
	}
	nobles_estate_target_satisfaction = medium_permanent_target_satisfaction
	global_raw_material_output = 0.1
	tolerance_heathen = 1
	army_heavy_cavalry_power = 0.1
	global_crown_estate_power = -0.05
}
