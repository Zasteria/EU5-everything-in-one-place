# Grants bureaucracy authority to BNG once the sultanate has been formed and
# the player has the bureaucracies game-rule enabled. Disappears if the
# player forms a successor formable (uses tag, not has_or_had_tag).
# Four slots by the slot rule of 2026-10-02: half of Bengal's seven
# institutions, rounded up. Vanilla gives the countries that form Bengal no
# slot of their own on that day, and none comes from the capital (only a
# megalopolis gives one).

nd_bng_bureaucracy_authority = {
	potential_trigger = {
		tag = BNG
		has_variable = nd_bng_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
