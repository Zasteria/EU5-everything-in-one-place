# YMP: Yamato Empire bureaucratic authority
#
# On formation of the Yamato Empire, the bakufu's continental
# administrative apparatus is recognized as a full institutional
# body. This auto modifier grants 3 slots, so 3 of the 6
# bureaucracies can run on the day the empire forms. No slot is
# counted from the capital: only a megalopolis gives one.
nd_ymp_bureaucracy_authority = {
	potential_trigger = {
		tag = YMP
		has_variable = nd_ymp_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
