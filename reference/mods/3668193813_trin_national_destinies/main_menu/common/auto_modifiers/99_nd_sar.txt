# Sardinia: three slots for its five institutions, from the day it forms
# (half of them, rounded up). Vanilla gives a Sardinian country none that day:
# no slot comes from a capital unless it is a megalopolis
# (location_ranks/00_default.txt), which Cagliari and Oristano are not, and
# the generic slots arrive with research in ages 4 and 6.
# tag, not has_or_had_tag: a country that goes on to form Italy takes its
# slots from Italy's own grant, so capacity does not stack.

nd_sar_bureaucracy_authority = {
	potential_trigger = {
		tag = SAR
		has_variable = nd_sar_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
