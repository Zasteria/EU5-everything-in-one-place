# National Destinies: KBU bureaucracy authority
#
# Grants 2 bureaucracy slots once KBU has been formed (nd_kbu_formed variable
# set by the formation event nd_kbu.1) and the player has the bureaucracies
# game-rule enabled. With the one slot a Muslim country has from sharia law
# (a capital gives one only as a megalopolis) the player may keep 3 of the 5
# Sayfawa institutions active at any given time.

nd_kbu_bureaucracy_authority = {
	potential_trigger = {
		tag = KBU
has_variable = nd_kbu_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
