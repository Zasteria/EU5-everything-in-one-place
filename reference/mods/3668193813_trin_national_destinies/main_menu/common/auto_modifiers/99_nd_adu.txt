# Al-Andalus bureaucracy slot grant.
# Applies after formation (nd_adu_formed) and only while the bureaucracies
# game rule is on. Grants 3 slots: Al-Andalus is Muslim under sharia law, which
# gives one slot from vanilla, and 4 slots hold half of its 7 institutions.

nd_adu_bureaucracy_authority = {
	potential_trigger = {
		tag = ADU
has_variable = nd_adu_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
