# DNM (Danubian Monarchy) bureaucracy slot grant.
# Activates after formation (nd_dnm_formed set by nd_dnm.1) so the player
# gains the bureaucracy-system permission once the multi-crown Danubian
# state has been forged, and only when the bureaucracies game-rule is
# enabled. DNM is a terminal formable (no successor in a formable chain),
# so both this slot grant and every bureaucracy potential use `tag = DNM`
# (current tag only): there is no later tag to preserve options across,
# and the validator enforces `tag = DNM` here as correct for DNM.
# Three slots for its six institutions, from the day it forms. No slot comes
# from a capital unless it is a megalopolis (location_ranks/00_default.txt),
# so these three are all the monarchy has that day.

nd_dnm_bureaucracy_authority = {
	potential_trigger = {
		tag = DNM
		has_variable = nd_dnm_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
