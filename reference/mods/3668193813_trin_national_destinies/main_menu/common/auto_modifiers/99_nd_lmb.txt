# itself (not has_or_had_tag), so forming a successor hands the slots to
# that formable instead of stacking them.
# Three slots for its six institutions, from the day it forms. No slot comes
# from a capital unless it is a megalopolis (location_ranks/00_default.txt),
# so these three are all Lombardy has that day.
nd_lmb_bureaucracy_authority = {
	potential_trigger = {
		tag = LMB
		has_variable = nd_lmb_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
