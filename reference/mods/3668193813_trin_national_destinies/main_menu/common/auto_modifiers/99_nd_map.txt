# Mapuche Confederation bureaucracy slot grant.
# Activates after formation (nd_map_formed) so the player only gains
# the bureaucracy-system permission once the federation has been
# proclaimed, and only when the bureaucracies game-rule is on.
# Two slots for its four institutions, from the day it forms. No slot
# comes from a capital unless it is a megalopolis
# (location_ranks/00_default.txt), so two councils stay
# uninstitutionalized, in keeping with the federation's preference
# for emergency convocations over standing courts.

nd_map_bureaucracy_authority = {
	potential_trigger = {
		tag = MAP
		has_variable = nd_map_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
