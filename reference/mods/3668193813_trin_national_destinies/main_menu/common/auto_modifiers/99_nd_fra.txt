# France (FRA): bureaucracy slot grant.
# Keys on nd_fra_formed, which nd_fra.1's gate sets for game-start France
# (appanages consolidated or the Hundred Years' War won) and INJECT:FRA_f
# sets for a country that forms France. Strict tag so the grant does not
# stack onto a successor formable; the bureaucracies themselves use
# has_or_had_tag and stay selectable there.
# Five bureaucracies, two slots here plus the capital's one: three of five.
nd_fra_bureaucracy_authority = {
	potential_trigger = {
		tag = FRA
		has_variable = nd_fra_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
