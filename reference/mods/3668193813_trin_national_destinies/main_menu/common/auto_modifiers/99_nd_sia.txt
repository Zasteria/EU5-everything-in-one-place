# National Destinies: SIA bureaucracy authority
#
# On formation (variable nd_sia_formed), Siam gains allow_bureaucracy and
# 3 slots to spend across the six SIA bureaucracies: half of its own
# institutions (slot rule of 2026-10-02). A capital adds a slot of its own
# only as a megalopolis, so none is counted from it.
nd_sia_bureaucracy_authority = {
	potential_trigger = {
		tag = SIA
has_variable = nd_sia_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
