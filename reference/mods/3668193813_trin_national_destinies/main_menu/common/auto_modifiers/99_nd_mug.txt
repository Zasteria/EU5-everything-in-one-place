# Grants bureaucracy authority to MUG once the empire has been formed
# and the player has the bureaucracies game-rule enabled. Disappears if
# the player forms a successor formable (uses tag, not has_or_had_tag).
# Three slots by the slot rule: half of the eight Mughal institutions is
# four, and a Muslim country under sharia law holds one slot from vanilla
# already. None comes from the capital (only a megalopolis gives one).

nd_mug_bureaucracy_authority = {
	potential_trigger = {
		tag = MUG
		has_variable = nd_mug_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
