# United Pueblo (PUE) bureaucracy slot grant.
# Activates after PUE is formed (nd_pue_formed) so the player only gains
# the bureaucracy-system permission once the federation has been
# explicitly proclaimed. Disappears if the player forms a successor
# formable (uses tag, not has_or_had_tag).
# Three slots for five institutions (half of its own, rounded up: the slot
# rule of 2026-10-02). Vanilla gives a country that is not Muslim no slot
# before age 4, and a capital adds one only as a megalopolis.

nd_pue_bureaucracy_authority = {
	potential_trigger = {
		tag = PUE
		has_variable = nd_pue_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
