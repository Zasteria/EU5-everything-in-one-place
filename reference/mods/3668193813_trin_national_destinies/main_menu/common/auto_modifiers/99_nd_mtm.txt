# Mataram bureaucracy slot grant.
# Activates after MTM is formed (nd_mtm_formed) so the player only gains the
# bureaucracy-system permission once the central-Javanese sultanate has been
# declared, and only when the bureaucracies game-rule is enabled.
# Three slots: half of the six MTM bureaucracies. No vanilla slot is counted
# for that day: a capital adds one only as a megalopolis
# (location_ranks/00_default.txt), and the country that becomes Mataram is
# not certain to be under sharia law (Java starts Hindu in 1337).

nd_mtm_bureaucracy_authority = {
	potential_trigger = {
		tag = MTM
		has_variable = nd_mtm_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
