# Aztec Empire bureaucracy slot grant.
# Activates after formation (nd_azt_formed) so the player only gains
# the bureaucracy-system permission once the Triple Alliance has
# been declared, and only when the bureaucracies game-rule is on.
# Four slots for seven institutions (half of its own, rounded up: the slot
# rule of 2026-10-02). Vanilla gives a country that is not Muslim no slot
# before age 4, and a capital adds one only as a megalopolis, which
# Tenochtitlan is not on day one.

nd_azt_bureaucracy_authority = {
	potential_trigger = {
		tag = AZT
		has_variable = nd_azt_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
