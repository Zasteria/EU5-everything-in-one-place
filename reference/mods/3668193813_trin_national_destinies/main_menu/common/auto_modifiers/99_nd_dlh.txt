# Grants bureaucracy authority to DLH once the sultanate has been formed
# and the player has the bureaucracies game-rule enabled. Disappears if
# the player forms a successor formable (uses tag, not has_or_had_tag).
# Three slots by the slot rule of 2026-10-02: half of Delhi's seven
# institutions is four, and a Muslim country under sharia law holds one
# slot from vanilla already. None comes from the capital (only a
# megalopolis gives one).

nd_dlh_bureaucracy_authority = {
	potential_trigger = {
		tag = DLH
		has_variable = nd_dlh_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
