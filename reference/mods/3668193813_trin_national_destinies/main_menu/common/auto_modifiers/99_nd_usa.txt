# The United States' bureaucracy slots.
#
# Four slots once the United States has been formed (nd_usa_formed is set
# in the form_effect of USA_f) and the bureaucracies game rule is on: half
# of the country's seven institutions, rounded up. Vanilla certainly gives
# a newly formed republic none (no sharia law, and a capital grants a slot
# only as a megalopolis, location_ranks/00_default.txt). The tag test is
# strict: the grant ends when the country forms a successor.
nd_usa_bureaucracy_authority = {
	potential_trigger = {
		tag = USA
		has_variable = nd_usa_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
