# Grants bureaucracy authority to DEC once the Deccan has been formed
# and the player has the bureaucracies game-rule enabled. Disappears if
# the player forms a successor formable (uses tag, not has_or_had_tag).
# 4 slots for seven institutions (slot rule of 2026-10-02: a country can run
# half of its own institutions, rounded up, on the day it forms). No vanilla
# slot is counted: no country stands ready to form the Deccan on day one, so
# none is certain to hold sharia law, and a capital gives a slot only as a
# megalopolis (location_ranks/00_default.txt).

nd_dec_bureaucracy_authority = {
	potential_trigger = {
		tag = DEC
		has_variable = nd_dec_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
