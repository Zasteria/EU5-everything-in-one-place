# Grants bureaucracy authority to Rum once the sultanate has been formed and
# the player has the bureaucracies game-rule enabled. Disappears if the
# player forms a successor formable (uses tag, not has_or_had_tag).
# Three slots: the country has seven institutions of its own and can run
# four of them on the day it forms; sharia law gives the fourth slot.

nd_rum_bureaucracy_authority = {
	potential_trigger = {
		# RUM_f creates the tag TUR; no country is ever RUM.
		tag = TUR
		has_variable = nd_rum_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
