# Grants MTP two bureaucracy slots once the empire has been formed and the
# bureaucracies game rule is on: half of its four institutions. Vanilla
# gives this country no slot at formation. Uses tag, so the grant ends when
# the country forms a successor.
nd_mtp_bureaucracy_authority = {
	potential_trigger = {
		tag = MTP
has_variable = nd_mtp_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
