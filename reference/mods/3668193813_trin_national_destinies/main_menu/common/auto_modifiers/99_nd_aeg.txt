# AEG (Aegyptus) bureaucracy slot grant.
# Activates after formation (nd_aeg_formed) so the player gains the
# bureaucracy-system permission once the Sultan-Pharaoh has been
# crowned in Cairo, only when the bureaucracies game-rule is enabled.
# 5 bureaucracies; half, rounded up, is 3. Two slots are vanilla's on the day
# Aegyptus forms (scripts/bureaucracy/apply_slot_rule.py counts sharia law and
# a megalopolis capital), so the grant here is 1. An ordinary capital gives
# no slot.

nd_aeg_bureaucracy_authority = {
	potential_trigger = {
		tag = AEG
		has_variable = nd_aeg_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
