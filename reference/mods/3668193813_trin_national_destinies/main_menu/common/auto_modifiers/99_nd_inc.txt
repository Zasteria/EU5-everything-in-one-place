# Grants bureaucracy authority to INC once Tawantinsuyu has been formed and
# the player has the bureaucracies game-rule enabled. Disappears if the
# player forms a successor formable (uses tag, not has_or_had_tag).
# Four slots for seven institutions (half of its own, rounded up: the slot
# rule of 2026-10-02). Vanilla gives a country that is not Muslim no slot
# before age 4, and a capital adds one only as a megalopolis.

nd_inc_bureaucracy_authority = {
	potential_trigger = {
		tag = INC
		has_variable = nd_inc_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
