# Grants bureaucracy authority to ETH once Ethiopia has been formed and the
# player has the bureaucracies game-rule enabled. Disappears if the player
# forms a successor formable (uses tag, not has_or_had_tag).
# Four slots: half of the seven institutions, rounded up, on the day the
# country forms. Vanilla gives it none that day, and the capital gives none.

nd_eth_bureaucracy_authority = {
	potential_trigger = {
		tag = ETH
		has_variable = nd_eth_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
