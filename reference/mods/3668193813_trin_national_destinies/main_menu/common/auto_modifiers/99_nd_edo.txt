# Grants bureaucracy authority to EDO once the Kingdom of Benin has been
# formed and the player has the bureaucracies game-rule enabled. Disappears
# if the player forms a successor formable (uses tag, not has_or_had_tag).
# Three slots: half of the country's six institutions (Uzama Nihinron,
# Eghaevbo n'Ogbe, Igun-Eronmwon, Igue Festival, Iyase, Ologbo'na). Vanilla
# gives EDO no slot on the day it forms (slot rule of 2026-10-02).

nd_edo_bureaucracy_authority = {
	potential_trigger = {
		tag = EDO
		has_variable = nd_edo_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
