# The Two Sicilies: three slots for its six institutions, from the day it
# forms. Vanilla gives the country none that day: no slot comes from a capital
# unless it is a megalopolis (location_ranks/00_default.txt), which Naples and
# Palermo are not in 1337, and the generic slots arrive with research in ages
# 4 and 6.
# tag, not has_or_had_tag: a country that goes on to form Italy takes its
# slots from Italy's own grant, so capacity does not stack.
nd_tws_bureaucracy_authority = {
	potential_trigger = {
		tag = TWS
		has_variable = nd_tws_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
