# Imerina (Madagascar) bureaucracy slot grant.
# Activates after MIR is formed (nd_mir_formed) so the player only gains the
# bureaucracy-system permission once the unified Malagasy kingdom has been
# proclaimed, and only when the bureaucracies game-rule is enabled. Two
# slots by the slot rule: half of the five Malagasy institutions is three,
# and the Muslim port republics that can form the country on day one hold
# one slot already from sharia law (validate_bureaucracy.py, slot_count).
# No slot comes from the capital: only a megalopolis gives one.

nd_mir_bureaucracy_authority = {
	potential_trigger = {
		tag = MIR
		has_variable = nd_mir_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
