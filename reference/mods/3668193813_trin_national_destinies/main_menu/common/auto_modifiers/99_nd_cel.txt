# Grants bureaucracy authority to CEL once the Celtic Union has been formed
# and the player has the bureaucracies game-rule enabled. Four slots: the
# country has seven institutions of its own and can run four of them on the
# day it forms. Vanilla gives it none that day.

nd_cel_bureaucracy_authority = {
	potential_trigger = {
		tag = CEL
has_variable = nd_cel_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
