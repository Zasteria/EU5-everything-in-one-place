# Westphalian bureaucracy slot grant.
# Activates after WES is formed (nd_wes_formed) so the player only gains the
# bureaucracy-system permission once the Westphalian state has been declared,
# and only when the bureaucracies game-rule is enabled.
# Three slots for five institutions: half of them, rounded up, on the day
# Westphalia forms.
# A capital adds a slot of its own only as a megalopolis
# (location_ranks/00_default.txt), which Muenster is not.

nd_wes_bureaucracy_authority = {
	potential_trigger = {
		tag = WES
has_variable = nd_wes_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
