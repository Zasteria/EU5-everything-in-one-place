# Grants bureaucracy authority to THE once Thessalonica has been formed and
# the player has the bureaucracies game rule enabled. Gates on tag rather
# than has_or_had_tag, so a player who goes on to form Greece or the Latin
# Empire loses this grant and takes the successor's own slot count instead:
# capacity never stacks along a formable chain. Two slots here, and none from
# the capital (only a megalopolis gives one, location_ranks/00_default.txt,
# and Thessalonica is not one): two seats against four institutions.

nd_the_bureaucracy_authority = {
	potential_trigger = {
		tag = THE
		has_variable = nd_the_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
