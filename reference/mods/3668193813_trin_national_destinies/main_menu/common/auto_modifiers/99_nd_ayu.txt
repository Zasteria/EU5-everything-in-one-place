# National Destinies: AYU bureaucracy authority
#
# Once Ayutthaya has been formed (nd_ayu_formed) and the bureaucracies game
# rule is on, three slots: half of the five AYU bureaucracies, rounded up.
# Vanilla gives the country none on that day. A capital adds a slot only as a
# megalopolis (location_ranks/00_default.txt), which Ayutthaya is not at the
# start, and the kingdom is not under sharia law.
nd_ayu_bureaucracy_authority = {
	potential_trigger = {
		tag = AYU
		has_variable = nd_ayu_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
