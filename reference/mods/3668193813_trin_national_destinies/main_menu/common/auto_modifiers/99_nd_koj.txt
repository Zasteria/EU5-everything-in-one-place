# Kingdom of Jerusalem bureaucracy slot grant: 3 slots, half of its five
# institutions rounded up. Vanilla gives a country that is not Muslim no
# slot on the day it forms, and a capital grants one only as a megalopolis.
# Applies once KOJ is formed (nd_koj_formed) and only while the
# bureaucracies game rule is on.

nd_koj_bureaucracy_authority = {
	potential_trigger = {
		tag = KOJ
has_variable = nd_koj_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
