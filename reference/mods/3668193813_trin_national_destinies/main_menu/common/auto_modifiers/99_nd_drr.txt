# Durrani Empire bureaucracy slot grant.
# Activates after formation (nd_drr_formed) so the player only gains the
# bureaucracy-system permission once the empire has been declared at the
# Kandahar loya jirga, and only when the bureaucracies game-rule is enabled.
# Two slots: half of the four Durrani institutions. Vanilla gives DRR no slot
# for certain on the day it forms (sharia law is a policy it may enact later),
# and a capital gives none unless it is a megalopolis (slot rule of 2026-10-02).

nd_drr_bureaucracy_authority = {
	potential_trigger = {
		tag = DRR
		has_variable = nd_drr_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
