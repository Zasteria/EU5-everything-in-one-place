# Grants bureaucracy authority to BAH once the sultanate has been formed
# and the player has the bureaucracies game-rule enabled. Disappears if
# the player forms a successor formable (uses tag, not has_or_had_tag).
# One slot from this grant and one from sharia law as the legal code
# (laws/01_legal_system.txt:119), which a Sunni country holds from the start:
# two slots on the day of formation, half of the four BAH bureaucracies. A
# capital adds a slot only as a megalopolis (location_ranks/00_default.txt).
# The same slots also hold vanilla's Islamic offices (bureaucracies/muslim.txt).

nd_bah_bureaucracy_authority = {
	potential_trigger = {
		tag = BAH
		has_variable = nd_bah_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
