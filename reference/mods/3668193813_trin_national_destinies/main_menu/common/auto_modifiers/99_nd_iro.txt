# Haudenosaunee (IRO) bureaucracy slot grant.
# Activates after IRO is formed (nd_iro_formed) so the player only gains
# the bureaucracy-system permission once the Confederacy has been
# explicitly declared. Disappears if the player forms a successor
# formable (uses tag, not has_or_had_tag).
# Three slots for six institutions (half of its own: the slot rule of
# 2026-10-02). Vanilla gives a country that is not Muslim no slot before
# age 4, and a capital adds one only as a megalopolis.

nd_iro_bureaucracy_authority = {
	potential_trigger = {
		tag = IRO
		has_variable = nd_iro_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
