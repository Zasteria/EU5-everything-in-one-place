# Greece: three slots for its six institutions, from the day it forms.
# Vanilla gives a Greek country none that day: no slot comes from a capital
# unless it is a megalopolis (location_ranks/00_default.txt), which Athens
# is not, and the generic slots arrive with research in ages 4 and 6.
# tag, not has_or_had_tag: a country that goes on to a larger formable takes
# its slots from that formable's own grant, so capacity does not stack.
nd_gre_bureaucracy_authority = {
	potential_trigger = {
		tag = GRE
		has_variable = nd_gre_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
