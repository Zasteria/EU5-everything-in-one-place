# Rajputana bureaucracy slot grant.
# Activates after formation (nd_rjp_formed) so the player gains the
# bureaucracy-system permission once Rajputana has been forged, and
# only when the bureaucracies game-rule is enabled.
# 7 bureaucracies, 4 slots granted here (slot rule of 2026-10-02: a country
# can run half of its own institutions, rounded up, on the day it forms). No
# vanilla slot is counted for that day, and a capital gives a slot only as a
# megalopolis (location_ranks/00_default.txt).

nd_rjp_bureaucracy_authority = {
	potential_trigger = {
		tag = RJP
		has_variable = nd_rjp_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
