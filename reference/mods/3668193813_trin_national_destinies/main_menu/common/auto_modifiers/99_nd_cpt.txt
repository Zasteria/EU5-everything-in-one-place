# CPT (Coptic Egypt) bureaucracy slot grant.
#
# Activates after formation (nd_cpt_formed), only when the bureaucracies
# game rule is enabled. 4 bureaucracies, 2 slots granted here (a capital
# gives a slot only as a megalopolis). The player picks 2 of 4.
#
# The ration is deliberately tight. A minority regime running a country
# four fifths of which does not share its confession cannot staff every
# apparatus at once, and the two it picks are the whole strategic
# argument of the campaign.
#
# Gated on `tag = CPT` rather than has_or_had_tag: forming Aegyptus
# hands the slot grant to AEG's own auto-modifier instead of stacking on
# top of this one. The bureaucracies themselves use has_or_had_tag, so a
# CPT that goes on to form AEG keeps the Coptic institutions in the
# picker without gaining capacity for them.

nd_cpt_bureaucracy_authority = {
	potential_trigger = {
		tag = CPT
		has_variable = nd_cpt_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
