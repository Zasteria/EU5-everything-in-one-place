# Swabian bureaucracy slot grant.
# Activates after SWA is formed (nd_swa_formed) so the player only gains the
# bureaucracy-system permission once unified Swabia has been declared, and
# only when the bureaucracies game-rule is enabled. Three slots for five
# institutions: half of them, rounded up, on the day Swabia forms. A capital
# adds a slot of its own only as a megalopolis
# (location_ranks/00_default.txt), which no Swabian city is at formation.

nd_swa_bureaucracy_authority = {
	potential_trigger = {
		tag = SWA
has_variable = nd_swa_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
