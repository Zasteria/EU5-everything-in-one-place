# Dalmatia: three slots for its five institutions, from the day it forms.
# Vanilla gives a Dalmatian country none that day: no slot comes from a
# capital unless it is a megalopolis (location_ranks/00_default.txt), which
# Dubrovnik is not, and the generic slots arrive with research in ages 4 and 6.
# tag, not has_or_had_tag: a country that goes on to a larger formable takes
# its slots from that formable's own grant, so capacity does not stack.
nd_dal_bureaucracy_authority = {
	potential_trigger = {
		tag = DAL
		has_variable = nd_dal_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
