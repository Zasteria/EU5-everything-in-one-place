# BNH (Beth Nahrain) bureaucracy slot grant.
# Activates after formation (nd_bnh_formed) so the player gains the
# bureaucracy-system permission once the Catholicos has been re-enthroned
# at Mosul, only when the bureaucracies game-rule is enabled.
# 4 bureaucracies, 2 slots granted here (half of the country's own
# institutions; a capital gives a slot only as a megalopolis).

nd_bnh_bureaucracy_authority = {
	potential_trigger = {
		tag = BNH
		has_variable = nd_bnh_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
