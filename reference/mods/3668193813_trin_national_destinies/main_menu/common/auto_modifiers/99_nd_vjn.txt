# Vijayanagara (Karnata Empire) bureaucracy slot grant.
# Activates after formation (nd_vjn_formed) so the player gains the
# bureaucracy-system permission once the empire has been restored, and
# only when the bureaucracies game-rule is enabled.
# Two slots for its four institutions, from the day it forms. No slot comes
# from a capital unless it is a megalopolis (location_ranks/00_default.txt),
# so these two are all the empire has that day.

nd_vjn_bureaucracy_authority = {
	potential_trigger = {
		tag = VJN
		has_variable = nd_vjn_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
