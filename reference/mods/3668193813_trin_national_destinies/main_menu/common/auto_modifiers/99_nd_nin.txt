# NIN: Assyrian Empire bureaucratic authority
#
# On formation of the restored Assyrian Empire, the Sargonid
# administrative apparatus is recognized as a full institutional
# body. This auto-modifier grants 3 slots, so 3 of the 6
# bureaucracies are active at once (a capital gives a slot only
# as a megalopolis).
nd_nin_bureaucracy_authority = {
	potential_trigger = {
		tag = NIN
		has_variable = nd_nin_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
