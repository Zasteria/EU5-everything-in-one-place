# Grants bureaucracy authority to ALM once the caliphate has been formed and
# the player has the bureaucracies game-rule enabled. Disappears if the
# player forms a successor formable (uses tag, not has_or_had_tag).
# One slot: with the one sharia law gives a Muslim country, two of the four
# institutions can run on the day the caliphate forms. A capital gives no
# slot unless it is a megalopolis (slot rule of 2026-10-02).

nd_alm_bureaucracy_authority = {
	potential_trigger = {
		tag = ALM
		has_variable = nd_alm_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
