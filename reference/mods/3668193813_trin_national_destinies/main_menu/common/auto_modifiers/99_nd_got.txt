# Formation is the unlock. A kingdom restored by legal argument inherits the
# offices that argument names, so the moment nd_got_formed is set the crown may
# host the institutions of the Gothic state: two slots, half of its four
# institutions (a capital gives a slot only as a megalopolis).
#
# The gate is the strict current tag, not has_or_had_tag: if this player goes on
# to form a successor at a higher level, that formable's own grant replaces this
# one and the slot counts do not stack. The bureaucracy file uses has_or_had_tag
# instead, so the four institutions stay selectable in the picker.
nd_got_bureaucracy_authority = {
	potential_trigger = {
		tag = GOT
		has_variable = nd_got_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
