# National Destinies: NPL bureaucracy authority
#
# Grants Nepal three bureaucracy slots by the slot rule of 2026-10-02:
# half of its five institutions, rounded up. Vanilla gives the countries
# that form Nepal no slot on that day, and none comes from the capital
# (only a megalopolis gives one).
# Gated on the formation event having fired (nd_npl_formed) and
# on the player's nd_bureaucracies_enabled game rule.
nd_npl_bureaucracy_authority = {
	potential_trigger = {
		tag = NPL
		has_variable = nd_npl_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
