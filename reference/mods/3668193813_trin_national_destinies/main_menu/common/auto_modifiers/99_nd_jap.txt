# Japanese bureaucracy slot grant.
# Activates after JAP is formed (nd_jap_formed) so the player only gains the
# bureaucracy-system permission once the unified shogunate state has been
# proclaimed, and only when the bureaucracies game-rule is enabled.
#
# Four slots: half of the eight JAP bureaucracies. Vanilla gives the country
# none on that day: a capital adds a slot only as a megalopolis
# (location_ranks/00_default.txt), which no Japanese city is at the start,
# and Japan is not under sharia law. From age 4 the same slots also hold
# vanilla's ten generic offices (bureaucracies/generic.txt).

nd_jap_bureaucracy_authority = {
	potential_trigger = {
		tag = JAP
		has_variable = nd_jap_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
