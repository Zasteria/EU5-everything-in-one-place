# China bureaucracy slot grant.
# Activates after formation (nd_chi_formed) so the player only gains the
# bureaucracy-system permission once the empire has been declared,
# and only when the bureaucracies game-rule is enabled.
# Three slots. With the one a megalopolis capital gives (Dadu in 1337) that
# is four of the seven institutions on the day the country forms; a smaller
# capital gives no slot (slot rule of 2026-10-02).

nd_chi_bureaucracy_authority = {
	potential_trigger = {
		tag = CHI
		has_variable = nd_chi_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
