# Grants bureaucracy authority to TDR once the Bosporan Kingdom has been
# formed and the player has the bureaucracies game-rule enabled.
# Disappears if the player forms a successor formable (uses tag, not
# has_or_had_tag).
# Two slots: half of the four Bosporan institutions. Vanilla gives TDR no
# slot for certain on the day it forms, and a capital gives none unless it
# is a megalopolis (slot rule of 2026-10-02).

nd_tdr_bureaucracy_authority = {
	potential_trigger = {
		tag = TDR
		has_variable = nd_tdr_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
