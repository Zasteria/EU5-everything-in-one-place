# National Destinies: MGE bureaucracy authority
#
# Once the empire is formed (nd_mge_formed), this auto modifier grants
# two slots. The main former is Muslim and starts under sharia law, which
# gives one slot of its own, so three of the six MGE bureaucracies can run
# on the day the empire forms (slot rule of 2026-10-02). A capital gives
# no slot unless it is a megalopolis.
nd_mge_bureaucracy_authority = {
	potential_trigger = {
		tag = MGE
		has_variable = nd_mge_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
