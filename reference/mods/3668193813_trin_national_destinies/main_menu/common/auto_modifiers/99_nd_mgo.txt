# Mongolia's bureaucracy slots.
#
# Three slots once Mongolia has been formed (nd_mgo_formed is set in the
# form_effect of MGO_f) and the bureaucracies game rule is on: half of the
# country's five institutions, rounded up. Vanilla gives a newly formed
# Mongolia none (no sharia law, and a capital grants a slot only as a
# megalopolis, location_ranks/00_default.txt), so the grant is all three.
# The tag test is strict: the grant ends when the country forms a successor.
nd_mgo_bureaucracy_authority = {
	potential_trigger = {
		tag = MGO
		has_variable = nd_mgo_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
