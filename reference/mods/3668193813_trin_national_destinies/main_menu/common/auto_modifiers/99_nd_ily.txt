# ILY (Illyria) bureaucracy slot grant.
# Activates after formation (nd_ily_formed is set by nd_ily.1) and only when
# the bureaucracies game rule is enabled. The slot grant gates on the strict
# current tag so capacity never stacks across a formable chain, while each
# bureaucracy's own potential uses has_or_had_tag so the institutions stay
# selectable for a player who forms a successor.
# 4 bureaucracies, 2 slots granted here (a capital gives a slot only as a
# megalopolis). Player picks 2 of 4.

nd_ily_bureaucracy_authority = {
	potential_trigger = {
		tag = ILY
		has_variable = nd_ily_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
