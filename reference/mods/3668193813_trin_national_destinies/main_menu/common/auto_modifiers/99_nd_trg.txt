# National Destinies: TRG bureaucracy slot
#
# One slot once the confederation has been formed (nd_trg_formed is set in the
# form_effect of TRG_f) and the bureaucracies game rule is on. The tag test is
# strict, so the grant ends when the country forms a successor; the four
# institutions stay selectable through has_or_had_tag in their own file.
#
# The country's second slot does not come from its capital: vanilla gives a
# capital slot to a megalopolis only (location_ranks/00_default.txt:49-51), and
# the confederation has no town at all. It comes from sharia law as the legal
# code, which every Tuareg tribe starts with
# (setup/templates/subsaharan_muslim_tribe.txt:6, laws/01_legal_system.txt:119).
# Two slots at formation, then, for four institutions of the mod and the nine
# Islamic offices of vanilla.
#
# allow_bureaucracy is granted to every country by vanilla's own base modifier
# on 1.4 (auto_modifiers/country.txt:78); it is repeated here so the grant does
# not depend on that line surviving a patch.
nd_trg_bureaucracy_authority = {
	potential_trigger = {
		tag = TRG
		has_variable = nd_trg_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
