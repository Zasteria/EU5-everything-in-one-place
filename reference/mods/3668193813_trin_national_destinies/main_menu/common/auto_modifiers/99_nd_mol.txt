# Moldavia: three slots for its six institutions, from the day it forms.
# Vanilla gives a Moldavian country none that day: no slot comes from a capital
# unless it is a megalopolis (location_ranks/00_default.txt), which neither
# Suceava nor Iasi is, and the generic slots arrive with research in ages 4 and 6.
# tag, not has_or_had_tag: a country that goes on to form Romania takes its
# slots from Romania's own grant, so capacity does not stack.
nd_mol_bureaucracy_authority = {
	potential_trigger = {
		tag = MOL
		has_variable = nd_mol_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
