# Grants Catalonia its bureaucracy slots once the country has been formed and
# the player has the bureaucracies game rule enabled.
# Uses the strict tag, not has_or_had_tag, so slot capacity does not stack when
# the player goes on to form the Aragonese Empire or Spain. The bureaucracies
# themselves gate on has_or_had_tag and stay selectable.
# allow_bureaucracy is already in vanilla's base values for every country
# (auto_modifiers/country.txt:78); it is restated because the mod's validator
# and its other 161 grants expect it, and it changes nothing.
# Two slots for four institutions. A capital adds a slot of its own only as a
# megalopolis (location_ranks/00_default.txt:49-50), which Barcelona is not
# for centuries, so these two are what a Catalan principality starts with.

nd_cat_bureaucracy_authority = {
	potential_trigger = {
		tag = CAT
		has_variable = nd_cat_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
