# Tuscany: four slots for its seven institutions, from the day it forms
# (half of them, rounded up). Vanilla gives a Tuscan country none that day:
# no slot comes from a capital unless it is a megalopolis
# (location_ranks/00_default.txt), which Florence is not in 1337, and the
# generic slots arrive with research in ages 4 and 6.
# tag, not has_or_had_tag: a country that goes on to form Italy takes its
# slots from Italy's own grant, so capacity does not stack.
nd_tus_bureaucracy_authority = {
	potential_trigger = {
		tag = TUS
		has_variable = nd_tus_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 4
}
