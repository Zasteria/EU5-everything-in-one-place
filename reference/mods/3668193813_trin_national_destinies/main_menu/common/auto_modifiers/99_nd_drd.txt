# Dalriada bureaucracy slot grant.
# Activates after formation (nd_drd_formed) so the player only gains the
# bureaucracy-system permission once the kingdom has been declared, and
# only when the bureaucracies game-rule is enabled.
# Three slots: the country has five institutions of its own and can run
# three of them on the day it forms. Vanilla gives it none that day.

nd_drd_bureaucracy_authority = {
	potential_trigger = {
		tag = DRD
has_variable = nd_drd_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
