# Grants bureaucracy authority to PON once the Pontic Empire has been
# refounded and the player has the bureaucracies game-rule enabled.
# Disappears if the player forms a successor formable (uses tag, not
# has_or_had_tag).
# Two slots for its four institutions, from the day it forms. No slot comes
# from a capital unless it is a megalopolis (location_ranks/00_default.txt),
# so these two are all the empire has that day.

nd_pon_bureaucracy_authority = {
	potential_trigger = {
		tag = PON
		has_variable = nd_pon_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
