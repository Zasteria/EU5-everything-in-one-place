# Grants bureaucracy slots to the country that formed Rome from Byzantium
# or Trebizond (ROM_BYZ_f). Three slots: half of its six institutions, by
# the slot rule (validate_bureaucracy.py, slot_count). No slot is counted
# from the capital: only a capital of rank megalopolis gives one
# (location_ranks/00_default.txt), which Constantinople is and Rome is not.
# The former Byzantium keeps the two slots of vanilla's advance
# byz_bureaucracy_advance (gated on has_or_had_tag = BYZ); ROM_f's own
# grant in 99_nd_rom.txt reads another formation marker and does not apply.
#
# Uses tag = ROM rather than has_or_had_tag so slots do NOT stack across
# any future successor formable; bureaucracies themselves persist via the
# has_or_had_tag pattern in their own potential blocks.

nd_rom_byz_bureaucracy_authority = {
	potential_trigger = {
		tag = ROM
		has_variable = nd_rom_byz_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
