# Bureaucratic authority for Maharlika
# Activates on formation, gated on the bureaucracies game rule.
# 5 bureaucracies. The grant is 3 slots, so about half of them can run on the day
# Maharlika forms. No slot is counted from the capital: only a megalopolis gives one.
nd_mrl_bureaucracy_authority = {
    potential_trigger = {
        tag = MRL
        has_variable = nd_mrl_formed
        has_game_rule = nd_bureaucracies_enabled
    }
    allow_bureaucracy = yes
    global_max_bureaucracy_slots = 3
}
