# National Destinies: KAZ bureaucracy authority
#
# Three slots once the Kazakh Khanate has been formed (nd_kaz_formed is
# set in the form_effect of KAZ_f) and the bureaucracies game rule is on:
# half of the khanate's five institutions, rounded up. Vanilla certainly
# gives a newly formed khanate none: a capital grants a slot only as a
# megalopolis (location_ranks/00_default.txt), and the tribes and hordes
# that form it do not all hold sharia law. The tag test is strict: the
# grant ends when the country forms a successor.
nd_kaz_bureaucracy_authority = {
	potential_trigger = {
		tag = KAZ
		has_variable = nd_kaz_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
