# Grants bureaucracy authority to DAH once Dahomey has been formed and the
# player has the bureaucracies game-rule enabled. Disappears if the player
# forms a successor formable (uses tag, not has_or_had_tag).
# Three slots: half of the five DAH bureaucracies, rounded up. Vanilla gives
# the kingdom none on that day: a capital adds a slot only as a megalopolis
# (location_ranks/00_default.txt), which Abomey is not, and Dahomey is not
# under sharia law.

nd_dah_bureaucracy_authority = {
	potential_trigger = {
		tag = DAH
		has_variable = nd_dah_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
