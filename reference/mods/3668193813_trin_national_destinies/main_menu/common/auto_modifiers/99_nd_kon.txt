# Grants bureaucracy authority to KON once Kongo has been formed and the
# player has the bureaucracies game-rule enabled. Disappears if the player
# forms a successor formable (uses tag, not has_or_had_tag).
# Three slots for five institutions; vanilla gives a country that is not
# Muslim none on the day it forms.

nd_kon_bureaucracy_authority = {
	potential_trigger = {
		tag = KON
has_variable = nd_kon_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
