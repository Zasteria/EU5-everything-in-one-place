##############################################################################
# Formation Event: The Kingdom of Lombardy
##############################################################################

nd_lmb.1 = {
	type = country_event
	title = nd_lmb.1.t
	desc = nd_lmb.1.d
	outcome = neutral
	# Per-country guard rather than fire_only_once, which is per CAMPAIGN
	# and would leave a re-formed Lombardy a content shell. Formation
	# markers themselves are set in form_effect, never only here.
	trigger = {
		NOT = { has_variable = nd_lmb_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_lmb_e1_done }
	}

	# The Iron Crown: the Visconti road. A crowned kingdom above the
	# communes, ducal centralization made royal.
	option = {
		name = nd_lmb.1.a
		custom_tooltip = { text = nd_lmb.1.a_tt }
		add_country_modifier = {
			modifier = nd_lmb_corona_del_regno
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_lmb_formed
			set_variable = nd_lmb_path_a
		}
	}

	# The Ambrosian Commune: the road of the League and the Golden
	# Republic. The cities rule themselves under Saint Ambrose.
	option = {
		name = nd_lmb.1.b
		custom_tooltip = { text = nd_lmb.1.b_tt }
		add_country_modifier = {
			modifier = nd_lmb_comune_ambrosiano
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_lmb_formed
			set_variable = nd_lmb_path_b
		}
	}
}
