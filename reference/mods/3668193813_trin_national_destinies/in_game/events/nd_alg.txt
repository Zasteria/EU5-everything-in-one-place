namespace = nd_alg

# Formation event: fires from ALG_f's form_effect.
nd_alg.1 = {
	type = country_event
	title = nd_alg.1.t
	desc = nd_alg.1.d
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that formed the
	# tag first would retire the popup for every later former.
	trigger = {
		NOT = { has_variable = nd_alg_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_alg_e1_done }
	}

	# Path A: the sea-facing city. The court goes to Algiers if it is ours.
	option = {
		name = nd_alg.1.a
		add_country_modifier = {
			modifier = nd_alg_the_well_guarded
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_right
		}
		if = {
			limit = {
				owns = location:algiers
				NOT = { capital = location:algiers }
			}
			set_capital = location:algiers
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_alg_formed
			set_variable = nd_alg_path_a
		}
	}

	# Path B: the emirate of the interior.
	option = {
		name = nd_alg.1.b
		add_country_modifier = {
			modifier = nd_alg_emir_of_the_tell
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_left
		}
		# The mysticism axis exists for Muslim countries only.
		if = {
			limit = { religion.group = religion_group:muslim }
			change_societal_value = {
				type = mysticism_vs_jurisprudence
				value = societal_value_minor_move_to_left
			}
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_alg_formed
			set_variable = nd_alg_path_b
		}
	}
}

# ===== DESTINY EVENTS =====
# One event per destiny advance, nd_alg.8 to nd_alg.14. The root and both
# first stages are acknowledgments. The second and third stages offer two
# readings of the milestone, each a timed modifier: 15 years at stage 2,
# 20 years at stage 3.
#
# Open question, reported to the orchestrator on 2026-10-01: these events
# name tag = ALG alone, as every destiny event in the mod names its own tag,
# and only a tag named there can get the event (events/readme.txt). The
# advances themselves survive a later formation (has_or_had_tag). A country
# that forms the Almohads or Carthage before finishing a path therefore keeps
# the advances and never receives the stage events or their modifiers.

# DHE 8: the destiny root
nd_alg.8 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = ALG
		tag = ALM
		tag = CTH
		tag = EUR
		tag = KHL
		tag = ROM
		tag = SPA
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_alg_destiny
	}

	title = nd_alg.8.t
	desc = nd_alg.8.d
	outcome = neutral

	option = {
		name = nd_alg.8.a
	}
}
