namespace = nd_val

# Formation event: fires from form_effect (99_nd_val.txt) for a Catalan
# country that formed the kingdom by the button, and from the four-yearly
# marker repair for a Valencia that was released or broke away and so never
# pressed it. The text speaks of the kingdom having a government of its own,
# which is true on both roads, and names no former.
nd_val.1 = {
	type = country_event
	title = nd_val.1.t
	desc = nd_val.1.d
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that held the tag
	# first would retire the popup for every later holder. Country variables
	# die with the country, so a Valencia that falls and rises again is asked
	# again.
	trigger = {
		NOT = { has_variable = nd_val_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_val_e1_done }
	}

	# The king's towns under the Furs.
	option = {
		name = nd_val.1.a
		add_country_modifier = {
			modifier = nd_val_furs_i_viles_reials
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_val_formed
			set_variable = nd_val_path_a
		}
	}
	# The Alfonsine jurisdiction: the lords inside the law, at a price.
	option = {
		name = nd_val.1.b
		add_country_modifier = {
			modifier = nd_val_jurisdiccio_alfonsina
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_val_formed
			set_variable = nd_val_path_b
		}
	}
}

# scripts/sync_history_events.py adds the tags of the larger formables a

# from every other holder of those tags.
