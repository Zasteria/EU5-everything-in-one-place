namespace = nd_fra

# nd_fra.1: the kingdom made whole.
# One event, two deliveries:
#   game-start France: the monthly gate in 99_nd_on_actions.txt fires it
#     once the appanages are consolidated (vanilla flavor_fra.1103,
#     DHE/flavor_FRA.txt:2912, sets the permanent variable
#     enable_centralized_fra_reform at :2953) or the Hundred Years' War is won
#     (the gate records nd_fra_won_hyw while victory_in_the_hyw_modifier holds);
#   any country that forms France: INJECT:FRA_f form_effect.
# The description follows how the moment was earned. If both deeds are on
# record at once, the war is the headline. The event sets NO destiny path
# variable: France's two destiny roads open through deeds, not this choice.
nd_fra.1 = {
	type = country_event
	title = nd_fra.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = nd_fra_won_hyw }
				desc = nd_fra.1.d.hyw
			}
			triggered_desc = {
				trigger = { has_variable = enable_centralized_fra_reform }
				desc = nd_fra.1.d.appanages
			}
			triggered_desc = {
				desc = nd_fra.1.d.formed
			}
		}
	}
	outcome = positive
	# Per-country guard, not fire_only_once: that is per CAMPAIGN
	# (events/readme.txt:62), so a France that falls and is formed again
	# would never see its founding moment. Country variables die with the
	# country, so this guard re-arms on re-formation.
	trigger = {
		NOT = { has_variable = nd_fra_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_fra_e1_done }
	}

	# Bodin's sovereignty: one king, one faith, one law.
	option = {
		name = nd_fra.1.a
		add_country_modifier = {
			modifier = nd_fra_un_roi_une_foi_une_loi
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = absolutism_vs_liberalism
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_fra_formed
		}
		ai_chance = { base = 1 }
	}
	# Seyssel's tempered monarchy: the three bridles of religion, justice
	# and police (La Grande Monarchie de France, 1519).
	option = {
		name = nd_fra.1.b
		add_country_modifier = {
			modifier = nd_fra_monarchie_temperee
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = absolutism_vs_liberalism
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_fra_formed
		}
		ai_chance = { base = 1 }
	}
}
