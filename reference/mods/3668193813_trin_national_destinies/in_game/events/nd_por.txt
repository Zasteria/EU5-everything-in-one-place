namespace = nd_por

# Formation event. Portugal holds its tag from day one and never runs
# form_effect: the four-yearly marker repair delivers this event to it, so
# the default text speaks to a kingdom that already exists. A Portuguese
# country that takes the name by the button (99_nd_por.txt) reads the other
# text.
nd_por.1 = {
	type = country_event
	title = nd_por.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { original_tag = POR } }
				desc = nd_por.1.d.restored
			}
			triggered_desc = {
				desc = nd_por.1.d
			}
		}
	}
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that held or
	# formed the tag first would retire the popup for every later former.
	# Country variables die with the country, so a Portugal that falls and
	# rises again is asked again.
	trigger = {
		NOT = { has_variable = nd_por_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_por_e1_done }
	}

	# The crusade across the strait: the party of conquest in Morocco.
	option = {
		name = nd_por.1.a
		add_country_modifier = {
			modifier = nd_por_alem_mar_em_africa
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_por_formed
			set_variable = nd_por_path_a
		}
	}
	# The Ocean Sea: the party of ships, pilots and partnerships.
	option = {
		name = nd_por.1.b
		add_country_modifier = {
			modifier = nd_por_mar_oceano
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_por_formed
			set_variable = nd_por_path_b
		}
	}
}

# The advance events below name Portugal's own tag only. The orchestrator's
# sync_history_events.py adds the tags of the larger formables a country can
# go on to (Spain and beyond), which keep these advances through
# has_or_had_tag = POR. The has_advance trigger keeps the events from every
# other holder of those tags.

# Advance event: the Cortes of Coimbra, 1385. The fork is whom the new
# dynasty leans on: the towns and lawyers who argued it onto the throne, or
# the captains who kept it there.
nd_por.2 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = POR
		tag = ADU
		tag = EUR
		tag = GOT
		tag = KHL
		tag = ROM
		tag = SPA
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_por_cortes_de_coimbra
	}

	title = nd_por.2.t
	desc = nd_por.2.d
	outcome = neutral

	option = {
		name = nd_por.2.a
		add_country_modifier = {
			modifier = nd_por_casa_dos_vinte_e_quatro
			years = 15
			mode = add_and_extend
		}
	}
	option = {
		name = nd_por.2.b
		add_country_modifier = {
			modifier = nd_por_nova_nobreza
			years = 15
			mode = add_and_extend
		}
	}
}
