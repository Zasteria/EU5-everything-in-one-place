# ==========================================================================
# DANUBIAN MONARCHY: the Habsburg dilemma made playable.
#
# The form_effect (in formable_countries/99_nd_dnm.txt) has ALREADY:
#   - set nd_dnm_origin_hab / _hun / _boh in hidden_effect
#   - welded any fully integrated union crown, opened the Compact of the
#     Crowns (opinion), stamped the formed marker, grace and chain timers
#   - added stability_severe_bonus
#   - fired nd_dnm.52, THE REICH QUESTION, whose options chain into
#     nd_dnm.1 (or, with no Reich on the map, fired nd_dnm.1 directly)
#
# THE REICH QUESTION (2026-08-24): the HRE exit is no longer a formation
# cost, it is a choice. nd_dnm.52 asks it; nd_dnm_stay_in_reich_effect
# and nd_dnm_break_with_reich_effect (scripted_effects/nd_dnm_effects.txt)
# answer it; nd_dnm.53 is the Diet's ratification of the Sanction inside
# the Reich, nd_dnm.54 the consequences of a departure by a vanilla road,
# nd_dnm.55 the Empire's dissolution around a Monarchy that stayed.
#
# This file owns the narrative, the founding path choice, the Reich
# question and the heritage DHE events.
#
# ONCE-PER-COUNTRY, NOT ONCE-PER-CAMPAIGN (2026-07-31, crisis-lifecycle
# audit): event fire_only_once means once per CAMPAIGN (events/readme.txt:
# 62), which turned a re-formed DNM into a content shell: every crisis and
# destiny beat consumed by the first incarnation was gone for the second.
# Every formerly fire_only_once CRISIS event now carries a per-country
# guard instead: trigger reads NOT has_variable = nd_dnm_e<N>_done, and
# an immediate hidden_effect sets it. Country variables die with the
# country, so a re-formed DNM gets the full arc again while the current
# incarnation still sees each beat exactly once.
#
# ENGINE EXCEPTION: dynamic_historical_event events REQUIRE
# fire_only_once (the engine logs "tagged as an dynamic historical
# event, but is not set as fire_only_once" without it; error.log run 10,
# 2026-07-31). The 14 DHE flavor beats (.2-.12, .48-.50) therefore stay
# once-per-campaign by engine design: a re-formed DNM keeps its crises,
# welding, Reich arc, and destiny MECHANICS, and loses only these
# acknowledgment popups. Converting them to pulse-fired events would
# recover per-country semantics at the cost of their event-tracker
# surfacing, which is deliberately not paid.
# ==========================================================================

# ===== FORMATION EVENT =====
# The Reich question has just been answered (nd_dnm.52), or there was no
# Reich to answer. Whatever the realm's standing toward the Empire, the
# question here is what kind of state the Danube monarchy will be: the
# sword of Christendom on the Ottoman line, or a single obedient
# Gesamtmonarchie. The description reads the answer already given.
nd_dnm.1 = {
	type = country_event
	trigger = {
		NOT = { has_variable = nd_dnm_e1_done }
	}
	title = nd_dnm.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { nd_dnm_is_in_reich = yes }
				desc = nd_dnm.1.d.reich
			}
			triggered_desc = {
				trigger = { NOT = { exists = international_organization:hre } }
				desc = nd_dnm.1.d.no_reich
			}
			triggered_desc = { desc = nd_dnm.1.d }
		}
	}
	outcome = neutral

	# Path A : Antemurale Christianitatis: the bulwark of Christendom.
	# The Militaergrenze, the Grenzer soldier-colonists, the perpetual
	# Ottoman war. The Danube line is the empire's reason to exist.
	immediate = {
		hidden_effect = { set_variable = nd_dnm_e1_done }
	}

	option = {
		name = nd_dnm.1.a
		add_country_modifier = {
			modifier = nd_dnm_antemurale_christianitatis
			years = 45
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_dnm_formed
			set_variable = nd_dnm_path_a
			# Grace: 10-year window before disasters can fire (was 25).
			set_variable = { name = nd_dnm_grace value = yes years = 10 }
			# Succession Question chain: 3 narrative events that fire at
			# fixed intervals via timed-variable expiry. Each `_pending` var
			# decays at the year shown; a monthly_country_pulse handler
			# (99_nd_on_actions.txt:376) then fires the matching event and
			# sets a `_fired` guard. NOT a yearly pulse: `on_country_
			# yearly_pulse` is not an engine on_action, and an unknown
			# on_action fires nothing and logs nothing.
			set_variable = { name = nd_dnm_chain_30_pending value = yes years = 2 }
			set_variable = { name = nd_dnm_chain_31_pending value = yes years = 5 }
			set_variable = { name = nd_dnm_chain_32_pending value = yes years = 8 }
		}
	}
	# Path B : Gesamtmonarchie: the total monarchy. The Hofkammer, the
	# Hofkriegsrat, the Pragmatic Sanction: three crowns welded into one
	# administrative body answerable to the Vienna court alone.
	option = {
		name = nd_dnm.1.b
		add_country_modifier = {
			modifier = nd_dnm_gesamtmonarchie
			years = 45
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_dnm_formed
			set_variable = nd_dnm_path_b
			set_variable = { name = nd_dnm_grace value = yes years = 10 }
			set_variable = { name = nd_dnm_chain_30_pending value = yes years = 2 }
			set_variable = { name = nd_dnm_chain_31_pending value = yes years = 5 }
			set_variable = { name = nd_dnm_chain_32_pending value = yes years = 8 }
		}
	}
}

# ===== THE REICH QUESTION (2026-08-24) =====
# Remain within the Holy Roman Empire, or break with it. Fired by
# form_effect right after formation whenever the Empire exists; each
# option chains into nd_dnm.1. The stay option is the historical one
# (`historical_option` for the highlight, and an ai_chance that reads
# has_game_rule = historical_ai_choices the way vanilla's own generic
# actions do, generic_actions/general_religion.txt:261, so the AI's
# behaviour under the rule does not depend on how the engine treats the
# flag on a non-DHE event). A Hungarian origin is not a member and may
# only stay by entering: the option is offered only where the Empire's
# own join rule admits it (nd_dnm_can_stay_in_reich), and a crown of the
# wrong faith is simply told the Reich is not theirs.
nd_dnm.52 = {
	type = country_event
	trigger = {
		NOT = { has_variable = nd_dnm_e52_done }
	}
	title = nd_dnm.52.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					is_member_of_international_organization = international_organization:hre
					is_emperor = yes
				}
				desc = nd_dnm.52.d.emperor
			}
			triggered_desc = {
				trigger = { is_member_of_international_organization = international_organization:hre }
				desc = nd_dnm.52.d.member
			}
			triggered_desc = {
				trigger = { nd_dnm_can_stay_in_reich = yes }
				desc = nd_dnm.52.d.outsider
			}
			triggered_desc = { desc = nd_dnm.52.d.barred }
		}
	}
	outcome = neutral
	immediate = {
		hidden_effect = { set_variable = nd_dnm_e52_done }
	}
	# Remain within the Reich.
	option = {
		name = nd_dnm.52.a
		historical_option = yes
		trigger = { nd_dnm_can_stay_in_reich = yes }
		ai_chance = {
			factor = 60
			modifier = {
				factor = 10
				is_emperor = yes
			}
			modifier = {
				factor = 2
				international_organization:hre ?= {
					country_has_special_status = {
						type = special_status:elector
						country = root
					}
				}
			}
			modifier = {
				factor = 0.5
				NOT = { is_member_of_international_organization = international_organization:hre }
			}
			modifier = {
				factor = 1000
				has_game_rule = historical_ai_choices
			}
		}
		custom_tooltip = nd_dnm.52.a_tt
		nd_dnm_stay_in_reich_effect = yes
		trigger_event_non_silently = nd_dnm.1
	}
	# Break with the Reich.
	option = {
		name = nd_dnm.52.b
		ai_chance = {
			factor = 40
			modifier = {
				factor = 0.2
				is_emperor = yes
			}
			modifier = {
				factor = 3
				NOT = { hre_country_fulfills_imperial_religion_requirement = { type = voting } }
			}
			modifier = {
				factor = 0
				has_game_rule = historical_ai_choices
				nd_dnm_can_stay_in_reich = yes
			}
		}
		custom_tooltip = nd_dnm.52.b_tt
		nd_dnm_break_with_reich_effect = yes
		trigger_event_non_silently = nd_dnm.1
	}
}

# The Diet ratifies the Pragmatic Sanction. Fired by the
# nd_dnm_seek_diet_ratification interaction on acceptance (or on an
# Emperor-Monarchy's own petition to the electors): the crowns that still
# answer to us are welded, and their land stays imperial, which is the
# whole point of asking.
nd_dnm.53 = {
	type = country_event
	title = nd_dnm.53.t
	desc = nd_dnm.53.d
	outcome = positive
	option = {
		name = nd_dnm.53.a
		add_prestige = prestige_mild_bonus
		nd_dnm_weld_crowns_effect = yes
		hidden_effect = {
			remove_variable = nd_dnm_sanction_awaits_diet
			remove_variable = nd_dnm_ps_upheld
		}
	}
}

# A departure by a road the mod does not own. The monthly pulse finds a
# Monarchy that chose the Reich and is no longer a member (the vanilla
# leave button, a peace term, the Swiss policy) and fires this; the
# consequences are the break option's, applied in immediate so they do
# not wait on the popup.
nd_dnm.54 = {
	type = country_event
	title = nd_dnm.54.t
	desc = nd_dnm.54.d
	outcome = negative
	immediate = {
		nd_dnm_break_with_reich_effect = yes
	}
	option = {
		name = nd_dnm.54.a
		custom_tooltip = nd_dnm.54.a_tt
	}
}

# The Empire is no more. A Monarchy that stayed is simply sovereign now:
# empire rank, the Hungarian grievance moot, no Reich left to react. State
# in immediate, for the same reason as nd_dnm.54.
nd_dnm.55 = {
	type = country_event
	title = nd_dnm.55.t
	desc = nd_dnm.55.d
	outcome = neutral
	immediate = {
		hidden_effect = {
			remove_variable = nd_dnm_in_reich
			if = {
				limit = {
					country_exists = c:HUN
					c:HUN = { has_country_modifier = nd_dnm_crown_of_st_stephen }
				}
				c:HUN = { remove_country_modifier = nd_dnm_crown_of_st_stephen }
			}
		}
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
	}
	option = {
		name = nd_dnm.55.a
		custom_tooltip = nd_dnm.55.a_tt
	}
}
