namespace = nd_tun

# ============================================================
# Formation event. Fires from TUN_f's form_effect, and is also delivered to
# a country that already holds the tag (the Hafsids at the 1337 start).
# So it has two descriptions: one for a ruler who took Tunis and did not
# inherit it, and one for the house that already sits there. The first is
# marked by nd_tun_newcomer, which TUN_f's form_effect sets only when the
# ruler is not of the Hafsid dynasty (formable_countries/99_nd_tun.txt).
# The second is the default. It is read by the Hafsids at the start, who
# never run the form effect, by a Hafsid governor whom vanilla's
# flavor_tun.16 turns into TUN when his ruler inherits the capital (that
# event calls form_country, so the form effect does run for him), and by a
# Hafsid of Bejaia or Gafsa who forms Tunis by conquest.
# The choice is the one every ruler of Tunis faced until the Husaynids: who
# holds the interior. Vanilla's own Tunis content (the Emirism event, the
# Ifriqiya Reunited decision) reads centralization, and each option moves it:
# the tribes' road pushes away from that decision, the customs road toward it.
# ============================================================
nd_tun.1 = {
	type = country_event
	title = nd_tun.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = nd_tun_newcomer }
				desc = nd_tun.1.d_newcomer
			}
			triggered_desc = {
				desc = nd_tun.1.d
			}
		}
	}
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), and the Hafsids, who hold
	# the tag from day one, would retire the popup for every later former.
	trigger = {
		NOT = { has_variable = nd_tun_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_tun_e1_done }
	}

	# Path A: the sultan of the tribes.
	option = {
		name = nd_tun.1.a
		add_country_modifier = {
			modifier = nd_tun_sultan_of_the_tribes
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tun_formed
			set_variable = nd_tun_path_a
		}
	}
	# Path B: the master of the customs house. Revoking the grants costs
	# stability at once, as it cost Abu al-Abbas a decade of revolts.
	option = {
		name = nd_tun.1.b
		add_country_modifier = {
			modifier = nd_tun_master_of_the_customs_house
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		add_stability = stability_mild_penalty
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tun_formed
			set_variable = nd_tun_path_b
		}
	}
}

# ===== DESTINY EVENTS =====
# One DHE per destiny advance. The root and the two stage 1 events are
# acknowledgements. The stage 2 and stage 3 events offer two readings of the
# same milestone, each a timed modifier (15 years at stage 2, 20 at stage 3).
# fire_only_once is kept because the engine requires it on a
# dynamic_historical_event (PROMPT_TEMPLATE.md, "Once Means Once Per
# Country"): it is once per campaign, so a second country that takes the tag
# after the Hafsids have passed a stage keeps the advance and its unlock and
# misses only the popup and its timed modifier.

# Root: the destiny opens.
nd_tun.8 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = TUN
		tag = ALM
		tag = CTH
		tag = EUR
		tag = KHL
		tag = KOA
		tag = OUT
		tag = ROM
		tag = SPA
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_tun_destiny
	}

	title = nd_tun.8.t
	desc = nd_tun.8.d
	outcome = neutral

	option = {
		name = nd_tun.8.a
	}
}
