namespace = nd_mll

# Formation event. It reaches a country by two roads: from form_effect
# (99_nd_mll.txt) for a Catalan country that took the islands and formed the
# kingdom by the button, and from the four-yearly marker repair for the
# Mallorca that is on the map from the first day and never pressed it. The
# text speaks of a kingdom whose government is styled "the City and
# Kingdom" and asks which of the two shall weigh more, which is true on
# both roads; it names no former, no overlord and no conquest.
nd_mll.1 = {
	type = country_event
	title = nd_mll.1.t
	desc = nd_mll.1.d
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that held the tag
	# first would retire the popup for every later holder. Country variables
	# die with the country, so a Mallorca that falls and rises again is asked
	# again.
	trigger = {
		NOT = { has_variable = nd_mll_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_mll_e1_done }
	}

	# The City: the merchants and jurats of Ciutat de Mallorca govern the
	# kingdom from inside the walls.
	option = {
		name = nd_mll.1.a
		add_country_modifier = {
			modifier = nd_mll_ciutat_e_regne
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mll_formed
			set_variable = nd_mll_path_a
		}
	}
	# The Part Forana: the villages are given their share of the council
	# and of the accounts.
	option = {
		name = nd_mll.1.b
		add_country_modifier = {
			modifier = nd_mll_part_forana
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = individualism_vs_communalism
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mll_formed
			set_variable = nd_mll_path_b
		}
	}
}

# The advance events below name the kingdom's own tag only. Only a tag named
# in the block is polled (events/readme.txt, dynamic_historical_event);
# scripts/sync_history_events.py adds the tags of the larger formables a
# Mallorca can go on to. The has_advance trigger is what keeps the events
# from every other holder of those tags.

# or do what the jurats did in 1425 and stop paying them.
