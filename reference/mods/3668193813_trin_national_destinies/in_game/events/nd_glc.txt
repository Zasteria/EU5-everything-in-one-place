namespace = nd_glc

# Formation event: fires from form_effect (99_nd_glc.txt) for a country of
# Galician culture that formed the kingdom by the button, and from the
# four-yearly marker repair for a Galicia that was released or broke away
# and so never pressed it (the start setup lists GLC without land). The text
# speaks of the kingdom having a king and a government of its own again,
# which is true on both roads, and names no former.
nd_glc.1 = {
	type = country_event
	title = nd_glc.1.t
	desc = nd_glc.1.d
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that held the tag
	# first would retire the popup for every later holder. Country variables
	# die with the country, so a Galicia that falls and rises again is asked
	# again.
	trigger = {
		NOT = { has_variable = nd_glc_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_glc_e1_done }
	}

	# The patrimony of the Apostle: the seat stays at Compostela.
	option = {
		name = nd_glc.1.a
		add_country_modifier = {
			modifier = nd_glc_patrimonio_do_apostolo
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_glc_formed
			set_variable = nd_glc_path_a
		}
	}
	# The king's town: the court moves to A Coruna, if the kingdom holds it.
	option = {
		name = nd_glc.1.b
		add_country_modifier = {
			modifier = nd_glc_cidade_do_rei
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_minor_move_to_right
		}
		if = {
			limit = {
				owns = location:coruna
				NOT = { capital = location:coruna }
			}
			set_capital = location:coruna
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_glc_formed
			set_variable = nd_glc_path_b
		}
	}
}

# The advance events below name the kingdom's own tag only. Only a tag named
# in the block is polled (events/readme.txt, dynamic_historical_event);
# scripts/sync_history_events.py adds the tags of the larger formables a
# Galicia can go on to. The has_advance trigger is what keeps the events
# from every other holder of those tags.

# Advance event: John of Gaunt at Ourense, 1386. The fork is the one the
# duke himself faced: press the claim with England and Portugal, or sell it.
nd_glc.2 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = GLC
		tag = ADU
		tag = EUR
		tag = GOT
		tag = KHL
		tag = ROM
		tag = SPA
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_glc_rei_de_alen_mar
	}

	title = nd_glc.2.t
	desc = nd_glc.2.d
	outcome = neutral

	option = {
		name = nd_glc.2.a
		add_country_modifier = {
			modifier = nd_glc_homenaxe_a_lancaster
			years = 15
			mode = add_and_extend
		}
	}
	option = {
		name = nd_glc.2.b
		add_country_modifier = {
			modifier = nd_glc_ouro_de_baiona
			years = 15
			mode = add_and_extend
		}
	}
}
