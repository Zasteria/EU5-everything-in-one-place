namespace = nd_cpt

# Formation event. Fired from form_effect via trigger_event_non_silently, so
# no dynamic_historical_event block.
#
# Per-country guard rather than fire_only_once: fire_only_once is per CAMPAIGN
# (events/readme.txt:62), so an AI that formed and lost the tag would burn the
# event for every later claimant. Country variables die with the country, so a
# re-formed CPT re-arms correctly.
nd_cpt.1 = {
	type = country_event
	title = nd_cpt.1.t
	desc = nd_cpt.1.d
	outcome = neutral

	trigger = {
		NOT = { has_variable = nd_cpt_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_cpt_e1_done }
	}

	option = {
		name = nd_cpt.1.a
		add_country_modifier = {
			modifier = nd_cpt_ledger_state
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cpt_formed
			set_variable = nd_cpt_path_a
		}
	}
	option = {
		name = nd_cpt.1.b
		add_country_modifier = {
			modifier = nd_cpt_patriarchs_land
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cpt_formed
			set_variable = nd_cpt_path_b
		}
	}
	option = {
		name = nd_cpt.1.c
		add_country_modifier = {
			modifier = nd_cpt_marsh_oath
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = offensive_vs_defensive
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cpt_formed
			set_variable = nd_cpt_path_c
		}
	}
}
