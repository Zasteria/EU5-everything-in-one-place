namespace = nd_koa

# Formation rewards run 30 years: vanilla 1.4 raised the formable from level 2 to
# level 3, and a level 3 formation lasts 25 to 35 years (PROMPT_TEMPLATE.md,
# "Timed Modifier Budget"). They ran 20 years until 2026-10-01.
nd_koa.1 = {
	type = country_event
	title = nd_koa.1.t
	desc = nd_koa.1.d
	outcome = neutral
	# Per-country guard, not fire_only_once: that is per CAMPAIGN, so a
	# game-start or AI-held tag burned the event before any later
	# former could ever see it (readme.txt:62). The guard is its own
	# variable, not nd_<tag>_formed, which the game-start bootstrap sets
	# directly so destiny visibility never depends on a delivered popup.
	trigger = {
		NOT = { has_variable = nd_koa_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_koa_e1_done }
	}

	option = {
		name = nd_koa.1.a
		add_country_modifier = {
			modifier = nd_sword_of_the_cross
			years = 30
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_koa_formed
			set_variable = nd_koa_path_a
		}
	}
	option = {
		name = nd_koa.1.b
		add_country_modifier = {
			modifier = nd_african_emporium
			years = 30
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_koa_formed
			set_variable = nd_koa_path_b
		}
	}
}
