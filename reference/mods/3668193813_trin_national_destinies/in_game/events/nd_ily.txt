##############################################################################
# Formation Event
##############################################################################

nd_ily.1 = {
	type = country_event
	title = nd_ily.1.t
	desc = nd_ily.1.d
	outcome = neutral
	# Per-country guard rather than fire_only_once, which is per CAMPAIGN and
	# would burn the event on the first tag to reach it, leaving any later
	# former with a content shell. The guard has its own variable so that
	# destiny visibility never depends on a popup being delivered.
	trigger = {
		NOT = { has_variable = nd_ily_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_ily_e1_done }
	}

	option = {
		name = nd_ily.1.a
		add_country_modifier = {
			modifier = nd_ily_confinia_regni
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = offensive_vs_defensive
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_ily_formed
			set_variable = nd_ily_path_a
		}
	}
	option = {
		name = nd_ily.1.b
		add_country_modifier = {
			modifier = nd_ily_sacrum_illyricum
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_ily_formed
			set_variable = nd_ily_path_b
		}
	}
}
