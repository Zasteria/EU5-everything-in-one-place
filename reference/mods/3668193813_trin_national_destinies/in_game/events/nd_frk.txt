namespace = nd_frk

# === Formation Event: Francia Restored ===
# The realm that the Treaty of Verdun split in 843 is whole again, over
# subjects who speak French, German, Dutch, Frisian and Lombard. The two
# options are the two legal principles the Carolingian realm lived under at
# once: capitularies that bound every subject, and the leges nationum that
# left each people its own law. Neither is army versus economy.
nd_frk.1 = {
	type = country_event
	title = nd_frk.1.t
	# A republic can form Francia (the potential has no government gate), so
	# the scene must not assume a crowned ruler.
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { government_type = government_type:republic }
				desc = nd_frk.1.d_republic
			}
			triggered_desc = {
				desc = nd_frk.1.d
			}
		}
	}
	outcome = positive
	# Per-country guard, not fire_only_once: that is per CAMPAIGN
	# (events/readme.txt:62), so an earlier former would burn the popup for
	# every later one. form_effect already sets nd_formed and nd_frk_formed,
	# so the content never depends on this popup being answered.
	trigger = {
		NOT = { has_variable = nd_frk_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_frk_e1_done }
	}

	# Path A: One law over all.
	option = {
		name = nd_frk.1.a
		add_country_modifier = {
			modifier = nd_frk_one_law_over_all
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_frk_formed
			set_variable = nd_frk_path_a
		}
	}

	# Path B: Each people under its own law.
	option = {
		name = nd_frk.1.b
		add_country_modifier = {
			modifier = nd_frk_law_of_the_peoples
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_frk_formed
			set_variable = nd_frk_path_b
		}
	}
}

# === DHE: The Shrine at Aachen ===
# From 1349 the relics of Aachen were shown to pilgrims every seventh year,
# and the crowds rivalled Rome's. The same church was the coronation church
# of the German kings from 936 to 1531. A shrine can serve the crowd or the
# crown, and the choice sets what Charlemagne's tomb means to the realm.

	type = country_event

# === DHE: Karl or Charles ===
# In 1501 Jakob Wimpfeling's Germania argued that Charlemagne was a German
# and that the left bank of the Rhine had never been French. In 1502 Thomas
# Murner answered from the same city, Strasbourg, that he was French. For
# every other realm this was a quarrel between neighbours. For Francia it is
