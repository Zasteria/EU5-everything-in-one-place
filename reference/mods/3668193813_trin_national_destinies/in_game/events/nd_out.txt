namespace = nd_out

# === Formation Event: The Crown of Outremer ===
# A coronation that inherits a problem. OUT_f can only be proclaimed while
# Catholics are at most half the realm's population, so the emperor crowned
# at the Sepulchre rules a majority he may not convert. The three options
# are three answers to that one question: on what basis do we govern people
# who are not us. They are deliberately NOT army versus economy.
nd_out.1 = {
	type = country_event
	title = nd_out.1.t
	desc = nd_out.1.d
	outcome = neutral
	# Per-country guard, not fire_only_once: that is per CAMPAIGN
	# (events/readme.txt:62), so an earlier former would burn the popup for
	# everyone else in the same game. The guard is its own variable so
	# destiny visibility never depends on a delivered popup.
	trigger = {
		NOT = { has_variable = nd_out_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_out_e1_done }
	}

	# Path A: The Compact of the Courts. The emperor swears to leave every
	# community's law standing: the Haute Cour for the barons, the Cour des
	# Bourgeois for the Frankish commons, the Cour de la Fonde for the
	# market, and the qadi, the bishop and the bet din for the rest. The
	# realm is governed through jurisdictions the crown does not own.
	option = {
		name = nd_out.1.a
		add_country_modifier = {
			modifier = nd_out_compact_of_the_courts
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_out_formed
			set_variable = nd_out_path_a
		}
	}

	# Path B: Custodian of the Sanctuaries. A Catholic emperor guarantees
	# the Hajj, garrisons the Darb al-Hajj, escorts the Karimi convoys and
	# taxes the Red Sea. The Latin Church calls it apostasy; the treasury
	# calls it solvency.
	option = {
		name = nd_out.1.b
		add_country_modifier = {
			modifier = nd_out_custodian_of_the_sanctuaries
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_out_formed
			set_variable = nd_out_path_b
		}
	}

	# Path C: The New Towns. The only lawful way to change the arithmetic
	# is to import people, not to convert them. Planted burgess villages on
	# the Magna Mahumeria model: free tenure, no labour dues, settlers from
	# Provence and Lombardy on the road out of Jerusalem.
	option = {
		name = nd_out.1.c
		add_country_modifier = {
			modifier = nd_out_the_new_towns
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = serfdom_vs_free_subjects
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_out_formed
			set_variable = nd_out_path_c
		}
	}
}
