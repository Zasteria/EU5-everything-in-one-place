# ==========================================================
# Formation event: the crown of Boniface taken up again
# ==========================================================
#
# Three options, because three bodies claimed this city and the crown had
# to choose which of them to rule through: the Lombard baronage that
# Boniface's knights founded in 1204, the guilds and dockhands of the
# harbour, and the Orthodox church with the Holy Mountain behind it.
nd_the.1 = {
	type = country_event
	title = nd_the.1.t
	desc = nd_the.1.d
	outcome = neutral
	# Per-country guard rather than fire_only_once, which is per CAMPAIGN
	# and would burn the event on the first AI former (events/readme.txt).
	trigger = {
		NOT = { has_variable = nd_the_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_the_e1_done }
	}

	option = {
		name = nd_the.1.a
		add_country_modifier = {
			modifier = nd_the_lombard_crown
			years = 12
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_the_formed
			set_variable = nd_the_path_a
		}
	}
	option = {
		name = nd_the.1.b
		add_country_modifier = {
			modifier = nd_the_charter_of_the_demos
			years = 12
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_the_formed
			set_variable = nd_the_path_b
		}
	}
	option = {
		name = nd_the.1.c
		add_country_modifier = {
			modifier = nd_the_two_rites
			years = 12
			mode = add_and_extend
		}
		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_the_formed
			set_variable = nd_the_path_c
		}
	}
}
