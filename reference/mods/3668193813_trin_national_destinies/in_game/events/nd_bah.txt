namespace = nd_bah

# Advance DHE event 1: Mahmud Gawan's Reforms
nd_bah.1 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = BAH
		tag = DEC
		tag = HIN
		tag = KHL
		tag = MHR
		tag = VJN
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_bah_gawan_reforms
	}

	title = nd_bah.1.t
	desc = nd_bah.1.d
	outcome = neutral

	option = {
		name = nd_bah.1.a
		add_country_modifier = {
			modifier = nd_bah_gawan_centralization
			years = 15
			mode = add_and_extend
		}
		# No markers or path variable here any more (2026-08-14). This is
		# an age-3 advance DHE, and it was carrying the formation state
		# because BAH had no formation event to carry it. nd_bah.10 does
		# now, and form_effect stamps the markers unconditionally, so
		# leaving the path variable here would let a choice made two
		# centuries later overwrite the one made at the founding.
	}
	option = {
		name = nd_bah.1.b
		add_country_modifier = {
			modifier = nd_bah_gawan_tolerance
			years = 15
			mode = add_and_extend
		}
	}
}

# ==========================================================================
# FORMATION EVENT (2026-08-14). Numbered .10 because .1 through .5 were
# already advance DHEs when this tag had no formation event of its own.
#
# It fires on the scripted route too. The Fall of Delhi builds its Bahmani
# by script and form_effect runs either way, so an AI sultanate answers this
# as well; trigger_event_non_silently on an AI country resolves an option
# without a popup.
# ==========================================================================
nd_bah.10 = {
	type = country_event

	# Guarded per COUNTRY, not per campaign. fire_only_once would let one
	# AI Bahmani born out of the Fall of Delhi burn the formation event for
	# the whole save, leaving a player who later restores the sultanate
	# with a content shell.
	trigger = {
		NOT = { has_variable = nd_bah_e10_done }
	}

	title = nd_bah.10.t
	desc = nd_bah.10.d
	outcome = positive

	illustration_tags = {
		10 = exterior
		10 = happy
	}

	immediate = {
		hidden_effect = { set_variable = nd_bah_e10_done }
	}

	# Path A: the Deccanis. The amirs who made the sultan keep the tarafs.
	option = {
		name = nd_bah.10.a
		add_country_modifier = {
			modifier = nd_bah_taraf_amirs
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_bah_formed
			set_variable = nd_bah_path_a
		}
	}

	# Path B: the Afaqis. The court buys the talent it cannot raise.
	option = {
		name = nd_bah.10.b
		add_country_modifier = {
			modifier = nd_bah_court_of_bidar
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_bah_formed
			set_variable = nd_bah_path_b
		}
	}
}
