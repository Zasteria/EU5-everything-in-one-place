# ----------------------------------------------------------------------------
# nd_occ.1: Formation
# ----------------------------------------------------------------------------
# Fires from form_effect via trigger_event_non_silently. The fork is the Midi's
# own constitutional question, not a war-or-trade menu: a country assembled out
# of consulates, fors and costumas has to decide whether sovereignty rises from
# the sworn charters underneath it or descends from a prince who holds from
# nobody. Toulouse gave one answer for six centuries. Gaston Febus gave the
# other at Orthez in 1347.
nd_occ.1 = {
	type = country_event
	title = nd_occ.1.t
	desc = nd_occ.1.d
	outcome = neutral
	# Per-country guard, not fire_only_once: that is per CAMPAIGN, so an
	# AI-held or game-start tag would burn the event before any later former
	# could see it (events/readme.txt:62). Country variables die with the
	# country, so a re-formed Occitania re-arms correctly.
	trigger = {
		NOT = { has_variable = nd_occ_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_occ_e1_done }
	}

	option = {
		name = nd_occ.1.a
		add_country_modifier = {
			modifier = nd_occ_lo_pacte_consular
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_right
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_occ_formed
			set_variable = nd_occ_path_a
		}
	}

	option = {
		name = nd_occ.1.b
		add_country_modifier = {
			modifier = nd_occ_tenc_de_degun
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_occ_formed
			set_variable = nd_occ_path_b
		}
	}
}
