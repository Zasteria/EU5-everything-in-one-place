# Formation Event
##############################

nd_trg.1 = {
	type = country_event
	title = nd_trg.1.t
	desc = nd_trg.1.d
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that formed the
	# tag first would retire the popup for every later former.
	trigger = {
		NOT = { has_variable = nd_trg_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_trg_e1_done }
	}

	# The drum groups keep their freedom: tribesmen stay tribesmen.
	option = {
		name = nd_trg.1.a
		add_country_modifier = {
			modifier = nd_trg_drum_of_the_free_tents
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_trg_formed
			set_variable = nd_trg_path_a
		}
	}
	# The amenokal settles in the oasis towns: tribesmen become peasants.
	option = {
		name = nd_trg.1.b
		add_country_modifier = {
			modifier = nd_trg_seat_in_the_oases
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_trg_formed
			set_variable = nd_trg_path_b
		}
	}
}
