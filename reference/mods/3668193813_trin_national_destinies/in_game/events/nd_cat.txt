namespace = nd_cat

# Formation event: fires from form_effect (99_nd_cat.txt), and from the
# four-yearly marker repair for a Catalonia that was released or broke away
# and so never pressed the button.
nd_cat.1 = {
	type = country_event
	title = nd_cat.1.t
	# A country that was the Crown of Aragon reads what it has just given up;
	# every other former reads the plain account.
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_or_had_tag = ARA }
				desc = nd_cat.1.d.crown
			}
			triggered_desc = {
				desc = nd_cat.1.d
			}
		}
	}
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that formed the
	# tag first would retire the popup for every later former. Country
	# variables die with the country, so a Catalonia that falls and rises
	# again is asked again.
	trigger = {
		NOT = { has_variable = nd_cat_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_cat_e1_done }
	}

	# The prince swears first: the pactist principality.
	option = {
		name = nd_cat.1.a
		add_country_modifier = {
			modifier = nd_cat_sworn_principality
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cat_formed
			set_variable = nd_cat_path_a
		}
	}
	# The land governs itself: the Diputacio as the state, as in 1462 and 1641.
	option = {
		name = nd_cat.1.b
		add_country_modifier = {
			modifier = nd_cat_govern_de_la_terra
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cat_formed
			set_variable = nd_cat_path_b
		}
	}
}

# The advance events below name three tags. Only a tag named in the block is
# polled for the event (events/readme.txt, dynamic_historical_event), and
# Catalonia is a stepping stone: a country that has gone on to the Aragonese
# Empire (CDA) or Spain (SPA) still researches these advances, which are
# gated on has_or_had_tag = CAT, and would otherwise never see their events.
# The has_advance trigger keeps them from every other CDA or SPA.

# Advance event: the Corts of Barcelona, 1283 ("Una vegada lo any")
nd_cat.2 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = CAT
		tag = CDA
		tag = SPA
		tag = ADU
		tag = CTH
		tag = EUR
		tag = GOT
		tag = ITA
		tag = KHL
		tag = OCC
		tag = ROM
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_cat_una_vegada_lo_any
	}

	title = nd_cat.2.t
	desc = nd_cat.2.d
	outcome = neutral

	option = {
		name = nd_cat.2.a
	}
}
