namespace = nd_tib

# Formation event, fires from form_effect
nd_tib.1 = {
	type = country_event
	title = nd_tib.1.t
	desc = nd_tib.1.d
	outcome = neutral
	# Per-country guard, not fire_only_once: that is per CAMPAIGN, so a
	# game-start or AI-held tag burned the event before any later
	# former could ever see it (readme.txt:62). The guard is its own
	# variable, not nd_<tag>_formed, which the game-start bootstrap sets
	# directly so destiny visibility never depends on a delivered popup.
	trigger = {
		NOT = { has_variable = nd_tib_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_tib_e1_done }
	}

	option = {
		name = nd_tib.1.a
		add_country_modifier = {
			modifier = nd_tib_ganden_phodrang
			years = 20
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tib_formed
			set_variable = nd_tib_path_a
		}
	}
	option = {
		name = nd_tib.1.b
		add_country_modifier = {
			modifier = nd_tib_dharma_kings
			years = 20
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tib_formed
			set_variable = nd_tib_path_b
		}
	}
}
