namespace = nd_rum

# DHE NOTE: tag = TUR is correct and load-bearing. Vanilla RUM_f forms
# into tag = TUR (the Ottoman tag, reused), and dynamic_historical_event
# only registers for tags known at startup: tag = RUM logs
# "unknown 'RUM' country" and the event is dropped (verified in
# error.log, 2026-08-04). Side effect: game-start Ottomans see these
# DHEs in the upcoming-events panel before forming Rum; the panel
# cannot be trigger-gated (DHE blocks only take tag/from/to/chance).

nd_rum.1 = {
	type = country_event
	title = nd_rum.1.t
	desc = nd_rum.1.d
	outcome = neutral
	# Per-country guard, not fire_only_once: that is per CAMPAIGN, so a
	# game-start or AI-held tag burned the event before any later
	# former could ever see it (readme.txt:62). The guard is its own
	# variable, not nd_<tag>_formed, which the game-start bootstrap sets
	# directly so destiny visibility never depends on a delivered popup.
	trigger = {
		NOT = { has_variable = nd_rum_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_rum_e1_done }
	}

	option = {
		name = nd_rum.1.a
		add_country_modifier = {
			modifier = nd_sultan_of_two_seas
			years = 40
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_rum_formed
			set_variable = nd_rum_path_a
		}
	}
	option = {
		name = nd_rum.1.b
		add_country_modifier = {
			modifier = nd_crossroads_of_civilizations
			years = 40
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_rum_formed
			set_variable = nd_rum_path_b
		}
	}
}
