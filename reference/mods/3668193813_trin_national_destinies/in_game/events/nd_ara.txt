namespace = nd_ara

# Formation event. It reaches two kinds of country:
#   the Aragon that carries the tag from day one, or was released later: it
#     never runs form_effect, and the four-yearly marker repair in
#     99_nd_on_actions.txt delivers this event to it;
#   a Catalan or Aragonese country that forms Aragon by the button
#     (99_nd_ara.txt).
# The first reads the account of the old Crown taking stock, the second the
# account of a new house being crowned. original_tag = ARA is true for the
# first kind only.
nd_ara.1 = {
	type = country_event
	title = nd_ara.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { original_tag = ARA }
				desc = nd_ara.1.d.crown
			}
			triggered_desc = {
				desc = nd_ara.1.d
			}
		}
	}
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt:62), so an AI that formed the
	# tag first would retire the popup for every later former. Country
	# variables die with the country, so an Aragon that falls and rises
	# again is asked again.
	trigger = {
		NOT = { has_variable = nd_ara_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_ara_e1_done }
	}

	# The kingdom in its own land: the king at Zaragoza, ruling with the
	# four arms and looking to the peninsula.
	option = {
		name = nd_ara.1.a
		add_country_modifier = {
			modifier = nd_ara_reino_en_su_tierra
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_ara_formed
			set_variable = nd_ara_path_a
		}
	}
	# The head of the Crown: Aragon as the title and seat of a house that
	# rules many realms across the sea.
	option = {
		name = nd_ara.1.b
		add_country_modifier = {
			modifier = nd_ara_cabeza_de_la_corona
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_ara_formed
			set_variable = nd_ara_path_b
		}
	}
}
