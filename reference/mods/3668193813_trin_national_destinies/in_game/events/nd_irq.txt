# === Formation Event: The City of Peace ===
# Baghdad was founded in 762 under the name Madinat al-Salam, the City of
# Peace, and for two centuries it was both the seat of the caliphate and the
# largest library in the world. The two things came apart long before the
# Mongols arrived. The question this formation asks is which of them the new
# state is trying to rebuild, because it cannot afford both at once.
nd_irq.1 = {
	type = country_event
	title = nd_irq.1.t
	desc = nd_irq.1.d
	outcome = neutral
	# Per-country guard rather than fire_only_once, which is per CAMPAIGN
	# (events/readme.txt:62) and would burn the popup on the first former.
	# The guard has its own variable so destiny visibility never depends on
	# the popup having been delivered.
	trigger = {
		NOT = { has_variable = nd_irq_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_irq_e1_done }
	}

	# Path A: Dar al-Khilafa. Baghdad as the political capital of Islam.
	option = {
		name = nd_irq.1.a
		add_country_modifier = {
			modifier = nd_irq_dar_al_khilafa
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_irq_formed
			set_variable = nd_irq_path_a
		}
	}

	# Path B: Madinat al-Salam. Baghdad as the civilizational capital.
	option = {
		name = nd_irq.1.b
		add_country_modifier = {
			modifier = nd_irq_madinat_al_salam
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_irq_formed
			set_variable = nd_irq_path_b
		}
	}
}
