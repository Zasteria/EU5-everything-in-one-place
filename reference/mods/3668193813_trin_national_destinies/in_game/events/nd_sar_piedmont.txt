namespace = nd_sar_piedmont

# Formation event, fires from form_effect.
nd_sar_piedmont.1 = {
	type = country_event
	title = nd_sar_piedmont.1.t
	desc = {
		first_valid = {
			# Montferrat kept the Palaiologos in Italy for two centuries. If
			# this crown is being raised by that house, say so.
			triggered_desc = {
				trigger = { has_or_had_tag = MFA }
				desc = nd_sar_piedmont.1.d.mfa
			}
			triggered_desc = {
				desc = nd_sar_piedmont.1.d
			}
		}
	}
	outcome = neutral
	# Per-country guard rather than fire_only_once, which is per campaign and
	# would burn the event on the first AI former.
	trigger = {
		NOT = { has_variable = nd_sar_piedmont_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_sar_piedmont_e1_done }
	}

	option = {
		name = nd_sar_piedmont.1.a
		add_country_modifier = {
			modifier = nd_sarp_wardens_of_the_passes
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = offensive_vs_defensive
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_sar_piedmont_formed
			set_variable = nd_sar_piedmont_path_a
		}
	}
	option = {
		name = nd_sar_piedmont.1.b
		add_country_modifier = {
			modifier = nd_sarp_two_codes_one_crown
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_sar_piedmont_formed
			set_variable = nd_sar_piedmont_path_b
		}
	}
	option = {
		name = nd_sar_piedmont.1.c
		# Only for a realm that carries the Montferrat inheritance. A country
		# that took the Greek road cannot reach this event at all, because the
		# formable requires an Italian culture, and that is deliberate.
		trigger = {
			has_or_had_tag = MFA
		}
		add_country_modifier = {
			modifier = nd_sarp_palaiologos_inheritance
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_sar_piedmont_formed
			set_variable = nd_sar_piedmont_path_c
		}
	}
}
