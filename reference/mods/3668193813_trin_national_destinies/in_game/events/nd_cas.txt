namespace = nd_cas

# Formation event. Castile carries its tag from day one and never runs
# form_effect, so for the kingdom of 1337 this event arrives through the
# four-yearly marker repair (99_nd_on_actions.txt); a Castilian country that
# restores the kingdom gets it from form_effect (99_nd_cas.txt). The two read
# different opening texts and face the same choice.
nd_cas.1 = {
	type = country_event
	title = nd_cas.1.t
	desc = {
		first_valid = {
			# original_tag: "a country is a specific historical tag or was
			# it". True for the Castile of the start, false for a Toledo or a
			# Sevilla that has just taken the name.
			triggered_desc = {
				trigger = { original_tag = CAS }
				desc = nd_cas.1.d
			}
			triggered_desc = {
				desc = nd_cas.1.d.restored
			}
		}
	}
	outcome = neutral
	# Once per COUNTRY, by its own guard variable. Not fire_only_once: that
	# is once per campaign (events/readme.txt), so the first Castile to
	# answer would retire the popup for every later one. Country variables
	# die with the country, so a Castile that falls and rises again is asked
	# again.
	trigger = {
		NOT = { has_variable = nd_cas_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_cas_e1_done }
	}

	# The king's justice: judges, accountants and the councils of the towns.
	option = {
		name = nd_cas.1.a
		add_country_modifier = {
			modifier = nd_cas_justicia_del_rey
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cas_formed
			set_variable = nd_cas_path_a
		}
	}
	# The host of the ricoshombres: the great lineages, paid in rents and
	# towns.
	option = {
		name = nd_cas.1.b
		add_country_modifier = {
			modifier = nd_cas_hueste_de_los_ricoshombres
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cas_formed
			set_variable = nd_cas_path_b
		}
	}
}

# The advance events below name the country's own tag only. The
# orchestrator's scripts/sync_history_events.py adds the tags of the larger
# formables a Castile can go on to; the has_advance trigger keeps the events
# from every other holder of those tags.

# Advance event: the alcabala, 1342. The fork is whether the tax stays a
# grant of the Cortes or becomes the king's by custom.
nd_cas.2 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = CAS
		tag = ADU
		tag = EUR
		tag = GOT
		tag = KHL
		tag = ROM
		tag = SPA
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_cas_alcabala
	}

	title = nd_cas.2.t
	desc = nd_cas.2.d
	outcome = neutral

	option = {
		name = nd_cas.2.a
		add_country_modifier = {
			modifier = nd_cas_alcabala_otorgada
			years = 15
			mode = add_and_extend
		}
	}
	option = {
		name = nd_cas.2.b
		add_country_modifier = {
			modifier = nd_cas_alcabala_perpetua
			years = 15
			mode = add_and_extend
		}
	}
}
