namespace = nd_got

# Formation Event: The Restoration of the Regnum
#
# The fork is the name, and the name is the claim. Latin continuity says the
# restored state is the juridical and ecclesiastical person of the old
# kingdom: the councils, the code, the primacy of Toledo. Gothic revivalism
# says it is a people with a tongue and a sworn king, and that the Latin
# apparatus was always borrowed clothing.
nd_got.1 = {
	type = country_event
	title = nd_got.1.t
	desc = nd_got.1.d
	outcome = neutral
	# Per-country guard rather than fire_only_once, which is per CAMPAIGN
	# (events/readme.txt:62) and would burn the event on the first former.
	trigger = {
		NOT = { has_variable = nd_got_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_got_e1_done }
	}

	option = {
		name = nd_got.1.a
		add_country_modifier = {
			modifier = nd_got_regnum_gothorum
			years = 30
			mode = add_and_extend
		}
		# One code, one court, one capital: the territorial principle of the
		# Liber Iudiciorum is a centralising doctrine before it is anything
		# else.
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_got_formed
			set_variable = nd_got_path_a
		}
	}
	option = {
		name = nd_got.1.b
		add_country_modifier = {
			modifier = nd_got_reiki_gutane
			years = 30
			mode = add_and_extend
		}
		# A crown refounded on the warband and the personal oath is a crown
		# that owes its magnates everything.
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_got_formed
			set_variable = nd_got_path_b
		}
	}
}
