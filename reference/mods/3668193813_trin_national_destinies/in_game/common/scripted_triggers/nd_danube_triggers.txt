# ==========================================================================
# THE SUCCESSION OF THE DANUBE: shared triggers (PLAN.md section 13).
# Crown identity is the three bare tags. DNM formation ends the situation
# (fulfilled), so DNM never needs crown membership here.
# ==========================================================================

nd_dnb_is_crown = {
	OR = {
		tag = HAB
		tag = HUN
		tag = BOH
	}
}

# ROOT sits in a union that has an ESTABLISHED senior partner, and it is
# not us: vanilla's own junior_union_with test (country_triggers.txt:929),
# without the partner argument.
#
# Seniority is a union LAW, not a fact of every union (Workshop reports
# 2026-09-15, minddrag and Marcell56). Every union opens under
# union_maintain_federal_status_policy (international_organizations/
# union.txt, laws block), the senior_partner status can only be bestowed
# once the union has voted union_establish_seniority_policy
# (international_organization_special_statuses/union.txt:8), that vote is
# locked for the union's first five years (laws/40_personal_unions.txt:9)
# and the AI keeps federal status at +50 ("not anyone's puppet",
# 40_personal_unions.txt:156). The old test asked country_is_senior_partner
# of every union member, which is false for BOTH members of a federal
# union: a player Austria that inherited any union lost every Danube
# action, read as a captive crown on the map and in the alerts, and the
# Contest lever opened against it. Vanilla's own idiom for "free to act"
# is NOT has_active_senior_partner OR is the senior
# (generic_actions/create_province_subject.txt:14-21); this follows it.
nd_dnb_union_junior = {
	exists = union
	union ?= {
		has_active_senior_partner = yes
		NOT = {
			country_has_special_status = {
				type = special_status:senior_partner
				country = prev
			}
		}
	}
}

# A crown that runs its own candidacy: alive, master of itself. A union
# SENIOR is independent (its juniors are the fold), a member of a union
# with no established senior is independent (nobody's junior), a junior
# under an established senior is not. `prev`, never `root`: this trigger
# runs inside every_country blocks of the situation, where root is the
# situation (readme: select_trigger visible/enabled bind root to the
# candidate, so root would have been fine there, but not here).
nd_dnb_crown_independent = {
	nd_dnb_is_crown = yes
	is_subject = no
	NOT = { nd_dnb_union_junior = yes }
}

# This crown's candidacy is merged into c:$WHO$ (subject of it, or union
# junior under its seniority): the two-of-three state.
nd_dnb_merged_under = {
	OR = {
		is_subject_of = c:$WHO$
		junior_union_with = { target = c:$WHO$ }
	}
}

# Under any fellow crown (candidacy merged, contest narrows).
nd_dnb_crown_merged = {
	OR = {
		AND = {
			country_exists = c:HAB
			NOT = { tag = HAB }
			nd_dnb_merged_under = { WHO = HAB }
		}
		AND = {
			country_exists = c:HUN
			NOT = { tag = HUN }
			nd_dnb_merged_under = { WHO = HUN }
		}
		AND = {
			country_exists = c:BOH
			NOT = { tag = BOH }
			nd_dnb_merged_under = { WHO = BOH }
		}
	}
}

# Under an OUTSIDE power (the Jagiellon case): subject or union junior of
# someone who is not a crown. The candidacy freezes and the Contest the
# Foreign Crown lever opens for the others.
nd_dnb_crown_foreign_backed = {
	nd_dnb_is_crown = yes
	NOT = { nd_dnb_crown_merged = yes }
	OR = {
		is_subject = yes
		nd_dnb_union_junior = yes
	}
}

# c:$WHO$ holds every other living crown: the fulfilled end. The self-case
# skips via the limit (c:HAB = { tag = $WHO$ } is true exactly when $WHO$
# is HAB, making NOT false and the trigger_if pass vacuously). A dead
# crown does not block.
nd_dnb_holds_all_crowns_p = {
	country_exists = c:$WHO$
	c:$WHO$ = { nd_dnb_crown_independent = yes }
	trigger_if = {
		limit = {
			country_exists = c:HAB
			NOT = { c:HAB = { tag = $WHO$ } }
		}
		c:HAB = { nd_dnb_merged_under = { WHO = $WHO$ } }
	}
	trigger_if = {
		limit = {
			country_exists = c:HUN
			NOT = { c:HUN = { tag = $WHO$ } }
		}
		c:HUN = { nd_dnb_merged_under = { WHO = $WHO$ } }
	}
	trigger_if = {
		limit = {
			country_exists = c:BOH
			NOT = { c:BOH = { tag = $WHO$ } }
		}
		c:BOH = { nd_dnb_merged_under = { WHO = $WHO$ } }
	}
}

# c:$WHO$ holds at least one OTHER LIVING crown: the union is a real
# union, not a survivorship. holds_all_crowns_p above passes VACUOUSLY for
# a lone surviving crown (every trigger_if limit goes false when the other
# crowns are dead), which is the correct reading for the fulfilled coda but
# the wrong one for the proclamation added 2026-08-07: a Hungary that
# merely outlived Austria and Bohemia, or watched Poland annex them, has
# not been elected anything and must not be turned into the Danubian
# Monarchy by a situation closing. Same idiom as above, self-case skipped
# the same way.
nd_dnb_holds_a_living_crown_p = {
	OR = {
		AND = {
			country_exists = c:HAB
			NOT = { c:HAB = { tag = $WHO$ } }
			c:HAB = { nd_dnb_merged_under = { WHO = $WHO$ } }
		}
		AND = {
			country_exists = c:HUN
			NOT = { c:HUN = { tag = $WHO$ } }
			c:HUN = { nd_dnb_merged_under = { WHO = $WHO$ } }
		}
		AND = {
			country_exists = c:BOH
			NOT = { c:BOH = { tag = $WHO$ } }
			c:BOH = { nd_dnb_merged_under = { WHO = $WHO$ } }
		}
	}
}

# Any crown holds all the others (fulfilled, whoever won it).
nd_dnb_question_answered = {
	OR = {
		any_country = { tag = DNM }
		nd_dnb_holds_all_crowns_p = { WHO = HAB }
		nd_dnb_holds_all_crowns_p = { WHO = HUN }
		nd_dnb_holds_all_crowns_p = { WHO = BOH }
	}
}

# The succession checkpoint's gate: a throne that cannot defend its own
# continuity. Every branch occurs in every campaign regardless of AI skill
# (user ruling 1, PLAN section 13).
nd_dnb_weak_succession = {
	OR = {
		has_regent = yes
		has_heir = no
		legitimacy < 40
		ruler ?= { is_adult = no }
	}
}

# c:$WHO$ still runs its own house: alive and not under an outside
# master. A crown merged under a fellow crown counts (mid-contest).
nd_dnb_crown_free_p = {
	country_exists = c:$WHO$
	c:$WHO$ = { NOT = { nd_dnb_crown_foreign_backed = yes } }
}

# The question can open: at least two crowns still run their own realms
# free of outside masters, and no one has answered it yet.
#
# Spelled as explicit pairs, NOT calc_true_if: the 1413-observer bug
# (2026-08-02). Repeated BLOCK-valued entries (AND x3) under one key
# evidently collapse inside calc_true_if, so the count never reached 2
# and this trigger was permanently false: no spark, including the 1437
# fallback, could ever fire, with a perfectly clean error.log (repeated
# SCALAR entries like `owns` x4 do count, per the 2026-07-28 Pintados
# Workshop repro; vanilla itself only ever feeds calc_true_if iterators
# or distinct keys). OR-of-ANDs is the everywhere-proven idiom.
nd_dnb_can_spark = {
	NOT = { any_country = { tag = DNM } }
	OR = {
		AND = {
			nd_dnb_crown_free_p = { WHO = HAB }
			nd_dnb_crown_free_p = { WHO = HUN }
		}
		AND = {
			nd_dnb_crown_free_p = { WHO = HAB }
			nd_dnb_crown_free_p = { WHO = BOH }
		}
		AND = {
			nd_dnb_crown_free_p = { WHO = HUN }
			nd_dnb_crown_free_p = { WHO = BOH }
		}
	}
	NOT = { nd_dnb_question_answered = yes }
}

# Every crown still running its own house has declared for the union
# (SITUATION scope). When this holds at the Congress, the acclamation
# becomes an ELECTION (user design 2026-08-03, the 1526 Bohemian
# model): the crowns are summoned to enter the compact of their own
# will, and the claim is drawn only against a crown that refuses. A
# dead or merged crown does not block unanimity.
nd_dnb_unanimous_union = {
	trigger_if = {
		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnb_crown_independent = yes }
		}
		var:nd_dnb_posture_hab = 1
	}
	trigger_if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { nd_dnb_crown_independent = yes }
		}
		var:nd_dnb_posture_hun = 1
	}
	trigger_if = {
		limit = {
			country_exists = c:BOH
			c:BOH = { nd_dnb_crown_independent = yes }
		}
		var:nd_dnb_posture_boh = 1
	}
}

# ROOT = the receiving crown, scope:actor = the proposing crown: this
# pair's pact tier is below 3. Before the situation starts every pair is
# implicitly tier 0, so the check passes.
nd_dnb_pair_tier_lt3_with_actor = {
	OR = {
		AND = {
			OR = {
				AND = { tag = HAB scope:actor = { tag = HUN } }
				AND = { tag = HUN scope:actor = { tag = HAB } }
			}
			OR = {
				NOT = { has_global_variable = nd_dnb_active }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_hun < 3 }
			}
		}
		AND = {
			OR = {
				AND = { tag = HAB scope:actor = { tag = BOH } }
				AND = { tag = BOH scope:actor = { tag = HAB } }
			}
			OR = {
				NOT = { has_global_variable = nd_dnb_active }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_boh < 3 }
			}
		}
		AND = {
			OR = {
				AND = { tag = HUN scope:actor = { tag = BOH } }
				AND = { tag = BOH scope:actor = { tag = HUN } }
			}
			OR = {
				NOT = { has_global_variable = nd_dnb_active }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hun_boh < 3 }
			}
		}
	}
}

# The pair pact tier between two explicit country scopes is at least
# $N$. $X$ and $Y$ are scope references (root, scope:actor,
# scope:recipient), passed textually; `root = { tag = HAB }` is valid
# scope-block form (vanilla ask_join_war_for_favors.txt:51). Written
# with explicit scopes because select_trigger enabled evaluates with
# root = candidate while ai_will_do evaluates with actor/recipient set.
nd_dnb_pact_tier_ge_p = {
	OR = {
		AND = {
			OR = {
				AND = {
					$X$ = { tag = HAB }
					$Y$ = { tag = HUN }
				}
				AND = {
					$X$ = { tag = HUN }
					$Y$ = { tag = HAB }
				}
			}
			situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_hun >= $N$ }
		}
		AND = {
			OR = {
				AND = {
					$X$ = { tag = HAB }
					$Y$ = { tag = BOH }
				}
				AND = {
					$X$ = { tag = BOH }
					$Y$ = { tag = HAB }
				}
			}
			situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_boh >= $N$ }
		}
		AND = {
			OR = {
				AND = {
					$X$ = { tag = HUN }
					$Y$ = { tag = BOH }
				}
				AND = {
					$X$ = { tag = BOH }
					$Y$ = { tag = HUN }
				}
			}
			situation:nd_danube_succession ?= { var:nd_dnb_tier_hun_boh >= $N$ }
		}
	}
}

# v2 HOUSE TRIGGERS (2026-08-10 respine, PLAN.md 13/v2). All house tests
# are LIVE pairwise dynasty comparisons: no dynasty scope is ever stored,
# so a save carries no stale house state to repair.

# Two named crowns currently share one ruling house. Regents do not count:
# a Habsburg regent in Buda is influence, not a Habsburg king.
nd_dnb_shared_house_p = {
	country_exists = c:$X$
	country_exists = c:$Y$
	c:$X$ ?= {
		has_ruler = yes
		ruler ?= { has_dynasty = yes }
	}
	c:$Y$ ?= {
		has_ruler = yes
		ruler ?= {
			has_dynasty = yes
			dynasty ?= { this = c:$X$.ruler.dynasty }
		}
	}
}

# Any pair of crowns under one house: the Ferdinand condition.
nd_dnb_house_on_two_thrones = {
	OR = {
		nd_dnb_shared_house_p = { X = HAB Y = HUN }
		nd_dnb_shared_house_p = { X = HAB Y = BOH }
		nd_dnb_shared_house_p = { X = HUN Y = BOH }
	}
}

# THIS is a crown whose king belongs to the ruling house of an OUTSIDE
# monarchy (the Jagiellon case): dynastic capture without political
# subjection, which nd_dnb_crown_foreign_backed cannot see. The size
# floor keeps a shared minor comital house from reading as capture.
# The crown is pinned with save_temporary_scope_as (vanilla
# religion_triggers.txt:72 idiom) because root is NOT the crown at two
# of the call sites: the situation's on_ended (root = situation, so
# root.ruler.dynasty resolved to nothing and the Foreign Crown coda
# never fired) and the Contest interaction's ai_will_do (root = actor,
# so the AI tested its own house instead of the target's).
nd_dnb_under_outside_house = {
	save_temporary_scope_as = nd_dnb_house_crown
	has_ruler = yes
	ruler ?= { has_dynasty = yes }
	any_country = {
		NOT = { nd_dnb_is_crown = yes }
		NOT = { this = scope:nd_dnb_house_crown }
		government_type = government_type:monarchy
		num_locations >= 10
		has_ruler = yes
		ruler ?= {
			has_dynasty = yes
			dynasty ?= { this = scope:nd_dnb_house_crown.ruler.dynasty }
		}
	}
}
