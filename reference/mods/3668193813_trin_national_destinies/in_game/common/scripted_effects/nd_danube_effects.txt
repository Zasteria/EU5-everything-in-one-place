# ==========================================================================
# THE SUCCESSION OF THE DANUBE: effects (PLAN.md section 13).
#
# The standing score follows the Italian Wars lesson: every source is a
# currency the AI already pursues on its own (strength, marriages, pacts,
# occupation), so the AI competes correctly without knowing the minigame
# exists. Weights are documented for the player in nd_dnb_standing_explain_tt
# (the iw_tension_explain_tt standard); change them in BOTH places.
# ==========================================================================

# Recompute one crown's standing. SITUATION scope. Parameters:
# TAG/O1/O2 = tags (self, other, other), tag/o1/o2 = lowercase,
# P1 = canonical pair suffix with O1, P2 = with O2 (hab_hun/hab_boh/hun_boh).
nd_dnb_recompute_standing_one = {
	if = {
		limit = {
			country_exists = c:$TAG$
			c:$TAG$ = { NOT = { nd_dnb_crown_merged = yes } }
		}
		set_variable = { name = nd_dnb_standing_$tag$ value = 20 }
		# Strength rank among living crowns (+20 first, +10 second).
		if = {
			limit = {
				c:$TAG$ = {
					trigger_if = {
						limit = { country_exists = c:$O1$ }
						military_strength >= c:$O1$.military_strength
					}
					trigger_if = {
						limit = { country_exists = c:$O2$ }
						military_strength >= c:$O2$.military_strength
					}
				}
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 20 }
		}
		else_if = {
			limit = {
				c:$TAG$ = {
					OR = {
						AND = {
							country_exists = c:$O1$
							military_strength >= c:$O1$.military_strength
						}
						AND = {
							country_exists = c:$O2$
							military_strength >= c:$O2$.military_strength
						}
					}
				}
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 10 }
		}
		# Royal marriages with the other crowns (+5 each).
		if = {
			limit = {
				country_exists = c:$O1$
				c:$TAG$ = { has_royal_marriage_with = c:$O1$ }
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 5 }
		}
		if = {
			limit = {
				country_exists = c:$O2$
				c:$TAG$ = { has_royal_marriage_with = c:$O2$ }
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 5 }
		}
		# Shared ruling dynasty (+10 each): the Habsburg method itself.
		if = {
			limit = {
				country_exists = c:$O1$
				c:$O1$ = {
					has_ruler = yes
					ruler ?= { has_dynasty = yes }
				}
				c:$TAG$ = {
					has_ruler = yes
					ruler ?= {
						has_dynasty = yes
						dynasty = c:$O1$.ruler.dynasty
					}
				}
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 10 }
		}
		if = {
			limit = {
				country_exists = c:$O2$
				c:$O2$ = {
					has_ruler = yes
					ruler ?= { has_dynasty = yes }
				}
				c:$TAG$ = {
					has_ruler = yes
					ruler ?= {
						has_dynasty = yes
						dynasty = c:$O2$.ruler.dynasty
					}
				}
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 10 }
		}
		# Pact tiers (+5 per tier per pair; vars initialized 0 in on_start).
		change_variable = {
			name = nd_dnb_standing_$tag$
			add = { value = var:nd_dnb_tier_$P1$ multiply = 5 }
		}
		change_variable = {
			name = nd_dnb_standing_$tag$
			add = { value = var:nd_dnb_tier_$P2$ multiply = 5 }
		}
		# Courted diets (+5 while the timed courtship holds).
		if = {
			limit = { has_variable = nd_dnb_courted_$tag$_$o1$ }
			change_variable = { name = nd_dnb_standing_$tag$ add = 5 }
		}
		if = {
			limit = { has_variable = nd_dnb_courted_$tag$_$o2$ }
			change_variable = { name = nd_dnb_standing_$tag$ add = 5 }
		}
		# Double betrothal seeds (+10 per active pair).
		if = {
			limit = { has_variable = nd_dnb_betrothal_$P1$ }
			change_variable = { name = nd_dnb_standing_$tag$ add = 10 }
		}
		if = {
			limit = { has_variable = nd_dnb_betrothal_$P2$ }
			change_variable = { name = nd_dnb_standing_$tag$ add = 10 }
		}
		# War position: a rival crown's capital in hand (+10 each), own
		# capital under a rival crown (-20). The Corvinus teeth: taking
		# Vienna is the strongest anti-union play there is.
		if = {
			limit = {
				country_exists = c:$O1$
				c:$O1$ = { capital ?= { controller = c:$TAG$ } }
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 10 }
		}
		if = {
			limit = {
				country_exists = c:$O2$
				c:$O2$ = { capital ?= { controller = c:$TAG$ } }
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 10 }
		}
		if = {
			limit = {
				c:$TAG$ = {
					capital ?= {
						OR = {
							AND = {
								country_exists = c:$O1$
								controller = c:$O1$
							}
							AND = {
								country_exists = c:$O2$
								controller = c:$O2$
							}
						}
					}
				}
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = -20 }
		}
		# Crowns already in the fold (+15 each).
		if = {
			limit = {
				country_exists = c:$O1$
				c:$O1$ = { nd_dnb_merged_under = { WHO = $TAG$ } }
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 15 }
		}
		if = {
			limit = {
				country_exists = c:$O2$
				c:$O2$ = { nd_dnb_merged_under = { WHO = $TAG$ } }
			}
			change_variable = { name = nd_dnb_standing_$tag$ add = 15 }
		}
		# A foreign-backed crown's candidacy freezes low (-25).
		if = {
			limit = { c:$TAG$ = { nd_dnb_crown_foreign_backed = yes } }
			change_variable = { name = nd_dnb_standing_$tag$ add = -25 }
		}
		# Clamp 0..100.
		if = {
			limit = { var:nd_dnb_standing_$tag$ > 100 }
			set_variable = { name = nd_dnb_standing_$tag$ value = 100 }
		}
		if = {
			limit = { var:nd_dnb_standing_$tag$ < 0 }
			set_variable = { name = nd_dnb_standing_$tag$ value = 0 }
		}
	}
	else = {
		# Dead or merged: the candidacy is folded, not competing.
		set_variable = { name = nd_dnb_standing_$tag$ value = 0 }
	}
}

# All three crowns. SITUATION scope.
nd_dnb_recompute_standings = {
	nd_dnb_recompute_standing_one = {
		TAG = HAB tag = hab O1 = HUN o1 = hun O2 = BOH o2 = boh
		P1 = hab_hun P2 = hab_boh
	}
	nd_dnb_recompute_standing_one = {
		TAG = HUN tag = hun O1 = HAB o1 = hab O2 = BOH o2 = boh
		P1 = hab_hun P2 = hun_boh
	}
	nd_dnb_recompute_standing_one = {
		TAG = BOH tag = boh O1 = HAB o1 = hab O2 = HUN o2 = hun
		P1 = hab_boh P2 = hun_boh
	}
}

# Menace at the Gates, 0-3. SITUATION scope. AI-agnostic by construction
# (user ruling 1): reads relative strength of whoever is actually big and
# adjacent, never a specific tag's performance. A trigger_if whose crown
# is dead passes vacuously, which is the correct reading ("stronger than
# every LIVING crown").
nd_dnb_recompute_menace = {
	set_variable = { name = nd_dnb_menace value = 0 }
	if = {
		limit = {
			any_country = {
				NOT = { nd_dnb_is_crown = yes }
				NOT = { tag = DNM }
				OR = {
					AND = {
						country_exists = c:HAB
						is_neighbor_of = c:HAB
					}
					AND = {
						country_exists = c:HUN
						is_neighbor_of = c:HUN
					}
					AND = {
						country_exists = c:BOH
						is_neighbor_of = c:BOH
					}
				}
				OR = {
					AND = {
						country_exists = c:HAB
						military_strength > c:HAB.military_strength
					}
					AND = {
						country_exists = c:HUN
						military_strength > c:HUN.military_strength
					}
					AND = {
						country_exists = c:BOH
						military_strength > c:BOH.military_strength
					}
				}
			}
		}
		set_variable = { name = nd_dnb_menace value = 1 }
	}
	if = {
		limit = {
			any_country = {
				NOT = { nd_dnb_is_crown = yes }
				NOT = { tag = DNM }
				OR = {
					AND = {
						country_exists = c:HAB
						is_neighbor_of = c:HAB
					}
					AND = {
						country_exists = c:HUN
						is_neighbor_of = c:HUN
					}
					AND = {
						country_exists = c:BOH
						is_neighbor_of = c:BOH
					}
				}
				trigger_if = {
					limit = { country_exists = c:HAB }
					military_strength > c:HAB.military_strength
				}
				trigger_if = {
					limit = { country_exists = c:HUN }
					military_strength > c:HUN.military_strength
				}
				trigger_if = {
					limit = { country_exists = c:BOH }
					military_strength > c:BOH.military_strength
				}
			}
		}
		set_variable = { name = nd_dnb_menace value = 2 }
	}
	if = {
		limit = {
			any_country = {
				NOT = { nd_dnb_is_crown = yes }
				NOT = { tag = DNM }
				trigger_if = {
					limit = { country_exists = c:HAB }
					military_strength > c:HAB.military_strength
				}
				trigger_if = {
					limit = { country_exists = c:HUN }
					military_strength > c:HUN.military_strength
				}
				trigger_if = {
					limit = { country_exists = c:BOH }
					military_strength > c:BOH.military_strength
				}
				OR = {
					AND = {
						country_exists = c:HAB
						is_at_war_with = c:HAB
					}
					AND = {
						country_exists = c:HUN
						is_at_war_with = c:HUN
					}
					AND = {
						country_exists = c:BOH
						is_at_war_with = c:BOH
					}
				}
			}
		}
		set_variable = { name = nd_dnb_menace value = 3 }
	}
}

# The Diet of the Three Crowns. SITUATION scope. The Diet is the
# stance-setting beat and nothing more: each crown declares which CAUSE
# it backs (the vote-streak mechanic died 2026-08-02 with the user's
# ruling that this is a causes-with-backers system, not a vote; the
# convene-spam cascade that instant-ended the 1444 game is impossible
# with no streak to race). The global convened-recently floor caps diet
# frequency at one per 5 years no matter who convenes.
nd_dnb_hold_diet = {
	change_variable = { name = nd_dnb_diets_held add = 1 }
	set_variable = { name = nd_dnb_diet_cooldown value = yes years = 10 }
	set_variable = { name = nd_dnb_convened_recently value = yes years = 5 }

	# The tally (2026-08-03 rework): the Diet weighs the DECLARED backers
	# by standing and hands the leading cause a surge. Convening early
	# while your side is ahead is the whole point of the button; the Diet
	# asks nobody anything (stances are the on-demand actions).
	set_variable = { name = nd_dnb_tally_union value = 0 }
	set_variable = { name = nd_dnb_tally_liberty value = 0 }
	if = {
		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnb_crown_independent = yes }
			var:nd_dnb_posture_hab = 1
		}
		change_variable = {
			name = nd_dnb_tally_union
			add = { value = var:nd_dnb_standing_hab }
		}
	}
	else_if = {
		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnb_crown_independent = yes }
			var:nd_dnb_posture_hab = 2
		}
		change_variable = {
			name = nd_dnb_tally_liberty
			add = { value = var:nd_dnb_standing_hab }
		}
	}
	if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { nd_dnb_crown_independent = yes }
			var:nd_dnb_posture_hun = 1
		}
		change_variable = {
			name = nd_dnb_tally_union
			add = { value = var:nd_dnb_standing_hun }
		}
	}
	else_if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { nd_dnb_crown_independent = yes }
			var:nd_dnb_posture_hun = 2
		}
		change_variable = {
			name = nd_dnb_tally_liberty
			add = { value = var:nd_dnb_standing_hun }
		}
	}
	if = {
		limit = {
			country_exists = c:BOH
			c:BOH = { nd_dnb_crown_independent = yes }
			var:nd_dnb_posture_boh = 1
		}
		change_variable = {
			name = nd_dnb_tally_union
			add = { value = var:nd_dnb_standing_boh }
		}
	}
	else_if = {
		limit = {
			country_exists = c:BOH
			c:BOH = { nd_dnb_crown_independent = yes }
			var:nd_dnb_posture_boh = 2
		}
		change_variable = {
			name = nd_dnb_tally_liberty
			add = { value = var:nd_dnb_standing_boh }
		}
	}
	if = {
		limit = { var:nd_dnb_tally_union > var:nd_dnb_tally_liberty }
		set_variable = { name = nd_dnb_diet_lean value = 1 }
		change_variable = { name = nd_dnb_cause_union add = 10 }
	}
	else_if = {
		limit = { var:nd_dnb_tally_liberty > var:nd_dnb_tally_union }
		set_variable = { name = nd_dnb_diet_lean value = 2 }
		change_variable = { name = nd_dnb_cause_liberty add = 10 }
	}
	else = {
		set_variable = { name = nd_dnb_diet_lean value = 0 }
	}

	if = {
		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnb_crown_independent = yes }
		}
		c:HAB = { trigger_event_non_silently = nd_danube.10 }
	}
	if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { nd_dnb_crown_independent = yes }
		}
		c:HUN = { trigger_event_non_silently = nd_danube.10 }
	}
	if = {
		limit = {
			country_exists = c:BOH
			c:BOH = { nd_dnb_crown_independent = yes }
		}
		c:BOH = { trigger_event_non_silently = nd_danube.10 }
	}
}

# The succession checkpoint. COUNTRY scope (root = the crown whose ruler
# just changed; chained from on_new_ruler). Every rival claimant crown
# with any pact standing on this pair is summoned to nd_danube.12/.13/.14
# (one static event per weak crown, so no scope plumbing survives event
# delivery). This is the deterministic lever the whole system exists for:
# every succession in a pact partner is a checkpoint, never a lottery.
nd_dnb_succession_checkpoint = {
	if = {
		limit = {
			has_global_variable = nd_dnb_active
			nd_dnb_is_crown = yes
			nd_dnb_weak_succession = yes
			NOT = { nd_dnb_crown_merged = yes }
		}
		# Weak Austria: Hungary and Bohemia may press.
		if = {
			limit = {
				tag = HAB
				country_exists = c:HUN
				c:HUN = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_hun >= 1 }
			}
			c:HUN = { trigger_event_non_silently = nd_danube.12 }
		}
		if = {
			limit = {
				tag = HAB
				country_exists = c:BOH
				c:BOH = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_boh >= 1 }
			}
			c:BOH = { trigger_event_non_silently = nd_danube.12 }
		}
		# Weak Hungary: Austria and Bohemia may press.
		if = {
			limit = {
				tag = HUN
				country_exists = c:HAB
				c:HAB = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_hun >= 1 }
			}
			c:HAB = { trigger_event_non_silently = nd_danube.13 }
		}
		if = {
			limit = {
				tag = HUN
				country_exists = c:BOH
				c:BOH = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hun_boh >= 1 }
			}
			c:BOH = { trigger_event_non_silently = nd_danube.13 }
		}
		# Weak Bohemia: Austria and Hungary may press.
		if = {
			limit = {
				tag = BOH
				country_exists = c:HAB
				c:HAB = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hab_boh >= 1 }
			}
			c:HAB = { trigger_event_non_silently = nd_danube.14 }
		}
		if = {
			limit = {
				tag = BOH
				country_exists = c:HUN
				c:HUN = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= { var:nd_dnb_tier_hun_boh >= 1 }
			}
			c:HUN = { trigger_event_non_silently = nd_danube.14 }
		}
	}

	# v2 GRIEVANCE WATCH (PLAN.md 13/v2). Same root: the crown whose ruler
	# just changed. A sealed double betrothal promised this succession to
	# the partner's house; if the new king is NOT of that house, the
	# partner was passed over and gets the grievance beat (.42/.43/.44,
	# one static event per subject crown). Ordinary father-to-son
	# successions in untied crowns trigger nothing: only a broken promise
	# does. Pair cooldown 10 years so one turbulent court cannot spam it.
	if = {
		limit = {
			has_global_variable = nd_dnb_active
			nd_dnb_is_crown = yes
			NOT = { nd_dnb_crown_merged = yes }
		}
		# Austria crowned someone outside a promised house.
		if = {
			limit = {
				tag = HAB
				country_exists = c:HUN
				c:HUN = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= {
					has_variable = nd_dnb_betrothal_hab_hun
					NOT = { has_variable = nd_dnb_grievance_cd_hab_hun }
				}
				NOT = { nd_dnb_shared_house_p = { X = HAB Y = HUN } }
			}
			situation:nd_danube_succession ?= {
				set_variable = { name = nd_dnb_grievance_cd_hab_hun value = 1 years = 10 }
			}
			c:HUN = { trigger_event_non_silently = nd_danube.42 }
		}
		if = {
			limit = {
				tag = HAB
				country_exists = c:BOH
				c:BOH = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= {
					has_variable = nd_dnb_betrothal_hab_boh
					NOT = { has_variable = nd_dnb_grievance_cd_hab_boh }
				}
				NOT = { nd_dnb_shared_house_p = { X = HAB Y = BOH } }
			}
			situation:nd_danube_succession ?= {
				set_variable = { name = nd_dnb_grievance_cd_hab_boh value = 1 years = 10 }
			}
			c:BOH = { trigger_event_non_silently = nd_danube.42 }
		}
		# Hungary crowned someone outside a promised house.
		if = {
			limit = {
				tag = HUN
				country_exists = c:HAB
				c:HAB = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= {
					has_variable = nd_dnb_betrothal_hab_hun
					NOT = { has_variable = nd_dnb_grievance_cd_hab_hun }
				}
				NOT = { nd_dnb_shared_house_p = { X = HAB Y = HUN } }
			}
			situation:nd_danube_succession ?= {
				set_variable = { name = nd_dnb_grievance_cd_hab_hun value = 1 years = 10 }
			}
			c:HAB = { trigger_event_non_silently = nd_danube.43 }
		}
		if = {
			limit = {
				tag = HUN
				country_exists = c:BOH
				c:BOH = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= {
					has_variable = nd_dnb_betrothal_hun_boh
					NOT = { has_variable = nd_dnb_grievance_cd_hun_boh }
				}
				NOT = { nd_dnb_shared_house_p = { X = HUN Y = BOH } }
			}
			situation:nd_danube_succession ?= {
				set_variable = { name = nd_dnb_grievance_cd_hun_boh value = 1 years = 10 }
			}
			c:BOH = { trigger_event_non_silently = nd_danube.43 }
		}
		# Bohemia crowned someone outside a promised house.
		if = {
			limit = {
				tag = BOH
				country_exists = c:HAB
				c:HAB = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= {
					has_variable = nd_dnb_betrothal_hab_boh
					NOT = { has_variable = nd_dnb_grievance_cd_hab_boh }
				}
				NOT = { nd_dnb_shared_house_p = { X = HAB Y = BOH } }
			}
			situation:nd_danube_succession ?= {
				set_variable = { name = nd_dnb_grievance_cd_hab_boh value = 1 years = 10 }
			}
			c:HAB = { trigger_event_non_silently = nd_danube.44 }
		}
		if = {
			limit = {
				tag = BOH
				country_exists = c:HUN
				c:HUN = { nd_dnb_crown_independent = yes }
				situation:nd_danube_succession ?= {
					has_variable = nd_dnb_betrothal_hun_boh
					NOT = { has_variable = nd_dnb_grievance_cd_hun_boh }
				}
				NOT = { nd_dnb_shared_house_p = { X = HUN Y = BOH } }

# The spark watch. COUNTRY scope (any living crown's monthly pulse).
# Earliest 1360 (the 1364 pact anchor; the early game breathes). The
# pact interaction itself is the player's match and sparks directly in
# its own effect; these are the world's own matches.
#
# SELF-HEALING (2026-08-02): gates on live state (situation active /
# concluded), never on a flag set before activation is confirmed. The
# old design stamped nd_dnb_started first, so any activation hiccup
# bricked the save's spark forever. If activation ever fails now, the
# watch simply retries next month. nd_dnb_started remains as an
# informational stamp set in on_start only.
#
# LOBBY RULE (2026-08-25): nd_danube_succession_rule gates this watcher
# and the pact interaction, the only two activation sites. Off means the
# question never opens, and with it the situation's forced proclamation
# (nd_dnb_proclaim_or_offer), the road that made AI Monarchies without a
# marriage. The AI Forms Nations pulse (99_nd_on_actions.txt) is the other
# AI road, governed by its own rule.
nd_dnb_spark_watch = {
	if = {
		limit = {
			nd_danube_succession_is_enabled = yes
			NOT = { has_global_variable = nd_dnb_active }
			NOT = { has_global_variable = nd_dnb_concluded }
			# The Hussite question is settled first (2026-08-29). The Danube
			# succession of 1437 followed the Basel Compacts of 1436, so the

			# still being fought. This only DEFERS the spark: the vanilla
			# Hussite Wars run to their own conclusion (they never subjugate
			# or annex a crown, so nothing here stops them), after which a
			# surviving Hussite Bohemia can still lead or win the Congress,
			# or leave to found the Tabor Republic. The second spark site
			# (nd_dnb_propose_the_pact) carries the same guard.
			NOT = { situation:hussite_wars ?= { situation_is_active = yes } }
			nd_dnb_is_crown = yes
			current_year >= 1360
			nd_dnb_can_spark = yes
			OR = {
				# Two crowns under one ruling house.
				AND = {
					country_exists = c:HAB
					country_exists = c:HUN

						has_ruler = yes

						has_ruler = yes

					}

						has_ruler = yes
						ruler ?= {
							has_dynasty = yes
							dynasty = c:HAB.ruler.dynasty
						}
					}
				}
				AND = {
					country_exists = c:HUN
					country_exists = c:BOH
					c:HUN = {
						has_ruler = yes
						ruler ?= { has_dynasty = yes }
					}
					c:BOH = {
						has_ruler = yes
						ruler ?= {
							has_dynasty = yes
							dynasty = c:HUN.ruler.dynasty
						}
					}
				}
				# A crown wobbles while another is married into it.
				AND = {
					country_exists = c:HAB
					c:HAB = { nd_dnb_weak_succession = yes }
					OR = {
						AND = {
							country_exists = c:HUN
							c:HUN = { has_royal_marriage_with = c:HAB }
						}
						AND = {
							country_exists = c:BOH
							c:BOH = { has_royal_marriage_with = c:HAB }
						}
					}
				}
				AND = {
					country_exists = c:HUN
					c:HUN = { nd_dnb_weak_succession = yes }
					OR = {
						AND = {
							country_exists = c:HAB
							c:HAB = { has_royal_marriage_with = c:HUN }
						}
						AND = {
							country_exists = c:BOH
							c:BOH = { has_royal_marriage_with = c:HUN }
						}
					}
				}
				AND = {
					country_exists = c:BOH

						AND = {
							country_exists = c:HAB

							country_exists = c:HUN

			}
		}
		activate_situation = situation:nd_danube_succession
	}

# ==========================================================================
# THE CAUSES, v2 (PLAN.md 13/v2, 2026-08-10). SITUATION scope, monthly.
# v1 let every declared backer FEED its cause monthly (standing x 0.006),
# which made the Congress a clock: the AI always declares union, so the
# bar always filled and the arc was a wait game (the live Workshop
# finding). v2 kills the flow: this effect only DECAYS, ticks the epoch
# display clock, prices a menace RISE once per rise, and runs the quiet
# clock toward Separate Ways. Causes now move on discrete happenings
# only, through nd_dnb_bump_cause: a throne changing house (+25, in
# nd_dnb_recompute_houses), a sealed betrothal (+8), a pact tier rise
# (+4), a diet verdict (+10, in nd_dnb_hold_diet), a declared stance
# (+8, in the stance actions), murmurs (+2, event options). Untended,
# the Question dies: that is the point.
# ==========================================================================
nd_dnb_drift_causes = {
	# Monthly-flow ledger: decay baseline; event bumps land on top and
	# mirror into the deltas so the panel can show WHY a bar moved.
	# v2.1 (1444 live-play): the pact web sustains the Union Cause;
	# parchment holds while the tiers stand. Fading runs full below a
	# web of 5 total tiers, half from 5, and stops at 8, so a court
	# that has actually bound the river never watches its work bleed
	# away, while a paper-thin web still fades.
	set_variable = { name = nd_dnb_web value = var:nd_dnb_tier_hab_hun }
	change_variable = { name = nd_dnb_web add = { value = var:nd_dnb_tier_hab_boh } }
	change_variable = { name = nd_dnb_web add = { value = var:nd_dnb_tier_hun_boh } }
	set_variable = { name = nd_dnb_delta_liberty value = -0.1 }
	# The epoch display clock (exit 4's live countdown).
	change_variable = { name = nd_dnb_epoch_years_left add = -0.0834 }
	if = {
		limit = { var:nd_dnb_epoch_years_left < 0 }
		set_variable = { name = nd_dnb_epoch_years_left value = 0 }
	}
	if = {
		limit = { var:nd_dnb_web >= 8 }
		set_variable = { name = nd_dnb_delta_union value = 0 }
	}
	else_if = {
		limit = { var:nd_dnb_web >= 5 }
		set_variable = { name = nd_dnb_delta_union value = -0.05 }
		change_variable = { name = nd_dnb_cause_union add = -0.05 }
	}
	else = {
		set_variable = { name = nd_dnb_delta_union value = -0.1 }
		change_variable = { name = nd_dnb_cause_union add = -0.1 }
	}
	change_variable = { name = nd_dnb_cause_liberty add = -0.1 }
	# Menace pricing, v2: a RISE to level 2 or higher is an event and is
	# priced once per rise (the Mohacs logic: estates move when the enemy
	# reaches the gates, not every month the map stays scary). Falling
	# menace re-arms the trigger. Lazy init covers live saves.
	if = {
		limit = { NOT = { has_variable = nd_dnb_menace_seen } }
		set_variable = { name = nd_dnb_menace_seen value = 0 }
	}
	if = {
		limit = {
			var:nd_dnb_menace >= 2
			var:nd_dnb_menace_seen < 2
		}
		nd_dnb_bump_cause = { CAUSE = union AMOUNT = 10 }
		every_country = {
			limit = { nd_dnb_crown_independent = yes }
			trigger_event_non_silently = nd_danube.53
		}
	}
	set_variable = { name = nd_dnb_menace_seen value = var:nd_dnb_menace }
	# The quiet clock: both causes dead, no house spanning two thrones,
	# at least one diet held, and the situation at least 40 years old
	# (the epoch display clock started at 160). Ten such years close the
	# Question as Separate Ways: reachable disappointment is the v2
	# acceptance bar, and historically the true base case.
	if = {
		limit = { NOT = { has_variable = nd_dnb_quiet_months } }
		set_variable = { name = nd_dnb_quiet_months value = 0 }
	}
	if = {
		limit = {
			var:nd_dnb_cause_union <= 5
			var:nd_dnb_cause_liberty <= 5
			var:nd_dnb_epoch_years_left <= 120
			var:nd_dnb_diets_held >= 1
			NOT = { nd_dnb_house_on_two_thrones = yes }
		}
		change_variable = { name = nd_dnb_quiet_months add = 1 }
	}
	else = {
		set_variable = { name = nd_dnb_quiet_months value = 0 }
	}
	if = {
		limit = {
			var:nd_dnb_quiet_months >= 120
			NOT = { has_variable = nd_dnb_separate_ways }
		}
		set_variable = nd_dnb_separate_ways
	}
	if = {
		limit = { var:nd_dnb_cause_union < 0 }
		set_variable = { name = nd_dnb_cause_union value = 0 }
	}
	if = {
		limit = { var:nd_dnb_cause_union > 100 }
		set_variable = { name = nd_dnb_cause_union value = 100 }
	}
	if = {
		limit = { var:nd_dnb_cause_liberty < 0 }
		set_variable = { name = nd_dnb_cause_liberty value = 0 }
	}
	if = {
		limit = { var:nd_dnb_cause_liberty > 100 }
		set_variable = { name = nd_dnb_cause_liberty value = 100 }
	}
}

		set_variable = { name = nd_dnb_cause_$CAUSE$ value = 0 }
	}
}

# Whom does the river acclaim? v2 (PLAN.md 13/v2): the crown whose
# ruling house sits on the most Danube thrones, tiebreak by standing;
# only when no house spans two thrones does it fall back to the
# strongest free crown. Postures no longer decide anything here: v1's
# posture tiers let the only stance-declaring AI (always Bohemia) win
# every fallthrough, which is the Workshop finding this respine exists
# to fix. Stamps nd_dnb_congress_held = 1 and nd_dnb_acclaimed
# (1 Austria, 2 Hungary, 3 Bohemia, 0 nobody); no other side effects
# beyond the transient house-count scratch variables.
nd_dnb_stamp_acclaimed = {
	set_variable = { name = nd_dnb_congress_held value = 1 }
	# The house flags may be absent when the save-repair path stamps a
	# Congress an old build consumed; the recompute lazy-inits them.
	if = {
		limit = { NOT = { has_variable = nd_dnb_shared_hab_hun } }
		nd_dnb_recompute_houses = yes
	}
	# Thrones per house, counted through the pair flags (a crown always
	# holds its own throne, so only the shared flags differentiate).
	set_variable = { name = nd_dnb_hc_hab value = var:nd_dnb_shared_hab_hun }
	change_variable = { name = nd_dnb_hc_hab add = { value = var:nd_dnb_shared_hab_boh } }
	set_variable = { name = nd_dnb_hc_hun value = var:nd_dnb_shared_hab_hun }
	change_variable = { name = nd_dnb_hc_hun add = { value = var:nd_dnb_shared_hun_boh } }
	set_variable = { name = nd_dnb_hc_boh value = var:nd_dnb_shared_hab_boh }
	change_variable = { name = nd_dnb_hc_boh add = { value = var:nd_dnb_shared_hun_boh } }
	if = {
		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnb_crown_independent = yes }
			var:nd_dnb_hc_hab >= 1
			var:nd_dnb_hc_hab >= var:nd_dnb_hc_hun
			var:nd_dnb_hc_hab >= var:nd_dnb_hc_boh
			var:nd_dnb_standing_hab >= var:nd_dnb_standing_hun
			var:nd_dnb_standing_hab >= var:nd_dnb_standing_boh
		}
		set_variable = { name = nd_dnb_acclaimed value = 1 }
	}
	else_if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { nd_dnb_crown_independent = yes }

			var:nd_dnb_hc_hun >= var:nd_dnb_hc_boh
			var:nd_dnb_standing_hun >= var:nd_dnb_standing_boh
		}
		set_variable = { name = nd_dnb_acclaimed value = 2 }
	}
	else_if = {
		limit = {
			country_exists = c:BOH
			c:BOH = { nd_dnb_crown_independent = yes }
			var:nd_dnb_hc_boh >= 1
		}
		set_variable = { name = nd_dnb_acclaimed value = 3 }
	}
	# House-first without the standing tiebreak: a spanning house whose
	# crown lost the standing comparison still outranks houses on one
	# throne (the cascade above required BOTH; these three require the
	# span only, so a Corvinus Hungary with modest standing but two
	# thrones is acclaimed over a strong single-throne Austria).

		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnb_crown_independent = yes }
			var:nd_dnb_hc_hab >= 1
			var:nd_dnb_hc_hab >= var:nd_dnb_hc_hun
			var:nd_dnb_hc_hab >= var:nd_dnb_hc_boh
		}
		set_variable = { name = nd_dnb_acclaimed value = 1 }
	}
	else_if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { nd_dnb_crown_independent = yes }
			var:nd_dnb_hc_hun >= 1
			var:nd_dnb_hc_hun >= var:nd_dnb_hc_boh
		}
		set_variable = { name = nd_dnb_acclaimed value = 2 }
	}
	else_if = {
		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnb_crown_independent = yes }
			var:nd_dnb_standing_hab >= var:nd_dnb_standing_hun
			var:nd_dnb_standing_hab >= var:nd_dnb_standing_boh
		}
		set_variable = { name = nd_dnb_acclaimed value = 1 }
	}
	else_if = {

	}
	else_if = {

			c:BOH = { nd_dnb_crown_independent = yes }
		}
		set_variable = { name = nd_dnb_acclaimed value = 3 }
	}
	else = {
