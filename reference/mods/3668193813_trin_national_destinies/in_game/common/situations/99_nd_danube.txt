# ==========================================================================
# THE SUCCESSION OF THE DANUBE (PLAN.md section 13, user-approved
# 2026-08-02): the road INTO the Danubian Monarchy. A three-way dynastic
# tug-of-war among Austria, Hungary and Bohemia: standings measured in
# AI-native currencies, pacts and betrothals as player levers, every
# succession in a pact partner a checkpoint, and a diet that can vote the
# whole question closed.
#
# can_start = { always = no }: started solely by activate_situation, from
# nd_dnb_spark_watch (world sparks: shared dynasty, weak succession under
# a marriage, the 1437 fallback) or from the pact interaction's own effect
# (the player's match). Single instance per campaign: the global
# nd_dnb_started flag never clears.
#
# Both activation sites read nd_danube_succession_rule (2026-08-25): with
# the situation disabled in the lobby the question never opens, and with
# it the forced proclamation in on_ended, the road that made AI
# Monarchies without a marriage. An AI can then reach DNM only through
# the AI Forms Nations pulse (99_nd_on_actions.txt), i.e. by meeting the
# formable's own requirements, marriage unions included.
#
# CRISIS LIFECYCLE CONTRACT (all four exits proven false at start):
#   1. Fulfilled: one realm holds every surviving crown (DNM counts).
#      False at start: spark requires >= 2 free crowns and no answer yet.
#   2. Settled by the sword: at most one crown left on the map.
#      False at start: same spark requirement.
#   3. The Separate Ways: recorded marker, set only by two consecutive
#      all-independence diets (first diet is 8 years out by on_start
#      timer) or the Renounce action. Cannot pre-exist: single instance.
#   4. Adjudication fallback: epoch timer set in on_start, 160 years
#      (the real arc was 1364-1526). Fallback only, never the sole exit.
# Every exit has a named on-demand lever: form the DNM / invoke the pact
# (1), war (2), Renounce the Question (3). No random_list gates any exit.
# ==========================================================================
nd_danube_succession = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_nd_danube_succession

	can_start = {
		always = no
	}

	# Vanilla self-scopes its own can_end checks explicitly
	# (italian_wars.txt:18); country checks wrap in any_country per the
	# Reaction of the Reich lesson (situation scope has no implicit owner).
	can_end = {
		OR = {
			custom_tooltip = {
				text = nd_dnb_end_fulfilled_tt
				nd_dnb_question_answered = yes
			}
			custom_tooltip = {
				text = nd_dnb_end_sword_tt
				# "At most one crown exists" as NOR of coexisting pairs:
				# calc_true_if cannot count repeated same-key entries (the
				# 1413-observer spark bug; see nd_danube_triggers.txt).
				NOR = {
					AND = {
						country_exists = c:HAB
						country_exists = c:HUN
					}
					AND = {
						country_exists = c:HAB
						country_exists = c:BOH
					}
					AND = {
						country_exists = c:HUN
						country_exists = c:BOH
					}
				}
			}
			custom_tooltip = {
				text = nd_dnb_end_separate_tt
				situation:nd_danube_succession ?= {
					OR = {
						# The dominance renounce (a player closing the book).
						has_variable = nd_dnb_separate_ways
						# The Old Liberties cause has triumphed: earned by
						# decades of backing, never by a vote (2026-08-02
						# rework; the streak mechanic is gone).
						var:nd_dnb_cause_liberty >= 100
					}
				}
			}
			custom_tooltip = {
				text = nd_dnb_end_epoch_tt
				situation:nd_danube_succession ?= {
					# diets_held is initialized by on_start, so this clause
					# is provably false until on_start has run even if the
					# engine evaluates can_end first (Lifecycle Contract:
					# every clause false at start by construction).
					has_variable = nd_dnb_diets_held
					NOT = { has_variable = nd_dnb_epoch_timer }
				}
			}
		}
	}

	# The three crowns and their neighborhood care; the wider Catholic
	# world watches (mirrors the Reaction of the Reich's visibility gate).
	visible = {
		OR = {
			nd_dnb_is_crown = yes
			tag = DNM
			is_member_of_international_organization = international_organization:hre
			has_presence_in = continent:europe
		}
	}

	on_start = {
		set_global_variable = nd_dnb_active
		# The pact interaction can activate before the watcher does;
		# stamping the started flag here too keeps both roads idempotent.
		set_global_variable = nd_dnb_started

		# Instance scoping (Lifecycle Contract): clear every marker a
		# prior activation could have left, so no end-state or once-guard
		# leaks across instances. The design is single-instance, but the
		# contract scopes anyway: a console or future-content
		# re-activation must start clean, never instant-end on a stale
		# Separate Ways marker.
		remove_variable = nd_dnb_separate_ways
		remove_variable = nd_dnb_congress_done
		remove_variable = nd_dnb_congress_held
		remove_variable = nd_dnb_acclaimed
		remove_variable = nd_dnb_convened_recently
		remove_variable = nd_dnb_murmur_cd
		remove_variable = nd_dnb_courted_hab_hun
		remove_variable = nd_dnb_courted_hab_boh
		remove_variable = nd_dnb_courted_hun_hab
		remove_variable = nd_dnb_courted_hun_boh
		remove_variable = nd_dnb_courted_boh_hab
		remove_variable = nd_dnb_courted_boh_hun
		remove_variable = nd_dnb_betrothal_hab_hun
		remove_variable = nd_dnb_betrothal_hab_boh
		remove_variable = nd_dnb_betrothal_hun_boh
		remove_variable = nd_dnb_merged_alert_hab
		remove_variable = nd_dnb_merged_alert_hun
		remove_variable = nd_dnb_merged_alert_boh
		remove_variable = nd_dnb_foreign_alert_hab
		remove_variable = nd_dnb_foreign_alert_hun
		remove_variable = nd_dnb_foreign_alert_boh
		remove_variable = nd_dnb_extinct_alert_hab
		remove_variable = nd_dnb_extinct_alert_hun
		remove_variable = nd_dnb_extinct_alert_boh
		# v2 house-watch and grievance state (PLAN.md 13/v2).
		remove_variable = nd_dnb_house_cd_hab_hun
		remove_variable = nd_dnb_house_cd_hab_boh
		remove_variable = nd_dnb_house_cd_hun_boh
		remove_variable = nd_dnb_grievance_cd_hab_hun
		remove_variable = nd_dnb_grievance_cd_hab_boh
		remove_variable = nd_dnb_grievance_cd_hun_boh
		# Throne war stamps (nd_danube_effects.txt, THE THRONE WAR).
		remove_variable = nd_dnb_throne_war_hab_hun
		remove_variable = nd_dnb_throne_war_hab_boh
		remove_variable = nd_dnb_throne_war_hun_hab
		remove_variable = nd_dnb_throne_war_hun_boh
		remove_variable = nd_dnb_throne_war_boh_hab
		remove_variable = nd_dnb_throne_war_boh_hun
		# A crown already dead when the question opens is old news, not an
		# alarm: pre-stamp its guard so no extinction beat fires for it.
		if = {
			limit = { NOT = { country_exists = c:HAB } }
			set_variable = nd_dnb_extinct_alert_hab
		}
		if = {
			limit = { NOT = { country_exists = c:HUN } }
			set_variable = nd_dnb_extinct_alert_hun
		}
		if = {
			limit = { NOT = { country_exists = c:BOH } }
			set_variable = nd_dnb_extinct_alert_boh
		}

		# Pact tiers per pair, postures, and bookkeeping: initialized so
		# every later var: read and change_variable is well-defined.
		set_variable = { name = nd_dnb_tier_hab_hun value = 0 }
		set_variable = { name = nd_dnb_tier_hab_boh value = 0 }
		set_variable = { name = nd_dnb_tier_hun_boh value = 0 }
		set_variable = { name = nd_dnb_posture_hab value = 0 }
		set_variable = { name = nd_dnb_posture_hun value = 0 }
		set_variable = { name = nd_dnb_posture_boh value = 0 }
		set_variable = { name = nd_dnb_diets_held value = 0 }
		# The two causes (the Guelphs-model rework, 2026-08-02): backers
		# push them monthly; Old Liberties at 100 is the Separate Ways,
		# the Union Cause at 100 convenes the Congress of the Danube.
		set_variable = { name = nd_dnb_cause_union value = 0 }
		set_variable = { name = nd_dnb_cause_liberty value = 0 }
		# 0 = no Congress yet; the panel swaps the capped union bar for
		# the acclamation line when this goes 1/2/3.
		set_variable = { name = nd_dnb_acclaimed value = 0 }
		set_variable = { name = nd_dnb_delta_union value = 0 }
		set_variable = { name = nd_dnb_delta_liberty value = 0 }
		set_variable = { name = nd_dnb_epoch_years_left value = 160 }
		# v2 bookkeeping: the menace-rise latch and the quiet clock.
		set_variable = { name = nd_dnb_menace_seen value = 0 }
		set_variable = { name = nd_dnb_quiet_months value = 0 }

		# The epoch clock (exit 4) and the first Diet's summons (8 years:
		# the courts need a generation's opening move first).
		set_variable = { name = nd_dnb_epoch_timer value = yes years = 160 }
		set_variable = { name = nd_dnb_diet_cooldown value = yes years = 8 }

		nd_dnb_recompute_standings = yes
		nd_dnb_recompute_menace = yes
		nd_dnb_recompute_houses = yes

		if = {
			limit = { country_exists = c:HAB }
			c:HAB = { trigger_event_non_silently = nd_danube.1 }
		}
		if = {
			limit = { country_exists = c:HUN }
			c:HUN = { trigger_event_non_silently = nd_danube.1 }
		}
		if = {
			limit = { country_exists = c:BOH }
			c:BOH = { trigger_event_non_silently = nd_danube.1 }
		}
	}

	on_monthly = {
		# First, before anything reads who holds whom: a crown that won a
		# war for a fellow crown's throne is the senior of that union.
		nd_dnb_throne_wars_monthly = yes
		nd_dnb_recompute_standings = yes
		nd_dnb_recompute_menace = yes
		nd_dnb_recompute_houses = yes
		nd_dnb_drift_causes = yes

		# The Congress of the Danube, once. Two roads (PLAN.md 13/v2):
		# the diplomatic route, the Union Cause invested all the way to
		# 100 (no longer reachable by waiting: v2 killed the drift); or
		# the Ferdinand moment, one house already on two thrones with the
		# cause at 60 or more, the world acknowledging a fait accompli.
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_congress_done }
				OR = {
					var:nd_dnb_cause_union >= 100
					AND = {
						var:nd_dnb_cause_union >= 60
						nd_dnb_house_on_two_thrones = yes
					}
				}
			}
			nd_dnb_convene_congress = yes
		}

		# Save repair (2026-08-03, the Tabor campaign): a Congress
		# convened by the pre-announce build set only the once-guard,
		# granted its instruments through the old event, and told nobody
		# else. Derive the acclaimed crown from current state (the best
		# available reconstruction) and send the announcement that build
		# never sent. New saves stamp at convening, so this never fires
		# for them.
		if = {
			limit = {
				has_variable = nd_dnb_congress_done
				NOT = { has_variable = nd_dnb_congress_held }
			}
			nd_dnb_stamp_acclaimed = yes
			nd_dnb_congress_announce = yes
		}
		# Second half of the same repair: pre-Congress saves from the old
		# build never initialized nd_dnb_acclaimed (new saves get it in
		# on_start), and the panel's bar-or-acclamation swap reads it.
		if = {
			limit = { NOT = { has_variable = nd_dnb_acclaimed } }
			set_variable = { name = nd_dnb_acclaimed value = 0 }
		}

		# Diet cadence: when the summons timer lapses, the crowns convene
		# (the Convene action can force it early through the same effect).
		if = {
			limit = { NOT = { has_variable = nd_dnb_diet_cooldown } }
			nd_dnb_hold_diet = yes
		}

		# Two-of-three: tell the holdout the fold is closing (once per
		# merged crown). The holdout is every remaining independent crown.
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_merged_alert_hab }
				country_exists = c:HAB
				c:HAB = { nd_dnb_crown_merged = yes }
			}
			set_variable = nd_dnb_merged_alert_hab
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.25
			}
		}
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_merged_alert_hun }
				country_exists = c:HUN
				c:HUN = { nd_dnb_crown_merged = yes }
			}
			set_variable = nd_dnb_merged_alert_hun
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.25
			}
		}
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_merged_alert_boh }
				country_exists = c:BOH
				c:BOH = { nd_dnb_crown_merged = yes }
			}
			set_variable = nd_dnb_merged_alert_boh
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.25
			}
		}

		# The Jagiellon alarm: a crown has fallen under an outside power
		# (once per crown; the Contest lever opens on the others' side).
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_foreign_alert_hab }
				country_exists = c:HAB
				c:HAB = { nd_dnb_crown_foreign_backed = yes }
			}
			set_variable = nd_dnb_foreign_alert_hab
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.17
			}
		}
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_foreign_alert_hun }
				country_exists = c:HUN
				c:HUN = { nd_dnb_crown_foreign_backed = yes }
			}
			set_variable = nd_dnb_foreign_alert_hun
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.18
			}
		}
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_foreign_alert_boh }
				country_exists = c:BOH
				c:BOH = { nd_dnb_crown_foreign_backed = yes }
			}
			set_variable = nd_dnb_foreign_alert_boh
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.19
			}
		}

		# The crown extinguished (2026-08-03, the Tabor campaign): a crown
		# has ceased to exist entirely, annexed away or transformed by a
		# tag switch (Hussite Bohemia becoming the Tabor Republic). Tell
		# the remaining crowns once instead of letting the panel rows
		# vanish silently. A DNM tag switch is fulfilment, not extinction,
		# and the situation ends that same tick, so it is excluded.
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_extinct_alert_hab }
				NOT = { country_exists = c:HAB }
				NOT = { any_country = { tag = DNM } }
			}
			set_variable = nd_dnb_extinct_alert_hab
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.31
			}
		}
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_extinct_alert_hun }
				NOT = { country_exists = c:HUN }
				NOT = { any_country = { tag = DNM } }
			}
			set_variable = nd_dnb_extinct_alert_hun
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.32
			}
		}
		if = {
			limit = {
				NOT = { has_variable = nd_dnb_extinct_alert_boh }
				NOT = { country_exists = c:BOH }
				NOT = { any_country = { tag = DNM } }
			}
			set_variable = nd_dnb_extinct_alert_boh
			every_country = {
				limit = { nd_dnb_crown_independent = yes }
				trigger_event_non_silently = nd_danube.33
			}
		}

		# Court murmurs: rare flavor, cooldown-gated so it never stacks.
		if = {
			limit = { NOT = { has_variable = nd_dnb_murmur_cd } }
			random_list = {
				98 = {}
				2 = {
					set_variable = { name = nd_dnb_murmur_cd value = yes years = 4 }
					random_country = {
						limit = { nd_dnb_crown_independent = yes }
						trigger_event_non_silently = nd_danube.30
					}
				}
			}
		}
	}

	on_ended = {
		remove_global_variable = nd_dnb_active
		# Single instance per campaign: the concluded stamp is what the
		# spark watch and the pact interaction check (the started flag is
		# informational only, set in on_start).
		set_global_variable = nd_dnb_concluded

		# v2 Foreign Crown coda (PLAN.md 13/v2): if the Question closes
		# with any crown under an outside power or an outside ruling
		# house, that is its own ending in the players' eyes and gets its
		# own beat, alongside whichever adjudication branch runs below.
		# Informational only: no durable state changes here.
		if = {
			limit = {
				country_exists = c:HAB
				c:HAB = {
					OR = {
						nd_dnb_crown_foreign_backed = yes
						AND = {
							nd_dnb_crown_independent = yes
							nd_dnb_under_outside_house = yes
						}
					}
				}
			}
			c:HAB = { trigger_event_non_silently = nd_danube.45 }
		}
		if = {
			limit = {
				country_exists = c:HUN
				c:HUN = {
					OR = {
						nd_dnb_crown_foreign_backed = yes
						AND = {
							nd_dnb_crown_independent = yes
							nd_dnb_under_outside_house = yes
						}
					}
				}
			}
			c:HUN = { trigger_event_non_silently = nd_danube.45 }
		}
		if = {
			limit = {
				country_exists = c:BOH
				c:BOH = {
					OR = {
						nd_dnb_crown_foreign_backed = yes
						AND = {
							nd_dnb_crown_independent = yes
							nd_dnb_under_outside_house = yes
						}
					}
				}
			}
			c:BOH = { trigger_event_non_silently = nd_danube.45 }
		}

		# Outcome adjudication, in priority order. Codas are informational
		# plus their payoffs; all durable state changes happen HERE, not in
		# event options (the delivery-independence rule).
		if = {
			limit = { any_country = { tag = DNM } }
			random_country = {
				limit = { tag = DNM }
				trigger_event_non_silently = nd_danube.20
			}
		}
		# One crown holds every surviving crown. THE QUESTION IS ANSWERED,
		# so the answer is the Monarchy: an AI holder proclaims it here, a
		# human holder is asked (nd_dnb_proclaim_or_offer). Until 2026-08-07
		# these three branches fired an informational coda and stopped,
		# which left the union standing and the Monarchy unreachable by
		# anyone, the AI included: the Congress hands the winner the other
		# crowns as SUBJECTS, and required_locations_fraction counts direct
		# ownership only, so the formable's own button can never light.
		#
		# Proclaim only when at least one other crown still lives under us
		# (nd_dnb_holds_a_living_crown_p). holds_all_crowns_p is vacuously
		# true for a lone survivor, and survivorship is the sword ending:
		# it keeps the informational coda and the ordinary formable button,
		# which conquest can actually satisfy.
		else_if = {
			limit = { nd_dnb_holds_all_crowns_p = { WHO = HAB } }
			if = {
				limit = { nd_dnb_holds_a_living_crown_p = { WHO = HAB } }
				c:HAB = { nd_dnb_proclaim_or_offer = yes }
			}
			else = {
				c:HAB = { trigger_event_non_silently = nd_danube.20 }
			}
		}
		else_if = {
			limit = { nd_dnb_holds_all_crowns_p = { WHO = HUN } }
			if = {
				limit = { nd_dnb_holds_a_living_crown_p = { WHO = HUN } }
				c:HUN = { nd_dnb_proclaim_or_offer = yes }
			}
			else = {
				c:HUN = { trigger_event_non_silently = nd_danube.20 }
			}
		}
		else_if = {
			limit = { nd_dnb_holds_all_crowns_p = { WHO = BOH } }
			if = {
				limit = { nd_dnb_holds_a_living_crown_p = { WHO = BOH } }
				c:BOH = { nd_dnb_proclaim_or_offer = yes }
			}
			else = {
				c:BOH = { trigger_event_non_silently = nd_danube.20 }
			}
		}
		else_if = {
			limit = {
				NOR = {
					AND = {
						country_exists = c:HAB
						country_exists = c:HUN
					}
					AND = {
						country_exists = c:HAB
						country_exists = c:BOH
					}
					AND = {
						country_exists = c:HUN
						country_exists = c:BOH
					}
				}
			}
			every_country = {
				limit = { nd_dnb_is_crown = yes }
				trigger_event_non_silently = nd_danube.21
			}
		}
		else_if = {
			limit = {
				situation:nd_danube_succession ?= {
					OR = {
						has_variable = nd_dnb_separate_ways
						var:nd_dnb_cause_liberty >= 100
					}
				}
			}
			every_country = {
				limit = { nd_dnb_is_crown = yes }
				add_country_modifier = {
					modifier = nd_dnb_proud_crown
					years = 25
				}
				set_variable = { name = nd_dnb_pacts_blocked value = yes years = 30 }
				trigger_event_non_silently = nd_danube.22
			}
		}
		else = {
			# Epoch adjudication: the strongest standing takes permanent
			# pretensions (claim CBs) on the others and the books close.
			if = {
				limit = {
					country_exists = c:HAB
					situation:nd_danube_succession ?= {
						var:nd_dnb_standing_hab >= var:nd_dnb_standing_hun
						var:nd_dnb_standing_hab >= var:nd_dnb_standing_boh
					}
				}
				c:HAB = {
					if = {
						limit = { country_exists = c:HUN }
						add_casus_belli = {
							type = casus_belli:cb_claim_throne
							years = 25
							target = c:HUN
						}
					}
					if = {
						limit = { country_exists = c:BOH }
						add_casus_belli = {
							type = casus_belli:cb_claim_throne
							years = 25
							target = c:BOH
						}
					}
					trigger_event_non_silently = nd_danube.23
				}
			}
			else_if = {
				limit = {
					country_exists = c:HUN
					situation:nd_danube_succession ?= {
						var:nd_dnb_standing_hun >= var:nd_dnb_standing_boh
					}
				}
				c:HUN = {
					if = {
						limit = { country_exists = c:HAB }
						add_casus_belli = {
							type = casus_belli:cb_claim_throne
							years = 25
							target = c:HAB
						}
					}
					if = {
						limit = { country_exists = c:BOH }
						add_casus_belli = {
							type = casus_belli:cb_claim_throne
							years = 25
							target = c:BOH
						}
					}
					trigger_event_non_silently = nd_danube.23
				}
			}
			else_if = {
				limit = { country_exists = c:BOH }
				c:BOH = {
					if = {
						limit = { country_exists = c:HAB }
						add_casus_belli = {
							type = casus_belli:cb_claim_throne
							years = 25
							target = c:HAB
						}
					}
					if = {
						limit = { country_exists = c:HUN }
						add_casus_belli = {
							type = casus_belli:cb_claim_throne
							years = 25
							target = c:HUN
						}
					}
					trigger_event_non_silently = nd_danube.23
				}
			}
		}
	}

	tooltip = {
		custom_tooltip = nd_dnb_situation_tt
	}

	# Crown-colored realms; everything else stays neutral. Fixed named
	# colors so legend chips can represent them (unlike the Reaction of
	# the Reich's dynamic country color).
	map_color = {
		if = {
			limit = {
				has_owner = yes
				owner = { tag = HAB }
			}
			value = nd_dnb_hab_color
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = { tag = HUN }
			}
			value = nd_dnb_hun_color
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = { tag = BOH }
			}
			value = nd_dnb_boh_color
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	# Stripes: a crown that no longer answers for itself (merged into a
	# rival, or under an outside master) is overlaid in the bound tone;
	# 8 of 22 vanilla situations use the secondary channel this way.
	secondary_map_color = {
		if = {
			limit = {
				has_owner = yes
				owner = {
					nd_dnb_is_crown = yes
					OR = {
						nd_dnb_crown_merged = yes
						nd_dnb_crown_foreign_backed = yes
					}
				}
			}
			value = nd_dnb_bound_color
		}
	}

	legend_key = {
		desc = "nd_dnb_legend_hab"
		color = nd_dnb_hab_color
		require_color_on_map = yes
	}
	legend_key = {
		desc = "nd_dnb_legend_hun"
		color = nd_dnb_hun_color
		require_color_on_map = yes
	}
	legend_key = {
		desc = "nd_dnb_legend_boh"
		color = nd_dnb_boh_color
		require_color_on_map = yes
	}
	legend_key = {
		desc = "nd_dnb_legend_bound"
		color = nd_dnb_bound_color
		require_color_on_map = yes
	}
}
