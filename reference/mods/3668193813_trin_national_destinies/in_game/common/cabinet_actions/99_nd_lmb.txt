# Kingdom of Lombardy: carry the Milanese integration window through formation.
#
# Vanilla's `mlo_restore_kingdom_lombardy` (cabinet_actions/, loc "Integrate
# Lombard Claims") is granted by flavor_mlo.26 when Milan moves its capital to
# Pavia: it sets `lombard_ambitions_variable` for 30 years and opens a cabinet
# action that integrates conquered Po valley and Tuscan provinces much faster.
# Its potential is `tag = MLO`, so forming any successor country drops it.
#
# For every other successor that is vanilla's own behaviour and we leave it
# alone. For LMB it is incoherent: the action is literally "restore the Kingdom
# of Lombardy", and forming Lombardy is how you restore it. A player reported
# losing it mid-window on the Workshop (2026-08-12).
#
# This is additive, not a REPLACE of the vanilla entry: the mod does not
# overwrite vanilla cabinet actions, so other mods touching that file are
# unaffected. The variable requirement is kept, so this grants nothing new. A
# Lombardy formed without ever taking the Milanese DHE has no window to carry,
# exactly as before; a Lombardy formed during one keeps the years it had left.
# The variable survives formation because forming a country keeps the same
# country object (the same reason `has_or_had_tag` works).
#
# Body mirrors the vanilla action, which is the point: the player should not be
# able to tell that the action changed hands.
nd_lmb_restore_kingdom_lombardy = {

	potential = {
		tag = LMB
		has_variable = lombard_ambitions_variable
	}
	ability = mil
	allow_multiple = yes

	allow = {
		any_province = {
			area = {
				OR = {
					this = area:lombardy_area
					this = area:emilia_area
					this = area:piemonte_area
					this = area:liguria_area
					this = area:veneto_area
					this = area:tuscany_area
				}
			}
			any_location_in_province = {
				integration_level = conquered
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "integrate_province_select_province"
		none_available_msg_key = "integrate_province_no_provinces"
		column = {
			data = name
		}
		column = {
			data = integration
		}
		column = {
			data = population
		}
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = nd_lmb_restore_kingdom_lombardy interaction_target = target } }
			any_location_in_province = {
				integration_level = conquered
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = integration_progress
			}
		}
	}

	is_finished = {
		scope:target = {
			any_location_in_province = {
				OR = {
					integration_level = integrated
					integration_level = core
				}
				count = all
			}
		}
	}

	province_modifier = {
		province_integration_speed = 1.5
	}

	location_modifier = {
		local_devastation_recovery = 0.025
		local_population_growth = 0.001
	}

	progress = {
		scope:target = {
			value = province_average_integration
		}
	}

	map_marker = {
		show_for_owner = yes
	}
}
