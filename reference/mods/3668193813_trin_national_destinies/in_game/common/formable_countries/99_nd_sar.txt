# One deliberate difference from vanilla: the fraction is 0.9, not 0.85
# (2026-05-07, tightened with the other small level 2 formables so a single
# island is not formed with a seventh of it still foreign). Everything else
# follows vanilla, which since 1.3.11 reserves this formable for Sardinian
# culture; the Savoyard road is SAR_piedmont_f.
REPLACE:SAR_f = {

	level = 2
	required_locations_fraction = 0.9
	rule = historical

	capital_required = no

	potential = {
		culture = culture:sardinian
	}

	allow = {
	}

	name = SAR
	flag = SAR
	adjective = SAR_ADJ
	tag = SAR
	color = map_sardinian

	areas = {
		sardinia_area
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_sar_formed
		}
		if = {
			# Vanilla tests Reich membership alone; this trigger also honours the
			# rank game rules (scripted_triggers/nd_game_rule_triggers.txt).
			limit = { nd_can_take_kingdom_rank = yes }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		trigger_event_non_silently = nd_sar.1
		add_stability = stability_mild_bonus
	}
}
