# Tabor Republic: radical Hussite communal republic (National Destinies)
# The Tabor Brotherhood (1419-1452) reimagined as a sovereign egalitarian
# theocratic-republican commune of the Hussite Wars: chalice, war-wagon, and
# Four Articles of Prague made into statecraft.

TAB_f = {

	# Level 3, not 2. `level` is a reachability gate, not a prestige rating:
	# "Players can only form countries of higher level" (formable_countries
	# readme). Bohemia is level 2, and Bohemia going Hussite is the entire
	# premise of this formable, so at level 2 the one country it was written
	# for could never form it. Reported three times on the Workshop
	# (2026-06-09, 2026-06-17, 2026-07-02) before the cause was found.
	level = 3
	required_locations_fraction = 0.85
	rule = plausible

	potential = {
		OR = {
			culture = culture:czech
			culture = culture:moravian
		}
		religion = religion:hussite
		# PLAYER ONLY (2026-08-07 user ruling, the 1.3.2 observer run).
		# The AI forms formables aggressively and has no sense of what a
		# tag switch costs the surrounding content: an AI Bohemia took the
		# chalice republic while it held Austria and Hungary as fiefdoms,
		# carried both vassals onto a tag the Danubian Monarchy could not
		# grow out of, and stranded the whole Danube arc. The Tabor
		# Republic is a deliberate player fork, not an AI reflex.
		# UNVERIFIED IDIOM: no vanilla formable uses is_ai in potential
		# (it is a valid country-scope trigger elsewhere, cf.
		# cabinet_actions/promote_culture.txt, country_interactions/hre.txt).
		# Confirm on an observer run that no AI Bohemia ever forms TAB.
		is_ai = no
		# THE DANUBE GATE (same ruling). While the Succession of the
		# Danube is open, the three crowns are answering one question and
		# a crown may not rebrand itself out from under it. Bohemia can
		# still take this road before the question opens, or after it
		# closes (including as a Danubian Monarchy origin: the Monarchy
		# has no religion gate, so the Tabor road remains open afterwards).
		NOT = { has_global_variable = nd_dnb_active }
		# THE TIMING GATE (2026-08-03 user ruling, the Tabor campaign):
		# Tabor is the radical wing WINNING the Hussite Wars, never a
		# mid-war or wars-skipping rebranding. Forming TAB tag-switches
		# BOH off the map, which voids the vanilla situation's can_start
		# (it needs c:BOH) or ends it abruptly mid-run, and the 1.3 AI
		# forms formables aggressively enough to do this in most
		# campaigns where Bohemia turned Hussite.
		NOT = {
			situation:hussite_wars ?= { situation_is_active = yes }
		}
		# The crown itself may only take the chalice-republic road out
		# of a WON war: vanilla stamps hussites_won_hussite_wars in the
		# situation's on_ended exactly when the wars close with Bohemia
		# alive and still Hussite. A later Hussite remnant
		# reconsolidating the lands (non-BOH former) is exempt, since
		# its wars may have ended in defeat for the old crown.
		trigger_if = {
			limit = { tag = BOH }
			has_global_variable = hussites_won_hussite_wars
		}
	}

	allow = {
		owns = location:husinec
	}

	# Not `TAB`: vanilla's interface text already owns that key (the Tab key).
	name = TAB_tabor
	flag = TAB
	adjective = TAB_tabor_ADJ
	tag = TAB
	color = map_TAB

	areas = {
		bohemia_area
		moravia_area
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_tab_formed

		}
		trigger_event_non_silently = nd_tab.1
		add_stability = stability_mild_bonus
		# Tabor Republic IS a peasant-egalitarian republic by founding
		# charter. Permanent governance modifier mirrors vanilla
		# peasant_republic_reform's country_modifier set so the player
		# gets the institutional benefits without forcing a major-reform
		# slot swap. Peasants ascendant, nobles diminished, drift toward
		# decentralization and free-subjects.
		add_country_modifier = {
			modifier = nd_tab_peasant_commune
			years = -1
		}
	}
}
