# Vanilla splits Ireland into three formables that all create the tag IRE:
# the generic road (IRE_f), the High Kingship's leader, and the Lord of
# Ireland. Each one carries the mod's formation lines, or a country formed
# by that road would get none of the mod's content.
REPLACE:IRE_f = {

	level = 2
	required_locations_fraction = 0.75
	rule = historical

	# Generic path. Mutually exclusive with the two IO variants: only shows when the country
	# belongs to neither the High Kingship nor the Lordship of Ireland.
	potential = {
		OR = {
			culture = { has_culture_group = culture_group:hibernian_group }
			tag = PLE
		}
		NOT = { is_member_of_international_organization_of_type = { type = high_kingship } }
		NOT = { is_member_of_international_organization = international_organization:lordship_of_ireland }
	}

	allow = {
		custom_tooltip = {
			text = neither_highkingship_nor_lordship_exist_tt
			NOT = {
				any_international_organization = {
					OR = {
						international_organization_type = international_organization_type:lordship_of_ireland
						international_organization_type = international_organization_type:high_kingship
					}
				}
			}
		}
	}

	name = IRE
	flag = IRE
	adjective = IRE_ADJ
	tag = IRE
	color = map_IRE

	regions = {
		ireland_region
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_ire_formed
		}
		trigger_event_non_silently = nd_ire.1
		add_stability = stability_mild_bonus

		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}

REPLACE:IRE_high_kingship_f = {

	level = 2
	required_locations_fraction = 0.5
	rule = historical

	capital_required = no
	potential_requires_own = no

	# Shown only to High Kingship members. The leader-of-the-IO check lives in allow.
	potential = {
		is_member_of_international_organization_of_type = { type = high_kingship }
	}

	allow = {
		custom_tooltip = {
			text = IRE_high_kingship_f_leader_tt
			any_international_organizations_member_of = {
				count = all
				filter = {
					international_organization_type = international_organization_type:high_kingship
					has_variable = high_kingship_region
					var:high_kingship_region = region:ireland_region
				}
				leader_country ?= root
			}
		}
	}

	name = IRE
	flag = IRE
	adjective = IRE_ADJ
	tag = IRE
	color = map_IRE

	regions = {
		ireland_region
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_ire_formed
		}
		trigger_event_non_silently = nd_ire.1
		add_stability = stability_mild_bonus

		# Leave the High Kingship before rank change to avoid block_from_change_to_kingdom_rank
		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:high_kingship
				has_variable = high_kingship_region
				var:high_kingship_region = region:ireland_region
			}
			remove_country_from_international_organization = root
		}

		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}

REPLACE:IRE_lordship_of_ireland_f = {

	level = 2
	required_locations_fraction = 0.5
	rule = historical

	capital_required = no
	potential_requires_own = no

	potential = {
		is_member_of_international_organization = international_organization:lordship_of_ireland
	}

	# You are the Lord of Ireland: the Anglo-Norman path to a sovereign Irish kingdom.
	allow = {
		custom_tooltip = {
			text = IRE_lordship_of_ireland_f_leader_tt
			international_organization:lordship_of_ireland = {
				leader_country ?= root
			}
		}
	}

	name = IRE
	flag = IRE
	adjective = IRE_ADJ
	tag = IRE
	color = map_IRE

	regions = {
		ireland_region
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_ire_formed
		}
		trigger_event_non_silently = nd_ire.1
		add_stability = stability_mild_bonus

		if = {
			limit = { unique_international_organization_type_exists = international_organization_type:lordship_of_ireland }
			destroy_international_organization = { target = international_organization:lordship_of_ireland }
		}

		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}
