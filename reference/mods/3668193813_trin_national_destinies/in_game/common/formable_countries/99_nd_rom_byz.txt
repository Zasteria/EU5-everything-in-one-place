# ROM_BYZ_f is vanilla 1.2's BYZ/TRE-only Italian Roman restoration, a
# smaller-scope counterpart to ROM_f. The vanilla form_effect already moves
# the capital to Rome and pushes latinization_vs_hellenization hard left.
# We wrap it with our formation event and a one-time stability reward.
#
# Both formables resolve to tag = ROM, so we use the variable nd_rom_byz_formed
# (set in the formation event) to gate ROM_BYZ-specific advances and avoid
# collision with ROM_f's nd_rom_formed gating.
REPLACE:ROM_BYZ_f = {
	level = 5
	required_locations_fraction = 0.25
	rule = historical

	capital_required = no
	potential_requires_own = no

	potential = {
		OR = {
			has_or_had_tag = BYZ
			has_or_had_tag = TRE
		}
	}

	allow = {
		owns = location:rome
		owns = location:constantinople
		country_rank = country_rank:rank_empire

		OR = {
			has_or_had_tag = BYZ
			NOT = { country_exists = c:BYZ }
		}
	}

	name = ROM
	flag = ROM
	adjective = ROM_ADJ
	tag = ROM
	color = map_ROM

	regions = {
		italy_region
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_rom_byz_formed
		}
		if = {
			limit = { NOT = { capital = location:rome } }
			move_capital_event_effect = {
				from = root.capital
				to = location:rome
			}
		}
		province_definition:marittima_province = {
			every_ownable_location_in_province_definition = {
				limit = { owner = root }
				change_integration_level = integrated
			}
		}
		if = {
			limit = { has_societal_value = societal_value_type:latinization_vs_hellenization }
			change_societal_value = { type = latinization_vs_hellenization value = societal_value_huge_move_to_left }
		}
		if = {
			limit = { has_dlc = "d008_fate_of_the_phoenix" }
			unlock_casus_belli_effect = { type = cb_restore_roman_borders }
		}
		trigger_event_non_silently = nd_rom_byz.1
		add_stability = stability_severe_bonus
	}
}
