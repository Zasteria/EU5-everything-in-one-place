# Thessalonica. REPLACE, not INJECT: vanilla THE_f already declares a
# form_effect (the rank bump to kingdom), and the engine rejects a second
# declaration merged in. Everything below the "vanilla content" marker is
# copied verbatim from 00_formable_countries.txt; only the mod block is new.
#
# Formation routes, walked before any content was designed. The potential
# gate is is_frankokratia_state, so the only countries that can ever form
# this are the small Latin lordships of Greece (Athens, Neopatras,
# Bodonitsa, Salona, Negroponte, Naxos, Crete, Cephalonia, Argos, Patras,
# the Knights) plus anyone who has held one of those tags. Albania is the
# one Frankokratia tag with a formable definition of its own (level 2), so
# the level gate excludes it; every other candidate carries no tier and
# qualifies. The allow block demands direct ownership of Thessaloniki and
# the fraction counts direct ownership only, so there is no personal-union
# or vassal route: every former arrives holding Macedonia and Thessaly in
# its own hand, with a population that is Greek Orthodox on the coast and
# Bulgarian Orthodox on the central plain. All content below assumes that
# realm shape and no other.
#
# LEVEL 2, NOT VANILLA'S 1 (2026-08-14, Workshop request by Ender). The
# engine gates formation on the forming country's own formable level and
# requires strictly higher, so at 1 this collided with HOS (also 1) and a
# player who had refounded the Order of Saint John could never take it.
# The Knights are in is_frankokratia_state's tag list and KNI is french
# culture, so every other condition here already passed for them; the tier
# was the only thing in the way.
#
# The change is strictly permissive. Level 2 admits formers at level 0 AND
# 1 where level 1 admitted only 0, so nothing that could form Thessalonica
# before loses it. Albania stays excluded exactly as designed, because
# ALB_f is itself level 2 and equal is not higher. Chain this completes:
# KNI, then HOS (1), then THE (2), then KOJ or LAT (3), then OUT (4).
REPLACE:THE_f = {
	level = 2
	required_locations_fraction = 0.80
	rule = plausible
	capital_required = no
	potential = {
		NOT = { country_exists = c:THE }
		NOT = { tag = LAT }
		is_frankokratia_state = yes
		NOT = { has_variable = halved_cost_to_form_thesslonica }
		NOT = { tag = ACH }
		NOT = { current_age_or_later = { age = age_3_discovery } }
	}
	allow = {
		NOT = { country_exists = c:THE }
		owns = location:thessaloniki
		religion = religion:catholic 
		OR = {
			culture = { has_culture_group = culture_group:french_group }
			culture = { has_culture_group = culture_group:iberian_group }
			culture = { has_culture_group = culture_group:italian_group }
		}
		OR = {
			current_age = age_1_traditions
			current_age = age_2_renaissance
		}
	}
	
	name = THE
	flag = THE
	adjective = THE_ADJ
	tag = THE
	color = map_THE

	provinces = {
		neopatras_province
		thessaly_province
		upper_macedonia_province
		central_macedonia_province
		chalkidiki_province
		lower_macedonia_province
	}
	form_effect = {
		# Vanilla content (verbatim from 00_formable_countries.txt).
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}		

		# Mod content. The formed markers live here rather than only in the
		# formation event's options, so no content layer depends on a popup
		# being delivered (set_variable is idempotent, so the event setting
		# them again is harmless).
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_the_formed
		}
		trigger_event_non_silently = nd_the.1
		add_stability = stability_mild_bonus
	}
}

# Vanilla's second road to the same country (half the territory, for Achaea
# or a claimant flagged by its own events). Hooked so that road also carries
# the mod's formation lines. Its level stays vanilla's 1: the raise to 2
# above was argued for the main road only.
REPLACE:THE_REDUCED_REQUIREMENTS_f = {
	level = 1
	required_locations_fraction = 0.50
	rule = plausible
	capital_required = no
	potential = {
		NOT = { country_exists = c:THE }
		NOT = { tag = LAT }
		is_frankokratia_state = yes
		OR = {
			has_variable = halved_cost_to_form_thesslonica
			tag = ACH
		}
		NOT = { current_age_or_later = { age = age_3_discovery } }
	}
	allow = {
		NOT = { country_exists = c:THE }
		owns = location:thessaloniki
		religion = religion:catholic
		OR = {
			culture = { has_culture_group = culture_group:french_group }
			culture = { has_culture_group = culture_group:iberian_group }
			culture = { has_culture_group = culture_group:italian_group }
		}
		OR = {
			current_age = age_1_traditions
			current_age = age_2_renaissance
		}	
	}
	
	name = THE
	flag = THE
	adjective = THE_ADJ
	tag = THE
	color = map_THE

	provinces = {
		neopatras_province
		thessaly_province
		upper_macedonia_province
		central_macedonia_province
		chalkidiki_province
		lower_macedonia_province
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_the_formed
		}
		trigger_event_non_silently = nd_the.1
		add_stability = stability_mild_bonus

		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}			
	}
}
