# Vanilla's TRG_f (added in 1.4) has an empty form_effect, so INJECT is safe:
# there is nothing of vanilla's to lose. It is the only vanilla formable that
# creates the tag TRG (00_formable_countries.txt:5407).
#
# No rank change: the confederation of the drum groups was never a kingdom.
# No government change: a tribe stays a tribe, and Timbuktu's Tuareg
# monarchy (TMB) keeps its crown.
INJECT:TRG_f = {
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script, and then the popup may never
		# be answered, leaving every advance gated on this variable
		# unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_trg_formed
		}
		trigger_event_non_silently = nd_trg.1
		add_stability = stability_mild_bonus
	}
}
