# Algiers (ALG_f, vanilla since 1.4): level 2, 75% of the Algiers area.
# Vanilla's form_effect is empty (00_formable_countries.txt:4973 region),
# so INJECT is safe. ALG_f is the only vanilla formable with tag = ALG.
INJECT:ALG_f = {
	form_effect = {
		# Formation markers live HERE, not only in the event's options: a
		# country can be formed by script, and then the popup may never be
		# answered, leaving every advance gated on the marker unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_alg_formed
		}
		# Europe knew the Regency as the Kingdom of Algiers, and the Zayyanid
		# realm it replaced was a sultanate. nd_can_take_kingdom_rank is false
		# only where the engine itself would refuse the rank.
		if = {
			limit = {
				nd_can_take_kingdom_rank = yes
				country_rank_level < 3
			}
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		trigger_event_non_silently = nd_alg.1
		add_stability = stability_mild_bonus
	}
}
