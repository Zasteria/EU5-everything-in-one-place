# Sardinia-Piedmont. REPLACE, not INJECT: the vanilla form_effect already
# carries a conditional rank change, and INJECT would overwrite the whole
# field. Everything above the form_effect is copied verbatim from
# 00_formable_countries.txt so this file cannot drift from vanilla.
# One deliberate difference: the kingdom rank is gated by
# nd_can_take_kingdom_rank, vanilla's own test (not a member of the Reich)
# widened to honour the rank game rules.
REPLACE:SAR_piedmont_f = {

	level = 2
	required_locations_fraction = 0.85
	rule = historical

	capital_required = no

	potential = {
		culture = { has_culture_group = culture_group:italian_group }
	}

	allow = {
	}

	name = SAR_piedmont
	flag = SAR_piedmont
	adjective = SAR_piedmont_ADJ
	tag = SRP
	color = map_sardinia

	areas = {
		sardinia_area
		piemonte_area
	}
	form_effect = {
		# Vanilla effect, preserved exactly. The crown of Sardinia is a royal
		# title, but the Empire holds its members below kingdom rank, so the
		# bump normally only applies outside the Empire. That block is
		# conditional rather than absolute, which is what the trigger reads:
		# see nd_can_take_kingdom_rank in scripted_triggers/.
		if = {
			limit = {
				nd_can_take_kingdom_rank = yes
			}
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		trigger_event_non_silently = nd_sar_piedmont.1
		# State markers are stamped here as well as in the event options, so a
		# content layer never depends on a popup being delivered.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_sar_piedmont_formed
		}
	}
}
