# Follows vanilla 1.4, which reshaped Russia: forming it now grants kingdom
# rank only (the empire is proclaimed afterwards by vanilla's own decision,
# decisions/flavor_decisions_rus.txt), needs Tver and Vladimir as well, an
# independent country with its capital inside, and covers the Russian
# region plus Severia and Kursk instead of the Urals and the steppe.
REPLACE:RUS_f = {
	level = 4
	required_locations_fraction = 0.5
	capital_required = yes
	content_priority = 800
	rule = historical

	potential = {
		culture = { has_culture_group = culture_group:russian_group }
	}

	allow = {
		is_subject = no
		owns = location:moscow
		owns = location:novgorod
		owns = location:tver
		owns = location:vladimir
	}
	
	name = RUS
	flag = RUS
	adjective = RUS_ADJ
	tag = RUS
	color = map_RUS

	regions = {
		russian_region
	}
	areas = {
		severia_area
		kursk_area
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_rus_formed
		}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		trigger_event_non_silently = nd_rus.1
		add_stability = stability_severe_bonus
	}
	ai_will_do = {
		value = 1000
	}
}
