# ==========================================================================
# THE SUCCESSION OF THE DANUBE: the four dynastic levers (PLAN.md
# section 13). All modeled on nd_dnm_sue_for_imperial_reconciliation and
# vanilla hre.txt interaction idioms. Message types suppressed for the
# same reason as the DNM interaction (gui/messagetypes.txt is
# non-additive); the PROPOSE_/INCOMING_OFFER_ loc surface renders.
# ==========================================================================

# --------------------------------------------------------------------------
# PROPOSE THE ERBVERBRUEDERUNG. The player's match: proposable BEFORE the
# situation exists (a signed pact is a spark, the 1364 anchor), and each
# acceptance raises the pair's tier one step (1 league, 2 betrothal
# compact, 3 full mutual succession). Pacts ride royal marriages, as the
# real ones did. Acceptance is where user ruling on senior-partner
# resistance lives: a stronger proposer with no menace at the gates gets
# a proud refusal.
# --------------------------------------------------------------------------
nd_dnb_propose_the_pact = {
	type = diplomacy

	show_message = no
	show_message_to_target = no

	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = {
			# The pact is one of the two activation sites of the
			# situation (with nd_dnb_spark_watch), so it reads the lobby
			# rule too (2026-08-25).
			nd_danube_succession_is_enabled = yes
			nd_dnb_crown_independent = yes
			NOT = { has_variable = nd_dnb_pacts_blocked }
			# A concluded question does not reopen; pacts lose their
			# object once the books are closed.
			NOT = { has_global_variable = nd_dnb_concluded }
			# Same Hussite deferral as the spark watcher (2026-08-29):
			# no pact opens the Danube question while the Hussite Wars
			# are being fought, so the dynastic union waits its turn
			# behind the Basel settlement.
			NOT = { situation:hussite_wars ?= { situation_is_active = yes } }
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			nd_dnb_crown_independent = yes
			NOT = { this = scope:actor }
			NOT = { has_variable = nd_dnb_pacts_blocked }
		}
		enabled = {
			# The 1364 anchor: the watcher opens the question no earlier
			# than 1360, and the pact (which sparks it directly) honors
			# the same floor, otherwise an AI crown with a game-start
			# royal marriage signs in 1338 and the early game never
			# breathes. Shown grayed with this reason until then.
			custom_tooltip = {
				text = nd_dnb_pact_year_req_tt
				current_year >= 1360
			}
			custom_tooltip = {
				text = nd_dnb_pact_rm_req_tt
				# The menace exception (2026-08-10, 1444 live-play finding):
				# every deed generator rode a cross-crown royal marriage and
				# nothing makes AI crowns marry each other, so AI worlds
				# starved. Historical anchor: the 1463 Frederick and
				# Corvinus Erbverbruederung was signed under Ottoman
				# pressure with no marriage behind it.
				OR = {
					has_royal_marriage_with = scope:actor
					situation:nd_danube_succession ?= { var:nd_dnb_menace >= 2 }
				}
			}
			custom_tooltip = {
				text = nd_dnb_pact_tier_cap_tt
				nd_dnb_pair_tier_lt3_with_actor = yes
			}
		}
	}

	cooldown = {
		type = nd_dnb_propose_the_pact
		years = 5
	}

	effect = {
		# A signed pact opens the question if nothing else has yet
		# (live-state gate, same self-healing rule as the spark watch).
		if = {
			limit = {
				NOT = { has_global_variable = nd_dnb_active }
			}
			scope:actor = {
				activate_situation = situation:nd_danube_succession
			}
		}
		scope:recipient = {
			nd_dnb_mark_pact_pending_by_actor = yes
			trigger_event_non_silently = { id = nd_danube.11 days = 5 }
		}
	}

	reject_effect = {
		scope:actor = {
			add_prestige = prestige_mild_penalty
			custom_tooltip = nd_dnb_pact_rejected_tt
		}
	}

	accept = {
		add = {
			desc = "ND_DNB_ACCEPT_BASE"
			value = -20
		}
		# The gates: danger makes brothers (AI-agnostic menace, 0-3).
		if = {
			limit = { has_global_variable = nd_dnb_active }
			add = {
				desc = "ND_DNB_ACCEPT_MENACE"
				value = situation:nd_danube_succession.var:nd_dnb_menace
				multiply = 15
			}
		}
		# The zeitgeist (2026-08-03): a cause past 50 sways the chanceries.
		if = {
			limit = {
				situation:nd_danube_succession ?= { var:nd_dnb_cause_union >= 50 }
			}
			add = {
				desc = "ND_DNB_ACCEPT_ZEITGEIST_UNION"
				value = 15
			}
		}
		if = {
			limit = {
				situation:nd_danube_succession ?= { var:nd_dnb_cause_liberty >= 50 }
			}
			subtract = {
				desc = "ND_DNB_ACCEPT_ZEITGEIST_LIBERTY"
				value = 25
			}
		}
		# A wobbling throne wants a strong pact partner.
		if = {
			limit = { scope:recipient = { nd_dnb_weak_succession = yes } }
			add = {
				desc = "ND_DNB_ACCEPT_WEAK"
				value = 25
			}
		}
		# A courted diet remembers the gold.
		if = {
			limit = {
				OR = {
					AND = {
						scope:actor = { tag = HAB }
						scope:recipient = { tag = HUN }
						situation:nd_danube_succession ?= { has_variable = nd_dnb_courted_hab_hun }
					}
					AND = {
						scope:actor = { tag = HAB }
						scope:recipient = { tag = BOH }
						situation:nd_danube_succession ?= { has_variable = nd_dnb_courted_hab_boh }
					}
					AND = {
						scope:actor = { tag = HUN }
						scope:recipient = { tag = HAB }
						situation:nd_danube_succession ?= { has_variable = nd_dnb_courted_hun_hab }
					}
					AND = {
						scope:actor = { tag = HUN }
						scope:recipient = { tag = BOH }
						situation:nd_danube_succession ?= { has_variable = nd_dnb_courted_hun_boh }
					}
					AND = {
						scope:actor = { tag = BOH }
						scope:recipient = { tag = HAB }
						situation:nd_danube_succession ?= { has_variable = nd_dnb_courted_boh_hab }
					}
					AND = {
						scope:actor = { tag = BOH }
						scope:recipient = { tag = HUN }
						situation:nd_danube_succession ?= { has_variable = nd_dnb_courted_boh_hun }
					}
				}
			}
			add = {
				desc = "ND_DNB_ACCEPT_COURTED"
				value = 15
			}
		}
		# Declared postures from the last Diet.
		if = {
			limit = {
				OR = {
					AND = {
						scope:recipient = { tag = HAB }
						situation:nd_danube_succession ?= { var:nd_dnb_posture_hab = 1 }
					}
					AND = {
						scope:recipient = { tag = HUN }
						situation:nd_danube_succession ?= { var:nd_dnb_posture_hun = 1 }
					}
					AND = {
						scope:recipient = { tag = BOH }
						situation:nd_danube_succession ?= { var:nd_dnb_posture_boh = 1 }
					}
				}
			}
			add = {
				desc = "ND_DNB_ACCEPT_UNION_POSTURE"
				value = 20
			}
		}
		if = {
			limit = {
				OR = {
					AND = {
						scope:recipient = { tag = HAB }
						situation:nd_danube_succession ?= { var:nd_dnb_posture_hab = 2 }
					}
					AND = {
						scope:recipient = { tag = HUN }
						situation:nd_danube_succession ?= { var:nd_dnb_posture_hun = 2 }
					}
					AND = {
						scope:recipient = { tag = BOH }
						situation:nd_danube_succession ?= { var:nd_dnb_posture_boh = 2 }
					}
				}
			}
			subtract = {
				desc = "ND_DNB_ACCEPT_INDEPENDENCE_POSTURE"
				value = 30
			}
		}
		# The senior-partner fear: a stronger proposer in a calm world
		# reads as a future master, not a brother.
		if = {
			limit = {
				scope:recipient = {
					NOT = { military_strength >= scope:actor.military_strength }
				}
				OR = {
					NOT = { has_global_variable = nd_dnb_active }
					situation:nd_danube_succession ?= { var:nd_dnb_menace = 0 }
				}
			}
			subtract = {
				desc = "ND_DNB_ACCEPT_PROUD_CROWN"
				value = 20
			}
		}
		if = {
			limit = { "scope:recipient.antagonism(scope:actor)" > 0 }
			subtract = {
				desc = "DIPLOREASON_ANTAGONISM"
				value = "scope:recipient.antagonism(scope:actor)"
			}
		}
	}

	ai_will_do = {
		value = 0
		add = {
			if = {
				limit = {
					scope:actor = { has_royal_marriage_with = scope:recipient }
				}
				value = 30
			}
		}
		add = {
			if = {
				limit = {
					has_global_variable = nd_dnb_active
					situation:nd_danube_succession ?= { var:nd_dnb_menace >= 2 }
				}
				value = 100
			}
		}
		add = {
			if = {
				limit = {
					scope:actor = { nd_dnb_weak_succession = yes }
				}
				value = 50
			}
		}
	}
}

# --------------------------------------------------------------------------
# COURT THE DIET. Gold into the other crown's magnate halls (the Corvinus
# election was bought the same way): a 10-year courtship marker worth +5
# standing and +15 on later pact acceptance. Unilateral spending, no
# refusal.
# --------------------------------------------------------------------------
nd_dnb_court_the_diet = {
	type = diplomacy

	show_message = no
	show_message_to_target = no

	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = {
			nd_dnb_crown_independent = yes
			has_global_variable = nd_dnb_active
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			nd_dnb_crown_independent = yes
			NOT = { this = scope:actor }
		}
	}

	cooldown = {
		type = nd_dnb_court_the_diet
		years = 5
	}

	effect = {
		scope:actor = {
			add_gold = {
				value = monthly_income_trade_and_tax
				multiply = -12
				max = -100
			}
		}
		scope:recipient = {
			custom_tooltip = {
				text = nd_dnb_courted_applied_tt
				nd_dnb_set_courted_by_actor = yes
			}
		}
	}

	ai_will_do = {
		value = 0
		add = {
			if = {
				limit = {
					scope:actor = {
						gold >= {
							value = monthly_income_trade_and_tax
							multiply = 24
						}
					}
					has_global_variable = nd_dnb_active
					situation:nd_danube_succession ?= { var:nd_dnb_menace >= 1 }
				}
				value = 60
			}
		}
	}
}

# --------------------------------------------------------------------------
# CONTEST THE FOREIGN CROWN (user ruling 2, the Jagiellon spoiler). When
# a crown sits junior under an outside power, any remaining crown can
# invoke the pact tradition against the foreign arrangement: a claim CB
# on the captive crown (war with its master follows from the union's own
# defense), and the third crown, if free, receives the same case plus
# the summons event. Modeled on break_union.txt's every-member CB grant.
# --------------------------------------------------------------------------
nd_dnb_contest_foreign_crown = {
	type = diplomacy

	show_message = no
	show_message_to_target = no

	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = {
			nd_dnb_crown_independent = yes
			has_global_variable = nd_dnb_active
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			nd_dnb_is_crown = yes
			NOT = { this = scope:actor }
		}
		# The captive test sits in `enabled`, not `visible` (Workshop
		# report 2026-09-15, Marcell56: a Hungary that wanted the
		# Austrian crown could not find the action at all, because a
		# free crown was filtered out of the list and the interaction
		# vanished with it). Now every other crown is listed, greyed
		# with the reason when it is not captive, so the player learns
		# what the lever is for instead of hunting for it.
		enabled = {
			custom_tooltip = {
				text = nd_dnb_contest_captive_req_tt
				# v2 (PLAN.md 13/v2): dynastic capture counts. A crown
				# whose king belongs to an outside monarchy's ruling
				# house is as contested as one bound by treaty, even
				# while politically independent (the Jagiellon case).
				OR = {
					nd_dnb_crown_foreign_backed = yes
					AND = {
						nd_dnb_crown_independent = yes
						nd_dnb_under_outside_house = yes
					}
				}
			}
		}
	}

	cooldown = {
		type = nd_dnb_contest_foreign_crown
		years = 10
	}

	effect = {
		scope:actor = {
			add_casus_belli = {
				type = casus_belli:cb_claim_throne
				target = scope:recipient
			}
		}
		# The third crown, if free, holds the same case.
		if = {
			limit = {
				NOR = {
					scope:actor = { tag = HAB }
					scope:recipient = { tag = HAB }
				}
				country_exists = c:HAB
				c:HAB = { nd_dnb_crown_independent = yes }
			}
			c:HAB = {
				add_casus_belli = {
					type = casus_belli:cb_claim_throne
					target = scope:recipient
				}
				trigger_event_non_silently = nd_danube.16
			}
		}
		if = {
			limit = {
				NOR = {
					scope:actor = { tag = HUN }
					scope:recipient = { tag = HUN }
				}
				country_exists = c:HUN
				c:HUN = { nd_dnb_crown_independent = yes }
			}
			c:HUN = {
				add_casus_belli = {
					type = casus_belli:cb_claim_throne
					target = scope:recipient
				}
				trigger_event_non_silently = nd_danube.16
			}
		}
		if = {
			limit = {
				NOR = {
					scope:actor = { tag = BOH }
					scope:recipient = { tag = BOH }
				}
				country_exists = c:BOH
				c:BOH = { nd_dnb_crown_independent = yes }
			}
			c:BOH = {
				add_casus_belli = {
					type = casus_belli:cb_claim_throne
					target = scope:recipient
				}
				trigger_event_non_silently = nd_danube.16
			}
		}
	}

	ai_will_do = {
		value = 0
		add = {
			if = {
				limit = {
					scope:recipient = {
						OR = {
							nd_dnb_crown_foreign_backed = yes
							AND = {
								nd_dnb_crown_independent = yes
								nd_dnb_under_outside_house = yes
							}
						}
					}
				}
				value = 100
			}
		}
	}
}
