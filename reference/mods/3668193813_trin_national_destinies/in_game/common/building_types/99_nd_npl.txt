# Herb Gatherers' Lodge: drying sheds where hill collectors bring musk pods,
# roots and herbs. It makes medicaments, which the game counts as a raw good,
# out of the hills' own game and plants, so it is bound to land whose own
# good is wild game (21 locations of Nepal): where the musk deer and the
# herbs are. Vanilla's apothecary makes the same good from the same stock in
# a town; this is the place it is gathered.
nd_npl_herb_lodge = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 2
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_npl_himalayan_herbs
	}

	location_potential = {
		raw_material = goods:wild_game
	}

	unique_production_methods = {
		nd_npl_herbal_trade = {
			wild_game = 0.2
			lumber = 0.1
			produced = medicaments
			output = 0.42
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
