# Thessalonica: custom buildings.
#
# Every one of these is bound to something that is physically in the six
# provinces the formable covers: the acropolis above the city, the Athonite
# estates on the Chalkidiki shore, the argentiferous ore of Sidirokausia,
# the myrrh of the martyr's tomb, the wool halls of the refugee
# congregations, the striking house at the mine, and the crocus fields of
# the Kozani plateau. Each carries a location_potential and a line saying
# why that reach is the correct one.

# The citadel on the acropolis. Its purpose is not to keep an army out; it
# is to keep a city in. Every ruler of Thessalonica from Boniface to Murad
# garrisoned this rock precisely because the town below could not be
# trusted, and the burgher power penalty is the honest price of that.
nd_the_eptapyrgion = {
	is_special = yes
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	employment_size = bastion_employment
	category = defense_category
	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = medium_fort_building

	country_potential = {
		has_advance = nd_the_crown_of_boniface
	}

	# Two restrictions, both deliberate. The tier flags above admit city
	# and megalopolis only: a citadel that overawes a commune needs a
	# commune to overawe, and there is nothing in a Macedonian village for
	# this to hold down (that tier is also what keeps local_max_control
	# inside the 0.10 city ceiling). The coastal gate is the second half:
	# this is the acropolis of a port, whose whole point was that it
	# commanded the sea wall and the harbour when the town below turned.
	location_potential = {
		is_coastal = yes
	}

	modifier = {
		local_garrison_size = 0.15
		local_max_control = 0.08
		local_unrest = -0.05
		local_defensive = 0.2
		local_burghers_estate_power = -0.5
	}

	unique_production_methods = {
		nd_the_eptapyrgion_maintenance = {
			stone = 0.2
			weaponry = 0.1
			lumber = 0.1
			tools = 0.1
			category = building_maintenance
		}
	}
	construction_demand = castle_construction
}

# An Athonite dependency: land, a chapel, a steward and a village of
# dependent peasants, sending grain, oil and spirits back to the mother
# house on the Mountain. Tsipouro, the marc spirit still made at Athos, is
# traditionally credited to fourteenth-century monks of exactly these
# houses.
nd_the_metochion = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 2
	employment_size = generic_peasant_building_employment
	category = consumer_goods_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_the_metochia_of_the_holy_mountain
	}

	# A metochion is an Orthodox monastery's estate. A Catholic crown can
	# license one, endow one and tax one, but it cannot conjure one among
	# its own Latins: the house, the steward and the paroikoi are all
	# Orthodox. This gate is the country's central fact expressed as reach,
	# and it also keeps the estate out of the crown's own core.
	location_potential = {
		dominant_religion ?= religion:orthodox
	}

	modifier = {
		local_clergy_estate_power = 0.5
		local_clergy_desired_pop = 0.05
		local_monthly_food_modifier = 0.025
		local_peasant_enfranchisment = -0.01
	}

	unique_production_methods = {
		# Profit check: fruit 0.3x1.0 + wheat 0.1x1.0 = 0.40 input cost
		# 0.40 x 1.20 / 2.5 = 0.192 liquor -> 20% margin
		nd_the_tsipouro_still = {
			fruit = 0.3
			wheat = 0.1
			produced = liquor
			output = 0.192
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
		# Profit check: fiber_crops 0.15x2.0 + lumber 0.1x1.5 = 0.45
		# 0.45 x 1.20 / 2.0 = 0.27 paper -> 20% margin
		nd_the_monastic_scriptorium = {
			fiber_crops = 0.15
			lumber = 0.1
			produced = paper
			output = 0.27
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}

# The workings of Sidirokausia. Argentiferous lead ore roasted in hundreds
# of small kilns on the Chalkidiki hills; by the sixteenth century the most
# productive mine complex in the Balkans.
nd_the_sidirokausia_works = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 2
	employment_size = manufactory_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = yes
	build_time = manufactory_build_time

	country_potential = {
		has_advance = nd_the_workings_of_sidirokausia
	}

	# Smelting is bound to the ore. Without this the works would coin
	# silver out of nothing in every location the crown holds, which is
	# the free-resource bug, not a mining district.
	location_potential = {
		is_produced_in_location_market = goods:silver
	}

	modifier = {
		local_laborers_desired_pop = 0.1
		local_max_rgo_size = 1
	}

	unique_production_methods = {
		# Input fix: tools (equipment only, no ore) replaced with lead, the
		# galena the localization already names as the roasted feedstock.
		# Profit check: lumber 0.2x1.5 + lead 0.05x2.0 = 0.40 input cost
		# 0.40 x 1.20 / 6.0 = 0.08 silver -> 20% margin
		nd_the_argyros_smelting = {
			lumber = 0.2
			lead = 0.05
			produced = silver
			output = 0.08
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
		# Profit check: lead 0.1x2.0 + lumber 0.2x1.5 + tools 0.05x3.0 = 0.65
		# 0.65 x 1.20 / 6.0 = 0.13 silver -> 20% margin
		nd_the_cupellation = {
			lead = 0.1
			lumber = 0.2
			tools = 0.05
			produced = silver
			output = 0.13
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = manufactory_construction
}

# The striking house at the mine. Bar silver went in and coin came out

	is_foreign = no
	pop_type = laborers
	max_levels = 1

	country_potential = {
		has_advance = nd_the_mint_of_the_mountain
	}

	# Follows the bullion. A striking house is put where the metal already
	# comes to market, which in this realm means the Chalkidiki ore road

# The Frankish tower at Platamon, raised between 1204 and 1222 where the
# coast road enters the Vale of Tempe. It is not a city citadel and it is
# not meant to be: the Eptapyrgion holds a town down, and this holds a road
# open. Everything that moves between Macedonia and Thessaly passes under
# it, which is why the crown puts a castellan of its own in it and why the
# baron who takes the post becomes the second man in the kingdom.
nd_the_platamon_castle = {
	is_special = yes
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	employment_size = castle_employment
	category = defense_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = yes
	build_time = small_fort_building

	country_potential = {
		has_advance = nd_the_dest_ravennika
	}

	# Two restrictions and a reason for each. The tier flags admit rural
	# settlements and towns only: the cities of this realm already have the
	# Eptapyrgion, and a castellany that duplicated it would just be a
	# second garrison in the same street. The topography gate is the rest:
	# a castellany is worth building where the ground does the work, on a
	# pass or a defile, and there is nothing for it to command on the open
	# Macedonian plain.
	location_potential = {
		OR = {
			topography = hills
			topography = mountains
		}
	}

	modifier = {
		local_nobles_estate_power = 0.5
		local_hostile_attrition = 0.1
		local_distance_from_capital_speed_propagation = 0.1
	}

	unique_production_methods = {
		nd_the_platamon_maintenance = {
			masonry = 0.2
			weaponry = 0.1
			tar = 0.1
			tools = 0.1
			category = building_maintenance
		}
	}
	construction_demand = castle_construction
}
