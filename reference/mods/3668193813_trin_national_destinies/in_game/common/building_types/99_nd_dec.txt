# Bidri Karkhana: the inlaid metalwork of Bidar.
# The craft came to Bidar under the Bahmani sultans. A vessel is cast in a zinc
# and copper alloy, inlaid with silver wire and blackened with a paste made of
# the soil of Bidar fort, which is what gives the ware its colour and what ties
# it to this ground: the building stands only in the province of Bidar,
# Gulbarga and Bijapur (kalyana_karnakassala, the Bahmani heartland).
# The game has no zinc; copper stands for the alloy. Vanilla's jewelry guild
# makes 1 jewelry from 1.3889 copper or 0.6944 silver; this takes both.
# Output from eu5-profit at the guild margin: cost 2.25, 0.54 jewelry.
# Three levels and no modifier: a plain producer in the one province it fits.
nd_dec_bidri_karkhana = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 3
	startup_ramp_target = guild_startup_ramp_target
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_dec_bidriware_craft
	}

	location_potential = {
		province_definition = province_definition:kalyana_karnakassala
	}

	unique_production_methods = {
		nd_dec_bidri_inlay = {
			copper = 0.5
			silver = 0.1
			tools = 0.05
			produced = jewelry
			output = 0.54
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Topkhana: the gun foundry of the sultanates.
# The Deccan cast its own ordnance in bell metal: the Malik-i-Maidan, cast at
# Ahmadnagar in 1549 by Muhammad bin Husayn Rumi and later mounted on the walls
# of Bijapur, weighs some fifty-five tons.
# No location gate: a foundry buys copper and tin in the market and makes a
# produced good, as vanilla's cannon maker does (bronze: copper 0.5207,
# tin 0.1045, tools 0.5207 for 1 cannon). Bell metal carries more tin, and
# this foundry spends fewer tools.
# Town and city only, two levels: it adds to the vanilla cannon maker and must
# not replace it. Output from eu5-profit at the guild margin: cost 2.7, 0.81.
nd_dec_topkhana = {
	is_special = yes
	is_foreign = no

	max_levels = 2

	town = yes

	country_potential = {
		has_advance = nd_dec_sultanate_artillery

	unique_production_methods = {

# Chintz Karkhana: the painted cottons of Golconda.
# Kalamkari, cloth drawn with the pen and dyed with mordants, was made on the
# Coromandel coast around Masulipatnam under the Qutb Shahis and sold to
# Persia and to the European companies as chintz.
# Bound to cotton country: it stands where the location's market grows cotton
# (27 locations of the Deccan have cotton as their own raw material).
# Woven cloth in, painted cloth out, with alum as the mordant, as vanilla's
# fine cloth guild takes cloth 0.8333 for 0.5 and alum in its second slot.
# Outputs from eu5-profit at the guild margin:
#   cloth 0.8, dyes 0.1, alum 0.05 costs 2.95 and gives 0.59;
#   cloth 1.2, dyes 0.15, alum 0.05, tools 0.1 costs 4.65 and gives 0.93.
# The second method (nd_dec_deccani_industrial_dawn, age 6) is a real step
# up: 58% more cloth from the same workers, at the price of tools.
nd_dec_chintz_karkhana = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 2
	startup_ramp_target = guild_startup_ramp_target
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_dec_textile_industry
	}

	location_potential = {
		is_produced_in_location_market = goods:cotton
	}

	unique_production_methods = {
		nd_dec_kalamkari_painting = {
			cloth = 0.8
			dyes = 0.1
			alum = 0.05
			produced = fine_cloth
			output = 0.59
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		nd_dec_cotton_milling_pm = {
			# unlocked by nd_dec_deccani_industrial_dawn
			cloth = 1.2
			dyes = 0.15
			alum = 0.05
			tools = 0.1
			produced = fine_cloth
			output = 0.93
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
