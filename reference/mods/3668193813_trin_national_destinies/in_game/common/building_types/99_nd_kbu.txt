# Natron extraction works: trade/production building
nd_kbu_natron_works = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 3
	employment_size = guild_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_kbu_natron_extraction
	}

	modifier = {
		local_merchant_power = 0.05
		local_raw_material_output = 0.25
		local_trades_per_burgher = 0.1
	}

	# Extraction is bound to the resource: you cannot mine, tap or
	# harvest salt where there is none. Without this the building
	# is a free resource generator in every location the owner holds.
	location_potential = {
		is_produced_in_location_market = goods:salt
	}

	unique_production_methods = {
		# Output at the rural margin (0.2 x 1.2 / 4). It stood at 0.15 with the
		# cap switched off, three times what the clay pays for.
		nd_kbu_natron_processing_pm = {
			clay = 0.4
			produced = salt
			output = 0.06
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		# Unlocked by nd_kbu_natron_monopoly (age 5). The brine of the pans
		# is run into moulds cut from palm trunks and dried into cones and
		# cakes that a camel can carry. Vanilla's salt collector makes salt
		# from clay as well; the building's gate on salt covers this method.
		nd_kbu_natron_export_pm = {
			clay = 0.4
			lumber = 0.2
			produced = salt
			output = 0.15
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
