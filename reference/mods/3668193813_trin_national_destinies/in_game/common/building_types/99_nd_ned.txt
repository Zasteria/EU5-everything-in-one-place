# Haringpakkerij: herring packing house of the Great Fishery
# Where the busses land their catch to be gibbed, salted and barrelled
# under the seal of the College of the Great Fishery (1567).
nd_ned_haringpakkerij = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = generic_peasant_building_employment
	category = basic_industry_category
	# Every tier: the herring ports ran from fishing villages (Vlaardingen,
	# De Rijp) to Enkhuizen and Amsterdam. The gate is the location, below.
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = rural_build_time

	# Only where the sea is fished: the same gate vanilla puts on its own
	# fish building (bubu_building), plus the coast the busses sail from.
	location_potential = {
		is_coastal = yes
		raw_material = goods:fish
	}

	country_potential = {
		has_advance = nd_ned_herring_fleets
	}

	modifier = {
		local_fish_output_modifier = 0.15
	}

	unique_production_methods = {
		nd_ned_haringpakkerij_upkeep = {
			salt = 0.1
			lumber = 0.1
			naval_supplies = 0.05
			category = building_maintenance
		}

		# Gibbed, salted and barrelled herring that keeps for a year: fish
		# for the market from salt and barrel staves, in the shape of
		# vanilla's fishing village (salt and tools for fish, rural margin).
		# Unlocked by the admiralties, whose convoy ships let the busses
		# stay at sea and cure on board.
		nd_ned_herring_curing = {
			salt = 0.2
			lumber = 0.1
			produced = fish
			output = 1.14
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
