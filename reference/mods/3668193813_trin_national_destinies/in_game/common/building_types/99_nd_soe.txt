# Niger River Port: the landing where the canoes of the river fleet load
# grain, salt and soldiers. Bound to a river: the map draws the Niger as a
# river and gives it no water a ship can sail (no sea zone, no lake and no
# port entry for Gao, Timbuktu, Kabara or Djenne), so until 2026-10-02 the
# port stood behind is_coastal and built ships where the empire has no shore.
# It now does what the fleet did on land: it carries the crown's reach and
# the merchants' goods along the river.
# Every settlement tier: Kabara, the landing of Timbuktu, is a village.
nd_soe_niger_port = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = dock_employment
	category = naval_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	build_time = trade_post_build_time

	country_potential = {
		has_advance = nd_soe_niger_fleet
	}

	location_potential = {
		has_river = yes
	}

	# Proximity value is vanilla's own for a waterway work (the pound lock
	# canal and the bridge both give 0.1).
	modifier = {
		local_merchant_power = 0.05
		local_distance_from_capital_speed_propagation = 0.1
	}

	unique_production_methods = {
		nd_soe_port_upkeep = {
			lumber = 0.213
			tar = 0.107
			tools = 0.107
			category = building_maintenance
		}
	}
	construction_demand = port_building_construction
}

# Sahel Stud (age 4, nd_soe_pastoral_wealth)
# Vanilla's horse breeders are closed to this country: their `allow` asks
# for a mediterranean, continental or oceanic climate
# (building_types/rural_buildings.txt, horse_breeders), and the Niger bend is
# arid. This is the Sahel's own stud, on millet instead of wheat.
# location_potential: vanilla's own gate for a horse breeder, horses
# produced somewhere in the location's market. Horses are a raw good, so the
# stud cannot stand where no breeding stock exists.
# Countryside and town, not city: pasture.
nd_soe_sahel_stud = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_soe_pastoral_wealth
	}

	location_potential = {
		market ?= {
			is_produced_in_market = goods:horses
		}
	}

	unique_production_methods = {
		nd_soe_horse_breeding = {
			millet = 0.5
			horses = 0.1
			salt = 0.025
			produced = horses
			output = 0.36
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = horse_breeders_construction
}

# Mud-Brick Town Wall (age 5, nd_soe_sahel_fortifications)

# cheap and has to be renewed after every rainy season.
