### Mehari Stud: riding camels bred in the valleys of the massif ###
# Produces camels, a raw good, so it is bound to a raw good: land whose own
# good is livestock. Hills and plateau only: the stud belongs to the massif
# and its valleys (Assode, Agadez, Kalfou, Garadoume), where the herds are
# kept between caravans. No vanilla building makes camels, and no location
# of the Air has them as its raw good (location_templates.txt); camels in
# the market are what vanilla's camelry asks for (unit_triggers.txt:502).
# Beside the Tuareg formable's salt cure grounds (nd_trg_salt_cure_grounds,
# arid salt or livestock land, pack and breeding herds) this stays the
# highland stud of riding camels: another place and another animal.
nd_air_mehari_stud = {
	is_special = yes
	is_foreign = no
	pop_type = tribesmen
	max_levels = 1
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_air_camel_mastery
	}

	location_potential = {
		raw_material = goods:livestock
		OR = {
			topography = hills
			topography = plateau
			topography = mountains
		}
	}

	unique_production_methods = {
		nd_air_camel_breeding = {
			millet = 0.4
			salt = 0.05
			produced = camels
			output = 0.24
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}
