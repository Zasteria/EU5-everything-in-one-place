##############################
# Tuareg (TRG): Custom Building Types
#
# Two decisions run through this whole file and are deliberate.
#
# 1. pop_type = tribesmen. Every one of the confederation's eleven locations
#    starts with tribesmen and next to nothing else
#    (main_menu/setup/1337/06_pops.txt), so a building staffed by burghers
#    or laborers would stand empty in the country it was written for.
#    Vanilla staffs buildings with tribesmen too (dun_fort, warrior_lodge,
#    safaviyya_order_hall). It also gives the formation choice teeth: a
#    country that settles its tribes (global_tribal_promotion) drains these
#    buildings of workers, one that keeps them nomad keeps them staffed.
#
# 2. rural_settlement = yes. None of the eleven locations is a town in 1337
#    (07_cities_and_buildings.txt), so a town-only building could not be
#    built at all. Reach is held down another way: every building here is
#    bound to the capital, to a raw good, to the desert, or to locations
#    where Tuareg culture is dominant.
##############################

### Court of the Drum: the amenokal's camp, where the ettebel is kept ###
# All four settlement tiers on purpose: the building follows the capital,
# whatever the capital is. One per country, removed when the capital moves.
nd_trg_drum_court = {
	is_special = yes
	is_foreign = no
	pop_type = tribesmen
	max_levels = 1
	employment_size = stockade_employment
	category = government_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = government_build_time

	country_potential = {
		has_advance = nd_trg_voice_of_the_drum
	}

	location_potential = {
		counts_as_capital = yes
	}

	allow = {
		is_capital = yes
		country_has_no_other_copy_of_building = { type = nd_trg_drum_court }
	}

	remove_if = {
		is_no_longer_capital = yes
	}

	# A country has one government power pool
	# (government_types/00_default.txt), so all five are granted side by side
	# as vanilla does (advances/country_tre.txt:109-113): a tribe is paid in
	# cohesion, Timbuktu's Tuareg monarchy in legitimacy, a Tuareg horde in
	# unity (vanilla's own capital building for a horde pays it the same way,
	# building_types/capital_buildings.txt:739).
	capital_country_modifier = {
		monthly_tribal_cohesion = 0.05
		monthly_legitimacy = 0.05
		monthly_republican_tradition = 0.05
		monthly_devotion = 0.05
		monthly_horde_unity = 0.05
		tribes_estate_target_satisfaction = small_permanent_target_satisfaction
		# The court's own grant (2026-10-02): the four older powers are also
		# what vanilla's trg_tin_hinan_legacy pays (advances/country_tle.txt:
		# 326-329). The host gathers at the drum, so asking the tribes for
		# men (generic_actions/estate_emergency_actions.txt:771, price
		# ask_tribes_for_manpower) costs less. Vanilla grants no modifier on
		# this price; its other price modifiers run from -0.1 to -0.25.
		ask_tribes_for_manpower_cost_modifier = -0.25
	}

	# The host musters where the drum is beaten: without this a country of
	# rural camps has nowhere to recruit the Agher Shieldmen.
	modifier = {
		can_recruit_regiment_in_this_location = yes
	}

	# 1.25 gold a month at base prices (livestock 1.5, millet 1.0, leather and
	# cloth 3.0, goods/*.txt). Its vanilla peers cost 1.95 (chieftain_hall,
	# capital_buildings.txt:805-814) and 1.30 (council_hall.txt:42); it read
	# 0.45 until 2026-10-02, when the owner ruled 1.0 to 1.3.
	unique_production_methods = {
		nd_trg_drum_court_upkeep = {
			livestock = 0.2
			millet = 0.2
			leather = 0.1
			cloth = 0.15
			category = building_maintenance
		}
	}
	construction_demand = stockade_construction
}

# ===== DESTINY BUILDINGS =====
#
# The two capstones answer the same fact of a Saharan realm, that the
# capital's reach dies in the sand long before it gets to the next town, in
# the two opposite ways the formation choice describes. Each is a small
# second source of proximity (20, the value of vanilla's seljuk_mint; a
# local_governor gives 80), and neither can stand in the capital.

### Camp Beyond the Walls: Path A capstone ###
# How Akil ag Malwal held Timbuktu from 1433: a camp outside a town that
# keeps its own governor and judges. Towns and cities of OTHER peoples only
# (the Tuareg's own towns are the other path's business), and staffed by
# tribesmen, so it works only where tribesmen live beside the town, as they
# do at Gao, Bamba, Kabara and Timbuktu in the 1337 setup, and only for a
# country that has kept its tribes in the tent. No region bound: the custom
# travels with the host. The tribes, not the crown, are the ones who grow
# strong on it.
nd_trg_dest_camp_beyond_the_walls = {
	is_special = yes
	is_foreign = no
	pop_type = tribesmen
	max_levels = 1
	employment_size = stockade_employment
	category = government_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = government_build_time

	country_potential = {
		has_advance = nd_trg_dest_tribute_of_the_pashas
	}

	location_potential = {
		is_capital = no
		NOT = { dominant_culture = culture:tuareg }
	}

	modifier = {
		local_proximity_source = 20
		local_unrest = -0.05
		local_tribes_estate_power = 0.25
	}

	unique_production_methods = {
		nd_trg_dest_camp_upkeep = {
			livestock = 0.2
			cloth = 0.134
			leather = 0.067
			weaponry = 0.067
			category = building_maintenance
		}
	}
	construction_demand = stockade_construction
}
