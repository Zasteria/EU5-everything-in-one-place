# Gefail: the forge of the gof, one of the three privileged crafts of
# Brythonic law. A plain producer with no modifier block, so it takes the
# level formula of vanilla's guilds.
nd_hen_gefail = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = weapons_industry_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_hen_brythonic_smiths
	}

	# No location gate: it makes nothing out of the land. The iron it
	# consumes has to be bought, and that is its limit (vanilla's weapon
	# guild has none either).
	unique_production_methods = {
		nd_hen_forge_work = {
			iron = 0.3
			coal = 0.1
			lumber = 0.1
			produced = weaponry
			output = 0.5
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_hen_northern_industry: nails, edge tools and
		# hardware (vanilla's iron foundry makes tools from iron and coal).
		nd_hen_ironmongery = {
			iron = 0.5
			coal = 0.15
			produced = tools
			output = 0.72
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Barcdy: the bark house. Hides lie for a year in pits of oak bark and

	pop_type = laborers
	max_levels = guild_max_level
	employment_size = generic_peasant_building_employment

	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_hen_clyde_trade
	}

	# Tan pits are dug beside running water, and the river carries the
	# leather to market.
	location_potential = {
		has_river = yes
	}

	unique_production_methods = {

		# countryside, so the method names the rural margin.

			lumber = 0.1
			tools = 0.05
			produced = leather
			output = 0.36
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction

# Bole Hill: a smelting hearth on a windy brow of the Pennine and Lakeland

	is_foreign = no

	max_levels = rural_building_cap
	employment_size = generic_peasant_building_employment

	# A hearth on open moor: rural settlements only.
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_hen_pennine_mines
	}

	# Only on the ore field itself: a bole smelts the lead the land
	# yields, it does not make lead where there is none.
	location_potential = {
		raw_material = goods:lead

		# Brushwood and the tools of the groove are what the work consumes.

			lumber = 0.3
			tools = 0.05
			produced = lead
			output = 0.36
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
