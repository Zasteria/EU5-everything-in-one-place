### Age 4: Imperial Fortress (Reichsfestung; the key keeps the older name) ###
# The Empire's own fortresses on the Rhine (Philippsburg, Kehl), garrisoned by
# the Circles. Re-themed 2026-10-02: as an arsenal it told of the same thing as
# vanilla's armory and hre_imperial_armory. Towns and cities only, one level:
# a fortress town, not a village work.
nd_hre_zeughaus = {
	is_special = yes
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	employment_size = castle_employment
	category = military_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = soldier_building

	country_potential = {
		has_advance = nd_hre_zeughaus
	}

	modifier = {
		local_manpower = 0.005
		local_garrison_size = 0.1
		can_recruit_regiment_in_this_location = yes
	}

	unique_production_methods = {
		nd_hre_arsenal_maintenance = {
			iron = 0.2
			weaponry = 0.1
			saltpeter = 0.05
			category = building_maintenance
		}

		# Unlocked by nd_hre_metallverarbeitung (age 4): the fortress's own
		# forges. Chosen instead of the plain upkeep, the fortress makes arms
		# for the market. Inputs cost 0.9 + 0.2 + 0.15 = 1.25; 0.5 weaponry
		# at 3 is 1.5, the guild margin of 20%.
		nd_hre_metalworks = {
			iron = 0.3
			coal = 0.1
			lumber = 0.1
			produced = weaponry
			output = 0.5
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		# Unlocked by nd_hre_manufaktur (age 6): the gun manufactory beside
		# the fortress (Potsdam and Spandau from 1722). It stood in the fair
		# grounds as an upkeep bill of iron and cloth until 2026-10-02.
		# Inputs cost 1.2 + 0.6 + 0.15 = 1.95; 0.78 firearms at 3 is 2.34.
		nd_hre_manufaktur_pm = {
			iron = 0.4
			tools = 0.2
			lumber = 0.1
			produced = firearms
			output = 0.78
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = soldier_building_construction
}
