# Ivory Workshop (age 1, nd_kws_ivory_trade; a second method in age 5)
# Tusks came down from the interior to the coastal towns and were sawn,
# turned and carved there: the side-blown siwa horns of the Swahili courts,
# combs, bangles, and hilts for the daggers of Arabia.
# Ivory carving was a method of the funduq until 2026-10-01, written as
# upkeep (game, dyes and tools in, nothing out). It is a workshop's
# production now, with ivory as its input, as vanilla's jewelry guild has it.
# location_potential: ivory has to be produced in the location's market.
# Ivory is the input, so the market form of the gate is the right one: the
# carvers follow the tusks. Two locations of Kilwa's own territory yield it.
# Settlement: town and city. The carvers worked where the ships loaded.
nd_kws_ivory_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_kws_ivory_trade
	}

	location_potential = {
		market ?= {
			is_produced_in_market = goods:ivory
		}
	}

	unique_production_methods = {
		nd_kws_ivory_carving_pm = {
			ivory = 0.2
			tools = 0.05
			produced = jewelry
			output = 0.228
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_kws_revived_gold_routes (age 5): ivory set with
		# the gold of the southern routes.
		nd_kws_gold_mounted_ivory_pm = {
			ivory = 0.2
			goods_gold = 0.05
			tools = 0.05
			produced = jewelry
			output = 0.324
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
