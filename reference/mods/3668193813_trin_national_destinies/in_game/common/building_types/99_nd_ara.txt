# ============================================================================
# Aragon (ARA): custom buildings
# National Destinies mod
# ============================================================================
#
# What is NOT here, because somebody already built it:
#   vanilla: the sheep farm, the mason, the pottery and weapon guilds, the
#   printing shop, the library, the art school, irrigation and the pound lock
#   canal are open to everyone; the Justicia de Aragon is an estate
#   privilege, the Fuero de Aragon a town right, the Diputacion a tag advance.
#   the mod's Aragonese Empire, Catalonia, Spain and Al-Andalus: the Llotja,
#   the Generalitat Hall, the Almudin, the Drassanes, the Mesta sheepwalk,
#   the Farga, the Taula de Canvi, the Masia, the acequia, the lustreware
#   kiln of Al-Andalus.
#
# What is here is the inland kingdom's own ground: the meeting house of the
# village communities, the stock breeders' brotherhood, the Mudejar kiln,
# the saffron field, the alabaster carvers, the swordsmiths, the Justicia's
# prison, a collector's cabinet and Pignatelli's canal.
#
# Every production method is inline and was sized with eu5-profit (margin 20
# percent, cap written as a name).

# ----------------------------------------------------------------------------
# Casa de la Comunidad (age 1)
# ----------------------------------------------------------------------------
# The house where the villages of a community met in plega general, kept
# their chest and their charters, and mustered their companies. The
# community was the villages WITHOUT the town: Calatayud, Daroca, Teruel and
# Albarracin never belonged to the communities that bore their names.
# location_potential: Aragonese-majority locations. The communities were an
# institution of the kingdom's own southern march and of its fueros; a
# conquered Castilian or Catalan district has no plega to house.
# Settlement: countryside and town, never a city, for the reason above.
# No local_max_control on purpose: the villages governed themselves, the king
# gained soldiers and lost nothing, and the lords lost their say.
# Upkeep: 1.2 gold a month at base prices (paper 2, weaponry 3, leather 3,
# wheat 1), most of it the arms and rations of the companies the villages owed
# the frontier. Below vanilla's bailiff at 1.50 because it grants no control.
nd_ara_casa_de_la_comunidad = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = estate_peasants_employment
	category = government_category
	rural_settlement = yes
	town = yes
	city = no
	build_time = government_build_time

	country_potential = {
		has_advance = nd_ara_comunidades_de_aldeas
	}

	location_potential = {
		dominant_culture ?= culture:aragonese
	}

	modifier = {
		local_peasants_estate_power = 0.25
		local_nobles_estate_power = -0.25
		local_levy_size_modifier = 0.1
	}

	unique_production_methods = {
		nd_ara_casa_de_la_comunidad_maintenance = {
			paper = 0.15
			weaponry = 0.15
			leather = 0.1
			wheat = 0.15
			category = building_maintenance
		}
	}
	construction_demand = town_building_construction
}

# ----------------------------------------------------------------------------
# Ligallo (age 1)
# ----------------------------------------------------------------------------
# The stock breeders' brotherhood of an Aragonese town: it kept the brand
# book, returned strays, hired the shepherds for the summer drive to the
# Pyrenees and tried rustlers before its own judge. Zaragoza's was the Casa
# de Ganaderos; Teruel, Daroca, Calatayud and Caspe had ligallos.
# Not a second sheep farm: it produces nothing itself. It makes the flocks
# that are already there worth more.
# location_potential: the location's own raw material is wool or livestock.
# A brotherhood of breeders needs breeders.
# Settlement: countryside only, one per location. The flocks are out there.
# Upkeep: salt for the flocks (a breeder's first privilege was cheap salt),
# shears and hurdles.
nd_ara_ligallo = {
	is_special = yes
	is_foreign = no

	max_levels = 1
	employment_size = generic_peasant_building_employment
	category = rgo_building_category
	rural_settlement = yes
	town = no
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_ara_casa_de_ganaderos
	}

	location_potential = {
		OR = {
			raw_material = goods:wool
			raw_material = goods:livestock
		}
	}

	modifier = {
		local_wool_output_modifier = 0.15

# ----------------------------------------------------------------------------
# Alfar Mudejar (age 1; a second method in age 4)

# ----------------------------------------------------------------------------
# Carcel de Manifestados (age 4)
# ----------------------------------------------------------------------------
# The Justicia's prison at Zaragoza. A prisoner who had himself "manifested"
# was taken from the king's gaol and lodged here, where no royal officer
# could put him to the question, until a lawful sentence was produced.
# One in the country, in the capital: there was one Justicia and one such
# prison (the gate is vanilla's own capital-building idiom, art_school).
# Not gated on the vanilla privilege justicia_de_aragon: that privilege is
# unlocked by an event a country may never get or may refuse, and the
# building would then be a dead unlock.
# It is a bargain and not a gift: every estate rests easier, and the crown's
# own hand is weaker everywhere (global_crown_estate_power is negative on
# purpose).
# Upkeep 1.3 gold a month at base prices: the court's paper and law books,
# the prisoners' keep, the guard's arms.
nd_ara_carcel_de_manifestados = {
	is_special = yes
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	employment_size = tiny_soldier_employment
	category = government_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = government_build_time

	country_potential = {
		has_advance = nd_ara_derecho_de_manifestacion
	}

	location_potential = {
		counts_as_capital = yes
	}

	allow = {
		is_capital = yes
		country_has_no_other_copy_of_building = { type = nd_ara_carcel_de_manifestados }
	}

	remove_if = {
		is_no_longer_capital = yes
	}

	modifier = {
		local_unrest = -0.05
	}

	capital_country_modifier = {
		global_estate_target_satisfaction = small_permanent_target_satisfaction
		global_crown_estate_power = -0.05
	}

	unique_production_methods = {
		nd_ara_carcel_de_manifestados_maintenance = {
			paper = 0.2
			books = 0.1
			cloth = 0.1
			weaponry = 0.1
			category = building_maintenance
		}
	}
	construction_demand = town_building_construction
}

# Real Palacio de la Aljaferia (destiny, path A capstone)
# ----------------------------------------------------------------------------
# The palace of the Banu Hud outside the walls of Zaragoza (most of it built
# between 1065 and 1081), the residence of the kings of Aragon from the
# conquest of 1118. The coronation procession to La Seo set out from it.
# Peter the Ceremonious enlarged it from 1336 and built the chapel of San
# Martin; Mudejar masters raised a second palace over the first for the
# Catholic Monarchs between 1488 and 1495, with the throne room.
# Not the heritage tree's Carcel de Manifestados (the Justicia's prison, in
# whatever location is the capital) and not vanilla's lieutenancy (which may
# not stand in a capital at all): this is the king's own house, and there is
# one Aljaferia.
# location_potential: Zaragoza and nowhere else, the idiom vanilla uses for
# its Moscow and Constantinople buildings.
# Settlement: Zaragoza is a town in 1337 and the stage asks it to grow, so
# every urban rank is open; the countryside is not, a palace needs a court.
# Values: crown power as vanilla's tax assessor (0.25), tradition at half its
# opera house (0.5), garrison as its stockade (0.1): the Aljaferia was a
# fortress before it was a palace and a barracks after.

	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	employment_size = normal_noble_employment
	category = government_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = large_capital_build_time

	country_potential = {
		has_advance = nd_ara_dest_aragon_y_pamplona
	}
