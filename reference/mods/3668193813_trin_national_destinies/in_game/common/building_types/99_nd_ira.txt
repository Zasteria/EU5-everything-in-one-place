# Age 2: Talambar, the raised, thatched silkworm rearing shed of Gilan and
# Mazandaran. A plain producer with no modifier block, so it takes the rural
# level formula as vanilla's sheep farms do.
nd_ira_talambar = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = rural_building_cap
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	# Rural only: the worms are reared beside the mulberry groves, by the
	# households that tend them.
	rural_settlement = yes
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_ira_caspian_silk
	}

	location_potential = {
		is_produced_in_location_market = goods:silk
	}

	unique_production_methods = {
		# Vanilla's own silk recipe is fiber crops (silk_production_guild);
		# the lumber is the shed and its trays.
		nd_ira_silkworm_rearing = {
			fiber_crops = 0.4
			lumber = 0.1
			produced = silk
			output = 0.285
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}
