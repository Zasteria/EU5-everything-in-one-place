		has_advance = nd_bno_landak_diamond_fields
	}

	modifier = {
		local_raw_material_output = 0.25
		local_monthly_prosperity = 0.001
		local_burghers_estate_power = 0.5
	}

	# Borneo's diamonds are washed from the same river gravels as its gold:
	# Landak (Ngabang), Sambas, Cempaka by Martapura. The map gives those
	# gravels as gold and gives the island no gems at all, so a gate on gems
	# in the market left the workshop with nowhere to stand in Borneo
	# (register row P131). It stands where the location's own raw material
	# is gold.
	location_potential = {
		raw_material = goods:goods_gold
	}

	unique_production_methods = {
		nd_bno_alluvial_mining = {
			lumber = 0.15
			pottery = 0.1
			produced = gems
			output = 0.0975
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		# Sized to the rural margin like the other two methods. It stood at
		# output 0.15 with no profit cap, six times its cost, while the
		# building could not be built; now that it can be, that is a mint.
		nd_bno_gold_panning_pm = {
			lumber = 0.2
			tools = 0.1
			pottery = 0.05
			produced = goods_gold
			output = 0.0975
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		nd_bno_antimony_smelting_pm = {
			lumber = 0.15
			copper = 0.1
			produced = tools
			output = 0.21
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = guild_construction
