# ============================================================================
# Mallorca (MLL): custom buildings
# National Destinies mod
# ============================================================================
#
# What is NOT here, because somebody already built it:
#   vanilla: Sa Llotja de Palma (a monument of the tag), the salt collector
#   (the pans themselves), the wharf and the dock ladder, the lazaretto, the
#   minting office, the monastery, the university, the coastal fort, the
#   fortress church and the distillers' guild.
#   the Aragonese Empire (99_nd_cda.txt): the Drassanes, the Casa Llotja de
#   Mar, the almadraba and the vineyard.
#   Catalonia (99_nd_cat.txt): the brandy still (fassina), the woollen
#   workshop, the glass furnace, the Taula de Canvi.
#   Valencia (99_nd_val.txt): the watchtower, the snow well, the sugar mill.
#
# What is here belongs to the islands: the college of Arabic that Llull
# asked a prince for, the house of the villages' syndics, the salt quay of
# Ibiza, the signal tower of Portopi, the oil press of the mountain estates,
# the yard in the harbour of Mahon, the gin still and the loom of tongues.
#
# Every production method is inline, names the margin of its tier, and was
# sized with eu5-profit (cap written as a name, never switched off).

# ----------------------------------------------------------------------------
# Col.legi de Llengues (age 1)
# ----------------------------------------------------------------------------
# A house of friars who study the language of the people they mean to
# convert. Miramar, endowed in 1276 with five hundred florins a year, kept
# thirteen Franciscans at their Arabic on a cliff above the sea; the Council
# of Vienne (1311) took up Llull's plea and ordered chairs of oriental
# tongues in the great schools.
# Not vanilla's monastery (clergy literacy and conversion at a cost in food):
# this is a school with one subject. It converts as a monastery does and
# draws new learning as a small library does, and teaches the parish nothing.
# location_potential: the shore. The friars trained here to cross the sea,
# and Miramar looked out on the water they would sail.
# Settlement: countryside only. Llull wanted his students away from towns.
# Upkeep 0.95 at base prices (paper 2, books 3, cloth 3, beeswax 2) against
# vanilla's madrasa at 0.80.
nd_mll_collegi_de_llengues = {
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = 1
	employment_size = religious_building_employment
	category = religious_category
	rural_settlement = yes
	town = no
	city = no
	expensive = yes
	build_time = religious_building_time

	country_potential = {
		has_advance = nd_mll_art_de_ramon_llull
	}

	location_potential = {
		is_coastal = yes
	}

	modifier = {
		local_pop_conversion_speed_modifier = 0.1
		local_institution_growth_modifier = 0.1
	}

	unique_production_methods = {
		nd_mll_collegi_de_llengues_maintenance = {
			paper = 0.2
			books = 0.05
			cloth = 0.1
			beeswax = 0.05
			category = building_maintenance
		}
	}
	construction_demand = school_construction
}

# ----------------------------------------------------------------------------
# Casa del Sindicat (age 1)
# ----------------------------------------------------------------------------
# The house where the syndics of the villages meet: ten chosen each
# Christmas, two of them holding the keys of the common chest. They sat by
# turns at Inca and at Sineu, apportioned the kingdom's taxes among the
# villages, and spoke for the countryside against the City.
# No location_potential on purpose: the settlement flags are the gate. Towns
# only. The syndics met in the market towns of the plain; a village was too
# small to host them and the City was the other party.
# It gives the peasants a voice and makes the levy of taxes orderly. It is
# not vanilla's council hall (the Crown's seat in a town): control here
# rises slowly and by consent, and the Crown's estate gains nothing.
# Upkeep 0.95 at base prices (paper 2, books 3, cloth 3) against vanilla's
# early council hall at 0.55 and its council hall at 1.30: the registers of
# every village's assessment, and the syndics' livery.

# ----------------------------------------------------------------------------
# Carregador de Sal (age 1)
# ----------------------------------------------------------------------------
# The loading quay of the salt pans: a stone mole, the heaped white hills
# drying beside it, mules and baskets, and a tower to cover the ships while
# they load. After 1235 the pans of Ibiza belonged to the islanders in
# common and their salt paid for the island's walls and its grain.
# Not the pans: vanilla's salt collector makes the salt. This is where it
# is weighed, sold and put aboard, and a place with a quay sells more of it.
# location_potential: the location's own raw material must be salt, and it
# must lie on the sea (is_coastal: a quay, not a harbour). In the kingdom
# that is Ibiza and nowhere else.
# Settlement: countryside and town, never a city. The quay stood at the
# pans, a league from the walls.
# Upkeep 0.40 at base prices (lumber 1.5, tools 3, tar 2) against vanilla's
# sawmill at 0.15, which gives the same output bonus and no trade.
nd_mll_carregador_de_sal = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = rural_peasant_enhance_employment

	rural_settlement = yes
	town = yes
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_mll_sal_d_eivissa
	}

	location_potential = {
		raw_material = goods:salt
		is_coastal = yes
	}

	modifier = {
		local_salt_output_modifier = 0.2
		local_merchant_capacity = 0.25

	unique_production_methods = {

# ----------------------------------------------------------------------------
# Tafona (age 4)
# ----------------------------------------------------------------------------
# The oil press of a mountain estate: a stone trough where a mule turns the
# millstone over the olives, a beam press as long as the room, and a row of
# settling jars. Every possessio of the Serra had one, and from the
# sixteenth century to the nineteenth their oil was most of what the island
# sold abroad.
# location_potential: hill or mountain ground, for the terraces, AND olives
# reaching the local market. The market form of the gate is used because no
# location of the Balearics has olives as its own raw material on this map
# (Palma wheat, Manacor wine, Pollensa fiber crops, Ibiza salt, Menorca
# fish), so the strict form would leave the building unbuildable at home.
# In the kingdom the slope test leaves Pollensa, which is the Serra.
# Settlement: countryside and town, never a city. A press stood on the
# estate.
# The inputs are the upkeep of the work: firewood for the scalding water,
# the ironwork of the press and the jars. Vanilla's rural shape:
# building_maintenance with produced, output and the rural margin.
nd_mll_tafona = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	startup_ramp_target = rural_startup_ramp_target
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_mll_oli_de_la_serra
	}

	location_potential = {
		OR = {
			topography = mountains
			topography = hills
		}
		is_produced_in_location_market = goods:olives
	}

	unique_production_methods = {
		nd_mll_tafona_oli = {
			lumber = 0.2
			tools = 0.05
			pottery = 0.1
			produced = olives
			output = 0.66
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}

# Arsenal de Mao (age 5)
# ----------------------------------------------------------------------------
# The yard in the harbour of Mahon: careening wharves on the north shore, a
# mast house, stores of cordage and shot, and a hospital on the island in
# the middle of the roadstead. A fleet that winters here is at sea a month
# before one that winters anywhere else.
# location_potential: Menorca, and only Menorca. The building is one place,
# a harbour five kilometres long that no other island of the kingdom has
# (the map gives this location alone a full natural harbour). is_port is
# written as well, as for every yard.
# Settlement: towns and cities.
# It refits and builds faster than a dry dock and berths as many keels. It
# raises no sailors: an anchorage is not a nursery of seamen.

# ----------------------------------------------------------------------------
# Destil.leria de Gin (age 6)
# ----------------------------------------------------------------------------
# A gin house on the quay: copper stills over a wood fire, wine spirit run
# through juniper berries, and the result sold by the jar to the garrison
# and the fleet. The island grew no grain to spare, so its gin was made
# from wine.
# location_potential: a port, AND wine reaching the local market. The trade
# existed because ships' companies drank it, and the still needs wine to
# burn (market form: the product is a made good).
# Settlement: towns and cities. A plain producer with no modifier block, on
# the guild level formula as vanilla's distillers' guild.
# Not Catalonia's fassina, a country still among the vines that ships raw
# brandy: this is a town trade with a recipe.
nd_mll_destilleria_de_gin = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	startup_ramp_target = guild_startup_ramp_target
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_mll_gin_de_mao
	}

	location_potential = {
		is_port = yes
		is_produced_in_location_market = goods:wine
	}

	unique_production_methods = {
		# The medicaments are the juniper berries, the lumber the fire
		# under the still.
		nd_mll_gin_de_ginebro = {
			wine = 0.8
			medicaments = 0.1
			lumber = 0.1
			produced = liquor
			output = 0.888
			category = guild_input
			debug_max_profit = guild_profit_margin

	}
	construction_demand = guild_construction
}

# bruixoler.

# Values: maritime presence as vanilla's buildings grant it (0.15), burgher
# literacy as its chancery (2.5), merchant power at its commonest step (0.1).
