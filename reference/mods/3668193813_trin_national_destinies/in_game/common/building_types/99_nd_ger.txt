# Bergwerk: German mining complex (silver, copper, iron)
nd_ger_bergwerk = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = castle_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = soldier_building

	country_potential = {
		has_advance = nd_ger_mining_tradition
	}

	modifier = {
		local_silver_output_modifier = 0.15
		local_copper_output_modifier = 0.1
		local_iron_output_modifier = 0.1
		local_pop_promotion_speed = 0.002
	}

	# Extraction is bound to the resource: you cannot mine, tap or
	# harvest iron where there is none. Without this the building
	# is a free resource generator in every location the owner holds.
	location_potential = {
		is_produced_in_location_market = goods:iron
	}

	unique_production_methods = {
		nd_ger_bergwerk_maintenance = {
			lumber = 0.2
			tools = 0.1
			produced = iron
			output = 0.24
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		nd_ger_silver_processing = {
			stone = 0.3
			tools = 0.15
			produced = silver
			output = 0.15
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		# The Saiger process (liquation, Nuremberg, about 1450): silver-bearing
		# copper is melted with lead, the lead carries the silver off and is
		# then burned away. The Fugger fortune stood on these works. The silver
		# is in the copper ore to begin with, so this is refining a metal the
		# ore already holds, not turning one raw good into another.
		nd_ger_saigerhuette = {
			copper = 0.3
			lead = 0.15
			lumber = 0.1
			produced = silver
			output = 0.27
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		# Coke replaces charcoal in the blast furnace (Gleiwitz 1796, the Ruhr
		# from 1849): the furnace is fed from the coal seam, not the forest.
		nd_ger_kokshochofen = {
			coal = 0.3
			tools = 0.1
			produced = iron
			output = 0.36
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = soldier_building_construction
}
