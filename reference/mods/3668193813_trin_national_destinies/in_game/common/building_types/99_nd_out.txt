# ===== Outremer (OUT) Buildings =====
# Outremer is an administration, not a manufactory. Seven of these eight are
# governance, sanctuary, road, port and court, because the country's own
# formation clause makes it a Latin crown over a non-Latin majority: what it
# has to build is the apparatus for governing other people's communities,
# not a second industrial base. The one production building is the tiraz
# house, and that is in the set precisely because the tiraz band carried the
# ruler's name woven in Arabic.

# --- The Ra'is's House ---
# The ra'is was the village headman the Franks found already in place: same
# language, same faith, same family as the tenants, confirmed by the lord
# rather than appointed by him. The lord's dragoman and scribe sat opposite
# him. Frankish estate administration in Palestine ran through this pair and
# almost never through a resident Frankish lord.
nd_out_rais_house = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = bailiff_employment
	category = government_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_out_rais_and_dragoman
	}

	# Only where the village is not of the crown's own faith. Where it is,
	# there is nothing for a ra'is to mediate and the office historically
	# did not exist: the Frankish new towns were run by a dispensator
	# instead. OUT_f gates on religion = catholic, so naming the religion
	# here is exact rather than a guess about the builder.
	location_potential = {
		NOT = { dominant_religion = religion:catholic }
	}

	# The bailiff bargain, deliberately: control over a village you do not
	# staff, bought by handing the peasantry real standing and taking the
	# crown's own agents out of the loop.
	modifier = {
		local_max_control = 0.05
		local_peasants_estate_power = 1.0
		local_crown_estate_power = -0.25
		local_unrest = -0.05
	}

	unique_production_methods = {
		nd_out_rais_maintenance = {
			paper = 0.1
			wheat = 0.1
			olives = 0.1
			category = building_maintenance
		}
	}
	construction_demand = village_construction
}

# --- Wardenship of the Sanctuary ---
# A crown officer posted at a shrine of a faith the emperor does not profess,
# to keep it open, keep it safe and keep the missionaries out. The realm
# contains the Kaaba, the Prophet's Mosque, the Holy Sepulchre's Greek and
# Armenian chapels, the Coptic patriarchate, the catholicos of the Church of
# the East at Baghdad, the Samaritan community at Nablus and the Mandaeans
# of the lower Euphrates. This building is the office that guards them.
nd_out_sanctuary_wardenship = {
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = 1
	employment_size = religious_building_employment
	category = religious_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = religious_building_time

	country_potential = {
		has_advance = nd_out_wardens_of_the_sanctuaries
	}

	# The reach is vanilla's own holy site list, which is the tightest and
	# most honest gate available: a warden can only be posted where there is
	# a sanctuary to guard, and only where that sanctuary belongs to someone
	# else. Catholic sites are not listed because the crown's own church
	# needs no protection from the crown.
	location_potential = {
		OR = {
			is_location_holy_site_for = religion:sunni
			is_location_holy_site_for = religion:shia
			is_location_holy_site_for = religion:ibadi
			is_location_holy_site_for = religion:orthodox
			is_location_holy_site_for = religion:miaphysite
			is_location_holy_site_for = religion:nestorianism
			is_location_holy_site_for = religion:judaism
			is_location_holy_site_for = religion:druzism
			is_location_holy_site_for = religion:samaritanism
			is_location_holy_site_for = religion:mandeaism
		}
	}

	# The conversion block is the point, not an oversight. A warden who
	# permits a mission is not a warden. Building this is a permanent
	# decision to stop converting that location in exchange for its quiet.
	modifier = {
		local_unrest = -0.1
		local_clergy_estate_power = 0.5
		local_monthly_prosperity = 0.001
		local_clergy_conversion_blocked = yes
	}

	unique_production_methods = {
		nd_out_wardenship_maintenance = {
			incense = 0.1
			olives = 0.1
			paper = 0.1
			category = building_maintenance
		}
	}
	construction_demand = important_building_construction
}

# --- Court of the Fonde ---
# The market court of the Latin East, and the only tribunal in medieval
# Christendom with a mixed-confession bench: two Frankish jurors and four
# Syrian, hearing commercial pleas between people of different faiths. Its
# procedure was written down in the Livre des Assises at Acre between 1240
# and 1244, alongside the Cour des Bourgeois and the maritime Cour de la
# Chaine.
nd_out_cour_de_la_fonde = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = generic_burgher_employment
	category = government_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = government_build_time

	country_potential = {
		has_advance = nd_out_livre_des_assises
	}

	# The fonde was the court OF the market. Vanilla's own market buildings
	# use exactly this gate, and it keeps the reach to the handful of
	# locations that actually carry a market rather than every town.
	location_potential = {
		is_market_center = yes
	}

	remove_if = {
		location = { is_market_center = no }
	}

	modifier = {
		local_market_access = 0.005
		local_trades_per_burgher = 0.1
		local_burghers_estate_power = 0.5
		local_unrest = -0.05
		local_max_control = 0.05
	}

	unique_production_methods = {
		nd_out_fonde_maintenance = {
			paper = 0.1
			cloth = 0.1
			olives = 0.1
			category = building_maintenance
		}
		nd_out_fonde_registry = {
			paper = 0.2
			books = 0.1
			cloth = 0.1
			wine = 0.1
			category = building_maintenance
		}
	}
	construction_demand = market_center_building_construction
}

# --- Station of the Pilgrim Road ---
# The Darb al-Hajj was a chain of cisterns, forts and provision markets
# running from Cairo across Sinai and from Damascus down the Hejaz, kept up
# at state expense for pilgrims going to Mecca. An Outremer that holds both
# ends of that road is a Catholic empire paying to maintain the Muslim
# pilgrimage, which is exactly what its formation clause commits it to.
nd_out_hajj_station = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1

	category = infrastructure_category
	rural_settlement = yes

	city = no
	expensive = no
	build_time = infrastructure_build_time

	country_potential = {

	# Egyptian road over Sinai, the Hejaz itself where the Egyptian and

# --- The Yard at Ayla ---
# In 1182 Raynald of Chatillon had five galleys built in Kerak, taken to
# pieces, carried by hired Bedouin camels two hundred kilometres across the
# Negev to Ayla at the head of the Gulf of Aqaba, and reassembled on a sea
# no Frankish ship had ever entered. The empire makes a permanent
# institution of the trick.
nd_out_ayla_yard = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = dock_employment
	category = naval_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = medium_port_building_time

	country_potential = {
		has_advance = nd_out_ships_carried_by_camel
	}

	# A port, and only on the Red Sea shore of the realm. Ayla itself is in
	# the Levant area, so Aqaba is named directly rather than admitting the
	# whole Mediterranean coast, which already has every shipyard it needs.
	location_potential = {
		is_port = yes
		OR = {
			area = area:hedjaz_area
			area = area:sinai_area
			area = area:south_yemen_area
			area = area:upper_egypt_area
			this = location:aqaba
		}
	}

	modifier = {
		can_build_ships_in_this_location = yes
		local_ship_build_speed = 0.2
		# Engine cap: sailors per employed pop may not exceed 0.01875 here
		# (dock_employment 1); the 2026-09-10 boot log flagged 0.15.
		local_sailors = 0.01875
		max_ships_built_at_same_time = 1
	}

	unique_production_methods = {
		nd_out_ayla_maintenance = {
			lumber = 0.2
			tar = 0.1
			naval_supplies = 0.1
			category = building_maintenance
		}
	}
	construction_demand = port_building_construction
}

nd_out_karimi_house = {
	is_special = yes
	is_foreign = no

	category = trade_category

# --- The Tiraz House ---
# The Fatimid state weaving houses of Tinnis and Damietta produced the most
# valuable linen in the medieval Mediterranean, and every bolt carried a
# woven band naming the reigning ruler. The crusader states took the
# workshops over intact. A Latin emperor's tiraz names him in Arabic, which
# is the whole country in a single object.
nd_out_tiraz_house = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_out_tiraz_of_tinnis
	}

	# The tiraz towns stood where the flax fields and the dye works were:
	# the Egyptian Delta and the Syrian coast. Two areas, and the building
	# refines flax rather than conjuring it, so its cost gate is its inputs.
	location_potential = {
		OR = {
			area = area:lower_egypt_area
			area = area:levant_area
		}
	}

	modifier = {
		local_burghers_estate_power = 0.5
		local_max_literacy = 2
	}

	unique_production_methods = {
		nd_out_tiraz_linen = {
			fiber_crops = 0.35

			produced = fine_cloth
			output = 0.236
			category = guild_input

# --- The Chartered Quarter (Path A capstone) ---
# The Latin East already ran on quarters: the Venetians, Genoese and Pisans
# held walled compounds inside the crown's own ports with their own church,
# oven, bath, weights and bench. Under the charter of the nations the same
# instrument is issued to any community that comes under the compact, and the
# grant is registered in the Livre rather than bought with a fleet. Vanilla
# builds the identical object twice, as the jewish_district and the
# phanariot_district, so the shape is the game's own.
nd_out_dest_chartered_quarter = {
	is_special = yes
	is_foreign = no
	pop_type = burghers

	employment_size = generic_burgher_employment
	category = cultural_category
	rural_settlement = no
	town = yes
	city = yes

	expensive = no
	build_time = large_cultural_building_time

	country_potential = {
		has_advance = nd_out_dest_charter_of_the_nations
	}

	# Ports only, which is where every chartered quarter in the real Latin
	# East stood: Acre, Tyre, Beirut, Alexandria, Famagusta. A community that

		is_port = yes
	}

	modifier = {
