# Clachan - Celtic cooperative farming settlement (production building)
nd_cel_clachan = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = generic_peasant_building_employment
	category = basic_industry_category
	rural_settlement = yes
	town = no
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_cel_rundale_system
	}

	modifier = {
		local_monthly_food_modifier = 0.025
		local_population_growth = 0.0005
	}

	# Extraction is bound to the resource: you cannot mine, tap or
	# harvest wool where there is none. Without this the building
	# is a free resource generator in every location the owner holds.
	location_potential = {
		is_produced_in_location_market = goods:wool
	}

	unique_production_methods = {
		# Open from the day the clachan is built. The wool method below was
		# the building's only one and is locked until nd_cel_wool_trade
		# (an age later), which left the clachan with no method to run.
		nd_cel_rundale_upkeep = {
			tools = 0.05
			lumber = 0.05
			livestock = 0.05
			category = building_maintenance
		}
		# Unlocked by nd_cel_wool_trade.
		nd_cel_communal_farming = {
			livestock = 0.2
			produced = wool
			output = 0.144
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}

# Blowing House: the tinners' smelting house on a moorland stream, where
# black tin is blown to white with charcoal and a water-driven bellows.
# A plain producer with no modifier block, so it takes the level formula of
# vanilla's rural producers.
nd_cel_blowing_house = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = generic_peasant_building_employment
	category = rgo_building_category
	# A works of the open moor: rural settlements only.
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_cel_cornish_mining
	}

	# Only on the tin ground itself: the house smelts what the streamers
	# bring in, it does not make tin where there is none.
	location_potential = {
		raw_material = goods:tin
	}

	unique_production_methods = {
		# Charcoal and the tools of the stream works are what it consumes.
		nd_cel_tin_blowing = {
			lumber = 0.3
			tools = 0.05
			produced = tin
			output = 0.36
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
