# National Destinies: Dahomey (DAH) buildings

# Palm Oil Processing Works: Dahomey's economic transition under Ghezo
nd_dah_palm_oil_works = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	# Not rural: this is a workshop, not a field. A per-location
	# bonus buildable in everything you own is a realm-wide bonus
	# wearing a local label, and vanilla keeps industry in towns.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_dah_palm_oil_economy
	}

	# Medicaments is a raw good, so the works stand only where the market
	# already yields it (vanilla's market idiom, production_glass.txt:22).
	# On the map that is the coast itself: Hulagan beside Whydah, Ukunzu
	# in Benin and Djougou inland all yield medicaments.
	location_potential = {
		is_produced_in_location_market = goods:medicaments
	}

	modifier = {
		local_production_efficiency = tiny_production_efficiency_bonus
		local_pop_promotion_speed = 0.002
		local_burghers_estate_power = 0.25
	}

	unique_production_methods = {
		# The game has no oil good. Palm oil left Whydah as the fat of
		# soap and salves, and the one good vanilla lets a town workshop
		# make from plants is medicaments (apothecary,
		# pure_one_level_productions.txt:22-32), so that is what it is
		# sold as. It was written as liquor, which nothing here distils.
		nd_dah_palm_oil_processing = {
			fruit = 0.3125
			produced = medicaments
			output = 0.375
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
