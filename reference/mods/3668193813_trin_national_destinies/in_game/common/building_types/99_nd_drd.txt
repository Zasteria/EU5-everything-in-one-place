# Buaile (cattle fold): where the cattle rendered to the chief are gathered,
# the Martinmas beasts slaughtered and their hides salted and cured.
# A plain producer with no modifier block, so it takes the level formula of
# vanilla's rural producers.
nd_drd_cattle_fold = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = rural_building_cap
	employment_size = generic_peasant_building_employment
	category = consumer_goods_category
	# A fold of the open hill country: rural settlements only.
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_drd_cattle_wealth
	}

	# Only where the land itself raises cattle: the herds are the tribute.
	location_potential = {
		raw_material = goods:livestock
	}

	unique_production_methods = {
		nd_drd_cattle_tribute = {
			livestock = 0.4
			salt = 0.05
			produced = leather
			output = 0.32
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}

}
