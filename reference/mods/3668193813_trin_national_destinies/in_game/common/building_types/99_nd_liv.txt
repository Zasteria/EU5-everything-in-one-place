# Packhaus: a merchant's packing house on the quay
# The goods of the hinterland came down the Daugava as they were cut and
# pulled; they were made fit for the sea in the port: masts dressed and
# sorted by the sworn graders, pitch barrelled, flax heckled and spun, furs
# dressed and baled.
# location_potential: a port. The work is done where the sea ships load,
# so the house stands in Riga, Reval, Pernau or Windau and not inland.
# Settlement: town and above. A merchant's house, never a village works.
nd_liv_packhaus = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = basic_industry_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_liv_baltic_timber
	}

	location_potential = {
		is_port = yes
	}

	unique_production_methods = {
		# Masts, spars and pitch. Vanilla's coastal boatyard makes 0.5 from
		# lumber 0.5 and tar 0.25.
		nd_liv_timber_trade = {
			lumber = 0.6
			tar = 0.2
			produced = naval_supplies
			output = 0.52
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_liv_flax_industry (age 2). Vanilla's cloth guild
		# makes 0.8 from fiber_crops 1.
		nd_liv_flax_processing = {
			fiber_crops = 0.9
			tools = 0.05
			produced = cloth
			output = 0.78
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_liv_fur_trade (age 3). Vanilla's fur trim makes
		# 0.2 from fur 0.5; the salt cures the pelts for the voyage.
		nd_liv_fur_processing = {
			fur = 0.5
			salt = 0.05
			produced = fine_cloth
			output = 0.24
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
