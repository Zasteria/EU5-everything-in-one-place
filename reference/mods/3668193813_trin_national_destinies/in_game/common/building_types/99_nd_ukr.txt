# Brotherhood Press: the printing house of an Orthodox lay brotherhood
# (the Dormition Brotherhood of Lviv printed from 1591). Unlocked by
# nd_ukr_orthodox_brotherhoods.
# A plain producer with no modifier block; two levels so a town's press
# stays a brotherhood's shop and not an industry.
nd_ukr_brotherhood_press = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 2
	employment_size = guild_employment
	category = consumer_goods_category
	# A press needs a town's craftsmen: the same tiers as vanilla's scriptorium.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_ukr_orthodox_brotherhoods
	}

	# The presses were the brotherhoods' own and stood beside their schools.
	allow = {
		has_building = building_type:nd_ukr_brotherhood_school
	}

	unique_production_methods = {
		# Books are a produced good made from paper, as in vanilla's
		# scriptorium (paper, dyes, lumber) and printing shop (paper, lead).
		# Lead is the cast type, dyes the red ink of the church rubrics.
		nd_ukr_church_slavonic_printing = {
			paper = 0.2
			lead = 0.05
			dyes = 0.05
			produced = books
			output = 0.28
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
