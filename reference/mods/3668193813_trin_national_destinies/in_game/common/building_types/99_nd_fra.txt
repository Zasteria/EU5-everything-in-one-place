# ============================================================================
# France (FRA): custom buildings
# ============================================================================
# Every building carries a location_potential and a line saying why that
# reach is correct. Vanilla has no buildable France building at all
# (versailles and fra_louvre_museum are event-only, country_potential
# always = no), so none of these shadows a vanilla one.

# ----------------------------------------------------------------------------
# Fabrique of Lyon (age 3): silk weaving
# ----------------------------------------------------------------------------
# location_potential: Lyon and Tours only. Francis I gave Lyon the kingdom's
# raw silk monopoly in 1540; Tours had woven silk for the crown since Louis
# XI moved the looms there in 1470. Nowhere else held the privilege.
# Produces fine_cloth, a manufactured good, so no raw material gate applies.
nd_fra_fabrique_lyonnaise = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	startup_ramp_target = guild_startup_ramp_target
	category = consumer_goods_category
	employment_size = guild_employment
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_fra_soieries_de_lyon
	}

	location_potential = {
		OR = {
			this = location:lyon
			this = location:tours
		}
	}

	unique_production_methods = {
		nd_fra_pm_soie_lyonnaise = {
			silk = 0.875
			produced = fine_cloth
			output = 0.7
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# The 1536 patent chartered "cloths of gold, silver and silk".
		nd_fra_pm_drap_d_or = {
			silk = 0.7
			silver = 0.1
			produced = fine_cloth
			output = 0.68
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
	custom_tags = { guild }
}

# ----------------------------------------------------------------------------
# Magnanerie (age 4): silkworm house
# ----------------------------------------------------------------------------
# Produces raw silk, so it is bound to the good (strict idiom, as vanilla
# glass_guild binds sand) AND to the silk country Henri IV's mulberry edicts
# built: the Languedoc (Cevennes, Vivarais, Gevaudan, Nimois), Provence and
# the Lyonnais. In 1.3.11 location_templates.txt that admits Florac and Ales
# in the Cevennes, Fenouillet, Tarascon and Lyon, five locations inside
# FRA_f's required region. The area clause keeps it French sericulture, not a
# silk booster for Genoa or Florence after a conquest. Rural and town tiers
# only: a silkworm house is a farm building, not a city one.
# fiber_crops stands in for mulberry leaf, as in vanilla's
# lucca_silk_production_guild (event_only_buildings.txt), at the same 0.6.
nd_fra_magnanerie = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = guild_max_level
	startup_ramp_target = rural_startup_ramp_target
	category = rgo_building_category
	employment_size = advanced_peasant_building_employment
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_fra_meuriers_du_roi
	}

	location_potential = {
		is_produced_in_location_market = goods:silk
		OR = {
			area = area:languedoc_area
			area = area:provence_area
			area = area:lyonnais_area
		}
	}

	unique_production_methods = {
		nd_fra_pm_magnanerie = {
			fiber_crops = 1.0
			produced = silk
			output = 0.6
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = village_construction
}

# list is the limit. max_levels = 1.

	max_levels = 1

		has_advance = nd_fra_dest_seconde_guerre_de_cent_ans

	location_potential = {
