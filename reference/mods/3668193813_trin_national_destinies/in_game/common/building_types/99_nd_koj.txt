# Pilgrim Workshop: the carvers who make rosaries, crosses and small models
# of the Holy Sepulchre from olive wood and mother of pearl, the craft the
# friars of the Custody taught at Bethlehem. A plain producer with no
# modifier block, in the shape of vanilla's guilds. Unlocked by
# nd_koj_holy_land_relics.
nd_koj_pilgrim_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = consumer_goods_category
	# A craft of the towns the pilgrims pass through: vanilla's guild tiers.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_koj_holy_land_relics
	}

	# The wares are worth their price because they come from the Holy Land:
	# only in the province of Jerusalem (vanilla gates a building on a
	# province the same way).
	location_potential = {
		province_definition = province_definition:filastin_province
	}

	unique_production_methods = {
		# Jewelry is a produced good. Vanilla makes it from pearls, stone,
		# silver and ivory; here the lumber is olive wood and the pearls
		# the shell that is cut for inlay.
		nd_koj_relic_trade = {
			lumber = 0.2
			pearls = 0.1
			produced = jewelry
			output = 0.168
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Sugar Press: Crusader sugar processing mill in the Levant
nd_koj_sugar_press = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = plantation_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = plantation_build_time

	country_potential = {
		has_advance = nd_koj_sugar_estates
	}

	modifier = {
		local_production_efficiency = small_production_efficiency_bonus
		local_monthly_food_modifier = 0.025
	}

	# Two gates, both load-bearing. Market supply: the press needs
	# cane, and Acre's own hinterland qualifies (acre's raw_material
	# is sugar in location_templates.txt), so the gate binds without
	# conjuring the good anywhere the owner holds. River: the
	# description promises a water-powered mill and the real Crusader
	# presses (Tawahin as-Sukkar by Jericho, the Acre and Tyre works)
	# ran on channeled water; without this the press was buildable on
	# any dry hilltop (workshop report, 2026-08-09).
	location_potential = {
		has_river = yes
		market ?= {
			is_produced_in_market = goods:sugar
		}
	}

	unique_production_methods = {
		nd_koj_mill_upkeep = {
			lumber = 0.1
			tools = 0.05
			stone = 0.05
			category = building_maintenance
		}
		nd_koj_sugar_processing = {
			lumber = 0.1
			tools = 0.05
			produced = sugar
			output = 0.12
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = plantation_construction
}
