# National Destinies mod

#
# What is NOT here, because somebody already built it:
#   vanilla: the Llotja de la Seda (event building, flavor_ara.106), the
#   irrigation system and the saquiyah advance that deepens it, the hospital,
#   the coastal fort, the salt collector, and the silk the vanilla VAL advance
#   of age 4 is about.
#   the Aragonese Empire (99_nd_cda.txt): the Almudin of Valencia, the
#   Drassanes, the Casa Llotja de Mar.
#   Catalonia (99_nd_cat.txt): the municipal Taula de Canvi, the paper mill,
#   the glass furnace, the woollen workshop.
#
# What is here is the kingdom's own ground: the irrigators' commune and its
# court, the commandery of Montesa, the board of walls and ditches, the
# potteries of Manises, the madhouse the merchants founded, the sugar mill of
# the Safor, the watchtower of the coast, the snow well, the faience works of
# l'Alcora, the hemp walk of the Plana and the mills of Alcoi.
#
# Every production method is inline, names the margin of its tier, and was
# sized with eu5-profit (cap written as a name, never switched off).

# ----------------------------------------------------------------------------
# Comuna de Regants (age 1)
# ----------------------------------------------------------------------------
# The irrigators of one canal as a body: they elect a syndic, keep the weir
# and the channels at their own cost, share the water by turn, and send the
# syndic to sit with the others as the Tribunal de les Aigues. A quarrel over
# water is judged within the week by men who farm the same ditch.
# Not the canal: that is vanilla's irrigation system, and the commune cannot
# exist without one (allow). It adds no food of its own. What it adds is that
# the peasants run the water themselves and do not fight over it.
# location_potential: the same river or lake vanilla's irrigation asks for.
# Settlement: countryside and town, never a city. The communes were bodies of
# farmers; the city of Valencia sat beside the Horta, not in it.
# Upkeep 0.40 at base prices (tools 3, lumber 1.5, stone 1) against vanilla's
# irrigation system at 0.30: the yearly cleaning of the channels and the
# repair of the weir.
nd_val_comuna_de_regants = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = estate_peasants_employment
	category = infrastructure_category
	rural_settlement = yes
	town = yes
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_val_tribunal_de_les_aigues
	}

	location_potential = {
		OR = {
			has_river = yes
			is_adjacent_to_lake = yes
		}
	}

	allow = {
		custom_tooltip = {
			text = nd_val_comuna_needs_canal_tt
			has_building = building_type:irrigation_systems
		}
	}

	modifier = {
		local_unrest = -0.05
		local_peasants_estate_power = 0.2
	}

	unique_production_methods = {
		nd_val_comuna_de_regants_maintenance = {
			tools = 0.05
			lumber = 0.1
			stone = 0.1
			category = building_maintenance
		}
	}
	construction_demand = village_construction
}
