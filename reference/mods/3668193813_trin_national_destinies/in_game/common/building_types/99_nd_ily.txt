##############################################################################
# Illyria (ILY): Custom Buildings
#
# Ten institutions (four of them bound to named places and written below
# the first six), and every one of them is a place where the Illyrian idea
# was actually made: the village where Mass was said in Slavonic, the toll
# station on the road down to the sea, the chapter press at Senj, the walled
# cathedral precinct, the college that made priests for the Ottoman lands,
# and the blockhouse on the cordon.
#
# Every location_potential below is a real restriction with a stated reason.
# Illyria cannot carpet its territory with any of these.
##############################################################################

# GLAGOLITIC PARISH (Glagoljaska zupa)
# Innocent IV licensed the Slavonic liturgy to the bishop of Senj in 1248.
# The glagoljasi who used it were village priests, poorer and far more
# numerous than the Latin canons, and they are the reason a Croatian
# peasant could follow his own Mass.
nd_ily_glagoljaska_zupa = {
	audio_tier = 1
	audio_category = religion
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = 1
	employment_size = religious_building_employment
	category = religious_category

	# RURAL AND SMALL-TOWN ONLY, on purpose. The Slavonic rite lived in the
	# countryside among married village priests; the cathedral cities kept
	# Latin and their own chapters, which is what nd_ily_kaptolska_utvrda
	# represents. City and megalopolis are therefore both excluded, and the
	# parish stays the cheap, numerous, rural institution it actually was.
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no

	build_time = religious_building_time
	graphical_tags = { small_religious }

	country_potential = {
		has_advance = nd_ily_glagoljasi
	}

	# The privilege of 1248 covers the Slavonic tongue, so it can only be
	# exercised where the congregation actually speaks it. A Romance
	# Dalmatian, Italian, Hungarian or German location hears Latin and gets
	# no parish of this kind.
	location_potential = {
		dominant_culture ?= culture:croatian
	}

	modifier = {
		local_clergy_desired_pop = 0.03
		local_peasants_max_literacy = 1
		local_cultural_tradition = 0.2
		local_unrest = -0.05
	}

	unique_production_methods = {
		nd_ily_zupa_maintenance = {
			wine = 0.1
			beeswax = 0.1
			paper = 0.1
			cloth = 0.05
			category = building_maintenance
		}
	}

	construction_demand = missionary_construction
}

# GLAGOLITIC PRESS (Glagoljska tiskara)
# Blaz Baromic learned the trade in Venice and brought it back to Senj,
# where the chapter's press ran from 1494 to 1508. Simun Kozicic Benja set
# up a second at Rijeka in 1530. Chapter-owned, small, and printing a
# liturgy nobody else in Europe was allowed to print.
nd_ily_glagoljska_tiskara = {
	audio_tier = 2
	audio_category = culture
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = guild_max_level
	startup_ramp_target = guild_startup_ramp_target
	employment_size = religious_building_employment
	category = consumer_goods_category

	# Urban only: a press needs a chapter, a bindery and buyers.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	build_time = guild_build_time
	graphical_tags = { small_religious }

	country_potential = {
		has_advance = nd_ily_senjska_tiskara
	}

	# A press consumes paper by the ream. It can only stand where paper
	# actually reaches the local market, which is the vanilla idiom for an
	# input-hungry workshop and stops the press from conjuring books in a
	# karst village.
	location_potential = {
		is_produced_in_location_market = goods:paper
	}

	modifier = {
		local_monthly_literacy = 0.05
		local_clergy_max_literacy = 1
	}

	unique_production_methods = {

			leather = 0.1
			produced = books
			output = 0.32
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Judita and what followed it: verse and history for laymen.
		nd_ily_pm_pucka_knjiga = {
			paper = 0.3
			dyes = 0.05
			produced = books
			output = 0.32
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Kasic's grammar was set in Roman letters so that a Jesuit in
		# Rome could read Illyrian without learning a second alphabet.
		nd_ily_pm_latinicka_slova = {
			paper = 0.3
			lead = 0.1
			tools = 0.05
			produced = books
			output = 0.38
			category = guild_input
			debug_max_profit = guild_profit_margin

			paper = 0.35
			lead = 0.1
			leather = 0.1
			beeswax = 0.05

			output = 0.52

		}
