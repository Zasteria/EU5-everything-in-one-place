# Abrisham Khaneh: Silk Workshop
# Gilan's silk production facility

	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 3
	employment_size = guild_employment
	category = consumer_goods_category
	# Not rural: this is a workshop, not a field. A per-location
	# bonus buildable in everything you own is a realm-wide bonus
	# wearing a local label, and vanilla keeps industry in towns.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_kry_caspian_sericulture
	}

	modifier = {
		local_silk_output_modifier = 0.25
		local_merchant_power = 0.05
		local_burghers_estate_power = 0.5
		local_pop_promotion_speed = 0.002
	}

	unique_production_methods = {
		nd_kry_silk_weaving = {
			dyes = 0.15
			silk = 0.1
			produced = fine_cloth
			output = 0.2
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		nd_kry_advanced_silk_processing = {
			dyes = 0.25
			silk = 0.2
			produced = fine_cloth
			output = 0.36
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_kry_industrial_silk (age 6): cocoons reeled over
		# steam-heated basins, so coal joins the bill.
		nd_kry_steam_filature = {
			silk = 0.4
			dyes = 0.3
			coal = 0.1
			produced = fine_cloth
			output = 0.6
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Shilat-khaneh: Caspian Fishery
nd_kry_shilat_khaneh = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 2
	employment_size = generic_peasant_building_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_kry_caspian_fisheries
	}

	modifier = {
		local_fish_output_modifier = 0.15
		local_monthly_food_modifier = 0.025
		local_laborers_desired_pop = 0.1
	}

	# Requires a real supply of fish: the formable's own
	# territory has no source of it, so this is bound to the
	# location's market rather than to the soil. Without a gate
	# the building conjures the good anywhere the owner holds.
	location_potential = {
		market ?= {
			is_produced_in_market = goods:fish
		}
	}

	unique_production_methods = {
		# The default method. The sturgeon method below was the fishery's
		# only one and is unlocked two ages after the building, which left
		# the building with nothing to run in between.
		nd_kry_lagoon_fishing = {
			salt = 0.05
			produced = fish
			output = 0.24
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
		# Unlocked by nd_kry_rice_fish_economy (age 5). Vanilla's fishing
		# village turns salt 0.0929 and tools 0.015 into fish 0.5.
		nd_kry_sturgeon_fishing = {
			salt = 0.1
			tools = 0.02
			produced = fish
			output = 0.552
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
