# Sardinia-Piedmont buildings. Two of them exist only on the island and two
# only in the mountains, because the kingdom's institutions were never shared
# between its halves: the Carta de Logu and the grain banks were Sardinian
# and stayed Sardinian, the toll houses and barrier forts were Alpine and
# had nothing to guard anywhere else.

# Domus de Logu: the judge's court house of a Sardinian village, where the
# majore read the Carta aloud in Sardinian so that the people bound by it
# could hear it. Never extended to the mainland, hence the island gate.

	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = small_peasant_employment
	category = government_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_sar_piedmont_domus_de_logu_tradition
	}

	# Island only: the Carta de Logu was the law of Sardinia and was never
	# in force on the mainland, where the Statuta Sabaudiae applied instead.
	location_potential = {
		area = area:sardinia_area
	}

	modifier = {
		local_unrest = -0.05
		local_peasants_estate_power = 0.5
		local_peasants_max_literacy = 1
	}

	unique_production_methods = {

			paper = 0.189
			lumber = 0.189
			beeswax = 0.095
			category = building_maintenance
		}
	}
	construction_demand = village_construction
}

# Filatoio: a water powered silk throwing mill. It does not weave. It twists
# raw silk into organzine warp thread, which Lyon and Spitalfields could not
# make for themselves and bought from Piedmont by weight.
nd_sar_piedmont_filatoio = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = mills_employment
	category = consumer_goods_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = mills_build_time

	country_potential = {
		has_advance = nd_sar_piedmont_filatoio_rosso
	}

	# Only where raw silk actually reaches the market, by local production or
	# by trade. A throwing mill consumes cocoon silk it cannot make, and
	# Piedmont itself grew none of it until the mulberry edicts, so the trade
	# branch is what lets the mills exist at all before that.
	location_potential = {
		OR = {
			is_produced_in_location_market = goods:silk
			market = { is_traded_in_market = goods:silk }
		}
	}

	modifier = {
		local_silk_output_modifier = 0.1
		local_laborers_desired_pop = 0.05
	}

	unique_production_methods = {
		nd_sar_piedmont_silk_throwing = {
			silk = 0.25
			produced = fine_cloth
			output = 0.2
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = mill_construction
}
