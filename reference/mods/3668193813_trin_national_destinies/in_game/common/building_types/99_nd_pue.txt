			copper = 0.15
			gems = 0.05

			output = 0.156
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		# Shown as Shell and Turquoise Inlay (age 5). Pueblo silverwork
		# began in the 1870s, after the game ends; shell (the game's pearls)
		# came up the old trails from the Gulf of California. Output from

			gems = 0.1
			pearls = 0.05
			produced = jewelry
			output = 0.144
			category = guild_input
			debug_max_profit = rural_profit_margin

# Horse Herd: the herds kept and bred after 1680. The horses the Spanish left
# behind were traded north and east from the pueblos, and the peoples of the
# plains were mounted within a generation.
nd_pue_horse_herd = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = generic_peasant_building_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_pue_horse_adoption
	}

	# A herd is bred from horses: it can be started only where they are already
	# raised or brought to market (vanilla's own gate for horse_breeders, with
	# trade allowed, since the Southwest has no horses of its own in 1337).
	location_potential = {
		market ?= {
			OR = {
				is_produced_in_market = goods:horses
				is_traded_in_market = goods:horses
			}
		}
	}

	unique_production_methods = {
		# vanilla's horse_breeders_maintenance with maize for wheat: breeding
		# stock and fodder in, more horses out. Sized by eu5-profit.
		nd_pue_horse_breeding = {
			maize = 0.8
			horses = 0.15
			produced = horses
			output = 0.5
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = horse_breeders_construction
}
