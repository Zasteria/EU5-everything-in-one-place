# Rangkhana: the dye house of the rangrez.
# Indian dyers fixed madder and indigo on cotton with mordants, so that the
# colour held through washing: the secret of the calicoes Europe could not
# copy for two centuries. The cloth is washed in running water between baths,
# so a dye house stands on a river (Sanganer, Ahmedabad on the Sabarmati,
# Burhanpur on the Tapti): has_river is the gate.
# Woven cloth in, dyed cloth out, with alum as the mordant, as vanilla's fine
# cloth guild takes cloth 0.8333 for 0.5 and alum in its second slot.
# Output from eu5-profit at the guild margin: cost 3.15, 0.63 fine cloth.
# Two levels and no modifier: a plain producer beside the vanilla guild. The
# Karkhana of age 4 is the court workshop with effects of its own; this is
# the trade of the riverbank.
nd_hin_rangkhana = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 2
	startup_ramp_target = guild_startup_ramp_target
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_hin_textile_mastery
	}

	location_potential = {
		has_river = yes
	}

	unique_production_methods = {
		nd_hin_textile_dyeing_pm = {
			cloth = 0.8
			dyes = 0.15
			alum = 0.05
			produced = fine_cloth
			output = 0.63
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
