# ============================================================================
# Galicia (GLC): custom buildings
# National Destinies mod
# ============================================================================
#
# What is NOT here, because somebody already built it:
#   vanilla: the Tower of Hercules (a monument at A Coruna), the cathedral,
#   the fishing village, the dock ladder from dock to naval base, the mason,
#   the winery, the jewelry guild, the stockade and the castle.
#   Leon (99_nd_lon.txt): the Frankish borough of the pilgrim road, the
#   raised granary, the linen works, the summer pasture.
#   Spain, Castile, Portugal, Valencia: the purveyor's yard, the royal salt
#   works, the king's pine wood, the hemp walk, the coast tower.
#
# What is here is the kingdom's own ground: the jet and badge workshop at
# the cathedral door, the mariners' confraternity and its seine, the wine
# priory of a monastery, the tower of a lord (which the brotherhood of 1467
# can pull down for good), the House of Spices, the pazo, the granite
# masons' yard, the royal dockyards and the works of Sargadelos.
#
# Every production method is inline, names the margin of its tier, and was
# sized with eu5-profit (cap written as a name, never switched off).

# ----------------------------------------------------------------------------
# Obradoiro de Acibeche (age 1)
# ----------------------------------------------------------------------------
# A workshop of the Acibecheria, the street on the north side of the
# cathedral of Santiago: jet from the Asturian coast carved into scallop
# shells, figures of the Apostle and hands closed in a fist against the evil
# eye, set in silver; and beside it the concheiros, casting the pilgrim's
# badge in tin. The guild refused soft jet that would crack under the knife.
# Jet is not a trade good: the silver is the setting and the tin the badge.
# allow: a cathedral must stand in the location. The trade lived off the
# people who came to a shrine and sold at its door; without one it is
# vanilla's jewelry guild and no more.
# Settlement: towns and cities. A plain producer with no modifier block, so
# it takes the guild level formula as vanilla's jewelry guild does.
nd_glc_obradoiro_de_acibeche = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	startup_ramp_target = guild_startup_ramp_target
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_glc_acibecheiros_e_concheiros
	}

	allow = {
		custom_tooltip = {
			text = nd_glc_acibeche_needs_cathedral_tt
			has_building = building_type:cathedral
		}
	}

	unique_production_methods = {
		nd_glc_acibeche_e_prata = {
			silver = 0.5
			tin = 0.2
			tools = 0.05
			produced = jewelry
			output = 0.852
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# ----------------------------------------------------------------------------
# Confraria de Mareantes (age 2; a second method in age 6)
# ----------------------------------------------------------------------------
# The mariners' confraternity of a fishing town, under the patronage of the
# Corpo Santo: it owns the great seine in shares, says who may shoot it and
# when, keeps the lamp of the chapel and buries its dead. The mariners of
# Pontevedra paid for the church of Santa Maria a Maior out of the sardine.
# location_potential: the shore (is_coastal, vanilla's own gate for the
# fishing village) in a market where fish is landed. The strict form
# (raw_material = fish) would shut out Pontevedra itself, whose raw material
# on this map is salt, so the market form is used.
# Settlement: countryside and town, never a city. The mariners lived in a
# quarter of their own outside the walls, the Moureira; a city's waterfront
# belongs to its merchants.
# Vanilla's rural shape: building_maintenance with produced, output and the
# rural margin. The salt is the cure, the lumber the boats, the fiber the
# net.
nd_glc_confraria_de_mareantes = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	startup_ramp_target = rural_startup_ramp_target
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_glc_cerco_dos_mareantes
	}

	location_potential = {
		is_coastal = yes
		is_produced_in_location_market = goods:fish
	}

	unique_production_methods = {
		# The cerco: the confraternity's own seine, shot from the boats of
		# the whole quarter, and the catch salted in open tubs.
		nd_glc_cerco_real = {
			salt = 0.1
			lumber = 0.1
			fiber_crops = 0.05
			produced = fish
			output = 0.78
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
		# Unlocked by nd_glc_fomentadores_da_salga (age 6): the Catalan
		# way. The sardine is packed in salt in casks and pressed under
		# weighted beams until the oil runs out, and it keeps for a year.
		# More salt, more casks, the ironwork of the press, twice the fish.
		nd_glc_salga_de_prensa = {
			salt = 0.25
			lumber = 0.15
			tools = 0.03
			produced = fish
			output = 1.578
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}

# ----------------------------------------------------------------------------
# Torre do Senor (age 2)
# ----------------------------------------------------------------------------
# The tower house of a Galician lord: a square keep of granite on a rock, a
# handful of squires and crossbowmen, and the right to make every tenant in
# sight of it carry stone, mount guard and pay. It gives the nobles a grip
# on the district and men for the host, and the district hates it (the
# positive local_unrest is a deliberate cost).
# It is not a fort: no fort level and no zone of control. Vanilla's stockade
# and castle are the fortresses; this is a lordship in stone.
# country_potential and remove_if: the Santa Irmandade can pull every tower
# in the kingdom down and forbid their rebuilding (event nd_glc.3, which
# sets nd_glc_torres_prohibidas). The towers then fall and can never be
# raised again.
# No location_potential on purpose: a lord's tower stood wherever a lord had
# tenants, on the coast and inland alike (the brotherhood counted some 130).
# The settlement flags are the gate: countryside and town, never a city,
# where the lord kept a house and not a keep.
# Upkeep 1.15 at base prices (stone 1, weaponry 3, horses 3, wine 2) against
# vanilla's peel tower at 1.00 and castle at 0.60.
nd_glc_torre_do_senor = {
	is_special = yes
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	employment_size = stockade_employment
	category = defense_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = yes
	build_time = small_fort_building

	country_potential = {
		has_advance = nd_glc_fernan_perez_o_boo
		NOT = { has_variable = nd_glc_torres_prohibidas }
	}

	remove_if = {
		location.owner ?= { has_variable = nd_glc_torres_prohibidas }
	}

	modifier = {
		local_nobles_estate_power = 0.5
		local_garrison_size = 0.1
		local_levy_size_modifier = 0.1
		local_unrest = 0.03
	}

	unique_production_methods = {
		nd_glc_torre_do_senor_maintenance = {
			stone = 0.2
			weaponry = 0.2
			horses = 0.05
			wine = 0.1
			category = building_maintenance
		}
	}
	construction_demand = peel_towers_construction
}

# ----------------------------------------------------------------------------
# Real Fabrica de Sargadelos (age 6)
# ----------------------------------------------------------------------------
# A blast furnace and a pottery in one wooded parish: ore from the hill,
# charcoal from the oak woods, a dammed stream for the bellows and white
# clay from the next valley. Cooking pots and shot first, then a fine
# earthenware printed from copper plates.
# location_potential: the location's own raw material must be iron or clay.
# The works stood on its ore and its kaolin; in the kingdom that is Viveiro,

# nothing is conjured.
# Settlement: countryside and town, never a city. It was built in a rural
# parish for the wood and the water. Because it can stand in the

	country_potential = {
		has_advance = nd_glc_ferro_e_louza
	}

	location_potential = {

			raw_material = goods:iron
