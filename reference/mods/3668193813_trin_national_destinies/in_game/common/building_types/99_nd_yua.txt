	is_foreign = no

	max_levels = 1

	# khan's seat is. The gate is the capital, below.
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = religious_building_time

	country_potential = {
		has_advance = nd_yua_genghisid_legitimacy

	# One shrine, at the seat of the khan (the gate vanilla writes on its
	# own capital buildings, capital_buildings.txt central_bank).

		counts_as_capital = yes

		is_capital = yes
		country_has_no_other_copy_of_building = { type = nd_yua_eight_white_yurts }

		is_no_longer_capital = yes

		local_cultural_tradition = 0.25
