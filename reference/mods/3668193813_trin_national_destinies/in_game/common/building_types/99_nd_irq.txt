# === Iraq: custom buildings ===
#
# Every building here carries a location_potential, and the comment above each
# one says why that reach is the correct one. Iraq's whole economic argument is
# that a specific kind of ground (irrigable, riverine, marshy, saline) does a
# specific kind of work, so a building that could go anywhere would be saying
# the opposite of what this country is about.

# ============================================================
# The Nahrawan head-works. Canal regulators (qanatir): masonry sluices at the
# offtake that let the engineer decide how much Tigris water enters the canal
# and shut it when the flood comes down. Restricted to river locations in the
# Fertile Crescent because a regulator with no river to regulate is nothing,
# and because the Tigris-Euphrates system plus its Levantine sisters is exactly
# the ground where head-works of this kind were built.
# ============================================================

	audio_tier = 2

# ============================================================
# The nakhil: the date-palm grove. Deliberately gated to arid river ground with
# desert or sparse cover, which is precisely the ground vanilla's fruit_orchard
# refuses (it wants grassland, farmland, woods or jungle). The palm is the tree
# that fruits on saline soil in an arid climate provided its roots reach water,
# and the Shatt al-Arab belt is the largest such grove on earth. This building
# therefore fills a real hole rather than shadowing the orchard.
# ============================================================
nd_irq_nakhil_grove = {
	audio_tier = 1
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	startup_ramp_target = rural_startup_ramp_target
	employment_size = rural_peasant_produce_employment

	category = rgo_building_category

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	build_time = rural_build_time

	country_potential = {
		has_advance = nd_irq_nakhil_husbandry
	}

	# The climate, vegetation and water gates describe the palm belt. The
	# market gate is the standard rule for a building that produces a raw
	# good: a grove is planted where dates are already a known crop, not
	# conjured onto ground that has never seen one.
	location_potential = {
		region = region:crescent_region
		climate = arid
		is_produced_in_location_market = goods:fruit
		OR = {
			has_river = yes
			is_adjacent_to_lake = yes
		}
		OR = {
			vegetation = desert
			vegetation = sparse
		}
	}

	unique_production_methods = {
		nd_irq_nakhil_harvest = {
			tools = 0.15
			produced = fruit
			output = 0.5399
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		nd_irq_nakhil_canal_fed = {
			# unlocked by nd_irq_hindiyya_channel
			tools = 0.25
			lumber = 0.15
			produced = fruit
			output = 1.17
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}

	construction_demand = farm_constructions
}
