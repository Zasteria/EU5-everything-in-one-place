# Date Palm Grove: the irrigated palm gardens of the oases. A plain producer
# of fruit with no modifier block, in vanilla's shape for a rural producer
# (fruit_orchard, which cannot stand in a desert, and the Maghreb's palm
# irrigation, which Arabia cannot build).
nd_aia_date_grove = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	# The gardens ring the oasis towns and villages; no city grows palms.
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_aia_date_palm_groves
	}

	# Only where the land itself bears fruit, and only in the dry country
	# where that fruit is the date.
	location_potential = {
		raw_material = goods:fruit
		climate = arid
	}

	unique_production_methods = {
		# Inputs are the upkeep of the harvest, as in vanilla's palm
		# irrigation (lumber, clay, tools): pruning knives, the jars the
		# dates are packed in, and palm rope for the climbers.
		nd_aia_date_processing = {
			tools = 0.05
			pottery = 0.1
			fiber_crops = 0.05
			produced = fruit
			output = 0.42
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}

		# Unlocked by nd_aia_oasis_agriculture: fruit trees and fodder under
		# the palms, iron tools and draught animals. A larger harvest.
		nd_aia_oasis_farming = {
			tools = 0.1
			livestock = 0.1
			pottery = 0.1
			fiber_crops = 0.05
			produced = fruit
			output = 0.78
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}
