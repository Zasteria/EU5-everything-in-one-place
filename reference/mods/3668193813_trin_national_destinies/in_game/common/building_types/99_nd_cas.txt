# What is NOT here, because somebody already built it:
#   vanilla: the Royal Atarazanas of Sevilla, the School of Salamanca, the
#   General Archive of Simancas, the Archive of the Indies, the Artillery
#   Academy of Segovia and the bullfighting school of Sevilla (event
#   buildings), the Brethren of the Marsh (unique_buildings.txt), the
#   aqueduct of Segovia, and, reserved for the Iberian DLC, the Consulado de
#   Bilbao.
#   the mod's Spain (99_nd_spa.txt): the alcazar, the Mesta sheepwalk, the
#   Casa de Contratacion, the proveeduria, the mint, the royal foundry, the
#   camino real, the Escorial and the Real Audiencia.
#   the mod's Catalonia (99_nd_cat.txt): the mountain forge. The Cantabrian
#   ferreria is the same works under another name and is left out for that
#   reason.
#
# What is here is the kingdom at ground level: the salt the Crown took for
# itself in 1338, the customs of Seville, the colleges that bred its judges,
# the office that kept its debts, the sword makers, the village grain bank,
# the playhouse yard, the great square, the potters of Talavera, the five
# guilds of Madrid, the colonies of the Sierra Morena and the bank of 1782.
#
# Every production method is inline and was sized with eu5-profit (margin 20
# percent, cap written as a name). Upkeep bills are set at or above the
# vanilla building of the same kind, named in each block.

# ----------------------------------------------------------------------------
# Salinas Reales (age 1)
# ----------------------------------------------------------------------------
# A royal saltworks: brine led from the spring over terraces of clay and
# timber and left to the sun, or the pans of a tidal marsh. In 1338 Alfonso
# XI took every saltworks of the kingdom for the Crown and obliged each
# household to buy its measure from the king's store.
# location_potential: the location's own raw material must be salt (the
# strict gate). Salt is a raw good and this building extracts it; it cannot
# be made where there is no brine. In Castile's lands of 1337 that admits
# Anana, Cuenca and the marshes of Cadiz, Huelva and Pontevedra.
# Settlement: countryside and town, never a city. The works stand at the
# spring or on the marsh, and the towns that grew beside them stayed small.
# Inputs are the upkeep of the pans: timber for the terraces and channels,
# tools for the rakes and sluices.
nd_cas_salinas_reales = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	startup_ramp_target = rural_startup_ramp_target
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_cas_estanco_de_la_sal
	}

	location_potential = {
		raw_material = goods:salt
	}

	unique_production_methods = {
		nd_cas_salinas_reales_sal_de_granjas = {
			lumber = 0.3
			tools = 0.05
			produced = salt
			output = 0.18
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}

# ----------------------------------------------------------------------------
# Real Chancilleria (destiny, path A capstone)
# ----------------------------------------------------------------------------
# The high court of the Crown of Castile, which heard appeals in the king's
# name and kept his seal. Henry II founded the Audiencia at the Cortes of
# Toro in 1371; it settled at Valladolid for good in 1442 and took its
# ordinances at Medina del Campo in 1489. A second was opened at Ciudad Real
# in 1494 and moved to Granada in 1505, with the Tagus as the line between
# them. Its Sala de Hijosdalgo decided who was noble and so who paid tax.
# Not vanilla's supreme court (the capital, once, crown power and reach) and
# not Spain's Real Audiencia (control and assimilation, anywhere): this is a
# law town. A city with a chancilleria lived off its suitors, proctors and
# notaries, which is what the three modifiers say.
# Settlement: cities only, one per location. Both historical seats were
# cities, and a court of appeal for half a kingdom does not sit in a market
# town. Bound to the two seats it kept, Valladolid and Granada, by a list of
# locations: vanilla's idiom for a building of a few named places (the
# aqueduct, town_buildings.txt:653-660). Ciudad Real held the southern court
# for eleven years only and is left out. Without the list it could stand in
# every city of the Crown, which history never saw.
# Values: unrest as vanilla's forts and theatre (0.05), burgher literacy at
# half a university's (10), promotion at one university step (0.001).
nd_cas_dest_chancilleria = {
	is_special = yes
