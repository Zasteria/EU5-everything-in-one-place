# ============================================================================
# Leon (LON): custom buildings
# National Destinies mod
# ============================================================================
#
# What is NOT here, because somebody already built it:
#   vanilla: the university, the scriptorium, the village granary, the cloth
#   guild and the cannon maker (every country's); the School of Salamanca
#   (an event building, raised here by event nd_lon.4 and not rewritten); the
#   Fuero Juzgo (a town right).
#   the mod's Spain: the Mesta sheepwalk and the alcazar. The mod's Castile:
#   the colegio mayor, the posito, the plaza mayor. The mod's Catalonia: the
#   farga, which is why Leon's forges are an advance and not a building.
#
# What is here is the kingdom's own ground: the kings' vault in San Isidoro,
# the Frankish borough on the pilgrim road, the village's council house, the
# summer pasture, the raised granary, the linen works, Jovellanos's
# institute and the royal works at Trubia.
#
# Every production method is inline and was sized with eu5-profit (the cap
# is always a name). The outputs are exact: 0.64, 0.76, 0.35, 1.4625, 1.625.

# ----------------------------------------------------------------------------
# Panteon de los Reyes (age 1)
# ----------------------------------------------------------------------------
# The narthex of the collegiate church of San Isidoro, where the kings of
# Leon lie under a vault painted about 1100 with the months of the farming
# year and the shepherds hearing the angel.
# location_potential: the city of Leon and nowhere else. There is one.
# Settlement: any urban tier, because the city may stand at any of them when
# the kingdom is restored; never the countryside.
# capital_country_modifier: the dynasty's dead add to the standing of a king
# who rules from beside them, and of nobody else. monthly_prestige and not
# legitimacy, so a Leon that is not a monarchy is not paid in a coin it
# cannot use.
# Upkeep 1.13 gold a month at base prices, level with vanilla's library
# (1.10): candles, hangings, the canons' books, glass, and gold leaf.
nd_lon_panteon_real = {
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = 1
	employment_size = cultural_employment
	category = cultural_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = large_cultural_building_time

	country_potential = {
		has_advance = nd_lon_reyes_en_san_isidoro
	}

	location_potential = {
		this = location:leon
	}

	modifier = {
		local_cultural_tradition = 0.25
		local_cultural_influence = 0.25
		local_clergy_estate_power = 0.25
	}

	capital_country_modifier = {
		monthly_prestige = 0.05
	}

	unique_production_methods = {
		nd_lon_panteon_real_maintenance = {
			beeswax = 0.15
			fine_cloth = 0.05
			books = 0.1
			glass = 0.05
			goods_gold = 0.01
			category = building_maintenance
		}
	}
	construction_demand = prestige_building_construction
}

# ----------------------------------------------------------------------------
# Burgo de Francos (age 1)
# ----------------------------------------------------------------------------
# A borough of settlers from beyond the Pyrenees, planted beside an older
# town on the pilgrim road and given its own charter and its own judge:
# money changers, innkeepers, cobblers, furriers. Sahagun had one from 1085,

# Castile, Leon and Galicia. Off the road there are no pilgrims to live on.

# The price is in the block: the newcomers are the king's men and not the
# local lord's or the abbot's, and they say so (local_unrest).

	country_potential = {
		has_advance = nd_lon_camino_frances

		OR = {
			area = area:leon_area

# ----------------------------------------------------------------------------
# Casa de Concejo (age 2)
# ----------------------------------------------------------------------------
# Where the village meets when it rains: a room, a bench round the wall, the
# chest with the ordinances and the tally sticks, and the jug of concejo
# wine that closes every sitting. The assembly allots the pasture and the
# water, fines the neighbour whose cattle strayed, and calls the facendera,
# the day of common labour on the road and the weir.
# location_potential: the Leon area, or any place whose people are Leonese.
# The open council is a custom of this people, not an office the crown can
# set up where it pleases.
# Settlement: countryside only. Towns were governed by closed councils of
# aldermen from the fourteenth century on.
# No local_max_control on purpose: the village governs itself, which is the
# opposite of the crown governing it. The price is in the block: where the
# neighbours rule, the king's officer does not (local_crown_estate_power).
# Upkeep 0.60 gold a month at base prices: paper, lumber, tools and wine. It
# grants no control, so it stays under vanilla's bailiff (1.50), inside the
# 0.5 to 0.8 the owner ruled for the same kind of building on 2026-10-02.
nd_lon_casa_de_concejo = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = estate_peasants_employment
	category = government_category
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	build_time = government_build_time

	country_potential = {
		has_advance = nd_lon_concejo_abierto
	}

	location_potential = {
		OR = {
			area = area:leon_area
			dominant_culture = culture:leonese
		}
	}

	modifier = {
		local_unrest = -0.05
		local_peasants_estate_power = 0.5
		local_monthly_control = 0.001
		local_crown_estate_power = -0.1
	}

	unique_production_methods = {
		nd_lon_casa_de_concejo_maintenance = {
			paper = 0.1
			lumber = 0.1
			tools = 0.05
			wine = 0.05
			category = building_maintenance
		}
	}
	construction_demand = village_construction
}

# ----------------------------------------------------------------------------
# Brana (age 2)
# ----------------------------------------------------------------------------
# The summer station: round stone huts under a thatch of broom, a walled
# fold, and the high grass that is under snow from November to May.
# Shape: vanilla's fulani_cattle_rearing_pen (unique_buildings.txt), an
# enhancer of what the land itself yields. It makes no good: until
# 2026-10-02 its one method made livestock out of lumber, tools and salt.
# location_potential: high ground (hills, mountains or the plateau) whose
# OWN raw good is livestock or wool. A brana takes the herd the land already
# has up to grass the valley cannot reach; it does not conjure a herd. The
# merino flocks that summered in the passes of Leon are the wool case.
# Settlement: countryside only. One level, because the grant is a local
# modifier: vanilla's pen has three and more and pays for them with slower
# assimilation, which a single level does not need.
# Upkeep: timber for the huts and the folds, tools, and salt for the licks.
nd_lon_brana = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = rural_peasant_produce_employment
	category = infrastructure_category
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_lon_branas_y_puertos
	}

	location_potential = {
		OR = {
			topography = mountains
			topography = hills
			topography = plateau
		}
		OR = {
			raw_material = goods:livestock
			raw_material = goods:wool
		}
	}

	modifier = {
		local_livestock_output_modifier = 0.2
		local_wool_output_modifier = 0.2
	}

	unique_production_methods = {
		nd_lon_brana_alzada = {
			lumber = 0.3
			tools = 0.05
			salt = 0.05
			category = building_maintenance
		}
	}
	construction_demand = farm_constructions
}
