# Zellige Kilns (age 2, nd_mor_zellige_craft)
# The tile makers' yards of Fez: squares of the grey Fez clay are fired,
# dipped in a tin and lead glaze, fired again, and then cut by hand with a
# sharpened hammer into the small shapes of the mosaic.
# No location_potential, and that is deliberate: pottery is a made good, the
# clay, the colours and the lead for the glaze are bought on the market, and
# those inputs are the gate.
# Settlement: town and city. The kilns stood at the edge of the medina.
# No modifier block, so the level formula of the guild tier multiplies nothing.
nd_mor_zellige_kiln = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_mor_zellige_craft
	}

	unique_production_methods = {
		# The building's only method: no advance has to unlock it.
		# eu5-profit -i "clay=0.4,dyes=0.05,lead=0.05" -o pottery (20%).
		nd_mor_zellige_workshop = {
			clay = 0.4
			dyes = 0.05
			lead = 0.05
			produced = pottery
			output = 0.6
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Moukahla Workshop (age 3, nd_mor_metalworking; a second method in age 5
# and a third in age 6)
# The gunsmiths' street. Andalusi exiles carried the craft to Tetouan and
# the Sous copied it at Taroudant: barrel forgers, lock makers and stock
# carvers each doing one part of the moukahla, the long musket with the
# snaphaunce lock that Morocco went on making into the nineteenth century.
# No location_potential, and that is deliberate: nothing here comes out of
# the ground. Iron, wood and tools are bought on the market and those inputs
# are the gate. The trade stood in any town with a souk.
# Settlement: town and city, never the countryside. A gun is the work of
# several guilds side by side.
nd_mor_moukahla_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = weapons_industry_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_mor_metalworking
	}

	unique_production_methods = {
		nd_mor_moukahla_gunsmithing = {
			iron = 0.25
			lumber = 0.15
			tools = 0.05
			produced = firearms
			output = 0.45
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_mor_corsair_fleet (age 5): the same street turns to
		# blades for the Sultan's ships, above all the nimcha, the short
		# sabre of the corsairs. Vanilla's weapon smith burns coal with wood
		# and tools the same way. Output from eu5-profit at the guild margin.
		nd_mor_arms_workshop = {
			iron = 0.25
			lumber = 0.15
			coal = 0.1
			tools = 0.05
			produced = weaponry
			output = 0.53
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_mor_industrial_stirrings (age 6): forges fired
		# with coal and boring benches on the European pattern.
		nd_mor_european_style_workshop = {
			iron = 0.3
			lumber = 0.15
			coal = 0.1
			tools = 0.1
			produced = firearms
			output = 0.65
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
