# Tunis (TUN): custom buildings.
#
# What the same country can already build, so nothing here repeats it:
# vanilla gives every Muslim country the madrasa and the funduq, the Maghrebi
# group the fortress granary and the palm irrigation, and everyone the
# hospital, the dock and shipyard, the cloth, fine cloth and pottery guilds.
# Carthage and the Almohads, which a Tunisian country can go on to form, add
# a shipyard, a citadel, an emporium, two funduqs, a ribat and a minaret.
# So there is no shipyard, fort, market or inn below. Each building here is
# tied to a place or a population that only this country's land has.

# ============================================================
# Ma'sara: the olive press of the Sahel.
# The Sahel from Sousse to Sfax was one olive grove, and its oil was the
# country's staple export from Hafsid times to the beys.
# Every settlement tier on purpose: the gate is the grove itself. Only a
# location whose raw material is olives may build one (13 of the 115
# locations of the Tunis and Tripolitania areas), and Sousse and Mahdia
# pressed their oil inside the walls.
# An enhancer and not a producer, on purpose. It has the shape of vanilla's
# windmill, sawmill and local smelters (rural_buildings.txt): one level,
# rural_peasant_enhance_employment, an upkeep-only method and an output
# modifier on the raw good the land already yields. The game has no oil good
# for a press to make, so a producing method could only make olives a second
# time on top of the modifier.
# ============================================================
nd_tun_olive_press = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = rural_peasant_enhance_employment
	category = infrastructure_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_tun_sahel_olive_groves
	}

	location_potential = {
		raw_material = goods:olives
	}

	modifier = {
		# Vanilla's one-level enhancers: the sawmill 0.2, the windmill and the
		# local smelters 0.1. Its plantations (0.1) and the Fulani cattle pen
		# (0.2) grant theirs per level and have several levels (the pen's cap
		# starts at 3, building_caps.txt:774). One level here.
		local_olives_output_modifier = 0.2
	}

	unique_production_methods = {
		nd_tun_olive_press_maintenance = {
			lumber = 0.1
			stone = 0.05
			pottery = 0.1
			tools = 0.05
			category = building_maintenance
		}
	}
	construction_demand = village_construction
}

# ============================================================

# ============================================================
# Zawiya: the marabout's lodge on the steppe.
# A saint's tomb, a hostel and a court of arbitration in one. The Hafsid
# sultans founded them along the roads, and the tribes of the interior took
# their quarrels, and in time their sons, to the shaykh who kept one.
# Bound to the tribes: only a location that holds tribesmen may build one.
# That is the whole point of the building (it settles tribesmen), and it
# keeps it out of the Sahel and the cities of the coast. Every settlement
# tier, because the tribes' land is mostly countryside and a few steppe towns.
# It has a price the player should feel: every tribesman it turns into a
# peasant is one fewer for the Ku'ub and Makhzen levies, which draw tribesmen.
# Muslim owners only, as vanilla's madrasa.
# ============================================================
nd_tun_zawiya = {
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = 1
	employment_size = religious_building_employment
	category = religious_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	build_time = religious_building_time

	country_potential = {
		has_advance = nd_tun_marabouts_of_the_steppe
		religion ?= {
			group = religion_group:muslim
		}
	}

	location_potential = {
		any_pop = {
			pop_type = pop_type:tribesmen
		}
	}

	modifier = {
		# Vanilla: madrasa and great_madrasa 0.1 tribal promotion, ribat -0.05 unrest.
		local_tribal_promotion = 0.1
		local_unrest = -0.05
	}

	unique_production_methods = {
		nd_tun_zawiya_maintenance = {
			wheat = 0.114
			cloth = 0.057
			beeswax = 0.023
			category = building_maintenance
		}
	}
	construction_demand = town_building_construction
}

# ============================================================
# Chechia Workshop: the red felt cap of Tunis.
# The Moriscos expelled from Spain in 1609 brought the craft, and by the
# eighteenth century the chaouachis were the richest guild of the country,
# selling from Algiers to Constantinople. Wool in, fulled and dyed caps out.
# No location gate: a guild that buys wool and dye in the market and makes a
# produced good, exactly as vanilla's fine cloth guild does. Town and city
# only, because the chaouachis were an urban corporation, and two levels,
# because this adds to the vanilla guild's capacity and must not replace it.
# Outputs from eu5-profit at the guild margin (20%):
#   wool 0.8, dyes 0.1 gives 0.48; wool 1.0, dyes 0.15, alum 0.05 gives 0.65.
# Both recomputed in exact decimals at the 1.4 prices in common/goods (wool
# 2.5, dyes 4, alum 3, fine cloth 6): cost 2.4 and 3.25, margin 20% each.
# The unlocked method is a real step up and not a second upkeep: 35% more
# caps and 35% more profit from the same workers, at the price of needing
# alum. A vanilla fine cloth guild level makes 0.6 to 0.9 over its two slots.
# ============================================================
nd_tun_chechia_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 2
	startup_ramp_target = guild_startup_ramp_target
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_tun_morisco_settlement
	}

	unique_production_methods = {
		nd_tun_chechia_felting = {
			wool = 0.8
			dyes = 0.1
			produced = fine_cloth
			output = 0.48
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		nd_tun_chechia_zaghouan_dyeing = {
			# unlocked by nd_tun_souk_of_the_chaouachis: fulled at the mills
			# of El Battan, mordanted with alum and dyed at Zaghouan
			wool = 1.0
			dyes = 0.15
			alum = 0.05
			produced = fine_cloth
			output = 0.65
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	custom_tags = { guild }
	construction_demand = guild_construction
}

# ============================================================
# Bagnio: the captives' prison of a corsair port.
# Tunis had about a dozen. The captives rowed the galleys, careened the hulls and
# waited for the Redemptionist friars to buy them out.
# Worked by slaves, so it does nothing in a port that holds none: the building
# needs captives, and the Maghrebi slave raids of vanilla bring them.
# Ports only (allow), slavery must be legal (allow and remove_if, as vanilla's
# slave_market), town and city only, one level.
# Five hundred captives a bagnio (employment 0.5): Tunis itself starts with
# under a thousand slaves, so a larger one would stand half empty.
# Sailors: the engine caps flat sailors per employed pop and logs it only
# when a game loads. Vanilla's dock grants 0.016 per 1.0 of employment, and
# this keeps to that ratio (0.008 at 0.5).
# ============================================================
nd_tun_bagnio = {
	is_special = yes
	is_foreign = no
	pop_type = slaves
	max_levels = 1
	employment_size = 0.5
	category = naval_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = medium_port_building_time

	country_potential = {
		has_advance = nd_tun_bagnios_and_redemption
	}

	location_potential = {
		is_coastal = yes
	}

	allow = {
		is_port = yes
		scope:actor = {
			has_slavery = yes
		}
	}

	remove_if = {
		scope:actor = {
			has_slavery = no
		}
	}

	modifier = {
		local_sailors = 0.008
		local_repair_speed = 0.02
		local_slave_pop_satisfaction = -0.025
	}

	unique_production_methods = {
		nd_tun_bagnio_maintenance = {
			wheat = 0.262
			cloth = 0.066
			lumber = 0.131
			tools = 0.066
			category = building_maintenance
		}
	}
	construction_demand = port_building_construction
}
