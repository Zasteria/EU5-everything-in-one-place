# Somono Landing: the boatmen's station on the Niger (age 5)
# The Somono are the boatmen of the middle Niger: a people the rulers of the
# river settled at its crossings and paid in privileges for ferrying the
# ruler's grain, salt and soldiers. The kings of Segu from Biton Kulibali
# (r. 1712-55) made a corps of them; tradition puts the first grant with the
# mansas of Mali.
# location_potential: a river. The landing is nothing without the Niger.
# Settlement: towns and cities only. The Somono quarters stood at the river
# towns (Segu, Sansanding, the port of Djenne, Kabara below Timbuktu).
# The proximity value is vanilla's own for a bridge (common_buildings.txt:74)
# and the supply value its grain house's (town_buildings.txt:335).

	max_levels = 1

	country_potential = {
		has_advance = nd_mal_niger_trade_network
	}

	location_potential = {
		has_river = yes
	}
