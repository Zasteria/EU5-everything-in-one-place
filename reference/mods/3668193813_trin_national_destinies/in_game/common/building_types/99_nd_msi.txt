# Weaving House - Economic building for Mossi cotton textile production

	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	# Not rural: this is a workshop, not a field. A per-location

	town = yes
	city = yes
	megalopolis = yes

	build_time = guild_build_time

	country_potential = {
		has_advance = nd_msi_cotton_weaving
	}

	modifier = {
		local_cloth_output_modifier = 0.05
		local_burghers_estate_power = 0.5
		local_production_efficiency = small_production_efficiency_bonus
	}

	unique_production_methods = {
		nd_msi_weaving_upkeep = {
			cotton = 0.2
			dyes = 0.1
			produced = cloth
			output = 0.40
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		# Unlocked by nd_msi_indigo_dyeing. The strips go through the dye
		# pits and come out as the dark blue cloth that sells as a luxury.
		nd_msi_indigo_processing = {
			cotton = 0.3
			dyes = 0.2
			produced = fine_cloth
			output = 0.34
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		# Unlocked by nd_msi_cotton_exports. Plain strip cloth in bulk, for
		# the caravans.
		nd_msi_export_cotton = {
			cotton = 0.5
			dyes = 0.1
			produced = cloth
			output = 0.76
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		# Unlocked by nd_msi_industrial_weaving.
		nd_msi_power_looms = {
			cotton = 1.0
			dyes = 0.15
			produced = cloth
			output = 1.44
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Zai Fields: planting pits dug into crusted ground before the rains
# Countryside only, and only on dry land whose own good is a food (hot
# semi-arid or arid climate: 15 of the 47 locations, the north of the
# country). Vanilla's irrigation needs a river or a lake; the pits are how
# land with neither is farmed, and they give the same bonus.
nd_msi_zai_fields = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = generic_peasant_building_employment
	category = infrastructure_category
	rural_settlement = yes
	town = no
	city = no
	megalopolis = no
	expensive = no
	build_time = infrastructure_build_time

	country_potential = {
		has_advance = nd_msi_volta_agriculture
	}

	location_potential = {
		OR = {
			climate = hot_semi_arid
			climate = arid
		}
		raw_material ?= { is_food = yes }
	}

	modifier = {
		local_monthly_food_modifier = monthly_food_productivity_weak_bonus
	}

	unique_production_methods = {
		nd_msi_zai_upkeep = {
			tools = 0.1
			livestock = 0.15
			category = building_maintenance
		}
	}
	construction_demand = village_construction
}
