# Coptic Egypt: custom buildings.
#
# Ten named institutions, every one of them dated and specific. Note what is
# NOT here: no glassworks, no cloth hall, no paper mill. Vanilla already ships
# a complete production ladder for every trade good, and adding a Coptic
# re-skin of the glass guild would say nothing about Copts. What CPT owns
# instead are the institutions of a minority that governs: the revenue office,
# the flood gauge, the desert keep, the school of canon law, the frontier
# exchange, the marsh redoubt, the monastery farm, the financier's household,
# the community endowment and the modern school.
#
# Three of these gate on where the community actually lives. That is deliberate
# and it is the country's central constraint: the map of Coptic Egypt's
# institutions is the map of its minority, and the only way to widen it is to
# convert the countryside faster than the countryside converts you.

# The Diwan of the Copts. The revenue bureau, staffed by the secretarial
# families that ran Egyptian finance under every dynasty since the conquest.
nd_cpt_diwan_office = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = generic_burgher_employment
	category = government_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = government_build_time

	country_potential = {
		has_advance = nd_cpt_diwan_of_the_copts
	}

	# The reach IS the mechanic. A Coptic revenue office needs Coptic clerks,
	# so it can only stand where the community is the dominant religion. In
	# 1337 that is a handful of locations. Widening the fiscal machine means
	# converting the countryside, which is the whole strategic problem.
	location_potential = {
		dominant_religion ?= religion:miaphysite
	}

	modifier = {
		local_crown_estate_power = 0.25
		local_burghers_max_literacy = 2.5
		local_monthly_control = 0.001
	}

	unique_production_methods = {

			paper = 0.3
			books = 0.15
			beeswax = 0.1
			incense = 0.1
			category = building_maintenance
		}
	}
	construction_demand = local_governor_construction
}

# ===== DESTINY BUILDINGS =====

# The Bureau of Endowments (diwan al-ahbas). The Mamluk state ran an office
# for the administration of Muslim pious foundations, and Coptic clerks
# staffed the fiscal bureaus that audited it. A Coptic crown keeps that office
# running, unchanged, over mosques and madrasas it has no share in.
#
# The gate is the exact mirror of the Diwan of the Copts. That office stands
# only where the community is dominant; this one stands only where it is not.
# Together they say the path's whole thesis: the fiscal machine covers the
# country because it treats both halves of it identically.
nd_cpt_diwan_al_ahbas = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = generic_burgher_employment
	category = government_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = government_build_time

	country_potential = {
		has_advance = nd_cpt_dest_indispensable
	}

	# The endowments this office administers are the waqf of congregational
	# mosques and madrasas, so it stands where those congregations are, which
	# in Egypt means Sunni towns and cities. Rural settlements are excluded
	# because a village has no waqf administration to take over.
	location_potential = {
		dominant_religion ?= religion:sunni
	}

	# The promise is the mechanic. Where this office stands the crown has
	# guaranteed that nobody's faith will be touched, and it means it: no
	# conversion effort is applied here at all, by us or by anyone. What the
	# guarantee buys is a town that does not riot and does not have to be
	# garrisoned into obedience.
	modifier = {
		local_unrest = -0.03
		local_max_control = 0.03
		local_monthly_prosperity = 0.001
		local_clergy_conversion_blocked = yes
	}

	unique_production_methods = {
		nd_cpt_ahbas_maintenance = {
			paper = 0.3
			books = 0.15
			beeswax = 0.1
			cloth = 0.1
			category = building_maintenance
		}
	}
	construction_demand = local_governor_construction
}
