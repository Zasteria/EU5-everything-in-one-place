# Alluvial Gold Works: Organized gold panning and washing operations
nd_mtp_gold_works = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = generic_peasant_building_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_mtp_alluvial_gold_tradition
	}

	modifier = {
		local_goods_gold_output_modifier = 0.25
		local_raw_material_output = 0.25
	}

	# Extraction is bound to the resource: you cannot mine, tap or
	# harvest goods_gold where there is none. Without this the building
	# is a free resource generator in every location the owner holds.
	location_potential = {
		is_produced_in_location_market = goods:goods_gold
	}

	unique_production_methods = {
		# Both methods stood with the cap switched off and an output of 0.15
		# gold from 0.1 lumber or 0.1 tools (seven and three times what the
		# inputs pay for). Outputs are now at the rural margin: sluice timber
		# and pans for the first, more of both for the second.
		nd_mtp_gold_works_upkeep = {
			lumber = 0.1
			tools = 0.05
			produced = goods_gold
			output = 0.045
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		# Unlocked by nd_mtp_gold_taxation (age 3): washing on a larger
		# scale under the court's officers, twice the yield of plain panning.
		nd_mtp_gold_preparation = {
			tools = 0.1
			lumber = 0.2
			produced = goods_gold
			output = 0.09
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}
