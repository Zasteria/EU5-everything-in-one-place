# Adze Makers' Workshop (the key keeps its old name). Vanilla gives every
# Maori country the pounamou_carver, which makes jewelry from gems, so this
# building is the other stone trade: argillite and basalt ground into adzes.
nd_ntr_pounamu_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_ntr_pounamu_mastery
	}

	modifier = {
		local_merchant_power = 0.05
		local_trades_per_burgher = 0.1
		local_cultural_tradition = 0.25
	}

	unique_production_methods = {
		# Adze grinding (the key keeps its old name). Recipe and output are
		# vanilla's stone_tools_guild_maintenance: stone makes tools at the
		# guild margin (20%).
		nd_ntr_pounamu_processing = {
			stone = 0.625
			produced = tools
			output = 0.25
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	
		nd_ntr_flax_weaving = {
			wool = 0.2
			dyes = 0.1
			produced = cloth
			output = 0.36
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	
		# Unlocked by nd_ntr_european_contact: pigs and potatoes go out to the
		# visiting ships and muskets come back. A barter, written as the
		# method it is in play: food in, firearms out.
		nd_ntr_provisioning_trade = {
			livestock = 0.2
			potato = 0.2
			produced = firearms
			output = 0.2
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Unlocked by nd_ntr_whalebone_artistry: the carvers work the bone and
		# teeth of stranded whales. The game has no whale good, so the whale
		# reaches the workshop as fish.
		nd_ntr_whalebone_carving = {
			fish = 0.3
			tools = 0.05
			produced = jewelry
			output = 0.108
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
