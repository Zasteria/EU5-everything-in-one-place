# Bergslagen Ironworks: industrial mining building for iron production
nd_sca_bergslagen_forge = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = manufactory_employment
	category = basic_industry_category
	# Not rural: this is a workshop, not a field. A per-location
	# bonus buildable in everything you own is a realm-wide bonus
	# wearing a local label, and vanilla keeps industry in towns.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = manufactory_build_time

	country_potential = {
		has_advance = nd_sca_bergslagen_ironworks
	}

	modifier = {
		local_iron_output_modifier = 0.1
		local_copper_output_modifier = 0.1
	}

	unique_production_methods = {
		nd_sca_osmund_iron_smelting = {
			iron = 0.3
			coal = 0.2
			produced = steel
			output = 0.312
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		# Lancashire forging (Sweden, from the 1830s): closed hearths fired
		# with charcoal, so the works runs on lumber and needs no coal, which
		# Scandinavia barely has. eu5-profit: 1.65 in, 0.396 steel out, 20%.
		nd_sca_lancashire_forging = {
			iron = 0.4
			lumber = 0.3
			produced = steel
			output = 0.396
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	# Profit: iron 0.3*3.0 + coal 0.2*2.0 = 1.30 input
	# output = 1.30*1.20/5.0 = 0.312 -> 0.31
	# revenue = 0.31*5.0 = 1.55; profit = (1.55-1.30)/1.30 = 19.2% (in 15-30% range)
	construction_demand = manufactory_construction
}
