# Ile Ise: the house of Yoruba weavers and dyers (all three methods make cloth)
nd_oye_ile_ise = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_oye_yoruba_guilds
	}

	modifier = {
		local_production_efficiency = tiny_production_efficiency_bonus
		local_pop_promotion_speed = 0.002
		local_burghers_estate_power = 0.5
	}

	# No location gate. The workshop is a weavers' and dyers' house: every
	# product is a made good and its inputs are its gate. Until 2026-10-02
	# it also held the kola method and was bound to markets that produce
	# medicaments, which shut the cloth methods out of every other market.
	# Kola now has a building of its own on kola land (nd_oye_kola_grove).

	unique_production_methods = {
		# Cotton, not wool: Yoruba cloth was woven from cotton. Output from
		# eu5-profit at the guild margin (20%).
		nd_oye_craft_maintenance = {
			dyes = 0.1
			cotton = 0.1
			produced = cloth
			output = 0.28
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		# Unlocked by nd_oye_textile_expansion: narrow strips of cotton
		# cloth woven with sanyan, the wild silk of the savanna.
		nd_oye_aso_oke_weaving = {
			cloth = 0.2
			silk = 0.1
			produced = fine_cloth
			output = 0.2
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		# Unlocked by nd_oye_indigo_trade: cloth dyed deep blue in the
		# pits of the dyers' quarter.
		nd_oye_indigo_dyeing = {
			cloth = 0.3
			dyes = 0.1
			produced = fine_cloth
			output = 0.26
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
