# Knights' Ribat: an old ribat of the coast held by brother knights. Named apart
# from vanilla's order_commandery, the foreign building a military order plants
# in a sponsor's land, which can stand in the same town.
nd_koa_commandery = {
	is_special = yes
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	employment_size = castle_employment
	category = military_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = soldier_building

	country_potential = {
		has_advance = nd_koa_order_commanderies
	}

	modifier = {
		local_manpower = 0.005
		local_garrison_size = 0.1
		local_defensive = 0.25
		can_recruit_regiment_in_this_location = yes
	}

	unique_production_methods = {
		nd_koa_commandery_maintenance = {
			stone = 0.2
			weaponry = 0.1
			leather = 0.1
			category = building_maintenance
		}
	}
	construction_demand = order_commandery_construction
}

# Coral Fishery: the boats that drag the red coral banks off Tabarka. The
# map gives those shores gems as their raw good, and the fishery stands
# only there. A plain producer with no modifier block, in vanilla's shape
# for a rural producer. Unlocked by nd_koa_coral_trade.
nd_koa_coral_fishery = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	# A fishing station of the coast: a village or a small port.
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_koa_coral_trade
	}

	# Only where the land's own raw good is the coral, and only on the sea.
	location_potential = {
		raw_material = goods:gems
		is_coastal = yes
	}

	unique_production_methods = {
		# Inputs are what the fishing consumes: timber for the boats and
		# the weighted cross that is dragged over the banks, hemp for its
		# nets, and tools.
		nd_koa_coral_harvesting = {
			lumber = 0.1
			fiber_crops = 0.1
			tools = 0.05
			produced = gems
			output = 0.15
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}

# Olive Press: the stone mills and presses of the Sahel of Tunis. A plain

	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = rural_peasant_produce_employment

	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_koa_olive_export
	}

	# Only where the land itself bears olives.
	location_potential = {
		raw_material = goods:olives
	}

	unique_production_methods = {
		# Inputs are what the pressing consumes: jars for the oil, tools
		# for the press, and the animals that turn the mill.
		nd_koa_olive_pressing = {
			pottery = 0.15
			tools = 0.05
			livestock = 0.05
			produced = olives
			output = 0.45
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}
