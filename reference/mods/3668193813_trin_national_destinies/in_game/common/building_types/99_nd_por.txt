# Portugal (POR): custom buildings
# National Destinies mod
# ============================================================================
#
# What is NOT here, because vanilla already built it:
#   the Feitoria (por_feitoria, unlocked by por_feitorias), the Tower of
#   Belem and the Ribeira das Naus of Lisbon (event buildings), the Casa da
#   India (a government reform and a decision), the Estudo Geral of Coimbra
#   and Lisbon (a law), the Jesuit college, the viceroyalty and the
#   lieutenancy, and the generic sugar plantation of the colonies.
# What is NOT here, because the mod's Spain already holds it for a country
#   that goes on from Portugal (99_nd_spa.txt): the colonial mission, the
#   house of trade, the victualling yard, the mint and the royal foundry.
#
# What is here is the kingdom's own ground: the king's pine forest behind the
# dunes, the merchants' common purse, the tower that keeps the crown's
# parchments, the cane mill of the islands, the house of mercy in every town,
# the walled terraces of the Douro, the tile kilns, the royal factories and
# the school that taught merchants to keep books.
#
# Every production method is inline and was sized with eu5-profit (output
# floored, margin equal to the named target, cap written as a name). A
# producer that can stand in the countryside names rural_profit_margin.

# ----------------------------------------------------------------------------
# Pinhal do Rei (age 1)
# ----------------------------------------------------------------------------
# The King's Pinewood: maritime pine sown behind the dunes of Leiria under
# Afonso III to stop the sand, enlarged by Dinis, and kept as a crown forest
# with its own wardens. It gave the yards of the Tagus their planking, masts,
# pitch and tar for five hundred years.
# location_potential: the location's own raw material is lumber AND it lies
# on the coast. A royal pinewood is a standing forest, not a workshop, so it
# goes where the trees are; and its whole worth is that timber and pitch can
# be floated to a shipyard, so an inland forest does not qualify. In the
# kingdom of 1337 that is Leiria and little else.
# Settlement: countryside and small town, never a city. A forest is not a
# street.
# No modifier block, so the per-location level formula of vanilla's rural
# producers is safe here.
nd_por_pinhal_do_rei = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = tiny_rural_peasant_produce_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_por_pinhal_de_leiria
	}

	location_potential = {
		raw_material = goods:lumber
		is_coastal = yes
	}

	unique_production_methods = {
		# Planking, spars and pitch made ready for the yards: the pine is
		# felled and squared, the stumps and resin burned down for tar.
		nd_por_pinhal_aprestos_navais = {
			lumber = 1.0
			tools = 0.1
			tar = 0.3
			produced = naval_supplies
			output = 0.96
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = guild_construction
}

# ----------------------------------------------------------------------------
# Casa da Bolsa (age 1)
# ----------------------------------------------------------------------------
# The common purse of the merchants. On 10 May 1293 King Dinis confirmed the
# fund the traders of the realm had agreed among themselves: every ship
# loading for Flanders, England or France paid a levy by tonnage into a
# chest, and the chest paid for lawsuits, ransoms and losses abroad. In 1380
# Fernando I turned the same idea into the Companhia das Naus: every decked
# ship above fifty tuns registered, two per cent of each freight paid in,
# and a lost hull replaced out of the purse kept at Lisbon and at Porto.
# location_potential and allow: vanilla's port authority pattern. The purse
# insured ships, so it stands where ships load.
# Settlement: towns and cities. The chest was kept by the merchants of a
# chartered port, not by a fishing hamlet.
# Upkeep: 1.2 gold a month at base prices (paper 2, naval supplies 3,
# cloth 3), between vanilla's merchants' quarters (1.1) and port authority
# (1.75 a level): ledgers, and the cordage and canvas the fund replaces.
nd_por_casa_da_bolsa = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = generic_burgher_employment
	category = trade_category
	rural_settlement = no

		has_advance = nd_por_bolsa_dos_mercadores

	}

	allow = {

# ----------------------------------------------------------------------------
# Engenho (age 3; a second method in age 4)
# ----------------------------------------------------------------------------
# The cane mill: rollers turned by oxen or by a water wheel, a boiling house
# with its row of copper kettles, a purging house where the loaves drain
# white. Madeira had them from the 1450s and had passed Cyprus by the 1490s;
# Sao Tome and then Pernambuco and Bahia copied Madeira.
# Not vanilla's sugar plantation: that one is the colonial estate worked by
# slaves and stands only overseas or in a colonial subject. The engenho is
# the mill itself, worked by hired hands, and may stand in any cane country
# its owner holds, the home islands included.
# location_potential: strict form, the location's own raw material is sugar.
# The mill makes a raw good, and cane cannot be carted far before it sours.
# Settlement: countryside and small town. The mill stands among the canes.

		has_advance = nd_por_engenhos_da_madeira

	location_potential = {
		raw_material = goods:sugar
	}

	unique_production_methods = {

# ----------------------------------------------------------------------------
# Santa Casa da Misericordia (age 3)
# ----------------------------------------------------------------------------
# The Holy House of Mercy: a lay brotherhood under the crown's protection,
# founded in Lisbon by Queen Leonor in 1498 and bound to the fourteen works
# of mercy. It runs the hospital, feeds prisoners, buries the poor, raises
# foundlings and dowers orphan girls, and is governed by brothers of the
# first and second condition in equal number, gentlemen and craftsmen. Within
# a generation there was one in every town of the kingdom, and then in Goa,
# Macau, Olinda, Luanda and Nagasaki.
# No location_potential on purpose: the Misericordia was the one institution
# every Portuguese town had, from Braganca to Macau, whatever its size or
# wealth. That is what set it apart from the hospital, which vanilla gates on
# development. The cost of that reach is paid in the numbers below: each is
# a fraction of the vanilla hospital's.
# Settlement: towns and cities. A brotherhood needs a town's worth of
# brothers.
# Upkeep: 0.8 gold a month at base prices (medicaments 1, cloth 3, wheat 1,
# wine 2, paper 2), just above the vanilla hospital (0.75): physic, shrouds
# and shirts, the prisoners' bread and wine, and the brotherhood's books.

# Real Fabrica (age 6)
# ----------------------------------------------------------------------------
# A royal factory: a works set up or taken over by the crown, housed in one
# building, given a monopoly, a tax exemption and foreign masters, and told
# to replace an import. The silk works of the Rato in Lisbon (1734), the
# woollen works of Covilha and Portalegre (1764, 1772) and the glass works of
# Marinha Grande, which William Stephens took over in 1769 with a royal loan
# and the right to cut his fuel in the King's Pinewood.
# location_potential: wool, silk or lumber must reach the local market. The
# crown planted each works beside its material or its fuel: wool in the
# Serra da Estrela, firewood at Marinha Grande, raw silk landed at Lisbon.
# A town with none of the three has nothing for a royal factory to work.
# Settlement: towns and cities.

# ----------------------------------------------------------------------------
# Paco da Ribeira (destiny, path B capstone)
# ----------------------------------------------------------------------------
# The Palace on the Riverbank. From 1498 Manuel I left the old castle on the
# hill of Lisbon and built his house on the quay itself, beside the
# shipyard, with the Casa da India and its pepper stores inside its walls.
# The court had moved in within a few years. A king who could watch the
# India fleet unload from his own windows was a new kind of king, and
# Francis I of France called him the grocer king for it. The earthquake of
# 1 November 1755 destroyed it and it was never rebuilt.
# Not vanilla's royal court (cabinet and crown power, any monarchy's
# capital), not its Ribeira das Naus (the shipyard next door, an event
# building), not the Casa da India (a reform and a decision) and not the
# heritage tree's Torre do Tombo (the archive in the castle the king left).
# location_potential and allow: the capital, on the coast, one in the realm.
# The whole point of the palace is that it stands on the waterfront; a
# capital inland has no riverbank to move the court down to.
# Settlement: town or city, whatever the capital is; never the countryside.
# modifier: merchants gather where the king lodges over the warehouse.
# Half of vanilla's merchants' guild (1 and 1), in one location.
# capital_country_modifier: prestige as vanilla's royal garden (0.05) and
# the selling margin of vanilla's Porto Pisano (the tiny named step).
# Upkeep: 2.2 gold a month at base prices, a court's bill (paper 2, fine
# cloth 6, furniture 3, glass 3).

	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = 1
