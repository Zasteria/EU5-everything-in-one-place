# Island Moat Fortress: Ayutthaya's unique water-defense system
# The city sat on an artificial island created by digging canals to connect
# the Chao Phraya, Pa Sak, and Lopburi rivers, making it virtually impregnable.
nd_ayu_island_fortress = {
	is_special = yes
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	employment_size = castle_employment
	category = defense_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = soldier_building

	country_potential = {
		has_advance = nd_ayu_water_defenses
	}

	modifier = {
		local_garrison_size = 0.15
		local_defensive = 0.2
		hostile_disembark_time_modifier = 0.5
		local_hostile_attrition = 0.25
	}

	unique_production_methods = {
		nd_ayu_fortress_upkeep = {
			lumber = 0.2
			stone = 0.15
			tools = 0.1
			iron = 0.1
			category = building_maintenance
		}
	}
	construction_demand = early_soldier_building_construction
}

# Deerskin Yard: where the hides of the upland forests were dried, salted,
# sorted by grade and baled for the Japan ships. The crown bought them in as
# tribute in kind and the Japanese and Dutch factors took them by the hundred
# thousand each monsoon.
# location_potential: wild game must reach the local market. Leather is a
# made good, so nothing is conjured; the gate keeps the yard on rivers the
# hides actually came down.
# Settlement: towns and cities. The hunting was done in the forest, the
# sorting and the sale at the river ports.
nd_ayu_deerskin_yard = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = basic_industry_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_ayu_deerskin_trade
	}

	location_potential = {
		market ?= { is_produced_in_market = goods:wild_game }
	}

	unique_production_methods = {
		nd_ayu_deerskin_processing = {
			wild_game = 0.8
			salt = 0.05
			lumber = 0.1
			produced = leather
			output = 0.46
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
