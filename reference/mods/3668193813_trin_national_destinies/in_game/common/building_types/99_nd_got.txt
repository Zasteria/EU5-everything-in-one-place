# Regnum Gothorum (GOT): Custom buildings
# National Destinies mod
#
# The institutional package of a late-antique state that never fell: the city
# court that applies one code to everyone, the joint council of crown and
# church, the pass forts that hold a realm together across a mountain range,
# the way-stations that carry the king's writ from Toledo to Narbonne, the
# primatial see, the chapel of a rite older than Rome's reformed liturgy, the
# episcopal hospital open to every faith, the mint that puts the king's name
# in eighty provincial cities, the royal goldsmiths, and a faculty that
# teaches a seventh-century law book as living law.
#
# Every building here carries an explicit location gate or an explicit
# settlement-tier restriction with the reason written down.

# Court of the Liber Iudiciorum. Recceswinth's code required a certified copy
# in every court and forbade citing any other law, foreign law included. The
# court was the comes civitatis and his iudex, sitting in a walled town; rural
# disputes came in to it. Hence town and above only, and one per location.
nd_got_liber_court = {
	audio_tier = 2
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = generic_burgher_employment
	category = government_category

	# The Gothic court sat where the count sat, in a chartered town or city.
	# There was no village jurisdiction: villages were judged at the town.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	build_time = government_build_time

	country_potential = {
		has_advance = nd_got_liber_iudiciorum
	}

	modifier = {
		local_pop_assimilation_speed_modifier = 0.01
		local_unrest = -0.05
		local_crown_estate_power = 0.25
		local_max_control = 0.05
	}

	unique_production_methods = {
		nd_got_liber_court_maintenance = {
			paper = 0.2
			books = 0.1
			leather = 0.1
			beeswax = 0.05
			category = building_maintenance
		}
	}

	construction_demand = local_governor_construction
}

# Aula Regia. The palatine court and the councils that met with it: bishops,

# together and promulgating their canons as royal law. Eighteen councils

# anywhere else would have been a schism, so this is capital-only by nature
# rather than by balance.

	is_foreign = no

# Royal Goldsmiths' Workshop. The Guarrazar hoard is twenty-six votive crowns

# hanging letters. This is the only production building in the tree and it is
# gated on the raw material: cloisonne work needs gold or stones in the local
# market, and the crowns were made where the metal was.

			is_produced_in_location_market = goods:goods_gold

			silver = 0.45

# ===== DESTINY BUILDINGS =====
# One per destiny path, and each states its path's thesis in stone.

# Thesaurus Publicus. The Eighth Council of Toledo (653) separated the
# treasury of the kingdom from the private patrimony of the king: what a king
# acquires after his accession is the realm's and does not pass to his heirs.
# A treasury that is not the king's purse needs an office, a strongroom and an
# audit, and the Visigothic fisc was kept by numerarii in the provincial
# capitals rather than in every chartered town. Hence city and above only, one
# per location.
nd_got_dest_thesaurus = {
	audio_tier = 3
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = large_burgher_employment
	category = government_category

	# The numerarius and the comes patrimonii sat where the provincial
	# administration sat. A village or a market town had a tax collector, not
	# a treasury of the kingdom.
	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes

	build_time = bank_office_build_time

	country_potential = {
		has_advance = nd_got_dest_fiscus_publicus
	}

	modifier = {
		local_monthly_prosperity = 0.001
		local_max_control = 0.1
		local_burghers_estate_power = 0.25
		# Revenue that visibly belongs to the kingdom is not resented as the
		# king's private taking.
		local_unrest = -0.05
	}

	unique_production_methods = {
		nd_got_dest_thesaurus_maintenance = {
			paper = 0.25
			fine_cloth = 0.1
			furniture = 0.1
			tools = 0.1
			category = building_maintenance
		}
	}

	construction_demand = bank_construction
}

# Thiudangardi. Wulfila's Gothic word for kingdom is thiudangardi, literally
# the king's enclosure: a household, not an office. That is Path B's whole
# argument in one noun, and it is the answer to the Latin regnum. The hall
# feeds the sworn men, quarters them where the crown holds land, and is where
# an heir is acclaimed in his father's lifetime while the men who would
# otherwise elect are standing in the room.
nd_got_dest_thiudangardi = {
	audio_tier = 3
	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	employment_size = normal_noble_employment
	category = military_category

	# A household hall needs a settlement to provision it. The countryside
	# supplies it; it does not sit in it.
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	build_time = big_soldier_building

	country_potential = {
		has_advance = nd_got_dest_gutthiuda
	}

	# The palatine court and its guard are already at Toledo in the heritage
	# tree. The thiudangardi is the provincial household hall, and keeping
	# sworn men in the country rather than at the palace is precisely what
	# defeats an elective court.
	location_potential = {
		is_capital = no
	}

	modifier = {
		local_levy_size_modifier = 0.1
		# Engine cap: manpower per employed pop may not exceed 0.0125 here
		# (normal_noble_employment 0.05); the 2026-09-10 boot log flagged 0.015.
		# 0.00062, not the exact 0.000625: the engine rejects a sixth decimal
		# as a malformed token and drops the value.
		local_manpower = 0.00062
		local_nobles_estate_power = 0.25
		can_recruit_regiment_in_this_location = yes
	}

	unique_production_methods = {

			livestock = 0.25
			wine = 0.15
