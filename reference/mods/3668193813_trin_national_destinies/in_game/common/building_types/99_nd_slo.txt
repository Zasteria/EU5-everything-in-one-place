# Tajch: a mining reservoir of the Stiavnica hills. Samuel Mikoviny, director
# and first professor of the mining school of 1735, and the two Hells laid
# out the system: at its height sixty reservoirs joined by over 100 km of
# channels. The water drove the pumps that drained the shafts, and the stamp
# mills. A drained mine can go deeper, which is what the building gives.
nd_slo_tajch = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = tiny_rural_peasant_produce_employment
	category = infrastructure_category
	# A dam in the hills above a mining town: never inside a great city.
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	expensive = yes
	build_time = infrastructure_build_time

	country_potential = {
		has_advance = nd_slo_technical_education
	}

	# Only where there is an ore mine to drain.
	location_potential = {
		OR = {
			raw_material = goods:silver
			raw_material = goods:goods_gold
			raw_material = goods:copper
			raw_material = goods:iron
			raw_material = goods:lead
			raw_material = goods:tin
		}
	}

	modifier = {
		# Vanilla's mining charter gives a mining town 0.35 (town_rights/00_traditions.txt:147).
		local_max_rgo_size_modifier = 0.25
	}

	unique_production_methods = {
		nd_slo_tajch_maintenance = {
			lumber = 0.15
			stone = 0.1
			tools = 0.05
			category = building_maintenance
		}
	}
	construction_demand = pound_lock_canal_infrastructure_construction
}
