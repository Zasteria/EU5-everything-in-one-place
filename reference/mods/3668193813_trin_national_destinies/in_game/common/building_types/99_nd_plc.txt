# Folwark: noble agricultural estate, grain export economy
nd_plc_folwark = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 3
	employment_size = estate_peasants_employment
	category = rgo_building_category
	rural_settlement = yes
	town = no
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_plc_folwark_economy
	}

	modifier = {
		local_wheat_output_modifier = 0.15
		local_max_rgo_size = 1
		local_peasants_estate_power = 0.25
	}

	# Extraction is bound to the resource: you cannot mine, tap or
	# harvest wheat where there is none. Without this the building
	# is a free resource generator in every location the owner holds.
	location_potential = {
		is_produced_in_location_market = goods:wheat
	}

	unique_production_methods = {
		nd_plc_folwark_maintenance = {
			lumber = 0.1
			livestock = 0.05
			produced = wheat
			output = 0.27
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}

# Oledry: a village "under Hollander law". Mennonites from the Low
# Countries diked and drained the Vistula delta from 1547, and the charter
# then went to anyone who would hold flood land for a money rent instead of
# labour dues. The counterpart of the folwark: free tenants, not serfs.
nd_plc_oledry = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 1
	employment_size = village_employment_size
	category = infrastructure_category
	# A Hollander village is a rural colony on flood land, never a town.
	rural_settlement = yes
	town = no
	city = no
	expensive = no
	build_time = infrastructure_build_time

	country_potential = {
		has_advance = nd_plc_refuge_of_nations
	}

	# The charter was given for land nobody else would farm: river flood
	# plain and marsh (vanilla gates its polders on wetlands and its
	# watermills on has_river).
	location_potential = {
		OR = {
			has_river = yes
			topography = wetlands
		}
	}

	# The first two lines are what vanilla's polders give for reclaimed
	# land; the third is the pull of a village without labour dues, at the
	# size vanilla gives a university.
	modifier = {
		local_population_capacity_modifier = 0.025
		local_monthly_food_modifier = monthly_food_productivity_weak_bonus
		local_migration_attraction = 0.05
	}

	unique_production_methods = {
		nd_plc_oledry_upkeep = {
			clay = 0.3
			lumber = 0.1
			tools = 0.05
			category = building_maintenance
		}
	}
	construction_demand = polders_construction
}
