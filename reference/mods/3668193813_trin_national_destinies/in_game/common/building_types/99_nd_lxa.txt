# Ivory Carvers' Workshop (age 3, nd_lxa_ivory_craft)
# Tusks from the royal herds and the upland tribute were carved into hilts,
# manuscript covers and the pairs of tusks set beside a Buddha image.
# location_potential: ivory or elephants reaching the local market. Ivory is
# the input here, not the product, so the market form of the gate is the
# right one: the carvers follow the tusks.
# No modifier block: a plain producer in vanilla's guild shape, so the level
# formula multiplies nothing but the method.
nd_lxa_ivory_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = consumer_goods_category

	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_lxa_ivory_craft
	}

		OR = {
			is_produced_in_location_market = goods:ivory
			is_produced_in_location_market = goods:elephants
		}

# Ho Tai (age 5, nd_lxa_manuscript_tradition)
# The scripture library of a wat. Monks incise the Pali texts on dried palm
# leaves, blacken the cuts with soot and bind the leaves between wooden
# covers. Vanilla's scriptorium makes books from paper, dyes and lumber; the
# palm leaf takes the paper's place, so the bill is leaf and cover wood
# (lumber), ink and lacquer (dyes) and the stylus (tools).
# No location_potential: every Lao town had its wat and every wat its
# manuscripts, and nothing in the bill is tied to the ground.
# No modifier block: a plain producer in vanilla's guild shape.
nd_lxa_ho_tai = {
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = guild_max_level
	employment_size = religious_building_employment
	category = consumer_goods_category

	build_time = guild_build_time

	country_potential = {
		has_advance = nd_lxa_manuscript_tradition
