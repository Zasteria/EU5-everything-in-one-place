# Himroo Looms: cotton and silk brocade (Age 3, production)
# Himroo is the Deccan's own cloth: a cotton warp under a silk weft, woven
# to look like the gold brocades of Persia at a fraction of the price. The
# weavers came south with the Tughluq court to Daulatabad and stayed when it
# went back.
# location_potential: cotton must reach the local market. Fine cloth is a
# made good, so nothing is conjured; the gate keeps the looms on the black
# cotton soil the craft grew up on.
# Settlement: towns and cities. A court craft sold through bazaars.
# No modifier block, so the building takes the level formula of its tier.
nd_bah_himroo_looms = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = guild_max_level
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_bah_deccan_textiles
	}

	location_potential = {
		is_produced_in_location_market = goods:cotton
	}

	unique_production_methods = {
		nd_bah_textile_pm = {
			cotton = 0.4
			silk = 0.2
			produced = fine_cloth
			output = 0.4
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		nd_bah_cotton_mill_pm = {
			cotton = 1.2
			dyes = 0.1
			tools = 0.05
			produced = cloth
			output = 1.66
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
