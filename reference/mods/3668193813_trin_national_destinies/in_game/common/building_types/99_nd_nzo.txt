# Salt Wells: Yunnan's salt extraction industry, a crucial revenue source
# for both Nanzhao and Dali kingdoms
nd_nzo_salt_wells = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 3
	employment_size = rural_peasant_produce_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_nzo_salt_extraction
	}

	modifier = {
		local_raw_material_output = 0.25
		local_crown_estate_power = 0.25
		local_monthly_prosperity = 0.001
	}

	# Extraction is bound to the resource: you cannot mine, tap or
	# harvest salt where there is none. Without this the building
	# is a free resource generator in every location the owner holds.
	location_potential = {
		is_produced_in_location_market = goods:salt
	}

	unique_production_methods = {
		# Output at the rural margin (0.2 x 1.2 / 4). It stood at 0.15 with the
		# cap switched off, three times what the inputs pay for, and above
		# the crown method an advance unlocks later.
		nd_nzo_salt_processing_pm = {
			lumber = 0.1
			clay = 0.1
			produced = salt
			output = 0.06
			category = guild_input
			debug_max_profit = rural_profit_margin
		}

		nd_nzo_crown_salt_pm = {
			lumber = 0.2
			clay = 0.1
			produced = salt
			output = 0.105
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
