# Dluwang Works: bark paper for the pesantren's books. The inner bark of the
# paper mulberry is soaked, beaten thin with a pounder, dried on banana leaf
# and smoothed with a shell. From the late sixteenth century it was the
# favoured sheet for Javanese Islamic writing, because it takes Pegon script
# better than palm leaf; it is also called Ponorogo paper.
# location_potential: a location that has a pesantren. The sheets were made
# for the schools' copyists.
# Settlement: countryside and towns, where the schools themselves stand. A
# garden craft of village households, hence laborers and a level cap that
# follows the school's own two levels.
nd_mtm_dluwang_works = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 2
	employment_size = rural_peasant_produce_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_mtm_pesantren_expansion
	}

	location_potential = {
		has_building = building_type:nd_mtm_pesantren
	}

	unique_production_methods = {
		nd_mtm_dluwang_beating = {
			fiber_crops = 0.4
			lumber = 0.1
			produced = paper
			output = 0.57
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = village_construction
}
