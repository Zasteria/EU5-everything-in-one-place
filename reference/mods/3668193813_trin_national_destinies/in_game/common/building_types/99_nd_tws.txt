# Masseria: fortified agricultural estate, the backbone of Mezzogiorno grain economy
# These were large-scale farming complexes that made southern Italy the granary of the Mediterranean
nd_tws_masseria = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 3
	employment_size = generic_peasant_building_employment
	category = basic_industry_category
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_tws_masseria_system
	}

	modifier = {
		local_monthly_food_modifier = 0.05
		local_wheat_output_modifier = 0.15
		local_population_capacity = 0.5
	}

	# Extraction is bound to the resource: you cannot mine, tap or
	# harvest livestock where there is none. Without this the building
	# is a free resource generator in every location the owner holds.
	location_potential = {
		is_produced_in_location_market = goods:livestock
	}

	unique_production_methods = {
		# From the start: the estate's own flock on its stubble. It was the
		# building's only method and stood locked behind an age 2 advance, so
		# an age 1 masseria had nothing to run.
		nd_tws_stubble_grazing = {
			tools = 0.05
			salt = 0.02
			produced = livestock
			output = 0.184
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
		# Unlocked by nd_tws_mediterranean_granary: the grain surplus fattens
		# herds kept for breeding. Fodder and breeding stock in, more stock
		# out, the shape of vanilla's horse_breeders (wheat and horses make
		# horses). Inputs cost 0.4 + 0.15 + 0.15 = 0.7; 0.56 livestock at 1.5
		# is 0.84, the rural margin of 20%.
		nd_tws_grain_export = {
			wheat = 0.4
			livestock = 0.1
			tools = 0.05
			produced = livestock
			output = 0.56
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}

# Locazione: a winter pasture of the Tavoliere, measured out and let to the
# flock masters of the Abruzzi by the Dogana della Mena delle Pecore (Foggia,
# 1447 to 1806).
nd_tws_locazione = {
	is_special = yes
	is_foreign = no
	pop_type = peasants
	max_levels = 2
	employment_size = generic_peasant_building_employment
	category = basic_industry_category
	# Open pasture: the Dogana barred the plough from its locazioni, so this
	# never stands in a town or a city.
	rural_settlement = yes
	town = no
	city = no
	expensive = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_tws_dogana_delle_pecore
	}

	# Wool is shorn where the flocks winter and summer. Inside the kingdom the
	# map grows it at Lucera (the Dogana's first seat), Celano, Cittaducale,
	# Lagonegro and Vallo.
	location_potential = {
		is_produced_in_location_market = goods:wool
	}

	modifier = {
		local_wool_output_modifier = 0.1

		# the same 0.25).
		local_crown_estate_power = 0.25

	unique_production_methods = {
		nd_tws_fida_flocks = {
			salt = 0.05
			tools = 0.05
			produced = wool
			output = 0.168
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}
