# ===== Francia (FRK) Buildings =====
# Two kinds. The Carolingian institutions (missaticum, fiscal villa,
# cathedral school, palace school) are buildable across the realm under
# gates that say where each actually stood. The monuments are bound to the
# one city that holds them: Aachen, Wetzlar, Frankfurt, Worms, Cologne and
# Mainz are all inside Francia's own territory, and Aachen and Frankfurt
# are formation requirements, so the player owns them the day Francia exists.

# Seat of the Missi
# The missaticum was an inspection circuit of several counties, ridden by a
# bishop and a count sent from outside it. One seat per province, at the
# province capital and never at the realm's capital, which the court already
# watches. Rural seats are allowed because many province capitals are still
# villages; that is why this uses the rahdar's slow control growth and not a
# flat control ceiling.
nd_frk_missaticum = {
	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	employment_size = normal_noble_employment
	category = government_category
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = government_build_time

	country_potential = {
		has_advance = nd_frk_missi_dominici
	}

	location_potential = {
		is_province_capital = yes
		is_capital = no
	}

	modifier = {
		local_monthly_control = 0.001
		local_unrest = -0.05
		local_crown_estate_power = 0.25
	}

	unique_production_methods = {
		nd_frk_missaticum_maintenance = {
			paper = 0.2
			horses = 0.1
			wine = 0.1
			category = building_maintenance
		}
	}
	construction_demand = local_governor_construction
}

# Royal Villa

# weaving house of women (the genitium), a brewhouse and a cellar. Arable
# land only, because the villas were grain and vine estates; the capitulary
# has nothing to say about forest, marsh or mountain. Every PM turns inputs
