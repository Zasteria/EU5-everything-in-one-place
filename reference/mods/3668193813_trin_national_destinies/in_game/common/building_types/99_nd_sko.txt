	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_sko_hausa_urbanism
	}

	modifier = {
		local_trades_per_burgher = 0.1
		local_production_efficiency = small_production_efficiency_bonus
	}

	unique_production_methods = {
		# The pits dye cloth that is already woven: plain cloth and indigo go
		# in, the glazed blue cloth of the caravan trade comes out. Until
		# 2026-10-02 this made cloth from dyes alone.
		nd_sko_dye_pit_maintenance = {
			cloth = 0.3
			dyes = 0.1
			produced = fine_cloth
			output = 0.26
			category = guild_input
			debug_max_profit = guild_profit_margin
		}

		# Unlocked by nd_sko_kano_textile_trade. Until 2026-10-02 this was a
		# copy of the method above, so the advance unlocked nothing new: now
		# the pits take in raw cotton as well and dye for export.
		nd_sko_indigo_dyeing = {
			cotton = 0.4
			dyes = 0.2
			produced = cloth
			output = 0.8
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
