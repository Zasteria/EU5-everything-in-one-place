# Solana: the commune's own salt pans. Those of Ston, bought with the
# peninsula in 1333, brought the Republic 15,900 ducats a year.
# One works to a place, sized like one level of vanilla's salt collector,
# and fed with timber and tools where the collector burns clay.
nd_dal_solana = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	startup_ramp_target = rural_startup_ramp_target
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	# Pans lie on the flats outside a town's walls, never inside a great city.
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_dal_adriatic_salt_routes
	}

	# Sea salt: only on a shore whose own good is salt.
	location_potential = {
		is_coastal = yes
		raw_material = goods:salt
	}

	unique_production_methods = {
		nd_dal_salt_panning = {
			lumber = 0.4
			tools = 0.15
			produced = salt
			output = 0.315
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = farm_constructions
}

# Friars' Pharmacy: the pharmacy the Friars Minor of Ragusa have kept open
# to the townspeople since 1317. Its remedies are made of what the coast
# grows (oil, wine, wax), where vanilla's apothecary uses game and flax.
# A plain producer with no modifier block, so it takes the guild formula.
nd_dal_friars_pharmacy = {
	is_special = yes
	is_foreign = no
	pop_type = clergy
	max_levels = guild_max_level
	startup_ramp_target = guild_startup_ramp_target
	employment_size = clergy_medical_employment
	category = basic_industry_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_dal_pharmacy_tradition
	}

	# The herbs have to be at hand: sage and the like are what the islands
	# sell (Pag's own good is medicaments).
	location_potential = {
		is_produced_in_location_market = goods:medicaments

	unique_production_methods = {
		nd_dal_friars_remedies = {
			olives = 0.4

			beeswax = 0.1
			produced = medicaments
			output = 1.2
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Lanterna: a harbour light. The first on this coast was lit at Savudrija in
# Istria in 1818. A ship that can find the harbour mouth at night lands its
# cargo more cheaply, and the coast is watched from the tower.
nd_dal_lanterna = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = tiny_rural_peasant_produce_employment
	category = infrastructure_category
	# A light stands on a cape as well as on a city's mole: every rank.
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = infrastructure_build_time

	country_potential = {
		has_advance = nd_dal_port_modernization
	}

	location_potential = {
		is_port = yes
	}

	modifier = {
		# Vanilla's rural harbour buildings give 0.02 to 0.05, its docks 0.1 and more.
		harbor_suitability = 0.05
		# Vanilla's fishing village gives 0.01, its coastal forts 0.15 and 0.3.
		local_maritime_presence = 0.05
	}

	unique_production_methods = {
		nd_dal_lanterna_maintenance = {
			stone = 0.1
			glass = 0.05
			olives = 0.1
			category = building_maintenance
		}
	}
	construction_demand = port_building_construction
}
