# Vitte: a herring station. The Hanse towns kept fenced compounds of booths
# on the herring coasts where the catch was gutted, salted and packed in
# barrels for the Lenten trade. Stralsund and Stettin had theirs at
# Falsterbo in Scania; Vitt on Ruegen and Vitte on Hiddensee keep the name.
nd_pom_vitte = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = guild_employment
	category = basic_industry_category
	# A Vitte stood on the beach by the fishing grounds, outside any town
	# wall, or below a small harbour town. Never in a city.
	rural_settlement = yes
	town = yes
	city = no
	expensive = no
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_pom_herring_trade
	}

	# Bound to the herring grounds: a sea shore whose own raw good is fish
	# (Ruegen, Kamien, Ueckermuende and Koeslin in Pomerania).
	location_potential = {
		is_coastal = yes
		raw_material = goods:fish
	}

	unique_production_methods = {
		# Vanilla's fishing village makes fish from salt and tools the same
		# way (fishing_village_maintenance); here the barrel staves take the
		# place of the tools.
		nd_pom_herring_salting = {
			salt = 0.15
			lumber = 0.05
			produced = fish
			output = 0.81
			category = guild_input
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = guild_construction
}
