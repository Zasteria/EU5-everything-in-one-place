# Marwari Stud: the horse of the Thar.
# The Rathores of Marwar bred the Marwari from the twelfth century for the
# desert: a hardy horse that marches far on little water. Vanilla's horse
# breeders stand only in a mediterranean, continental or oceanic climate
# (common_buildings.txt, horse_breeders, allow), so no stud of vanilla's can
# be built in Rajputana. This one is its desert counterpart: the same market
# gate (horses must be raised in the location's market: Nagaur and Ramgar are
# horse country) and the dry climates in place of the temperate ones.
# Fed on millet, the bajra of the Thar, where vanilla's breeders take wheat
# (wheat 1.563, horses 0.313 for 1). Output from eu5-profit at the rural
# margin: millet 1.2, horses 0.25 costs 1.95 and gives 0.78.
# A plain producer with no modifier, so it takes vanilla's rural level formula
# as horse_breeders does. Countryside and towns: a stud farm is not a city.
nd_rjp_marwari_stud = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	startup_ramp_target = rural_startup_ramp_target
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = no
	build_time = rural_build_time

	country_potential = {
		has_advance = nd_rjp_marwari_horse_tradition
	}

	location_potential = {
		is_produced_in_location_market = goods:horses
	}

	allow = {
		OR = {
			climate = arid
			climate = hot_semi_arid
		}
	}

	unique_production_methods = {
		nd_rjp_marwari_breeding = {
			millet = 1.2
			horses = 0.25
			produced = horses
			output = 0.78
			category = building_maintenance
			debug_max_profit = rural_profit_margin
		}
	}
	construction_demand = horse_breeders_construction
}

# Suratkhana: the painters' department of a Rajput court.
# Every great house kept one: Mewar's school is dated from the Chawand
# Ragamala of 1605, Bundi and Kota followed, and Kishangarh's Nihal Chand
# painted for Savant Singh in the 1730s and 1740s. The folios were kept as
# loose leaves in cloth wrappers, which is why cloth stands where vanilla's
# scriptorium takes leather.
# Town and city: the courts sat in small hill towns as often as in cities.
# Two levels and no modifier: a plain producer. Paper, pigments and cloth in,
# books out, as vanilla's scriptorium (dyes 0.0503, paper 0.1995 and lumber
# or leather for 0.3). Outputs from eu5-profit at the guild margin:
#   paper 0.2, dyes 0.08, cloth 0.05 costs 0.87 and gives 0.348;
#   paper 0.3, dyes 0.15, cloth 0.05, gold 0.02 costs 1.51 and gives 0.604.
# The second method (nd_rjp_court_patronage, age 5) is a real step up: 74%
# more from the same painters, at the price of gold leaf.
nd_rjp_suratkhana = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 2
	startup_ramp_target = guild_startup_ramp_target
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = guild_build_time

	country_potential = {
		has_advance = nd_rjp_miniature_painting
	}

	unique_production_methods = {
		nd_rjp_folio_painting = {
			paper = 0.2
			dyes = 0.08
			cloth = 0.05
			produced = books
			output = 0.348
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		nd_rjp_court_arts = {
			# unlocked by nd_rjp_court_patronage
			paper = 0.3
			dyes = 0.15
			cloth = 0.05
			goods_gold = 0.02
			produced = books
			output = 0.604
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}

# Poppy Fields: the opium of Mewar, Kota and the Malwa border.
# Vanilla files opium under medicaments (goods/00_raw_materials.txt:446,
# "Medicinal plants, Pills, Opium"). A raw good, so the building is bound to
# where it exists: the location's market must already produce medicaments.
# The market form of the gate is used because Rajputana's own land has one
# source of it (Sidhmukh).
# The capsules are lanced and the sap is scraped, kneaded in earthen pans and
# packed in wooden chests: lumber, pottery and tools are the upkeep of the
# harvest, never another crop turned into this one.
# Output from eu5-profit at the rural margin: cost 0.55, 0.66 medicaments.
# Countryside only, two levels, no modifier.
