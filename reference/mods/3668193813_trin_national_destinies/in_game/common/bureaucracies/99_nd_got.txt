# Regnum Gothorum (GOT): bureaucracies
# National Destinies mod
#
# Four institutions of the Toledan state that the heritage and destiny trees
# deliberately left on the table. The trees already hold the palatine office,
# the Aula Regia, the councils, the Liber Iudiciorum and its town courts, the
# primacy of Toledo, the mint network, and (on the destiny side) the doctrine
# separating the treasury of the kingdom from the chamber of the king. What is
# left is the machinery underneath all of that: who collects the tax, who
# teaches the men who write, what binds a subject to a king whose office was
# elective, and what law reaches a village the count's court never visits.
#
# ON THE SUCCESSION. The heritage tree poses the elective question at the
# unction of Wamba and refuses to answer it; the destiny tree answers it twice,
# toward the office on one path and toward the blood on the other. The oath
# below is deliberately neutral between them. It does not say who becomes king.
# It says the men who swore may not unmake one, which both answers still need.
#
# On the day of forming the country's own grant gives 2 places: 2 of 4
# institutions can run.
#
# All IDs carry the got_ prefix. The engine loads 99_nd_*.txt alphabetically and
# the last definition of a key wins, so an unprefixed id would silently
# overwrite (or be overwritten by) any other tag's institution of the same name.

# 1. Numerarii of the fisc. The comes patrimonii at court appointed the
#    numerarius of each city directly, for a term of one year, bypassing the
#    provincial authority entirely: the Visigothic fisc was the most
#    centralised part of the kingdom's administration. The two Barcelona
#    documents of 592, the De fisco Barcinonensi, record the bishops of
#    Tarraconensis fixing what the collectors might take and what they might
#    charge for taking it, and the Third Council of Toledo (589) had already
#    ordered an annual provincial synod at which the fiscal agents answered
#    for their conduct. Hence clergy among the likers: the men who assessed
#    the land were audited by the men who baptised it.
#
#    The destiny tree's Thesaurus Publicus is the other half of this and not
#    the same half. That one is the doctrine of the Eighth Council (653) and
#    the strongroom it needs; this one is the circuit of collectors in the
#    provinces, which existed whether or not the doctrine did.
got_numerarii_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = GOT
		has_game_rule = nd_bureaucracies_enabled
	}

	# The everyday fiscal establishment. A restored kingdom has it from the
	# first day it has a treasury to fill.
	allow = {}

	estates_that_like = {
		crown_estate
		clergy_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# A census that records what every estate holds is a census that fixes
	# what may be demanded of it. The ceiling moves whether or not the crown
	# chooses to reach for it.
	neutral_modifier = {
		peasants_estate_max_tax = 0.05
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		tax_income_efficiency = 0.05
		# The collectors' circuits are how a court on the meseta sees the
		# land it taxes.
		global_monthly_rural_control = 0.003
		# Appointed from Toledo, answering to Toledo, replaced yearly by
		# Toledo. The fisc pulled the kingdom inward.
		monthly_towards_centralization = 0.05
	}

	# The collectors exceed the fixed measure, and free cultivators put
	# themselves under whoever can shield them from the assessment. That is
	# how a late Roman fisc eats its own tax base.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_pop_demotion_speed_modifier = 0.15
		peasants_estate_satisfaction_decay = 0.02
	}
}

# 2. Episcopal schools. Canon 1 of the Second Council of Toledo (527) ordered
#    that boys given to the church be raised in a house attached to the
#    cathedral, under the bishop's eye and a teacher's, until they were old
#    enough to choose their own life. It is the earliest cathedral school on
#    record anywhere in the Latin West, two centuries before the Carolingian
#    capitularies, and the Fourth Council (633) made the rule general. What
#    came out of those houses was Isidore's Etymologiae, the treatises of
#    Ildefonsus and Julian, and the notarial hands that drafted the code.
#
#    The heritage tree teaches the script (littera toletana) and the destiny
#    era teaches the code in a faculty (Catedra del Fuero). This is the boys'
#    school underneath both, and it is why the kingdom's learning was legal
#    learning rather than monastic contemplation.
got_schola_episcopalis_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = GOT
		has_game_rule = nd_bureaucracies_enabled
	}

	# A cathedral school is an episcopal foundation. A crown that has left the
	# Christian communion has no sees to keep one, whatever else it keeps of
	# the Gothic inheritance.
	allow = {
		religion.group = religion_group:christian
	}

	estates_that_like = {
		clergy_estate
		burghers_estate
	}
	# The offices these boys grew into were offices the magnates expected for
	# their own sons.
	estates_that_dislike = {
		nobles_estate
	}

	# Every see keeping a school is every see recruiting its own successors.
	neutral_modifier = {
		global_clergy_pop_growth = 0.05
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		# Grammar first, then law. A realm whose schools are run to produce
		# the men who draft charters ends up reasoning about everything the
		# way it reasons about the code.
		monthly_towards_jurisprudence = 0.05
		country_child_education = 0.2
		global_clergy_max_literacy = 5
	}

	# Starved, the schools stop teaching anyone the church does not intend to
	# keep. Letters retreat into the order that owns them, and the order that
	# owns the letters can name its price for using them.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_burghers_max_literacy = -5
		global_clergy_estate_power = 0.10
	}
}

# 3. Sacramentum fidelitatis. The general oath of fidelity sworn to the king by
#    the palatine office, the bishops and the people at every accession, set
#    down in the written form the lawyers called conditiones sacramentorum.
#    Recceswinth's code voided any oath sworn to a lord other than the king,
#    which is a claim no feudal monarchy north of the Pyrenees could make:
#    there, the whole political order was built out of exactly the private
#    oaths this law annulled. Canon 75 of the Fourth Council of Toledo (633)
#    put the anathema of the church behind it, and Egica laid his own oath
#    before the Fifteenth Council in 688 to ask which part of it still bound
#    him, which is how we know it bound the king too.
#
#    DELIBERATELY NEUTRAL ON THE SUCCESSION. The oath says nothing about who
#    becomes king. It says the men who swore may not unmake one. An elective
#    crown conferred by the assembly needs it exactly as much as an heir
#    crowned in his father's lifetime does, so this neither restates nor
#    forecloses either destiny path.
got_sacramentum_fidelitatis_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}
	# Telling a whole realm that its oaths are void costs more than swearing
	# them did. An oath machinery is dismantled at the price of everything it
	# was holding together.
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = GOT
		has_game_rule = nd_bureaucracies_enabled
	}

	# Canon 75 has the king raised in the royal city, and the councils that
	# pronounced the anathema sat at Toledo. A crown that has lost Toledo has
	# lost the place where the oath is administered and enforced.
	allow = {
		owns = location:toledo
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# While the oath monopoly stands, no magnate may gather sworn men of his
	# own. That is a permanent structural ceiling on the aristocracy, and it
	# applies whether or not the institution is being funded.
	neutral_modifier = {
		base_nobles_estate_power_modifier = -0.05
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		# A usurper needs men who will swear to him, and every one of them has
		# already sworn elsewhere under anathema.
		monthly_pretender_rebel_growth = -0.01
		pop_leave_rebels_threshold = -0.05
		# The crown recovers from its bad years because the oath outlasts them.
		crown_estate_satisfaction_recovery = 0.005
	}

	# Perjury stops being sacrilege and becomes procedure. Frankish writers
	# named the result the morbus Gothorum, the Gothic disease, and it was an
	# outsider's verdict on the kingdom as much as a description of it.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_nobles_estate_rebel_growth = 0.005
		diplomatic_reputation = -1
	}
}

# 4. Conventus publicus vicinorum. The public assembly of the neighbours, the
#    only jurisdiction the Liber Iudiciorum recognises below the count's court.
#    A runaway had to be produced before it, a boundary proved before it,
#    straying stock declared to it, and the lesser wrongs of a village settled
#    in it by men who owed no lord for the right to judge. Its descendants are
#    the village concilia of tenth century northern Iberia, the best evidence
#    in Latin Europe for village decision making in that century.
#
#    This is the tier the heritage tree cannot reach. The Liber court is the
#    comes and his iudex sitting in a chartered town, and villages came in to
#    it; the conventus is the law that stays in the village. A realm of
#    seigneurial justice has no equivalent, because there the lord judges.
got_conventus_vicinorum_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = GOT
		has_game_rule = nd_bureaucracies_enabled
	}

	# Written into the code itself, so it is available wherever the code runs.
	allow = {}

	estates_that_like = {
		peasants_estate
	}
	# Every matter the neighbours settle is a matter the local magnate does
	# not settle, and does not charge for settling.
	estates_that_dislike = {
		nobles_estate
	}

	# A standing recognised by the law is a standing whether or not anyone is
	# paying for the assembly to meet.
	neutral_modifier = {
		global_peasants_estate_power = 0.10
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_towards_free_subjects = 0.05
		# Wamba's muster law of 673 obliged every free subject to march,
		# Goth or Roman, freeman or manumitted. The assembly is where the
		# village knows which of its men are free, so it is also where the
		# roll comes from.
		peasants_estate_levy_size = 0.25
		# The counterweight to the fisc, and the councils said so out loud:
		# a man with neighbours who will speak for him at law does not have
		# to put himself under a lord who will. Deliberately the same lever
		# the numerarii push the other way when they are left unwatched.
		global_pop_demotion_speed_modifier = -0.10
	}

	# Unfunded, the assembly still meets, but nothing it decides is written
	# down and nothing written elsewhere reaches it. The village closes: it
	# judges its own by memory, takes in no stranger, and lets nobody go.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_monthly_control_decline = 0.003
		global_migration_speed_modifier = -0.10
	}
}
