##############################
# Finland: Historical Bureaucracies
##############################
# Four institutions of the Swedish-Finnish realm (c. 1300-1721). The
# auto modifier grants 2 slots at formation (a capital gives one only as a
# megalopolis), so two of the four can run on the day Finland forms.
#
#  - Häradshövding Ting (rural assize courts: district judge + 12 farmer
#    lay judges; the working bedrock of medieval Finnish self-rule)
#  - Bishopric of Åbo (Diocese of Turku: Cathedral chapter at Åbo, the
#    spiritual and educational backbone of medieval Finland; bishop also
#    secular leader for centuries)
#  - Rusthåll (key fin_hakkapeliitta_levy_bureaucracy, kept for saved
#    games): the cavalry farms. A farm that keeps a horse and an armed
#    rider for the crown is freed of its land tax. Re-themed 2026-10-02:
#    vanilla gives Finland the Hakkapeliitta themselves as an age 5
#    advance and unit (country_swe.txt, hakkapelitta), so the mod's
#    institution is the farm system that mounted them.
#  - Birkarl Lapp Tribute (Pirkkalaiset: royally-licensed merchants holding
#    monopoly on Sami trade and tax extraction in Tornio, Lule, and Pite
#    river valleys, 13th-17th centuries)
##############################

fin_haradshovding_ting_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = FIN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		peasants_estate
		burghers_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# An active assize court system means royal decrees pass through panels
	# of seven-or-more local farmers before they bind a hundred. Customary
	# Scandinavian law constrains the throne by simply existing.
	neutral_modifier = {
		stability_cost_efficiency = 0.1
	}

	# Working as designed: the häradshövding rides circuit, the ting convenes
	# at fixed seasons, twelve lay judges hear cases alongside the district
	# judge, and rural disputes resolve without escalating. Peasants own
	# their forum; literate yeomen rise; the realm coheres without coercion.
	positive_modifier = {
		scale = { value = scope:maintenance }
		legislative_efficiency = 0.10
		global_peasants_max_literacy = 8
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: ting attendance lapses, judgments become arbitrary, and
	# unresolved disputes feed banditry and rural rebellion. The lay-judge
	# tradition that bound communities to Stockholm hollows out into a
	# grievance against absent royal authority.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_decay = 0.00025
		crown_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		global_separatism = 0.10
	}
}

fin_abo_bishopric_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = FIN
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Cathedral of Åbo, consecrated 1300, was the seat of the bishop
	# whose diocese was the entirety of Finland. The institution requires
	# either possession of Åbo or an explicit Catholic disposition of the
	# realm.
	allow = {
		OR = {
			owns = location:abo
			religion = religion:catholic
		}
	}

	estates_that_like = {
		clergy_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# An active bishopric means the parish network doubles as the local
	# administration. Tithes flow, education is centralized, and the Latin
	# clergy fills offices the lay magnate class might otherwise contest.
	neutral_modifier = {
		nobles_estate_blocked_from_cabinet = yes
	}

	# Working as designed: Bishop and cathedral chapter run the schools at
	# Åbo, the Dominican convent (founded 1249) trains clergy in the
	# vernacular, native Finnish bishops from 1291 forward speak for the
	# realm in Uppsala, and the parish network reaches every village.
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_devotion = 0.03
		monthly_legitimacy = 0.03
		monthly_republican_tradition = 0.03
		global_max_literacy = 8
		global_pop_conversion_speed_modifier = 0.1
	}

	# Decaying: parish records fall behind, vacant sees go unfilled, and
	# the lay magnate class reasserts itself against an enfeebled church.
	# The clergy that ran the kingdom's recordkeeping no longer have the
	# institutional reach to bind clergy and crown.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		stability_decay = 0.00025
	}
}

fin_hakkapeliitta_levy_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = FIN
		has_game_rule = nd_bureaucracies_enabled
	}

	# The crown offers the bargain when it needs riders or can afford to
	# forgo the tax: a warlike realm or a settled one.
	allow = {
		OR = {
			societal_value:belligerent_vs_conciliatory > 0
			stability >= 1
		}
	}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The cavalry farms are a standing arrangement: the wealthier
	# farmsteads of each county keep a horse and rider against their tax,
	# and the throne supplies the colonel.
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# Working as designed: the farms are inspected at the yearly muster,
	# horse and rider turn out fit, and a rider fed from his own farm
	# marches light and lives off little.
	positive_modifier = {
		scale = { value = scope:maintenance }
		army_light_cavalry_power = 0.10
		army_initiative = 0.10
		army_food_gathering = 1
	}

	# Decaying: musters lapse, the farms keep the tax freedom and send a
	# nag and a farmhand, and the crown has given away revenue for riders
	# it does not get. The lineage nobility, bypassed when the throne
	# raised horsemen directly from the soil, reasserts its grievance.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		monthly_inflation = 0.002
		stability_decay = 0.00025
	}
}

fin_birkarl_lapptribute_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = FIN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# An active birkarl monopoly means a small caste of royally-licensed
	# Tornio-valley traders holds exclusive rights to commerce with and
	# tribute extraction from the Sami of Lappmarken. The license is
	# revenue. The license is also a cultural wedge.
	neutral_modifier = {
		estate_enrichment = 0.05
	}

	# Working as designed: birkarl families ride the winter routes north
	# to the Sami siida, exchange iron and cloth for furs, collect crown
	# taxes alongside their own profit, and hold the trade monopoly that
	# made Tornio, Luleå, and Piteå the wealthiest northern entrepôts
	# before their 1621 town charters formalized the system.
	positive_modifier = {
		scale = { value = scope:maintenance }
		trade_sea_efficiency = 0.10
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
		global_pop_assimilation_speed_modifier = 0.10
	}

	# Decaying: the birkarl families turn to outright extortion of Sami
	# producers, the fur supply collapses as the herds disperse, and the
	# crown's revenue from Lappmarken evaporates while the coastal yeoman
	# resents a privileged caste living off colonial extraction.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		monthly_inflation = 0.002
	}
}
