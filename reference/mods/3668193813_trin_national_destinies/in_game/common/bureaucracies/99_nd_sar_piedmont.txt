# National Destinies: Sardinia-Piedmont bureaucracies
# Four institutions of the Savoyard composite crown, spanning the dynastic
# order of 1572, the concordat settlement of 1727, the school reform of
# 1717-1729 and the provincial intendancies of 1696. On the day of forming the
# country's own grant gives 2 places: 2 of 4 institutions can run.
#
# Identity: this crown was assembled by treaty out of an Alpine duchy and a
# Mediterranean island, and its institutional question is how far a single
# administration can be pushed over two peoples who keep two laws. Two of the
# four are the crown's instruments against an estate (the intendant against
# the feudal bench, the economato against the church), one is the crown's
# instrument for buying an estate (the order and its commende), and one builds
# the lettered class that the other three need in order to work.
#
# IDs are sar_piedmont_-prefixed. 99_nd_sar.txt (Sardinia) owns the sar_
# prefix and the island institutions of the Stamenti, the curatoria, the
# viceregal court and the salt regalia; nothing here touches them, nor the
# heritage and destiny content of this tag.

# 1. Intendenti di Provincia. Victor Amadeus II put a salaried royal
#    commissioner into each province from 1696 on the French model. He
#    assessed and collected taxes the provincial estates had formerly voted,
#    audited every community's accounts, absorbed the appeals that had gone to
#    the feudal bench, and kept the recruitment rolls. The single sharpest
#    instrument of Savoyard centralisation, and the reason a small state could
#    mobilise like a large one.
sar_piedmont_intendenti_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# A commissioner in every province is a payroll that grows with the realm.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = SRP
		has_game_rule = nd_bureaucracies_enabled
	}

	# An intendancy is not an office a decentralised realm can staff: the
	# whole point of the man is that he answers to the capital and to nobody
	# where he sits. Left of centre on the axis means leaning centralised.
	allow = {
		societal_value:centralization_vs_decentralization < -20
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The crown's writ reaches the countryside because a salaried man lives
	# there, whether or not anything is spent on him this year.
	neutral_modifier = {
		global_monthly_rural_control = 0.002
	}

	# Working: local exemptions are revoked at will, the nobility is assessed
	# rather than asked, and the provincial rolls fill the regiments quickly.
	positive_modifier = {
		scale = { value = scope:maintenance }
		revoke_privilege_cost_modifier = -0.15
		nobles_estate_max_tax = 0.05
		regiment_recruit_speed = 0.10
	}

	# Decaying: an unpaid commissioner still holds his commission, so he lives
	# off the communities whose books he audits. The towns are squeezed and
	# the provinces quietly stop answering the capital at all.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_monthly_control_decline = 0.003
		burghers_estate_satisfaction_decay = 0.02
	}
}

# 2. Regio Economato dei benefici vacanti. By the concordat of 29 May 1727
#    Benedict XIII conceded that the fruits of every vacant bishopric and
#    abbey be kept under royal custody and administered by economi the crown
#    named, and confirmed the House of Savoy's right of nomination to the
#    sees. The island was organised the same way from 1733 under a conservator
#    general, and Charles Emmanuel III renewed the settlement in 1741. With
#    the royal placet on top of it, no bull reached a Savoyard pulpit that the
#    crown had not read first.
sar_piedmont_regio_economato_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = SRP
		has_game_rule = nd_bureaucracies_enabled
	}

	# The whole apparatus rests on a concordat with Rome. A crown outside the
	# Roman communion has no vacant benefices to administer and no nomination
	# right to bargain over.
	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		clergy_estate
	}

	# The regio placet: nothing is published in the realm unread.
	neutral_modifier = {
		can_ignore_papal_bulls = yes
	}

	# Working: nomination and sequestration together give the crown a standing
	# voice inside its own church, a share of its wealth, and a hierarchy that
	# owes its mitres to Turin rather than to Rome.
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_religious_influence = 0.15
		clergy_estate_max_tax = 0.10
		global_clergy_estate_power = -0.05
	}

	# Decaying: the economo stops administering and starts waiting. Sees are
	# left vacant for years because the vacancy is worth more than the bishop,
	# parishes go unserved, and in a realm with preachers in the high valleys
	# an unserved parish does not stay empty for long.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_religious_rebel_growth = 0.02
		clergy_estate_satisfaction_decay = 0.02
	}
}

# 3. Ordine dei Santi Maurizio e Lazzaro. The bull Pro Commissa Nobis of
#    13 November 1572 joined the Savoyard Order of Saint Maurice to the
#    hospitaller Order of Saint Lazarus and vested the grand mastership in the
#    House of Savoy in perpetuity from January 1573. Emmanuel Philibert
#    recovered the Lazarite commanderies as endowed livings for his knights,
#    opened the order's hospital at Turin in 1575, kept a convent house in the
#    citadel of Nice for sea service and another at Turin for land service,
#    and armed two galleys, the Piemontese and the Margarita, which sailed
#    under the order's banner in the papal service against the Turks.
sar_piedmont_ordine_mauriziano_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# Dissolving an order means unpicking endowed land held by living men with
	# papal title to it. It costs more the richer the realm that granted it.
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = SRP
		has_game_rule = nd_bureaucracies_enabled
	}

	# The magistral seat, the hospital of 1575 and the convent for land
	# service were all at Turin. An order whose grand master has lost his own
	# capital has nowhere to hold a chapter.
	allow = {
		owns = location:turin
	}

	estates_that_like = {
		nobles_estate
		clergy_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# A commenda is granted for life out of endowed land. Whatever the crown
	# spends on the order, that income has left the fisc for good.
	neutral_modifier = {
		estate_enrichment = 0.05
	}

	# Working: the cross and the commenda give the aristocracy an honourable
	# career inside the dynasty's gift, so what the great houses want carries
	# further, favours cost the crown less to hand out, and the order's
	# galleys bring back a reputation from every action they fight.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_agenda_impact = 0.10
		grant_privilege_cost_modifier = -0.10
		navy_tradition_from_battle = 0.25
	}

	# Decaying: the commende become absentee sinecures held by courtiers who
	# never see them, the hospital and the convents fall on the treasury, and
	# the endowment turns into a private base of power the crown funded.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		base_nobles_estate_power_modifier = 0.05
		building_upkeep_efficiency = -0.10
	}
}

# 4. Magistrato della Riforma. Victor Amadeus II rebuilt the University of
#    Turin between 1717 and 1720 and set a Magistrato della Riforma over all
#    teaching in his states, expressly to keep the religious orders out of it.
#    Its regulation of 1729 put every school under the university and required
#    each schoolmaster to hold a royal diploma; the Collegio delle Province
#    boarded students from outside Turin at crown expense, and they came back
#    to staff the intendancies and the benches. Scholars have called the 1729
#    settlement the archetype of the Italian public school.
sar_piedmont_magistrato_riforma_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = SRP
		has_game_rule = nd_bureaucracies_enabled
	}

	# A licensed schoolmaster in every town presupposes cheap books to teach
	# out of. Before the press, schooling is whatever the local chapter keeps.
	allow = {
		has_embraced_institution = institution:printing_press
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		clergy_estate
	}

	# Town schools under royal licence, open to whoever can pay the fees.
	neutral_modifier = {
		global_burghers_max_literacy = 5
	}

	# Working: the Collegio delle Province turns provincial boys into salaried
	# men of law and account, the court can find qualified servants without
	# bidding for them, and a schooled realm takes up new ways cheaply.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_pop_promotion_speed_modifier = 0.10
		hire_advisor_cost_modifier = -0.10
		embrace_institution_cost_modifier = -0.10
	}

	# Decaying: the royal diploma becomes a thing to be had rather than earned,
	# offices go to whichever estate can place its own men in them, and the
	# religious orders quietly walk back into the classrooms they were shut
	# out of, which suits them very well.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		estate_power_from_cabinet = 0.05
		clergy_estate_satisfaction_recovery = 0.001
	}
}
