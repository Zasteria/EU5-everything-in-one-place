# National Destinies: Iraqi bureaucracies
#
# Four offices of Mesopotamia, picked so that none of them repeats what the
# rest of this tag already owns. The heritage tree digs (the misaha cadastre,
# the Nahrawan head-works, the muqasama assessment, the Hindiyya channel) and
# the destiny tree builds (the Register of Waters, the Dar al-Barid, the Dulab
# Works, the Marsad). Neither leaves behind a standing office with a payroll.
# KHL, the caliphate this tag chains upward into, already holds the mazalim
# court, the bayt al-mal, the ulama and the Hajj, so none of those appear here.
#
# The spine is still water, but here it is the turn list rather than the
# earthwork: who may draw, in what order, and who is standing between the
# treasury and the ground when nobody keeps the list. The other three are the
# three ways Baghdad actually financed and legitimised itself: land assigned
# to soldiers, credit advanced by protected subjects, and a register of holy
# descent that runs across the Sunni and Shia line rather than along it.
#
# A player who takes the Dar al-Khilafa destiny branch reaches 3 of 4 at its
# capstone. On the day of forming the country's own grant gives 1 place and
# vanilla 1: 2 of 4 institutions can run.

# ============================================================
# 1. The Mirab and the Nawba.
#    The nawba is the turn: a measured span in which one village, and inside
#    it one family, may draw from the canal. The mirab is the watermaster who
#    keeps the turn list, opens and shuts the offtake, and calls out the
#    yearly clearance that stops silt from closing the intake. Harun al-Rashid
#    had the Nahrawan mouths shut, dug clean and reopened on exactly that
#    principle. This is the smallest unit of Mesopotamian government and the
#    one the Abbasid land revenue actually rested on.
#    Estates: the cultivator gets a turn he can plant against, and the crown
#    gets a register of the countryside it otherwise only visits. The great
#    holders at the head of the canal lose the right to take what they like.
#    Gated on the head-works advance, because a turn list means nothing until
#    somebody has rebuilt the regulators that make the turn enforceable.
# ============================================================
irq_mirab_nawba_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# Dearer than the mod's usual 0.003: canal clearance was the largest
	# recurring public work in the Sawad and the first thing a broke state cut.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = IRQ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		has_advance = nd_irq_nahrawan_breach
	}

	estates_that_like = {
		peasants_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The watermaster is the state's routine errand into rural ground.
	neutral_modifier = {
		global_monthly_rural_control = 0.002
	}

	# The list is kept, the clearance is called, the turn arrives on time.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_max_rural_control = 0.02
		global_peasants_rural_desired_pop_scaled = 0.02
		peasants_estate_satisfaction_recovery = 0.001
	}

	# Nobody calls the clearance. The intakes silt, the tail villages get
	# nothing, and the water that does not move stands and breeds fever.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_prosperity_decay = 0.02
		rural_disease_resistance = -0.075
	}
}

# ============================================================
# 2. Soyurghal.
#    The Mongol word for a grant in perpetuity, and the thing the Jalayirids
#    left behind them in Iraq. Qara Yusuf paid emirs and kinsmen out of the
#    revenue of assigned land instead of in coin, and within a generation the
#    assignment had hardened into a hereditary holding with its own immunity
#    from tax. Uzun Hasan spent a reign trying to pull that revenue back into
#    direct collection and did not finish.
#    Estates: the holders are Turkoman, Kurdish and Lur lords who now owe
#    service, so nobles and tribes gain. The crown pays for the army with the
#    one asset it cannot make more of.
#    The removal price is raised deliberately. A grant made in perpetuity is
#    not revoked by a decree, it is revoked by a war.
# ============================================================
irq_soyurghal_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# Cheap to keep, because the holders feed themselves off the land.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}
	# Dearer to undo than the mod's usual 0.004.
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.006
	}

	potential = {
		has_or_had_tag = IRQ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		tribes_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Contentment that does not drift, because the grant is theirs for life.
	neutral_modifier = {
		nobles_estate_satisfaction_decay = -0.01
	}

	# The holders muster. Horsemen raised off assigned land and out of peoples
	# the court does not speak for cost the treasury almost nothing.
	positive_modifier = {
		scale = { value = scope:maintenance }
		wrong_culture_levy_size = 0.15
		tribes_estate_levy_size = 0.20
		army_light_cavalry_maintenance_cost_modifier = -0.10
	}

	# Service lapses, the grant does not. The holder stops being a soldier and
	# becomes a landlord standing between the treasury and the ground.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_max_rural_control = -0.05
		global_nobles_estate_power = 0.25
	}
}

# ============================================================
# 3. The Jahbadhiyya.
#    From 908 to 932 the wazirs of Baghdad kept two Jewish bankers on standing
#    retainer, Yusuf ibn Finhas and Harun ibn Imran. They advanced the treasury
#    a fixed sum on the first of each month against the revenue of Ahvaz, held
#    the deposits of the merchant houses, and settled between Baghdad and the
#    provinces on written order rather than by mule.
#    Estates: the money is protected-community money and the paper is merchant
#    paper, so dhimmi and burghers gain. The jurists lose an argument about
#    interest they had already lost in practice.
#    Gated on Baghdad, because the jahbadh sat in the wazir's own diwan.
# ============================================================
irq_jahbadhiyya_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = IRQ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:baghdad
	}

	estates_that_like = {
		dhimmi_estate
		burghers_estate
	}
	estates_that_dislike = {
		clergy_estate
	}

	# Somebody reliable is holding the state's paper, so the state's paper is
	# worth holding.
	neutral_modifier = {
		bond_interest = -0.01
	}

	# Pooled deposits, not one man's fortune, which is why the advance is large.
	# The transit effect is the suftaja: goods and debts cross Iraq on paper
	# instead of on a mule, which is the only reason a land bridge between the
	# Gulf and the Mediterranean pays better than the sea route around it.
	positive_modifier = {
		scale = { value = scope:maintenance }
		total_loan_capacity_modifier = 0.20
		merchant_maintenance_efficiency = 0.15
		trade_land_efficiency = 0.10
	}

	# The retainer goes unpaid and the state does what states do to creditors
	# it cannot dismiss: musadara, confiscation, and then it clips its own coin.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_dhimmi_estate_rebel_growth = 0.005
		monthly_inflation = 0.003
	}
}

# ============================================================
# 4. Naqib al-Ashraf.
#    The naqib kept the shajara, the register of those descended from the
#    Prophet. He entered their births and deaths, judged their quarrels, paid
#    their stipends, kept them out of trades held unworthy of the descent and
#    struck out the false claimants. Baghdad's niqaba was held around the year
#    1000 by Abu Ahmad al-Husayn ibn Musa, father of al-Sharif al-Radi, and the
#    office ran on under the Ottomans in the hands of the Gaylani house.
#    Estates: the register is the senior corporation of the religious class and
#    the ruler is the one who confirms its head, so clergy and crown gain. The
#    towns pay for a caste that is exempt from what they are not.
#    Gated on the two Islamic confessions this formable admits, since the
#    niqaba is an office of the Muslim community and IRQ can be either.
# ============================================================
irq_naqib_al_ashraf_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = IRQ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		OR = {
			religion = religion:sunni
			religion = religion:shia
		}
	}

	estates_that_like = {
		clergy_estate
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# The standing price of hosting the descent: you cannot tax it as hard.
	neutral_modifier = {
		clergy_estate_max_tax = -0.05
	}

	# A register both confessions are inside is a register both confessions
	# will argue in front of instead of in the street.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_estate_target_satisfaction = 0.025
		monthly_religious_rebel_growth = -0.01
		clergy_estate_satisfaction_recovery = 0.002
	}

	# Certificates get sold, the descent multiplies past counting, and the
	# corporation that was meant to be governed starts leading the quarter.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_clergy_estate_rebel_growth = 0.025
		burghers_estate_satisfaction_decay = 0.04
	}
}
