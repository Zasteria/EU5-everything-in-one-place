# National Destinies: Thessalonica bureaucracies
#
# Thessalonica had three masters and none of them was the crown. The Lombard
# baronage that Boniface of Montferrat's knights founded in 1204, the guilds
# and dockhands of the harbour who ran the city as a commune from 1342 to
# 1350, and the Orthodox church with the Holy Mountain behind it. The first
# three institutions below hand one of those bodies a formal seat and pay for
# it out of the other two, which is why their estate lists all read "my body
# likes it, the other two resent it" and why the crown estate is absent from
# every one of them: there was no crown party in this city to reward.
#
# The fourth is the exception that states the rule. The sekreton is the crown's
# own bench, the only institution here that belongs to none of the three, and
# it is the only one the crown estate likes.
#
# Four bureaucracies against two slots, both from the auto modifier
# (main_menu/common/auto_modifiers/99_nd_the.txt). Seat two masters and shut one out, or take the crown's
# court and seat only one master while the other two are told what they may not
# do. Which claimant goes without is the whole of the country's politics.
#
# Scale discipline. This is one city, its gulf and the Macedonian plain behind
# it. A Frankish caste ruling a continent is Outremer's subject and the Latin
# Empire's; here the same problem fits inside one set of walls, and none of
# these institutions reaches past the six provinces of the kingdom.
#
# Nothing here re-uses the heritage tree's material (the Zealot revolt and its
# exits, the appeal of the Athonite houses to Rome, Palamas kept out of his
# see, the Venetian handover, the Sephardic arrival, the press of 1512, the
# Union of Florence, the mining villages) or the destiny tree's (Platamon,
# the kommerkion house, the marches, the islands).
#
# IDs are the_-prefixed so they cannot collide with another tag's file.

# ---------------------------------------------------------------------------
# 1. Curia of the Lombards
# ---------------------------------------------------------------------------
# Boniface of Montferrat died at Mosynopolis in September 1207 and left an
# infant heir, Demetrius, so the kingdom fell to the court of Lombard fief
# holders around the regent Oberto II of Biandrate. In 1208 they refused the
# suzerainty of Emperor Henry of Flanders, shut the gates of the city against
# him, and proposed to depose the child in favour of his half-brother William
# VI of Montferrat; Henry outmanoeuvred them, crowned Demetrius in January
# 1209, and had Oberto imprisoned at Serres. That court is the institution:
# a baronage that holds the castles of the passes and the summer host, and
# that governs in the crown's name whenever the crown is weak.
#
# Working: the fiefs answer the summons, the castellanies are kept up, and
# the Greek villages of the fiefs march under Lombard banners. Neglected:
# the barons look for another Montferrat to crown and quarter themselves on
# the paroikoi who feed them.
the_curia_of_the_lombards_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = THE
		has_game_rule = nd_bureaucracies_enabled
	}

	# The curia is the court of a Latin king holding a Greek city by knight
	# service. A crown that has gone over to the Greek rite has no Lombard
	# baronage left to convene.
	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		burghers_estate
		clergy_estate
	}

	# The council of the fief holders is closed to the men of the city.
	# Vanilla grants every country burghers in the cabinet by default, so
	# this revokes something real rather than restating it.
	neutral_modifier = {
		burghers_estate_blocked_from_cabinet = yes
	}

	# The host mustered, the passes garrisoned, and a Greek countryside
	# levied by Frankish lords who never learned its language. The crown
	# pays for all of it in standing: a curia that governs in the king's
	# name spends the king's claim.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_levy_size = 0.25
		wrong_culture_levy_size = 0.10
		fort_maintenance_efficiency = 0.15
		monthly_legitimacy = -0.02
		monthly_devotion = -0.02
		monthly_republican_tradition = -0.02
	}

	# Unsummoned and unpaid, the curia goes looking for a better king, and
	# knights with no campaign to go on live off the villages of their fiefs.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_pretender_rebel_growth = 0.02
		peasants_estate_satisfaction_decay = 0.015
	}
}

# ---------------------------------------------------------------------------
# 2. Archon of the Demos
# ---------------------------------------------------------------------------
# In the summer of 1342 the parathalassioi, the sailors and dockhands of the
# waterfront, rose with the Zealots, and for eight years the city had two
# archontes at once. The governors sent from the capital (Michael Monomachos
# in 1342, John Batatzes in 1343, the megas primikerios John Apokaukos in
# 1344) held the landward side; the harbour answered to an archon the demos
# chose for itself, with Michael and Andreas Palaiologos at the head of it.
# The office, not the rising, is what is modelled here: a permanent elected
# magistracy of the waterfront sitting beside the crown's man.
#
# Working: the quarters trade on their own account, keep their own streets,
# and speak in the realm's business. Neglected: the officers of the harbour
# stop being the crown's and become the city's, and the archontes shut out of
# their own town treat with whoever is camped outside the walls.
the_archon_of_the_demos_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# The commune pays its own officers, so the crown's share of the cost
	# is the lightest in this file.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = THE
		has_game_rule = nd_bureaucracies_enabled
	}

	# The demos is this city's body. Lose Thessalonica and there is no
	# waterfront to elect anybody.
	allow = {
		owns = location:thessaloniki
	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
		clergy_estate
	}

	# The harbour sets business before the realm's assembly. Vanilla seats
	# only the nobility there by default, so this is a genuine admission.
	neutral_modifier = {
		burghers_estate_can_participate_in_parliament = yes
	}

	# Merchants trading on their own account, a city that polices itself,
	# and a burgher body that recovers from every setback. The bargain is
	# that the crown's reach ends at the land gate: under the commune the
	# archontic estates of the plain answered to nobody at all.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_trades_per_burgher = 0.15
		global_monthly_urban_control = 0.003
		burghers_estate_satisfaction_recovery = 0.001
		global_monthly_rural_control = -0.003
	}

	# An unfunded magistracy is still a magistracy, and it starts governing
	# for itself. The dispossessed archontes go looking for an army.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_nobles_estate_rebel_growth = 0.02
		global_burghers_estate_power = 0.08
	}
}

# ---------------------------------------------------------------------------
# 3. Synaxis of Karyes
# ---------------------------------------------------------------------------
# The Holy Mountain has governed itself from Karyes since the tenth century
# through a Protos and an assembly of abbots, the arrangement fixed by the
# typikon John Tzimiskes gave the monks in 972 and revised by a chrysobull of
# Andronikos II in November 1312 that made the Protos an appointee of the
# Patriarch rather than of the emperor. The Protos answered to the Patriarch,
# the judge and the Metropolitan of Thessalonica; the houses of the Mountain
# held metochia across Macedonia and Thessaly, and their granaries were the
# most reliable food reserve in the region.
#
# Working: the Orthodox countryside accepts a Latin king whose name is
# commemorated at Karyes, and the storehouses of the metochia stand behind
# the city. Neglected: the abbots take offence at promises the crown did not
# keep, and the houses go on eating whether or not they serve.
the_synaxis_of_karyes_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# Endowments, commemorations and the upkeep of the Protaton make this
	# the dearest in this file to hold open.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = THE
		has_game_rule = nd_bureaucracies_enabled
	}

	# There is no Synaxis to seat unless the Mountain itself is in the realm.
	allow = {
		owns = location:mount_athos
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# Recognising the assembly makes the church a standing power in the
	# realm, which is the point of it and the price of it.
	neutral_modifier = {
		global_clergy_estate_power = 0.05
	}

	# The metochia's granaries and cellars, an Orthodox population that does
	# not rise for its faith against a king the Mountain prays for, and
	# abbots who forgive the crown quickly. Against that stands the reason
	# the arrangement was worth having to them: the houses hold their
	# exemptions, and a Catholic king who wants a church tax has to buy his
	# way past them.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_food_capacity_modifier = 0.10
		monthly_religious_rebel_growth = -0.01
		clergy_estate_satisfaction_recovery = 0.001
		demand_church_tax_price_cost_modifier = 0.20
	}

	# A concordat the crown stops funding is an insult with a signature on
	# it. The abbots cool, and the Mountain keeps its mouths and its lands
	# while giving the realm nothing back for them.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_satisfaction_decay = 0.015
		global_clergy_food_consumption = 0.10
	}
}

# ---------------------------------------------------------------------------
# 4. Sekreton of Thessalonike
# ---------------------------------------------------------------------------
# Yolanda Irene of Montferrat took up residence in Thessalonica in 1303 and
# left the city an imperial court of record, the evages sekreton, which the
# provincial General Judges of the region later replaced. The bench reached
# its final shape when Andronikos III cut the tribunal to four katholikoi
# kritai in 1329, at least one of them a bishop; it took written complaints
# only, heard expert testimony where it thought fit, deliberated in seclusion
# and decided by majority. It was founded to curb the abuses of the powerful,
# and in 1337 three of the four judges at the capital were convicted of taking
# bribes and exiled, which is exactly what a bench costs when nobody watches
# it.
#
# This is the crown's instrument and no master's, so it is the only entry in
# this file the crown estate likes and the only one that gives away nothing.
# The clergy is on neither list because a bishop sat among the judges: the
# church is inside this institution rather than for or against it.
#
# Working: privileges are hard to grant and cheap to take back, and a villager
# with a written petition can be heard against the man who owns his fields.
# Neglected: verdicts go to whoever pays for them, and the crown's own law
# stops running.
the_sekreton_of_thessalonike_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# Salaried judges, notaries and a permanent archive. Dearer than the
	# commune's magistracy, cheaper than the Mountain's endowments.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = THE
		has_game_rule = nd_bureaucracies_enabled
	}

	# A court that judges in the sovereign's name needs a sovereign whose
	# authority is his own. A realm that has stopped governing by legitimacy
	# has nothing to seat these judges under. Vanilla uses this same trigger
	# to gate a bureaucracy (byz.txt, nomos_empsychos).
	allow = {
		uses_government_power = legitimacy
	}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
		burghers_estate
	}

	# Written complaints only, which puts a man who can draft a petition in
	# reach of every village that wants to be heard.
	neutral_modifier = {
		global_peasants_max_literacy = 5
	}

	# A standing court of record cuts both ways on privilege: what the crown
	# conceded can be recovered before judges who keep the deed, and what the
	# crown would rather concede quietly now has to survive the same record.
	positive_modifier = {
		scale = { value = scope:maintenance }
		revoke_privilege_cost_modifier = -0.10
		global_peasant_enfranchisment = 0.05
		grant_privilege_cost_modifier = 0.10
	}

	# Judges who are not paid are judges who are bought, and the men with the
	# money to buy them are the ones the bench was raised to restrain. The
	# crown's law stops meaning anything, and the realm loosens with it.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		legislative_efficiency = -0.10
		stability_decay = 0.0003
	}
}
