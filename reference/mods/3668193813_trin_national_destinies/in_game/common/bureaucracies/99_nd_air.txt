# National Destinies: Aïr / Agadez bureaucracies
# Four institutions of the Tuareg Sultanate of Agadez and the Kel Tuareg
# confederations of the Aïr massif (1449 onward). The sultanate emerged
# as a kingmaker compromise: when the Itesen, Kel Owey, Kel Tagidda, and
# their associates could no longer agree on precedence, they imported
# Yunus from Istanbul-via-the-Maghreb in 1449 to be a "neutral" sultan
# whose office was meant to bind them without empowering any one of them.
# That tension (ritual-religious sovereignty at Agadez against the real
# political power of the imajeghen drum-chiefs) is the institutional
# backbone of every choice in this set.
#
# The two paired tradeoffs (sultanate vs Cure Salée; caravan compact vs
# ineslemen tariqat) push the player to commit to either confederation
# legitimacy or merchant-clerical legitimacy as the basis of the state. On the
# day of forming the country's own grant gives 1 place and vanilla 1: 2 of 4
# institutions can run.

# 1. Scribes of the Agadez Chronicle (key says agadez_sultanate): the
#    sultan's Arabic secretaries, who kept the chronicle of the house (the
#    list of sultans and reigns from Yunus onward) and wrote his letters to
#    Bornu, the Hausa kings and the scholars of Cairo. Vanilla 1.4 tells
#    the founding of the sultanate itself in the Kel Ayr advance
#    kay_agadez_sultanate, and the mod's advance nd_air_sultanate_founding
#    tells the household officers, so the bureaucracy tells the written
#    side of the court. A court that rules by the written word needs the
#    drum chiefs' memory less, which is why the tribes dislike it.
air_agadez_sultanate_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		# AIR_f sets no tag: the marker is the only thing that identifies Air.
		has_variable = nd_air_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:agadez
	}

	estates_that_like = {
		crown_estate
		clergy_estate
	}
	estates_that_dislike = {
		tribes_estate
	}

	neutral_modifier = {
		monthly_prestige = 0.05
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_legitimacy = 0.10
		monthly_devotion = 0.1
		monthly_republican_tradition = 0.1
		diplomatic_reputation = 1
		country_cabinet_efficiency = 0.075
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		tribes_estate_satisfaction_decay = 0.025
		global_separatism = 0.05
	}
}

# 2. Cure Salée Compact: the annual congress at the salt-bearing wells of
#    In-Gall and the Tegidda-n-Tesemt salt flats, where the Kel Tuareg
#    confederations gathered at the end of the rains to water their herds,
#    settle disputes, marry across drum-groups, and confirm the sultan in
#    his office. The festival was the working constitution of the Aïr: each
#    drum-chief brought tribute of salt, dates, and animals, and each took
#    away the right to migrate north into the Tamesna plains for the dry
#    season. No sultan could ignore the Cure Salée; no ineslemen tribe could
#    escape its tariffs. The compact entrenches itself: the longer it runs
#    the harder it is to set aside, and pulling out wrecks the migrations.
air_cure_salee_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		# AIR_f sets no tag: the marker is the only thing that identifies Air.
		has_variable = nd_air_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		OR = {
			owns = location:in_gall
			owns = location:agadez
		}
	}

	estates_that_like = {
		tribes_estate
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	neutral_modifier = {
		tribes_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		tribes_estate_levy_size = 1
		global_livestock_output_modifier = 0.10
		global_monthly_rural_control = 0.005
	}

	negative_modifier = {
		scale = {
			value = 1
			# entrenchment is 0-100, not 0-1 (vanilla hire_advisor.txt steps it
			# >= 10 through >= 100, and default_values.txt sets its penalties as
			# integers 1..25). A bare `add = scope:entrenchment` therefore scaled
			# this block by up to 101x instead of the intended 2x. Divided by 100
			# on 2026-08-07 to restore the 1.0x to 2.0x range the values were
			# written for.
			add = { value = scope:entrenchment divide = 100 }
			multiply = { value = 1 subtract = scope:maintenance }
		}
		crown_estate_satisfaction_decay = 0.020
		monthly_legitimacy = -0.05
		monthly_devotion = -0.05
		monthly_republican_tradition = -0.05
	}
}

# 3. Kel Owey Caravan Compact: the Tuareg caravan-guarantor system, by
#    which the Kel Owey and Kel Geres confederations led, watered, and
