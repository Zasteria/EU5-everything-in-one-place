# National Destinies: Russian bureaucracies
# 6 historically-grounded institutions of the Muscovite/Russian state.
# The auto modifier grants 3 slots, half of the six (the slot rule of
# 2026-10-02). No slot comes from the capital: only a megalopolis gives one.

# 1. Chamber of Genealogy (the key keeps the name Mestnichestvo, which stood
#    here until 2026-10-02: vanilla has the Mestnichestvo as an advance and a
#    government reform of the Russian culture group). This is the later
#    stage: when Feodor III burned the precedence books in 1682, a chamber
#    was set up to write the clans into one genealogy, the Velvet Book of
#    1687. The clans keep their honor and lose their quarrels. All from
#    memory, not fetched.
mestnichestvo_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = RUS
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	neutral_modifier = {
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Precedence kept the clans content and the throne respected. It never
	# enlarged a muster (the quarrels at every campaign did the opposite), so
	# the noble levy size that stood here is gone: with the service estates
	# advance it doubled the noble levy.
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_legitimacy = 0.05
		monthly_devotion = 0.05
		monthly_republican_tradition = 0.05
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		land_morale_modifier = -0.05
		country_cabinet_efficiency = -0.10
	}
}

# 3. The D'iaki (the key keeps the name Prikazy, which stood here until
#    2026-10-02: vanilla has the Prikazi as a government reform with seven
#    cabinet actions). This is the corps of clerks who staffed them: men of
#    common birth, trained in the offices, paid in coin and land, the
#    highest of them sitting in the Duma as dumnye d'iaki. From memory.

		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = RUS
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The bureaus were staffed by literate clerks of common birth, the d'iaki
	# and pod'iachie. The cabinet seat is theirs: the dumnye d'iaki sat in the
	# Duma. It was taken out on 2026-10-02 and put back the same day by the
	# owner's ruling that seats may stay; on 1.4 an empty seat lowers political
	# influence, which is the seat's own cost, so no penalty pays for it.

			subtract = scope:maintenance

	}
