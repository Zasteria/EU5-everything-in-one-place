# National Destinies: Egyptian bureaucracies
# 7 historically-grounded institutions of the Mamluk and post-Mamluk Egyptian
# state. The auto modifier grants 4 slots: half of the seven, rounded up.
# The formable asks nothing of the former's religion, so no slot from
# sharia law is certain, and a capital grants one only as a megalopolis.
# The player runs 4 of 7. Spans Bahri/Burji Mamluk diwans, Fatimid-rooted al-Azhar, the
# Coptic fiscal clerical class, and the Khedival Nizam of Muhammad Ali.
#
# Vanilla 1.4 ships generic Islamic bureaucracies for every Muslim country
# (bureaucracies/muslim.txt). The Egypt formable has no religion gate, but
# a Muslim Egypt, which is the usual one, holds them beside these. Two
# blocks here carried a vanilla office: the army register (vanilla's Diwan
# al-Jund) and the market inspector (vanilla's Hisba). Each now carries
# what the Mamluks did with that office that nobody else did. Their KEYS
# are unchanged, because saves and the impact modifier registry refer to
# them.

# 1. Rawk al-Nasiri (key says diwan_al_jaysh): the Army Bureau's great
#    work. Between 1313 and 1325, in Egypt in 1315, al-Nasir Muhammad had
#    every village surveyed and every grant reassigned. The sultan's share
#    of the land rose from 4 parts in 24 to 10, the amirs' grants shrank
#    and were scattered, and a crowd of petty market taxes was abolished.
#    Vanilla's Diwan al-Jund is the register that pays and musters
#    (manpower, army tradition), and this block used to share the army
#    tradition key. The survey is the opposite motion: the crown taking
#    land back from its own officers.
egy_diwan_al_jaysh_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = EGY
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# While the register stands, the amirs' grants lie scattered across the
	# country and none holds a province together (vanilla grants or takes
	# 0.1 to 0.25 of an estate's power).
	neutral_modifier = {
		global_nobles_estate_power = -0.10
		government_size = 1
	}

	# Funded: grants are sized to the contingent each must field, the
	# sultan holds ten parts in twenty-four, the market taxes stay abolished.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		global_crown_estate_power = 0.10
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		levy_recovery_modifier = -0.20
		country_cabinet_efficiency = -0.10
	}
}
