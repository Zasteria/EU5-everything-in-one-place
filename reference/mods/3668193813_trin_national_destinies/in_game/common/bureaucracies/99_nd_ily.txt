# National Destinies: Illyrian bureaucracies
#
# 4 institutions of the Catholic Slavic crown of the eastern Adriatic, none of
# them repeating the heritage tree, the destiny tree or the buildings. The
# heritage tree already spends the Glagolitic privilege, the Vinodol Code, the
# banderia, banal authority, the tricesima, the Senj press, the chapter walls,
# the colleges, the Military Frontier and the parish schools; the destiny tree
# spends Propaganda Fide, the union of Marca and the Statuta Valachorum. What
# is left, and what a state of this shape actually ran on, is paper: a church
# that held the public seal, a diet that voted the money and the men, an
# archdeacon's circuit that counted the souls, and a bench that decided who
# owned the ground.
#
# The identity is the period Illyrian one, the learned and ecclesiastical self
# designation of the Catholic Slavs of the eastern Adriatic between roughly
# 1450 and 1650. The Napoleonic provinces and the romantic movement of the
# 1830s are out of scope and appear nowhere.
#
# Deliberately NOT Dalmatia and deliberately NOT the Danubian Monarchy. DAL
# holds the maritime commune: the Senate of Ragusa, the quarantine of 1377,
# the communal statutes, the stradioti, tributary diplomacy. DNM holds the
# Vienna organs and the Graz-financed cordon. Illyria is the inland crown, and
# its institutions are clerical, conciliar and judicial.
#
# On the day of forming the country's own grant gives 2 places: 2 of 4
# institutions can run.
#
# All IDs are ily_-prefixed. The engine loads 99_nd_*.txt alphabetically and
# the last definition of a key wins, so an unprefixed id would silently
# override (or be overridden by) another tag's institution.

# 1. Loca credibilia, the places of authentic faith. From the decree of 1231
#    the recording of a legal act before a chapter or convent was compulsory,
#    and Louis I fixed the roll of authorised seals in 1351. The chapter of
#    Zagreb, sealing under the image of Saint Stephen since 1297, and the
#    collegiate chapter of Cazma, founded in 1232 by bishop Stjepan II, drew,
#    sealed and archived the charters of the realm, perambulated disputed
#    boundaries, served summonses and executed judgments for the king. Western
#    Europe hired lay notaries for this; here canons did it, and went on doing
#    it until feudal tenure was abolished in the 1840s.
#    Clergy and crown win, the towns and their displaced notariate lose.
ily_loca_credibilia_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}
	# Six centuries of title deeds sit in the chapter sacristies. Dismantling
	# the registry means repudiating every land title in the kingdom, so this
	# one is dear to be rid of.
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = ILY
		has_game_rule = nd_bureaucracies_enabled
	}

	# A chapter of Roman canons is what makes an instrument authentic. A crown
	# outside that communion has no such body to seal for it.
	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		clergy_estate
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# The clergy may raise archive houses: the registry needs somewhere to put
	# six hundred years of parchment.
	neutral_modifier = {
		allow_clerical_archives = yes
	}

	# Working: every boundary walked and written, every conveyance sealed and
	# filed. The crown's writ runs in the countryside because the countryside
	# is on paper, and land whose title nobody can dispute is land worth
	# improving. The price is paid at once: the church becomes indispensable.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_monthly_rural_control = 0.003
		global_rural_build_buildings_efficiency = 0.10
		global_clergy_estate_power = 0.05
	}

	# Decaying: the seal becomes a trade. Canons farm the fees, the archive
	# fills with instruments nobody will vouch for, and the towns whose own
	# notaries were pushed aside long ago hold the whole arrangement in
	# contempt.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		estate_enrichment = 0.05
		burghers_estate_satisfaction_decay = 0.015
	}
}

# 2. Sabor Kraljevine, the congregation of the realm. First recorded at Zagreb
#    in 1273 and merged on 1 September 1558 into a single assembly for Croatia
#    and Slavonia. Prelates, magnates, the ban, the viceban, the protonotary
#    and the deputies of the counties sat together; the diet voted the dika
#    laid on each hearth and called out the noble insurrection, and the
#    prelates brought banderia of their own. It answered a summons wherever
#    the ban sent it, which is how it came to elect a king in a frontier
#    castle at Cetin on 1 January 1527.
#    Nobles and prelates win, the crown pays in consent.
ily_sabor_kraljevine_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = ILY
		has_game_rule = nd_bureaucracies_enabled
	}

	# The everyday institution of the kingdom: available from the day it is
	# proclaimed, because it is older than the proclamation.
	allow = {}

	estates_that_like = {
		nobles_estate
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# A diet with no fixed seat. Krizevci, Topusko, Bihac, a castle at Cetin:
	# it assembles where it is called, not where the court happens to be.
	neutral_modifier = {
		can_call_rural_parliaments = yes
	}

	# Working: the insurrection is proclaimed and the counties answer, the
	# bishops bring their banderia, and a diet that has voted the dika itself
	# will let the collector into its own barns.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_levy_size = 0.20
		clergy_estate_levy_size = 0.20
		nobles_estate_max_tax = 0.05
	}

	# Decaying: a diet that is not summoned is a grievance with a membership
	# list. The great houses arrange the kingdom's business among themselves,
	# and the counties simply decline to publish what the ban sends them.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_nobles_estate_rebel_growth = 0.003
		global_monthly_control_decline = 0.002
	}
}

# 3. Visitationes canonicae, the archdeacon's circuit. From 1615 the
#    archdeacons of Zagreb rode every parish in turn and wrote a protocol on
#    each: the priest and what he could read, the church and its vessels, the
#    parish fields and their yield, the confraternities, the number of houses
#    and souls, the conduct of the people in them. Two hundred and fourteen
#    bound volumes survive from 1615 onward. It is the nearest thing this
#    kingdom had to a census, kept by clergymen for a crown that could not pay
#    surveyors, and on the frontier it is also the register of which villages
#    are of the Greek rite and which have come over.
#    The hierarchy and the parishes win, the patron nobility is audited.
ily_visitationes_canonicae_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = ILY
		has_game_rule = nd_bureaucracies_enabled
	}

	# A circuit of inspection needs inspectors who can write a protocol and
	# parish clergy worth inspecting. The colleges of 1580 supplied both.
	allow = {
		religion = religion:catholic
		has_advance = nd_ily_ilirski_kolegiji
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A clergy that is visited is a clergy that is answered to and mended.
	neutral_modifier = {
		clergy_estate_satisfaction_recovery = 0.001
	}

	# Working: the crown knows to the acre what every benefice is worth and
	# taxes it accordingly; the same circuit that counts the souls presses the
	# villages of the Greek rite toward Rome; and an absentee priest can be
	# reported by the people he is supposed to serve.
	positive_modifier = {
		scale = { value = scope:maintenance }
		clergy_estate_max_tax = 0.10
		global_heretic_pop_conversion_speed_modifier = 0.10
		peasants_estate_satisfaction_recovery = 0.001
	}

	# Decaying: the circuit shrinks to a fee hunt. The archdeacon comes for
	# his procuration and leaves, the schismatic villages learn that the
	# register is a denunciation and nothing else, and the parishes that had
	# one channel of complaint no longer have it.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_religious_rebel_growth = 0.010
		peasants_estate_satisfaction_decay = 0.015
	}
}

# 4. Tabula Banalis, the Banal Table. Charles VI erected it in 1723 in place
#    of the octaval terms, under which a suit could outlive the man who
#    brought it. The ban or his viceban presided over seven to nine judges
#    with the protonotary of the realm as reporter; it heard the great causes
#    at first instance and the appeals of the county courts, and from 1725 the
#    king named its members. Its docket was land, in a kingdom that had been
#    overrun, emptied and granted out again, and nobody clears a field in
#    Slavonia until the question of who owns it has an answer.
#    Crown and nobility win, the tenantry has no standing before it.
ily_tabula_banalis_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = ILY
		has_game_rule = nd_bureaucracies_enabled
	}

	# The bench sits at Zagreb, with the protonotary and the kingdom's seal.
	allow = {
		owns = location:zagreb
	}

	estates_that_like = {
		crown_estate
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# While the bench sits, a privilege is a document that can be read out and
	# ruled on, and the privileged pay a floor they cannot plead away.
	neutral_modifier = {
		nobles_estate_min_tax = 0.05
	}

	# Working: title is settled, so the emptied country is worth moving into;
	# a magnate's suit is heard within his own lifetime; and a grievance goes
	# to the bench instead of to the retainers.
	positive_modifier = {
		scale = { value = scope:maintenance }
		rural_migration_attraction = 0.10
		nobles_estate_satisfaction_recovery = 0.001
		monthly_rebel_growth = -0.0005
	}

	# Decaying: the terms are not kept, the appeals pile up, and the county
	# courts in magnate hands become the only jurisdiction that meets. The
	# tenant, who never had standing at the Table, now has nowhere at all.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_nobles_estate_power = 0.10
		monthly_peasants_estate_rebel_growth = 0.002
	}
}
