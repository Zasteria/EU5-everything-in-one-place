# National Destinies: Spanish bureaucracies
# 7 historically-grounded institutions of post-Reconquista Spain.
# The auto modifier grants 4 slots: half of the 7 institutions, rounded up, on
# the day the country forms. A capital gives no slot unless it is a megalopolis.
#
# All IDs are tag-prefixed (spa_) to avoid id_collision with other tags
# whose history overlaps (e.g. inquisicion, encomienda, audiencia, mesta).

# 1. Colegios Mayores: the six endowed colleges of Salamanca, Valladolid and
#    Alcala whose graduates filled the councils, high courts and sees of the
#    monarchy. Displayed under that name since 2026-10-02; the key keeps its
#    first name. Before that this was the Consejo Real, which vanilla gives
#    Spain as the reform royal_council_of_castile and the advance
#    cas_consejo_de_castilla. The grants stay: they were written for the
#    letrados, and the colleges are where the letrados came from.
spa_consejo_real_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = SPA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Letrados: a standing pool of university jurists to fill the crown's offices.
	neutral_modifier = {
		set_cabinet_member_cost_modifier = -0.2
		government_size = 1
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		global_max_urban_control = 0.03
		global_burghers_max_literacy = 10
		annexation_speed_modifier = 0.10
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}

# 5. Tribunal del Santo Oficio: the Spanish Inquisition, authorised by

#    enforced the orthodoxy of conversos and moriscos, censored the press,
#    and handled property confiscations that filled the royal treasury. The
#    auto-da-fe was as much a political theatre of monarchical authority

# 6. Encomienda: the Indies labour grant defined in law 1503, awarding

#    promulgate the New Laws of 1542 abolishing inheritance, but colonial

#    rotational repartimiento: a labour draft administered directly by
