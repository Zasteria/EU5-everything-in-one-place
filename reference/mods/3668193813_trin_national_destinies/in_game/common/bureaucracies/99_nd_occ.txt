# ============================================================================
# Occitania (OCC): bureaucracies
# National Destinies mod
# ============================================================================
#
# The heritage tree already spent this country's famous written instruments:
# the Consistori and the Leys d'Amors, the capitouls and the consulates, the
# pays de droit ecrit, the bastide charters, the Parlement of Toulouse, the
# civil dioceses, the Chambre de l'Edit and the works of the estates. What is
# left, and what these four institutions model, is the machinery that decided
# who paid, how much, and on whose authority.
#
# 1. Estats de Lengadoc: the province votes, apportions, collects and borrows
#    in its own name. A pays d'etats never admitted royal elus.
# 2. Cort dels Ajudas: the sovereign court of the fisc at Montpellier, the one
#    bench that could void a claimed exemption.
# 3. Lo Compes: the village land register, from which the taille falls on land
#    rather than on persons.
# 4. Cort Major de Bearn: twelve barons who take the prince's oath before he
#    takes theirs, and whose consent Bearn needs before a foreign war.
#
# No two share an estate alignment: the estates are burgher and clerical
# against the crown, the fiscal court is the crown against the nobility, the
# compoix is peasant and burgher against the nobility, and the Bearnais court
# is noble and peasant against the crown.
#
# On the day of forming the country's own grant gives 2 places: 2 of 4
# institutions can run.
#
# All IDs are occ_-prefixed to prevent collisions with other tags (the engine
# loads 99_nd_*.txt alphabetically and the last definition of a key wins).
#
# Catharism is deliberately absent. The last perfectus of the Languedoc burned
# at Villerouge-Termenes in 1321, sixteen years before the game starts, and the
# heritage tree treats the Cathar inheritance strictly as archive and land
# settlement. Nothing here reopens it.

# ----------------------------------------------------------------------------
# 1. Los Estats de Lengadoc
# ----------------------------------------------------------------------------
# The provincial estates, in annual session from the 1340s until 1789. Three
# orders: twenty-three bishops presided by the archbishop of Narbonne as
# president born, twenty-three barons, and sixty-eight deputies of the towns,
# whose weight was held to parity with the other two. Languedoc was a pays
# d'etats and never admitted the royal elus: the crown asked, the estates
# bargained the don gratuit down to a lump sum, and then the province itself
# apportioned that sum through the assiettes of its civil dioceses, whose
# composition it was recognised as regulating in 1553 and whose accounts it
# won the right to audit in January 1658. Its own receivers collected. Its
# syndics general lived at court to defend its privileges. And the bourse of
# the estates borrowed in the province's name on paper that lenders rated
# above the crown's, which is why the estates could put up four fifths of the
# cost of the Canal des Deux Mers while the treasury could not.
occ_estats_de_lengadoc_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = OCC
		has_game_rule = nd_bureaucracies_enabled
	}

	# The estates were already sitting in 1337. Nothing gates them.
	allow = {}

	estates_that_like = {
		burghers_estate
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# The bourse of the estates is standing credit in the province's own name,
	# separate from the treasury and better rated than it.
	neutral_modifier = {
		monthly_loan_capacity_investment = 5
	}

	# Full session: the syndics resident, the assiettes audited, the don gratuit
	# voted rather than extracted. Subsidy the orders have consented to is
	# subsidy that arrives, the province's rentes carry a low coupon, and the
	# sixty-eight town deputies acquire leverage that is the price of the
	# arrangement.
	positive_modifier = {
		scale = { value = scope:maintenance }
		bond_interest = -0.01
		clergy_estate_max_tax = 0.05
		global_burghers_estate_power = 0.05
	}

	# Neglected, the session becomes a formality. The bishops who preside the
	# assiettes apportion unwatched, the receivers keep the float, and the
	# arithmetic slides downward onto the land that cannot claim exemption.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_clergy_estate_power = 0.10
		estate_enrichment = 0.10
	}
}

# ----------------------------------------------------------------------------
# 2. La Cort dels Ajudas
# ----------------------------------------------------------------------------
# The Cour des Aides, created by Charles VII on 20 April 1437 as the second in
# the kingdom and given the whole Midi: Languedoc, the Rouergue, the Quercy and
# part of Guyenne. Louis XI fixed it at Montpellier in 1467. The Parlement of
# Toulouse and the estates combined against it and got it suppressed in 1484;
# it was back in 1486, and the edict of July 1629 merged it with the chamber of
# accounts of 1523 into the Cour des comptes, aides et finances de Montpellier.
# Its business was the fisc: registering fiscal edicts, judging the aides and
# the gabelle in last resort, and hearing the exemption claims. It was the one
# bench in the Midi that could rule a man's nobility usurped and his exemption
# void, which is why the crown liked it and the country did not.
occ_cort_dels_ajudas_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = OCC
		has_game_rule = nd_bureaucracies_enabled
	}

	# A sovereign court of the fisc needs the seat Louis XI gave it and a legal
	# culture that will treat a fiscal ruling as binding on a lord.
	allow = {
		owns = location:montpellier
		has_embraced_institution = institution:legalism
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A standing bench for exemption cases is the cheapest way to take a
	# privilege back: the claim is tried rather than bought out.
	neutral_modifier = {
		revoke_privilege_cost_modifier = -0.10
	}

	# Working court: usurped nobility struck off the rolls, farmers of the aides
	# audited on their receipts, and the urban consumption duties actually
	# collected in the towns where they fall.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_max_tax = 0.05
		estate_enrichment = -0.10
		global_max_urban_control = 0.02
	}

	# Starved, the docket jams and the offices are sold. A bench of hereditary
	# proprietors cannot be reformed and will not be dismissed, everything else
	# the country has built calcifies around it, and lenders price the crown's
	# paper by whether its revenue is enforceable.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_bureaucracy_entrenchment_speed_modifier = 0.10
		bank_interest = 0.02
	}
}

# ----------------------------------------------------------------------------
# 3. Lo Compes
# ----------------------------------------------------------------------------
# The compoix, the community land register of the Midi. The oldest surviving
# one is of 1320 and most communities kept at least one down to the Revolution.
# Every parcel in the terroir taillable was walked, measured, graded for
# quality and entered in allivrement against the name of its holder, and the
# register was reworked roughly once a lifetime. From it came the taille
# reelle: the tax attaches to the land and not to the person, so a parcel
# entered as taillable stays taillable in a noble's hands. North of the Loire
# the taille was personal and a noble's purchase left the roll. Three registers
# stack above one another, the provincial cadastre dividing the levy between
# dioceses, the diocesan cadastre between communities, and the compoix of the
# village between neighbours who can all hear it read out.
occ_lo_compes_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = OCC
		has_game_rule = nd_bureaucracies_enabled
	}

	# Registers of this kind were already being kept in 1320. Nothing gates it.
	allow = {}

	estates_that_like = {
		peasants_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A countryside where every parcel has a name against it is a countryside
	# the state can see.
	neutral_modifier = {
		global_monthly_rural_control = 0.002
	}

	# Kept current: the surveyors go out, the allivrement is revised, and a
	# holding entered under a man's own name is a holding he keeps the yield
	# of. Land that changes hands upward keeps paying, so the seigneury stops
	# being a way to move wealth out of the roll, and a village whose register
	# is honest is a village worth moving to.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_peasant_enfranchisment = 0.05
		global_nobles_estate_power = -0.10
		rural_migration_attraction = 0.10
	}

	# Left unrevised, the register becomes fiction. It taxes vines that were
	# grubbed up two generations ago and misses the ones planted since, the
	# quality grades no longer match the ground, and the assessment that ruins
	# a smallholder is the one nobody has checked in eighty years.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_pop_demotion_speed_modifier = 0.10
		peasants_estate_satisfaction_decay = 0.020
	}
}

# ----------------------------------------------------------------------------
# 4. La Cort Major de Bearn
# ----------------------------------------------------------------------------
# The Cour Majour, the high court of Bearn, which Gaston Febus reduced to
# twelve barons sitting as jurats at Orthez and which followed the court to Pau
# after 1464. It judged in last resort, it had to be heard before Bearn
# engaged itself in a war outside its own borders, and it received the
# viscount's oath at his accession: the lord swore to keep the Fors first, and
# only then did Bearn swear to him. The Fors themselves were compiled from the
# eleventh century onward and expanded under Febus, and what they guarantee is
# the free household against arbitrary levy. That is the constitutional ground
# Febus was standing on in 1347 when he answered Philip VI that he would do
# homage for Foix, but that he held Bearn from God and from no man in the world.
occ_cort_major_de_bearn_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = OCC
		has_game_rule = nd_bureaucracies_enabled
	}

	# The court sat at Orthez under Febus and at Pau from 1464. It is a
	# Bearnais body and cannot be convened anywhere else.
	allow = {
		OR = {
			owns = location:orthez
			owns = location:pau
		}
	}

	estates_that_like = {
		nobles_estate
		peasants_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# The Fors are sworn before homage is taken, and what they guarantee first
	# is that the free households are not taxed at the prince's pleasure.
	neutral_modifier = {
		peasants_estate_max_tax = -0.05
	}

	# The court in session: a war outside the borders is argued and voted
	# before it is fought, so the country's case is made in writing and its
	# quarrels are entered rather than nursed. The barons have a bench to be
	# angry at instead of a grievance to keep, and a country whose law was
	# sworn to before it was obeyed does not rise against it.
	positive_modifier = {
		scale = { value = scope:maintenance }
		casus_belli_creation_speed_modifier = 0.15
		nobles_estate_satisfaction_decay = -0.010
		monthly_peasants_estate_rebel_growth = -0.005
	}

	# When the court stops meeting, the twelve go on judging, only now in their
	# own castles and on their own account. The oath becomes a ceremony nobody
	# can call in, and the writ of Orthez thins out one valley at a time.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_nobles_estate_power = 0.10
		global_monthly_control_decline = 0.005
	}
}
