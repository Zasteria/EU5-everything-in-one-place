# National Destinies: Al-Andalus bureaucracies
# Seven historically-grounded institutions of the Andalusi state, drawn
# from the Caliphate of Cordoba (929-1031), the taifa kingdoms, and the
# Almoravid/Almohad emirates that ruled al-Andalus. The auto modifier grants
# 3 slots and sharia law gives the fourth: half of the 7 institutions, rounded
# up, on the day the country forms. A capital gives no slot unless it is a
# megalopolis.
#
# All IDs prefixed with `adu_` to avoid cross-file collisions.
#
# Vanilla 1.4 ships generic Islamic bureaucracies for every Muslim country
# (bureaucracies/muslim.txt), and Al-Andalus is Muslim by its formable's
# own gate, so it holds those as well. Three blocks here once carried the
# same office as a vanilla one (Diwan al-Jund, Hisba, Barid) and now carry
# the Andalusi form instead. Their KEYS are unchanged, because saves and
# the impact modifier registry refer to them, so adu_barid_bureaucracy is
# now the watchtower line and not a postal service.

# 1. Hijaba: the Caliphal chief-ministry. The hajib stood at the head
#    of the chancery, received petitioners, drafted decrees, and from
#    the 10th century onward (Almanzor) effectively ruled in the
#    caliph's name. Centralized administration through the qasr.
adu_hijaba_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = ADU
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The hajib is the head of the ministers: the office exists to name one.
	neutral_modifier = {
		head_of_cabinet_promotion_cost_modifier = -0.25
		government_size = 1
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		global_crown_estate_power = 0.10
		monthly_legitimacy = 0.05
		monthly_devotion = 0.05
		monthly_republican_tradition = 0.05
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}

# 2. Syrian Junds (key says diwan_al_jund): the regiments of Syria that
#    the governor Abu al-Khattar settled across the south in 743 (Damascus
#    in Elvira, Jordan in Rayyu, Palestine in Sidonia, Hims in Seville,
#    Qinnasrin in Jaen, Egypt in Beja and Tudmir). Vanilla's Diwan al-Jund
#    is the eastern register that pays soldiers a stipend (manpower, food,
#    likes the crown). This is the opposite arrangement: regiments living

# 6. Atalayas of the Marches (key says barid): the chains of round
#    watchtowers the emirs and caliphs of Cordoba built along the three
#    marches, from Muhammad I in the ninth century to Abd al-Rahman III.
#    Fire by night and smoke by day carried a warning from tower to tower:
#    word from Soria could reach Cordoba in about five hours. Vanilla's
#    Barid is the relay post of the east (spy network, proximity speed),
#    which is what this block used to copy key for key. The towers do a
#    different job: the frontier musters and mans its walls before the
#    raid arrives.
adu_barid_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = ADU
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# "Levy Recruitment Speed": how quickly levies assemble. The towers
	# stand whether they are well kept or not, and a warning is a warning.
	neutral_modifier = {
		global_levy_recruitment_speed_modifier = 0.10
	}

	# Manned towers: garrisons fill before the enemy arrives, strangers on
	# the frontier are seen, and the march reports to Cordoba the same day.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_garrison_growth = 0.01
		counter_espionage = 0.20
		global_monthly_control = 0.003
	}

	# Empty towers: raids arrive unannounced and wear the frontier down, and
	# the marcher families (Tujibids, Banu Qasi) look to their own defence.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_nobles_estate_power = 0.10
		monthly_war_exhaustion = 0.008
	}
}
