# ===== Outremer (OUT) Bureaucracies =====
# Six institutions of the Latin East, all of them apparatus the crown either
# inherited, licensed, or paid for, and none of it apparatus the crown simply
# owns. That is the country. OUT_f can only be proclaimed while Catholics are
# at most half the realm's population, so the emperor crowned at the Sepulchre
# governs a majority he may not convert. There is no conversion modifier here,
# no holy war line and no tolerance key used as decoration, exactly as in the
# heritage and destiny trees. The religious question is reached through the
# estates instead, because estate religion (game concept) means each estate's
# satisfaction and power are set by the crown's tolerance of that estate's
# faith, and in this realm the clergy estate contains the qadi, the Coptic
# priest and the rabbi as well as the Latin bishop.
#
# The dhimmi estate is deliberately absent from every block below. The
# scripted trigger is_dhimmi requires the owner's religion group to be Muslim,
# so a Catholic Outremer has no dhimmi estate and every key in that family is
# inert. Nothing here can be fixed by using them.
#
# On the day of forming the country's own grant gives 3 places: 3 of 6
# institutions can run.
#
# Nothing here restates the heritage tree. The ra'is has a building, the Fonde
# has a building, the communes have an advance and a decision event, the
# secrete and the capitation are advances, the Darb al-Hajj is a building and
# three advances. These six are the institutions those did not cover: the
# native courts rather than the native villages, the harbour court rather than
# the market court, the standing charters rather than the one-off bargain, the
# native cavalry corps, the planted settlements, and the crown's horse debt.

# 1. Cour des Syriens: the native Christians' own tribunal, sitting under the
#    ra'is of the Syrians, one of the lower jurisdictions the Franks left
#    standing beside the qadi's court and the bet din. The principle was the
#    personality of law, and each community swore on its own book. The jurists
#    of Acre named these subordinate courts when they wrote the Livre des
#    Assises between 1240 and 1244. A crown that is a minority among its own
#    subjects rules by guaranteeing other people's judges.
out_cour_des_suriens_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = OUT
		has_game_rule = nd_bureaucracies_enabled
	}

	# There is nothing for these benches to do in a realm the crown's own
	# church has taken over. The threshold sits above OUT_f's own formation
	# clause so the institution survives a generation of Catholic growth
	# rather than vanishing the moment the arithmetic shifts.
	allow = {
		religion_percentage_in_country = {
			religion = religion:catholic
			value <= 0.75
		}
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A community that can be heard does not have to rise to be heard. This
	# is the constant return on the whole arrangement, and it is the one
	# number that makes an unconvertible realm governable at all.
	neutral_modifier = {
		monthly_religious_rebel_growth = -0.01
	}

	# Funded, the benches sit, the ra'is is paid and confirmed, the oaths are
	# taken on the right book, and a grievance goes to a judge instead of to
	# the hills. The community's own clergy become the crown's easiest
	# interlocutors, and the writ of the towns runs further than the walls.
	positive_modifier = {
		scale = { value = scope:maintenance }
		peasants_estate_satisfaction_recovery = 0.001
		clergy_estate_agenda_impact = 0.15
		global_max_urban_control = 0.03
	}

	# Unfunded, the benches sit anyway. That is the danger. The qadi, the
	# bishop and the ra'is go on judging without the crown's licence, fee or
	# record, and the countryside stops being administered and starts being
	# merely inhabited.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_clergy_estate_power = 0.10
		global_monthly_rural_control = -0.003
	}
}

# 2. Cour de la Chaine: the harbour court. The catena was the chain that
#    closed the port of Acre and the customs house beside it, and the court
#    that sat there heard mariners' and shippers' pleas under a bailli who
#    presided for the lord without judging. Its procedure was written down at
#    Acre between 1240 and 1244 alongside the market court of the Fonde. Every
#    Italian commune in the port claimed exemption from it.
out_cour_de_la_chaine_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = OUT
		has_game_rule = nd_bureaucracies_enabled
	}

	# A chain court needs a chain. These are the three harbours where the
	# crusader states actually kept one: the kingdom's capital and its last
	# mainland holding, and the two Egyptian customs ports the Frankish
	# expeditions of 1219 and 1249 were launched to take.
	allow = {
		OR = {
			owns = location:acre
			owns = location:alexandria
			owns = location:damietta
		}
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# The court keeps the roll of every hull and every mariner that comes
	# under the chain, which is also the only register of seamen the crown
	# has.
	neutral_modifier = {
		global_navy_levy_size_modifier = 0.2
	}

	# Funded, the tariff is assessed where the cargo lands rather than where
	# the merchant chooses to declare it, foreign houses trade under the
	# crown's bailli instead of around him, and the burgess quarter pays what
	# the roll says it owes.
	positive_modifier = {
		scale = { value = scope:maintenance }
		trade_sea_efficiency = 0.05
		foreign_export_from_market_efficiency = 0.10
		burghers_estate_max_tax = 0.10
	}

	# Unfunded, the chain is farmed to whoever will advance the cash. The
	# farmer keeps the tenth, the escort stops sailing because nobody is
	# paying for it, and the crown's own household discovers that the
	# harbour it built belongs to a syndicate.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_trade_protection_factor = -0.15
		crown_estate_satisfaction_decay = 0.020
	}
}

# 3. The Charters of the Communes: the standing privileges of the Venetian,
#    Genoese and Pisan quarters. The Pactum Warmundi of 1123 gave Venice a
#    third of Tyre with its own church, oven, bath, weights, court and
#    gallows; Genoa and Pisa took comparable quarters at Acre through the
#    twelfth and thirteenth centuries. Venice claimed over everyone resident
#    in its quarter the same jurisdiction and taxation the king held over his
#    own. Baldwin II refused to concede they were autonomous and the treaty
#    ran until 1291 regardless. This is the one institution in the file the
#    crown did not create, cannot direct and can barely afford to end.
out_charters_of_the_communes_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}
	# A charter granted for a fleet that has already sailed is not bought
	# back cheaply. Revoking one costs several times what any other office
	# here costs to close, which is the institutional inertia of the thing
	# expressed as a bill.
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.006
	}

	potential = {
		has_or_had_tag = OUT
		has_game_rule = nd_bureaucracies_enabled
	}

	# The crown has to have recognised the quarters before it can institute
	# their charters as a standing arrangement.
	allow = {
		has_advance = nd_out_communes_of_the_ports
	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Not a benefit and not a penalty the player can switch off. A third of
	# the capital is under a foreign republic's bench, permanently, for as
	# long as the charters stand.
	neutral_modifier = {
		global_burghers_estate_power = 0.10
	}

	# Honoured, the charters are worth what the crown gave up for them. The
	# quarters bring the shipping, the credit and the factors, and the
	# realm's commerce reaches ports its own flag never visits. It still
	# cannot be governed: the ceiling of the crown's control in its own
	# cities stays lowered whatever the maintenance.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_trades_per_burgher = 0.20
		trade_range_modifier = 0.15
		global_max_urban_control = -0.02
	}

	# Neglected, the quarters settle their own quarrels. The War of Saint
	# Sabas was fought between Venetians and Genoese inside the kingdom's own
	# capital from 1256, with towers, siege engines and fleets, while the
	# crown watched from the other side of a wall it had granted away.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_burghers_estate_rebel_growth = 0.02
		global_monthly_urban_control = -0.003
	}
}

# 4. The Office of the Turcopolier: the command of the turcopoles, the light
#    horse and mounted archers raised in the East from Syrian, Greek and
#    Armenian Christians and from Turks who had come over. Byzantine
#    turcopoles under Tatikios rode with the First Crusade and gave the model.
#    The Turcopolier outranked the sergeants in the field, and both the Temple
#    and the Hospital kept the office among their senior dignities. This is
#    the corps that let a Frankish army move; the knights could not do it.
out_turcopolier_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = OUT
		has_game_rule = nd_bureaucracies_enabled
	}

	# Turcopoles were recruited from the Eastern Christian communities before
	# they were recruited from anyone else, so the office needs those
	# communities inside the realm.
	allow = {
		any_owned_location = {
			OR = {
				dominant_religion = religion:miaphysite
				dominant_religion = religion:orthodox
				dominant_religion = religion:nestorianism
			}
		}
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		clergy_estate
	}

	# Light horse that does not need a road, which in this theatre is most
	# of the country.
	neutral_modifier = {
		movement_speed_if_no_road = 0.05
	}

	# Paid, the office is the answer to this realm's actual military problem:
	# a Latin crown drawing horse from a population that is not Latin. The
	# muster rolls fill from communities no Frankish levy could reach, and
	# they ride before the knights have finished assembling.
	positive_modifier = {
		scale = { value = scope:maintenance }
		wrong_culture_levy_size = 0.15
		army_light_cavalry_power = 0.10
		global_levy_recruitment_speed_modifier = 0.10
	}

	# Unpaid, the corps becomes exactly what the Latin clergy always said it
	# was. Saladin had the captured turcopoles put to death after Hattin in
	# 1187 as apostates, and the Church of the realm never quite stopped
	# agreeing with him. The horse still has to be bought, at the price a
	# seller names to a crown in arrears.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_clergy_estate_rebel_growth = 0.02
		army_light_cavalry_maintenance_cost_modifier = 0.15
	}
}

# 5. The New Towns: the planted burgess villages. Magna Mahumeria, now
#    al-Bira, stood sixteen kilometres north of Jerusalem and was founded by
#    the canons of the Holy Sepulchre on one of the twenty-one casalia
#    Godfrey of Bouillon granted them. Ninety-two burgesses swore allegiance
#    in 1156 and fifty more over the next thirty years, settlers out of
#    southern France, Italy and Spain, holding in burgage tenure for a tithe
#    and a share, with three smiths, three carpenters, a mason, a shoemaker
#    and a goldsmith among them. Parva Mahumeria and Bethgibelin were the same
#    instrument. It is the only lawful way this crown can change the
#    arithmetic of its own realm.
out_new_towns_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# Planting a village and keeping it planted is the most expensive office
	# in the file after the horse debt.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = OUT
		has_game_rule = nd_bureaucracies_enabled
	}

	# The third founding settlement of the coronation, which is the one that
	# has no destiny branch. A crown that swore the compact or took the
	# guardianship instead did not put settlers on the Jerusalem road, and
	# cannot institute an office for a policy it never adopted.
	allow = {
		custom_tooltip = {
			text = out_new_towns_path_c_tt
			has_variable = nd_out_path_c
		}
	}

	estates_that_like = {
		burghers_estate
		clergy_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A planted village is the rare thing in this realm: land the crown
	# actually administers, because it laid out the plots itself.
	neutral_modifier = {
		global_max_rural_control = 0.02
	}

	# Funded, the plots are surveyed, the charters are issued and the ships
	# keep arriving from Provence and Lombardy. Burgage tenure makes a
	# burgess of a man who arrived a peasant, the smiths and masons of the
	# new towns are the realm's literate trades, and the settlements owe
	# sergeants rather than rent.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_pop_promotion_speed_modifier = 0.10
		burghers_estate_levy_size = 0.10
		global_laborers_max_literacy = 2.5
	}

	# Unfunded, the plantation stops and the villages are absorbed. The
	# seigneurs whose rents the free tenure undercut take the casalia back
	# one charter at a time, and the burgesses who came for a plot of their
	# own take ship home.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_nobles_estate_power = 0.10
		burghers_estate_satisfaction_decay = 0.020
	}
}

# 6. The Restor of the Host: the crown's obligation to replace a horse killed
#    or crippled in its service, and the constable's jurisdiction that
#    administered it. Captured Arab horses were crown spoil and were
#    redistributed through the restor to knights and turcopoles who had lost
#    their mounts. Europe had nothing on this scale because Europe could breed
#    its own warhorses; the Latin East could not, and the obligation was the
#    difference between a host that could survive a mauling and one that could
#    not. Its arrears were a standing baronial grievance for two centuries.
out_restor_of_the_host_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# The crown is buying horses it cannot breed, every season, forever.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = OUT
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Remount depots, standing herds and the crown's own studs, because an
	# obligation to replace horses is first of all an obligation to have
	# them.
	neutral_modifier = {
		global_horses_output_modifier = 0.10
	}

	# Honoured, a knight who loses his horse in the crown's service is
	# mounted again by the crown, so he serves in the field instead of
	# husbanding a beast he cannot replace. The host reforms after a defeat
	# that would otherwise end it, which is precisely what the kingdom
	# needed after Hattin and did not have.
	positive_modifier = {
		scale = { value = scope:maintenance }
		army_heavy_cavalry_reinforce_cost_modifier = -0.15
		nobles_estate_levy_size = 0.10
		levy_recovery_modifier = 0.10
	}

	# In arrears, the restor becomes a claim rather than a service. Horses
	# are billed for that never died, the barons hold the arrears against
	# every request for taxation, and the crown pays the market price for
	# remounts it was already owed.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		army_heavy_cavalry_maintenance_cost_modifier = 0.15
		nobles_estate_max_tax = -0.10
	}
}
