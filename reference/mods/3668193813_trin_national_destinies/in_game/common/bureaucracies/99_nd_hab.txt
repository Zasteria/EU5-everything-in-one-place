# 2. Geheime Konferenz: the Privy Conference, evolved from the
#    sixteenth-century Geheimer Rat under Ferdinand I and given its
#    classical form by Leopold I after 1665. A small inner council
#    of five to seven hand-picked grandees (the Auersperg, Lobkowitz,
#    Schwarzenberg, later Kaunitz) deliberated on dynastic marriages,
#    war and peace, and the disposition of the Spanish succession.
#    Its protocols, kept in the Haus-, Hof- und Staatsarchiv, became
#    the chief register of Habsburg high diplomacy.

# 5. Directorium in publicis et cameralibus: the central administrative
#    body imposed by Friedrich Wilhelm Graf von Haugwitz on 2 May 1749
#    after the loss of Silesia exposed the bankruptcy of estate-based
#    administration. The Directorium subordinated the Bohemian and
#    Austrian Hofkanzleien and the Hofkammer into a single college,
#    centralizing tax assessment, recruit-quota allocation and the
#    Theresian cadaster. The provincial Landstaende lost their
#    constitutional veto over the Contributionale; permanent salaried
#    Kreishauptleute replaced noble tax-collectors on the ground.
hab_directorium_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = HAB
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		has_embraced_institution = institution:enlightenment
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# One college in Vienna over the chanceries and the chamber, and its own
	# salaried district captains in every Kreis: the monarchy's centralizing
	# reform. Vanilla's Board of Revenue moves a societal value the same way.
	neutral_modifier = {
		monthly_towards_centralization = societal_value_monthly_move
		government_size = 1
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		tax_income_efficiency = 0.05
		global_manpower_modifier = 0.10
		global_monthly_control = 0.003
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}
