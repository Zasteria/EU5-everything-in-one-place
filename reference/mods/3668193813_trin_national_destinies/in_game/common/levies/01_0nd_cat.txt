# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Catalonia (CAT): age 5 levies
#
# Two levies in one age, drawing different pop types: the hills send
# fusiliers, the guilds man the walls. Soldier pops raise vanilla's militiamen.

# Fusellers Voluntaris: successor of the miquelet levy for the same pops. It
# must out-power both vanilla's militiamen and the age 4 miquelets.
levy_nd_cat_fusellers_voluntaris = {
	size = levy_generic_infantry_size
	unit = a_nd_cat_fusellers_voluntaris

	allowed_pop_type = peasants
	allowed_pop_type = laborers
	allowed_culture = catalan

	country_allow = {
		has_or_had_tag = CAT
	}
}

# Coronela: the guild battalions. Burghers only, because a company of the
# Coronela WAS a guild under arms: masters and journeymen of one trade under
# that trade's banner. Catalan pops only, for the same reason the guilds of
# Barcelona, Girona and Perpinya had no chapters in a conquered town.
# Not carried into age 6 on purpose: the Nueva Planta abolished it in 1714.
levy_nd_cat_coronela = {
	size = levy_generic_infantry_size
	unit = a_nd_cat_coronela

	allowed_pop_type = burghers
	allowed_culture = catalan

	country_allow = {
		has_or_had_tag = CAT
	}
}
