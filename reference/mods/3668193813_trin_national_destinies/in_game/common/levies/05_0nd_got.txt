# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Regnum Gothorum (GOT): age 1 levies
# National Destinies mod
#
# Newer GOT levies live in 02_0nd_got.txt (age 4), 01_0nd_got.txt (age 5) and
# 00_0nd_got.txt (age 6), so each is superseded in turn.

# Gardingi: the king's sworn men. Nobles only, at the noble cavalry rate.
levy_nd_got_gardingi = {
	size = levy_generic_noble_cavalry_size
	unit = a_nd_got_gardingi

	allowed_pop_type = nobles

	country_allow = {
		has_or_had_tag = GOT
	}
}

# Thiufa: the muster of the whole free population. The pop list is the
# content, not boilerplate. Ervig's revision of Wamba's law (681) binds
# "whether general, count or gardingus, Goth or Roman, freeman or manumitted
# slave, or any fiscal slave", and Wamba's original text bound bishops and
# their retainers too, which is why clergy appear here and appear in no other
# infantry levy in this mod.
levy_nd_got_thiufa = {
	size = levy_generic_infantry_size
	unit = a_nd_got_thiufa

	allowed_pop_type = soldiers
	allowed_pop_type = peasants
	allowed_pop_type = laborers
	allowed_pop_type = clergy

	country_allow = {
		has_or_had_tag = GOT
	}
}
