# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.

levy_nd_iro_war_party = {
	size = levy_generic_infantry_size
	unit = a_nd_iro_war_party

	allowed_pop_type = peasants
	allowed_pop_type = soldiers
	allowed_pop_type = tribesmen
	allowed_pop_type = laborers

	country_allow = {
		has_or_had_tag = IRO
	}
}

# Canoe raiders are raised where there is water to paddle on: a river, a
# lake or the coast (the location test is vanilla's own for water access,
# building_types/common_buildings.txt:101). Pops inland raise the war party.
# Both levies draw the same pop types and the game keeps only the stronger
# one for a pop, so the narrower levy has to be the stronger of the two.
levy_nd_iro_canoe_raiders = {
	size = levy_generic_infantry_size
	unit = a_nd_iro_canoe_raiders

	allowed_pop_type = peasants
	allowed_pop_type = soldiers
	allowed_pop_type = tribesmen
	allowed_pop_type = laborers

	allow = {
		location ?= {
			OR = {
				has_river = yes
				is_coastal = yes
				is_adjacent_to_lake = yes
			}
		}
	}

	country_allow = {
		has_or_had_tag = IRO
	}
}
