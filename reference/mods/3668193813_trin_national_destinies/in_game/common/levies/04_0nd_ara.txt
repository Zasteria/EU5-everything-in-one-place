# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Aragon (ARA): age 2 levies

# Caballeros Villanos: the horse of the frontier towns. Burghers only,
# because that is who they were: townsmen whom the fuero freed from tax for
# keeping a horse. Drawn at the infantry size, never the noble cavalry size:
# burghers are a large pop type. Aragonese pops only. Vanilla has no levy for
# burghers before age 3, so nothing has to be beaten; from age 3 the burghers
# raise vanilla's matchlock levy, which is the intended end of this one.
levy_nd_ara_caballeros_villanos = {
	size = levy_generic_infantry_size
	unit = a_nd_ara_caballeros_villanos

	allowed_pop_type = burghers
	allowed_culture = aragonese

	country_allow = {
		has_or_had_tag = ARA
	}
}
