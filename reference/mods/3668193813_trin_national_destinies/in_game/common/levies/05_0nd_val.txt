# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Valencia (VAL): age 1 levies

# Frares de Montesa: the knight brethren of the kingdom's own military order.
# Clergy, at the size vanilla uses for its crusader knights
# (levy_crusader_knights, 05_traditions_levies.txt:28). Only the clergy of a
# location where a Comanda de Montesa stands: the brethren lived in their
# commanderies, and a parish priest was no knight. The same shape as
# vanilla's gallowglass and their sept (unit_gallowglass_pop_trigger).
levy_nd_val_frares_de_montesa = {
	size = levy_generic_noble_cavalry_size
	unit = a_nd_val_frares_de_montesa

	allowed_pop_type = clergy

	allow = {
		location = { has_building_with_at_least_one_level = nd_val_comanda_de_montesa }
	}

	country_allow = {
		has_or_had_tag = VAL
	}
}
