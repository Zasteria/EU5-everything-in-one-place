# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Catalonia (CAT): age 4 levies

# Miquelets: mountain irregulars raised by the Diputacio from 1640. Peasants
# and laborers, never soldiers or burghers: these were charcoal burners,
# shepherds and day workers of the hill country, not a standing force and not
# a town militia. Catalan pops only, because the miquelet was a creature of
# the Catalan hills and of the usatge Princeps namque that bound a Catalan to
# defend the land; a conquered Aragonese or Occitan village raises vanilla's
# levy instead.
levy_nd_cat_miquelets = {
	size = levy_generic_infantry_size
	unit = a_nd_cat_miquelets

	allowed_pop_type = peasants
	allowed_pop_type = laborers
	allowed_culture = catalan

	country_allow = {
		has_or_had_tag = CAT
	}
}
