# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Portugal (POR): age 2 levies

# Besteiros do Conto: the crossbowmen each concelho owed by tally. Laborers
# and burghers, never peasants or soldiers: the rolls were filled from the
# crafts of the towns (shoemakers, tailors, carpenters, smiths), men with a
# trade and the means to keep a crossbow, not from field hands and not from
# the paid garrison. No culture test: the tally was owed by the concelho, and
# a town the crown conquered and chartered owed it like any other.
# No successor in age 3 on purpose: Manuel I abolished them in 1498, so their
# pops pass to vanilla's matchlock levy when it arrives.
levy_nd_por_besteiros_do_conto = {
	size = levy_generic_infantry_size
	unit = a_nd_por_besteiros_do_conto

	allowed_pop_type = laborers
	allowed_pop_type = burghers

	country_allow = {
		has_or_had_tag = POR
	}
}
