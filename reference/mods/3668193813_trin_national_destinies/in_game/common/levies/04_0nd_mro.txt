# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.

# Kalis warriors: Moro warrior-noble class with the wavy-bladed kris
levy_nd_mro_kalis_warriors = {
	size = levy_generic_infantry_size
	unit = a_nd_mro_kalis_warriors

	allowed_pop_type = soldiers
	allowed_pop_type = peasants
	allowed_pop_type = laborers

	country_allow = {
		has_or_had_tag = MRO
	}
}

# Iranun raiders: coastal pirate levy of Illana Bay. Raised on the coast
# only; pops inland raise the kalis warriors. Both levies unlock in the same
# age and draw the same pop types, and the game keeps only the stronger one
# for a pop, so the narrower levy has to be the stronger of the two.
levy_nd_mro_iranun_raiders = {
	size = levy_generic_infantry_size
	unit = a_nd_mro_iranun_raiders

	allowed_pop_type = soldiers
	allowed_pop_type = peasants
	allowed_pop_type = laborers

	allow = {
		location ?= {
			is_coastal = yes
		}
	}

	country_allow = {
		has_or_had_tag = MRO
	}
}
