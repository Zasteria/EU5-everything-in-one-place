# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.

# Thessalian lancers: the horse-owning lords of the Larissa plain, raised
# from the nobility as cavalry levies always are.
levy_nd_the_thessalian_lancers = {
	size = levy_generic_noble_cavalry_size
	unit = a_nd_the_thessalian_lancers

	allowed_pop_type = nobles

	country_allow = {
		has_or_had_tag = THE
	}
}

# The boats of the parathalassioi. The burghers of the harbour own the
# hulls; the labourers and peasants of the shore pull the oars. This is
# the same body of men that ran the city as a commune from 1342.
levy_nd_the_parathalassioi = {
	size = levy_ship_size
	unit = n_nd_the_parathalassioi_galley

	allow = {
		pop_type = pop_type:burghers
	}
	allow_as_crew = {
		OR = {
			pop_type = pop_type:laborers
			pop_type = pop_type:peasants
		}
	}

	country_allow = {
		has_or_had_tag = THE
	}
}
