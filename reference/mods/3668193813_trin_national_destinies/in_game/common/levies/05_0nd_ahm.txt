# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.

# Hill Auxiliary: Garo/Naga/Bodo recruits paid in salt and corvee exemption.
# Raised from the hill peoples themselves (Kachari is the Dimasa culture),
# never from the valley's own peasants, who serve as paiks. The game keeps
# only the stronger levy for a pop, so this narrower levy has to be stronger
# than the paik militia or no hill pop would ever raise it.
levy_nd_ahm_hill_auxiliary = {
	size = levy_generic_infantry_size
	unit = a_nd_ahm_hill_auxiliary

	allowed_pop_type = tribesmen
	allowed_pop_type = peasants

	allowed_culture = garo_culture
	allowed_culture = naga_culture
	allowed_culture = bodo_culture
	allowed_culture = dimasa_culture

	country_allow = {
		has_or_had_tag = AHM
	}
}
