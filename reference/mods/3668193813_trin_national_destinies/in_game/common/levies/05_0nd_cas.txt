# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Castile (CAS): age 1 levies

# Milicia Concejil: the host of a council and its villages. Every pop type
# that kept a house in a Castilian town or its land owed the muster, the
# craftsman and the shopkeeper beside the ploughman, which is why burghers
# are drawn with the rest. No culture gate: the fuero bound whoever settled
# under it, and the frontier towns were settled by all comers.
levy_nd_cas_milicia_concejil = {
	size = levy_generic_infantry_size
	unit = a_nd_cas_milicia_concejil

	allowed_pop_type = peasants
	allowed_pop_type = laborers
	allowed_pop_type = soldiers
	allowed_pop_type = burghers

	country_allow = {
		has_or_had_tag = CAS
	}
}
