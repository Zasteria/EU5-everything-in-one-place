# Levies are chosen by total combat power: each pop raises the strongest
# levy it is eligible for, wherever the levy is written (EU5 1.4,
# common/levies/readme.txt). The numeric prefix of this file name decides
# nothing and is kept only so paths stay stable. Check a levy with
# eu5-validate --checks levies; scripts/fix_levy_power.py repairs one.
#
# Mallorca (MLL): age 3 levies

# Forans: the villagers. Peasants alone, the shape vanilla uses for a rural
# specialist levy: the rising of 1450 was the countryside's, and the
# laborers of the City stood on the other side of the wall. Stronger than
# the late almogavars a Catalan peasant could still raise (1.126).
levy_nd_mll_forans = {
	size = levy_generic_infantry_size
	unit = a_nd_mll_forans

	allowed_pop_type = peasants

	country_allow = {
		has_or_had_tag = MLL
	}
}

# Agermanats: the guilds in arms. Burghers and laborers, the masters and
# the journeymen of the City's trades. No peasants: the villages joined the
# Germania, but they are the forans above.
levy_nd_mll_agermanats = {
	size = levy_generic_infantry_size
	unit = a_nd_mll_agermanats

	allowed_pop_type = burghers
	allowed_pop_type = laborers

	country_allow = {
		has_or_had_tag = MLL
	}
}
