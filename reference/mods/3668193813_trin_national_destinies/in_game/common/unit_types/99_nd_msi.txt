# Nakomsé Horsemen - Age 1 levy cavalry
# Elite mounted warriors of the Mossi ruling class
# Fast raiders with grassland bonus, representing the Nakomsé cavalry tradition
a_nd_msi_nakomse_horsemen = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Base template: max_strength=0.2, combat_power=2.6
	# The spear charge on open grass is the signature (see the combat block).
	morale_damage_done = 0.1
	movement_speed = 1
	flanking_ability = 0.2
	# South of the Sahel the tsetse fly kills horses. The herds were restocked
	# from Yatenga and the north, and wasted quickly on campaign.
	attrition_loss = 0.25
	max_strength = 0.205	# levy power, scripts/fix_levy_power.py

	combat = {
		grasslands = 0.15
		sparse = 0.1
	}

	country_potential = {
		has_or_had_tag = MSI
	}

	gfx_tags = { spear_tag light_tag }
}
