# Knights Hospitaller - Age 1 heavy infantry knight-monks (levy)
# Sworn brothers of the Order, the combat brothers of the Convent.
# Better morale and resilience than common infantry, fewer bodies.
# Base age 1 heavy infantry: max_strength=0.5, combat_power=1
a_nd_hos_knights = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_1_traditions_heavy_infantry
	buildable = no
	levy = yes

	# Fewer bodies, as the header says: the line read +0.1 (more men than a
	# common company) until the Tercio was given the same reading.
	max_strength = -0.1
	combat_power = 0.2
	morale_damage_done = 0.15
	strength_damage_taken = -0.1

	country_potential = {
		has_or_had_tag = HOS
	}

	gfx_tags = { medium_tag }
}

# Tercio of the Order - Age 4 heavy infantry levy
# Veterans of the Great Siege institutionalized as a permanent garrison
# infantry. Blends Spanish tercio drill with the Order's discipline.
# Base age 4 heavy infantry: max_strength=2.0, combat_power=1.5
a_nd_hos_tercio = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_4_reformation_heavy_infantry
	buildable = no
	levy = yes

	# The Order never had numbers: its companies are small, and what they
	# have instead is drill and a square with no open flank. The flank value
	# is vanilla's largest (a_wagenburg); the rest of the edge is steadiness.
	max_strength = -0.2
	combat_power = 0.1
	morale_damage_done = 0.15
	secure_flanks_defense = 0.025
	morale_damage_taken = -0.05

	country_potential = {
		has_or_had_tag = HOS
	}

	gfx_tags = { medium_tag }
}

# Order's Galley - Age 1 war galley levy
# The slim, fast galley of the Hospitaller fleet. Crewed by knight
# captains and Mediterranean sailors. The Order's signature weapon.
# Base age 1 galley needed first - vanilla provides n_age_1_traditions_galley
n_nd_hos_galley = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_1_traditions_galley
	buildable = no
	levy = yes

	# No vanilla ship changes morale damage: the knights' boarding parties
	# show as damage done and the slim hull as speed.
	cannons = 1
	strength_damage_done = 0.15
	movement_speed = 0.25

	country_potential = {
		has_or_had_tag = HOS
	}

	gfx_tags = { galley_tag }
}

# Sacra Capitana - Age 4 buildable signature galley

n_nd_hos_capitana = {

	country_potential = {
		has_or_had_tag = HOS
	}

# Maltese Frigate - Age 5 buildable light heavy ship
# The late-period Hospitaller sailing warship. Built at the Valletta
# arsenal as galleys gave way to sail. Faster and lighter than a ship
# of the line; designed to hunt corsairs across the Mediterranean.
# Base age 5 heavy ship: speed=8, hull=50, cannons=100, crew=0.750
n_nd_hos_frigate = {
	is_special = yes
	category = navy_heavy_ship
	copy_from = n_age_5_absolutism_heavy_ship
	upgrades_to = n_threedecker
	buildable = yes

	# Speed and crew saving are vanilla's largest for a ship (the early
	# Iberian galleon, the Italian galleass). With the crew saving that
	# small, the lighter hull and battery cost less than they did, and the
	# corsair hunter's edge is its gunnery.
	hull_size = -10
	cannons = -20
	crew_size = -0.03
	movement_speed = 0.5
	strength_damage_done = 0.15
	blockade_capacity = -0.25

	country_potential = {
		has_or_had_tag = HOS
	}

	gfx_tags = { heavy_tag }
}
