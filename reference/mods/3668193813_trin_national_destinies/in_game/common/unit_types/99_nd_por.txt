# ============================================================================
# Portugal (POR): custom unit types
# National Destinies mod
# ============================================================================
#
# Vanilla already gives Portugal its ships and its regulars: the Iberian
# caravels of ages 1 to 3, the Iberian galleons of ages 3 and 4, the galiot,
# the tercio and the cacadores (advances/region_iberia.txt), and the marines
# of the Terco da Armada as a naval law. Nothing here repeats those.
#
# What vanilla leaves out is the kingdom's own muster and its two workhorses
# overseas: the crossbowmen every concelho owed the king, the three lines of
# the territorial militia that replaced them (ordenancas, auxiliaries,
# militia regiments), the light horse of the Moroccan fortresses, and the
# nau, the great cargo carrack of the India run.
#
# Every stat under copy_from is a DELTA added to the template and then to the
# unit category's own value (common/levies/readme.txt: "EVERY STAT LINE IS
# ADDITIVE, NEVER AN ASSIGNMENT"). Template totals are quoted for the reader
# only; none of them is written in a block.
#
# The militia line is one levy per age (4, 5, 6) because the engine picks the
# strongest levy again every age and upgrades_to does nothing on a levy. The
# crossbowmen of age 2 have no successor on purpose: Manuel I abolished them
# in 1498, and in age 3 their pops fall back to vanilla's matchlock levy
# until the ordenancas are raised in age 4.

# ----------------------------------------------------------------------------
# Age 2: Besteiros do Conto (levy light infantry)
# ----------------------------------------------------------------------------
# "Crossbowmen of the tally": every concelho owed the king a fixed count of
# them, picked from its craftsmen, enrolled by an anadel and excused certain
# dues in return. The tally keeps the companies full and the feast-day butts
# keep them quick to the mark; they shoot hard and reload slowly
# (combat_speed), which is what a windlass crossbow does.
# Template a_age_2_renaissance_light_infantry: max_strength 1, combat_power 1.
a_nd_por_besteiros_do_conto = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_2_renaissance_light_infantry
	buildable = no
	levy = yes

	max_strength = 0.05
	initiative = 1
	strength_damage_done = 0.1
	combat_speed = -0.5

	country_potential = {
		has_or_had_tag = POR
	}

	gfx_tags = { crossbow_tag light_tag }
}

# ----------------------------------------------------------------------------
# Age 3: Fronteiros (buildable light cavalry)
# ----------------------------------------------------------------------------
# The horse of the Moroccan fortresses: Ceuta, Alcacer Ceguer, Arzila,
# Tangier, Safim, Mazagao. Their trade was the entrada, a ride out of the
# gate before dawn to lift cattle and captives and be back inside by dark,
# riding short stirrups in the Moorish fashion. Quick and hard to catch;
# light mail and a leather shield do not stop much (strength_damage_taken).
# Template a_age_3_discovery_light_cavalry: max_strength 0.6, combat_power 2.6.
a_nd_por_fronteiros = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	upgrades_to = a_pistoleers
	buildable = yes

	initiative = 2
	movement_speed = 0.25
	morale_damage_done = 0.1
	strength_damage_taken = 0.05

	country_potential = {
		has_or_had_tag = POR
	}

	gfx_tags = { spear_tag light_tag }
}

# ----------------------------------------------------------------------------
# Age 3: Nau da Carreira (buildable heavy ship)
# ----------------------------------------------------------------------------
# The carrack of the Carreira da India: three and four decks of pepper,
# cinnamon and porcelain, castles fore and aft, six heavy guns below and a
# row of falconets above. Built to survive the Cape and carry a fortune, not
# to chase or to fight in line: she has a deeper hull than any warship of her
# day, fewer guns for her size, and she is slow.
# Vanilla's Early Iberian Galleon (same age) is the fighting escort; this is
# the ship it escorts.
# Template n_age_3_discovery_heavy_ship: speed 6.5, hull 25, cannons 50,
# crew 0.3, blockade 0.75.
n_nd_por_nau_da_carreira = {
	is_special = yes
	category = navy_heavy_ship
	copy_from = n_age_3_discovery_heavy_ship
	upgrades_to = n_galleon
	buildable = yes

	hull_size = 8
	cannons = -10
	crew_size = 0.05
	movement_speed = -0.5

	country_potential = {
		has_or_had_tag = POR
	}
}
