# Dromon: Age 1 war galley, the backbone of Byzantine naval power
# Equipped with a siphon for Greek fire projection. Slower than standard
# galleys but sturdier and far more lethal in close engagements.
# Base age 1 galley: speed=4, hull=6, cannons=8, crew=0.03, blockade=0.1
n_nd_byz_dromon = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_1_traditions_galley
	buildable = no
	levy = yes

	cannons = 2
	strength_damage_done = 0.15

	country_potential = {
		has_or_had_tag = BYZ
	}

	gfx_tags = { galley_tag }
}

# Pronoiarioi: Age 2 cavalry of the pronoia holders
# A pronoia was the revenue of a district assigned to a soldier for his
# service. Its holders rode in mail and lamellar with lance and bow: less
# plate than the Western knight they stand in for (vanilla's plated
# knights take a quarter less damage again), steadier under orders, and
# deadlier before the lines meet.
# Base age 2 cavalry: max_strength=0.4, combat_power=2.6
a_nd_byz_pronoiarioi = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_2_renaissance_heavy_cavalry
	buildable = no
	levy = yes

	strength_damage_taken = -0.25
	morale_damage_taken = -0.25
	# The bow is the edge: they loose before the lines meet. Strength damage
	# is at vanilla's largest for heavy cavalry (0.1, the banner cavalry; it
	# read 0.2), and the difference went into initiative (4, vanilla's
	# largest, the sowar) and into the shock of the charge.
	initiative = 4
	strength_damage_done = 0.1
	morale_damage_done = 0.2

	country_potential = {
		has_or_had_tag = BYZ
	}

	gfx_tags = { heavy_tag }
}

# Stradioti: Age 3 light cavalry raiders
# Albanian and Greek light horsemen who served as scouts and raiders.
# Fast, lethal on the flanks, excellent at looting and harassment.
# Speed is the signature; the flanking edge matches vanilla's akinji, the
# same Balkan light horse in Ottoman service. Unarmoured and paid in
# plunder, they scatter when pressed (they looted the French baggage at
# Fornovo in 1495 instead of closing), so they lose morale faster.
# Base age 3 cavalry: max_strength=0.6, combat_power=2.6
a_nd_byz_stradioti = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	buildable = no
	levy = yes

	movement_speed = 1
	flanking_ability = 0.2
	initiative = 5
	morale_damage_taken = 0.1
	max_strength = 0.65	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = BYZ
	}

	gfx_tags = { light_tag }
}

n_nd_byz_fire_dromon = {

	upgrades_to = n_galleass

	country_potential = {

# Pyrpolikon: Age 6 fireship

# burned the enemy's: Kanaris took a brig packed with pitch and powder

# Base age 6 light ship: speed=14, hull=20, cannons=25, crew=0.125

	hull_size = -5
	cannons = -10

	country_potential = {
		has_or_had_tag = BYZ
	}

	gfx_tags = { light_tag }
}
