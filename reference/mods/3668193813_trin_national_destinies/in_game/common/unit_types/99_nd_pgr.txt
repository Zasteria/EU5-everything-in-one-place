# Base age 1 infantry: max_strength=0.5, combat_power=1

	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_1_traditions_heavy_infantry
	upgrades_to = a_men_at_arms
	buildable = yes

	# Silat practitioners are agile, deal more morale damage but are lightly armored.

	country_potential = {
		has_or_had_tag = PGR
	}

# Age 4, Silat Masters: advanced martial artists, feared in close combat
# Base age 4 infantry: max_strength=2.0, combat_power=1.5
a_nd_pgr_silat_masters = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	upgrades_to = a_hunters
	buildable = yes

	# Close-combat fighters. Silek is footwork: the fighter steps off the line
	# and strikes first from the side. Flanking stands at the largest vanilla
	# gives a light infantry unit (0.25; it was 0.75), and the edge it lost
	# is carried by initiative (the old line added 0.15 to a base of 5, which
	# did nothing).
	# They fight unarmoured and in loose order, so they take more losses.
	morale_damage_done = 0.15
	initiative = 3
	flanking_ability = 0.25
	strength_damage_taken = 0.05

	country_potential = {
		has_or_had_tag = PGR
	}

	gfx_tags = { medium_tag }
}
