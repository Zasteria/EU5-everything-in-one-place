##############################################################################
# Illyria (ILY): Custom Units
#
# FOUR LAND UNITS AND NO SHIPS, deliberately. Illyria is the inland
# consolidation of the Catholic Slavic world, and its coast is a terminus
# for land traffic, not a fleet base. The Adriatic navy belongs to Dalmatia
# (DAL), which already carries the argosy and the uskok bracera; duplicating
# them here would make Illyria a bigger Dalmatia instead of a different
# country. What Illyria has instead is a permanent land frontier, and all
# four units come off it.
#
# The uskoks of Senj are likewise DAL's, so the frontier infantry here is the
# haramije, the paid foot of the Croatian and Slavonian forts, which is the
# historically correct and mechanically distinct institution.
##############################################################################

# BANDERIJ: magnate heavy cavalry (levy), Age 1
# The banderium was a great house's own banner: Subic, Nelipic, Kurjakovic,
# Frankopan. Splendidly mounted, few in number, and answering to its lord
# before its king. Harder-hitting than common feudal horse and slower to
# come when called.
a_nd_ily_banderij = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_1_traditions_heavy_cavalry
	levy = yes
	buildable = no

	# Base age 1 heavy cavalry: max_strength = 0.2, combat_power = 2.6.
	# Terrain block deliberately NOT overridden: the template's jungle,
	# wetland, mountain and river penalties stay, because a banderium is
	# still heavy horse and the Dinaric karst is still bad ground for it.
	combat_power = 0.34	# levy power, scripts/fix_levy_power.py
	strength_damage_done = 0.1
	# The tradeoff: a magnate's retinue travels with a magnate's baggage.
	supply_weight = 1.15
	morale_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	strength_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = ILY
	}

	gfx_tags = { heavy_tag }
	age = "age_1_traditions"
}

# HARAMIJA: frontier light infantry (levy), Age 4
# Raised in 1539 for the Croatian and Slavonian forts, specifically to chase
# down the Ottoman martolos raiders who crossed to take captives. Pursuit
# troops: they lived in the woods and river country of the Sava and Una and
# they were paid to catch people, not to hold a line.
a_nd_ily_haramija = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	levy = yes
	buildable = no

	# Base age 4 light infantry: max_strength = 2.0, combat_power = 1.5.
	max_strength = -0.2
	movement_speed = 1.15
	flanking_ability = 0.25	# levy power, scripts/fix_levy_power.py
	combat_power = 0.24	# levy power, scripts/fix_levy_power.py
	# Woods and broken riverine country, which is where the raiding roads
	# actually ran. The mountain war is the uskoks' and DAL keeps it.
	combat = {
		forest = 0.15
		hills = 0.1
		river = 0.1
	}

	country_potential = {
		has_or_had_tag = ILY
	}

	gfx_tags = { light_tag }
	age = "age_4_reformation"
}

# GRANICAR: Slavonian frontier foot (levy), Age 5
# After Karlowitz the recovered Slavonian bank of the Sava was organised as a
# frontier of its own in 1702: households settled on crown land, free of
# rent, on the single condition that their men turned out armed. Marksmen
# who knew their own woods, numerous because every household owed a man,
# and not trained to press an attack in the open.
a_nd_ily_granicar = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_5_absolutism_light_infantry
	levy = yes
	buildable = no

	# Base age 5 light infantry: max_strength = 2.5, combat_power = 2.25.
	strength_damage_done = 0.1
	morale_damage_taken = -0.1
	# The tradeoff: skirmishers, not a storming column.
	morale_damage_done = -0.05
	combat_power = 0.2

	country_potential = {
		has_or_had_tag = ILY
	}

	gfx_tags = { light_tag }
	age = "age_5_absolutism"
}

# CRABAT: Croat light horse (buildable), Age 5
# At the height of the German war some twenty thousand Croats rode in
# Imperial service, and the French borrowed the scarf they wore at the
# throat and called it after them. Unmatched at scouting, pursuit and
# burning a countryside; not asked to stand in a line of battle, and it
# shows when they have to.
a_nd_ily_crabat = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_5_absolutism_light_cavalry
	upgrades_to = a_light_dragoons
	buildable = yes

	# Base age 5 light cavalry: max_strength = 1.0, combat_power = 3.25.
	movement_speed = 1.2
	# Flanking at vanilla's ceiling for cavalry, and first into contact: the
	# two things scouts and pursuers are for.
	flanking_ability = 0.3
	initiative = 1
	morale_damage_done = 0.1
	# The price of riding light: they are shot to pieces in a stand-up
	# fight.
	strength_damage_taken = 0.1
	# The template's own penalties are restated so the override does not
	# silently hand a raider free mountain and marsh performance; the
	# forest bonus is the German campaign the Croats actually fought.
	combat = {
		jungle = -0.10
		wetlands = -0.10
		mountains = -0.10
		forest = 0.10
	}

	country_potential = {
		has_or_had_tag = ILY
	}

	gfx_tags = { medium_tag }
	age = "age_5_absolutism"
}
