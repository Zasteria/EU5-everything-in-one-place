# Kingdom of Benin: Custom Units

# Eben Warriors: Edo sword infantry, organized guild warriors
# Age 2 buildable unit. Benin's warriors were noted for discipline
# and use of the eben (short sword). Slower but more resilient.
a_nd_edo_eben_warriors = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_2_renaissance_light_infantry
	upgrades_to = a_early_arquebusiers
	buildable = yes

	# Base: max_strength=1.0, combat_power=1
	morale_damage_taken = -0.1
	strength_damage_taken = -0.05
	combat_speed = -0.05

	country_potential = {
		has_or_had_tag = EDO
	}

	gfx_tags = { medium_tag }
}

# Edo War Canoe: the canoe fleet that carried the Oba's armies along the
# creeks and lagoons to Eko (Lagos). Age 3 galley, buildable.
# Base age 3 galley: speed 4, hull 8, cannons 12. A dugout is fast in shoal
# water and carries fighting men, but it is no gun platform.
n_nd_edo_war_canoe = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_3_discovery_galley
	upgrades_to = n_galleass
	buildable = yes

	# Transport at vanilla's largest for a galley (n_long_fada_2, 0.03); the
	# hull gives up one point instead of two.
	movement_speed = 2
	transport_capacity = 0.03
	hull_size = -1
	cannons = -5

	country_potential = {
		has_or_had_tag = EDO
	}

	gfx_tags = { light_tag }
}
