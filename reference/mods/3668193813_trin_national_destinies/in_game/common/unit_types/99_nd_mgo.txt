# ===== LEVY UNITS (raised through levy system) =====

# Khorchin light cavalry: the horde's own riders, age 1
a_nd_mgo_khorchin_rider = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Base: max_strength=0.2, combat_power=2.6
	# A horde unit (ruling of 2026-10-02). Every country that can form
	# Mongolia is a steppe horde, and under that government each pop raises
	# vanilla's horde levy (a_steppe_horde, total combat power 1.752) unless
	# this one is stronger. So the unit is built as vanilla builds that one:
	# three times the riders of an ordinary light horse regiment, with its
	# speed and its initiative. What is the Khorchin's own is the raid:
	# riders who come round a flank (0.2, the best of vanilla's age 1 light
	# horse, the Iberian jinetes) and break an enemy's nerve.
	# Weakness: herdsmen on a raid break off when the enemy stands, they do
	# not hold a line.
	# Total combat power 1.882, 7 percent over the horde levy. Set by hand:
	# the levy script does not place a levy against the horde levy.
	max_strength = 0.4
	movement_speed = 1
	initiative = 2
	flanking_ability = 0.2
	morale_damage_done = 0.15
	morale_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = MGO
	}

	gfx_tags = { steppe_tag }
}

# Mangudai horse archer: the signature Mongol mounted archer, age 1
a_nd_mgo_mangudai = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	upgrades_to = a_cavalrymen
	buildable = yes

	# Base: max_strength=0.2, combat_power=2.6
	# Signature: the feigned retreat, wheeling to shoot into a pursuer's
	# flank. Flanking stands at vanilla's ceiling (0.3, ruling of
	# 2026-10-02; it was 0.75). What the unit lost there it gets back as a
	# horde unit does, in riders and in loosing first: every country that
	# can form Mongolia is a steppe horde and already builds vanilla's
	# steppe horse archers (a_steppe_horse_archers: three times the riders,
	# quicker to engage, a fifth more morale damage, faster on the march;
	# total combat power 2.147), beside which a unit with a third of the
	# riders was never worth building. The mangudai stand at 1.926: fewer
	# kills by fright, more by the flank.
	# Weakness: the vanguard that baits the enemy rides unarmoured and takes
	# the losses when it is caught in the melee.
	max_strength = 0.4
	initiative = 3
	combat_speed = 0.1
	strength_damage_done = 0.1
	flanking_ability = 0.3
	strength_damage_taken = 0.1

	combat = {
		desert = 0.05
	}

	country_potential = {
		has_or_had_tag = MGO
	}

	gfx_tags = { steppe_tag }
}
