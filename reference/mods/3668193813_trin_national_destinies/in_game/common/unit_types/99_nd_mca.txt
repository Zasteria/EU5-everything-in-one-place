# Highland Slingers: ranged skirmishers using slings and blowpipes
# Age 3 levy light infantry
# Light infantry, not an auxiliary: from 1.4 a pop raises the strongest levy
# it can reach (common/levies/readme.txt) and a supply unit never is.
a_nd_mca_highland_slingers = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_3_discovery_light_infantry
	buildable = no
	levy = yes

	# Deltas on the age 3 light infantry template (1.5 men, combat power 1).
	# Flanking stays at vanilla's largest for light infantry (0.25,
	# a_caterans). The rest of the edge is the sling itself: stones reach
	# the enemy before the lines meet (initiative) and unsettle him (morale
	# damage). The weakness is the melee: no cotton armour and no shield.
	# Levy power 1.7397, 1.24 times vanilla's matchlock levy of the age.
	max_strength = 0.05
	flanking_ability = 0.25
	initiative = 3
	morale_damage_done = 0.1
	strength_damage_taken = 0.05

	combat = {
		mountains = 0.1
	}

	country_potential = {
		has_or_had_tag = MCA
	}

	gfx_tags = { medium_tag }
}
