# Pancerni: medium armored cavalry, backbone of Commonwealth cavalry
# Age 3 Discovery: cheaper than hussars, versatile medium cavalry
a_nd_plc_pancerni = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	upgrades_to = a_pistoleers
	buildable = yes

	# Deltas on a_age_3_discovery_light_cavalry. The name means "mailed":
	# the mail shirt and misiurka helmet are the signature (less strength
	# damage taken, the number vanilla gives a_hakkapelitta), morale damage
	# is at vanilla's best for this template (a_light_camelry_3). Mail on a
	# cossack-style horse makes them the slowest of the light cavalry
	# (-0.25 is what vanilla takes from a_byzantine_cataphracts).
	strength_damage_taken = -0.1
	morale_damage_done = 0.05
	movement_speed = -0.25

	country_potential = {
		has_or_had_tag = PLC
	}

	gfx_tags = { medium_tag }
}

# Lisowczycy: irregular raider cavalry, fast and devastating on open ground

	copy_from = a_age_4_reformation_light_cavalry
	upgrades_to = a_hussars
	buildable = yes

	# Deltas on a_age_4_reformation_light_cavalry. They rode without a

	flanking_ability = 0.3
	strength_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = PLC
	}

	gfx_tags = { medium_tag }
}
