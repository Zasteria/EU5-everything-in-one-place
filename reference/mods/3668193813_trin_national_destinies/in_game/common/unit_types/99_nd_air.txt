### Age 1: Tuareg Mehari Raiders, levy camel cavalry ###
a_nd_air_mehari_raiders = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Base cavalry: max_strength=0.2, combat_power=2.6
	# Desert raiders: fast, devastating in open terrain, weaker in rough
	movement_speed = 1
	morale_damage_done = 0.1
	strength_damage_taken = -0.05
	# Riders: the levy draws tribesmen, who raise vanilla's tribal camelry
	# (a_tribal_camelry_levy_1, total combat power 0.678) wherever camels
	# are in the market. This many riders put the unit 1.19 times over it.
	# The levy script measures only the unconditional tribesmen levy
	# (0.469), so this number is set by hand.
	max_strength = 0.05

	combat = {
		desert = 0.15
		mountains = -0.15
	}

	country_potential = {
		has_variable = nd_air_formed
	}

	gfx_tags = { medium_tag }
}

### Age 3: Kel Ayr Lancers, buildable camel cavalry ###
a_nd_air_kel_ayr_lancers = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	upgrades_to = a_pistoleers
	buildable = yes

	# Base cavalry: max_strength=0.6, combat_power=2.6
	# Camel lancers: hurling spears from camelback before charging.
	# Signature: desert mobility, the same +1 vanilla gives its own Tuareg
	# veiled cavalry, and the javelin volley that opens the fight
	# (initiative, that unit's own +3). Flanking and shock sit at the best
	# of vanilla's age 3 light cavalry (akinji 0.2, camelry 0.05).
	# Weakness: Tuareg riders wore robes and carried a hide shield, no body
	# armour, so they bleed when a formed line holds against them.
	morale_damage_done = 0.05
	flanking_ability = 0.2
	movement_speed = 1
	initiative = 3
	strength_damage_taken = 0.1

	combat = {
		desert = 0.15
		mountains = -0.1
	}

	country_potential = {
		has_variable = nd_air_formed
	}

	gfx_tags = { medium_tag }
}
