# Nusantara Custom Units

# Kris Warriors: elite Javanese infantry wielding the sacred kris dagger
# Emphasize shock and morale impact in close combat, representing the spiritual
# and martial tradition of silat and the kris as a weapon of prestige
a_nd_nus_kris_warriors = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_2_renaissance_heavy_infantry
	upgrades_to = a_halberdiers
	buildable = yes

	# Base: max_strength=1.0, combat_power=1
	# The sudden rush to close quarters is the signature.
	morale_damage_done = 0.15
	strength_damage_done = 0.1
	initiative = 2
	# A dagger fighter closes bare chested, with no armour and no shield wall.
	strength_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = NUS
	}

	gfx_tags = { medium_tag }
}
