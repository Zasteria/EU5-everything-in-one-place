# Age 3: Siege Topchi (advanced siege artillery, mine warfare)
a_nd_bah_siege_topchi = {
	is_special = yes
	category = army_artillery
	copy_from = a_age_3_discovery_artillery
	upgrades_to = a_chambered_cannon
	buildable = yes

	# Base template: max_strength=0.6, combat_power=4, bombard_efficiency=0.15, artillery_barrage=4
	# Specialized in siege and mining operations. No stat stands beyond the
	# largest vanilla gives an artillery unit (combat power +0.25, bombard
	# +0.1, barrage +1); the edge is in breaching, which is what the unit is.
	# A siege train with its miners moves slower than field guns.
	combat_power = 0.2
	bombard_efficiency = 0.1
	artillery_barrage = 0.8
	movement_speed = -0.25

	country_potential = {
		has_or_had_tag = BAH
	}

	gfx_tags = { medium_tag }
}

# Age 5: Reformed Sipahi (modernized heavy cavalry)
# Heavy cavalry in name, text and role, so it stands in the heavy category
# (it was filed as light cavalry until 2026-10-01).
a_nd_bah_reformed_sipahi = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_5_absolutism_heavy_cavalry
	upgrades_to = a_cuirassiers
	buildable = yes

	# Base template: max_strength=1.0, combat_power=3.25
	# The armoured charge is the signature; the pistols add a little fire.
	morale_damage_done = 0.15
	strength_damage_done = 0.05
	# The Deccan bred no war horses. Every mount came by sea from Arabia and
	# Persia through Dabhol, Chaul and Goa, so the corps was always small.
	max_strength = -0.1

	country_potential = {
		has_or_had_tag = BAH
	}

	gfx_tags = { medium_tag }
}
