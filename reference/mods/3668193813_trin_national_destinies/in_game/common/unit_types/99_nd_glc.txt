# ============================================================================
# Galicia (GLC): custom unit types
# National Destinies mod
# ============================================================================
#
# Vanilla already gives every Iberian the caravels, the galleons and the
# tercio (advances/region_iberia.txt). Nothing here repeats those. This file
# holds what was the kingdom's own: the host of the archbishop's lordship,
# the sworn brotherhood of 1467, the coast militia that held A Coruna, the
# clinker coaster that went privateering, the two-deckers of Esteiro, and the
# peasants and students of 1808 and 1809.
#
# Every stat under copy_from is a DELTA added to the template and then to the
# unit category's own value (common/levies/readme.txt: "EVERY STAT LINE IS
# ADDITIVE, NEVER AN ASSIGNMENT"). Template totals are quoted for the reader
# only; none of them is written in a block.
#
# Levy lines, each ending where history ended it. Powers are the levy
# readme's total combat power; the figure to beat is vanilla's open levy for
# the same pops in the same age (VANILLA_REFERENCE.md):
#   age 1 Hoste da Mitra   0.587 against 0.538 (soldiers): 1.09
#   age 3 Irmandinos       1.709 against 1.408: 1.21
#   age 4 Milicia do Reino 3.349 against 2.816: 1.19
#   age 6 Paisanaxe       10.114 against 8.447: 1.20
# Nothing in age 2 (the brotherhood of 1431 was crushed and left no force)
# and nothing in age 5 (the Bourbon provincial militia of 1734 was Castile's
# institution, and the mod's Castile has it): in those ages the pops raise
# vanilla's levy.

# ----------------------------------------------------------------------------
# Age 1: Hoste da Mitra (levy heavy infantry)
# ----------------------------------------------------------------------------
# The tenants of the Terra de Santiago called out under the Pertigueiro
# Maior: shield, spear and iron cap, sworn on the Apostle's altar and quick
# to form. They are a lord's tenants and not his household: pressed hard,
# they go home (morale_damage_taken).
# Template a_age_1_traditions_heavy_infantry: max_strength 0.5, combat_power 1.
a_nd_glc_hoste_da_mitra = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_1_traditions_heavy_infantry
	buildable = no
	levy = yes

	initiative = 2
	strength_damage_done = 0.15
	morale_damage_taken = 0.05

	country_potential = {
		has_or_had_tag = GLC
	}

	gfx_tags = { spear_tag medium_tag }
}
