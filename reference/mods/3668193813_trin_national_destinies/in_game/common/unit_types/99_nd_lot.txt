# Scara cavalry: Carolingian elite heavy cavalry tradition, Age 1 levy
# Base template: max_strength=0.2, combat_power=2.6
# Heavy cavalry since 2026-10-01: the scara were mailed household horsemen,
# and the unit had been left in the light category (with a flanking bonus
# no vanilla unit has) by the split of the old cavalry category.
a_nd_lot_scara_cavalry = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_1_traditions_heavy_cavalry
	buildable = no
	levy = yes

	# Signature: the shock of the charge. Weakness: a household corps of
	# mailed riders travels with remounts and grooms, so it weighs more on
	# the land it crosses than a feudal levy does.
	morale_damage_done = 0.1
	supply_weight = 0.25
	morale_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	strength_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	combat_power = 0.05	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = LOT
	}

	gfx_tags = { medium_tag }
}
