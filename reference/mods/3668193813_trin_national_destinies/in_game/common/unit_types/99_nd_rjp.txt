# Camel Risala: Age 4 camel cavalry for desert operations
# Base age 4 cavalry: max_strength=0.8, combat_power=2.85
a_nd_rjp_camel_risala = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_4_reformation_light_cavalry
	upgrades_to = a_hussars
	buildable = yes

	# Speed, morale shock and initiative are vanilla's own camelry values
	# (a_light_camelry_4, 3_camel_units.txt:172: 0.2, 0.05 and 5). The
	# initiative was missing, which left the unit below its own template in
	# total combat power (2.443 against 2.452) and well below the vanilla
	# camelry the same country can build.
	movement_speed = 0.2
	morale_damage_done = 0.05
	initiative = 5
	# Shutarnal: the swivel gun fired from the camel's saddle.
	strength_damage_done = 0.05
	supply_weight = -0.25
	# A camel wheels slowly: poor at turning a flank.
	flanking_ability = -0.2
	combat = {
		desert = 0.15
	}

	country_potential = {
		has_or_had_tag = RJP
	}

	gfx_tags = { medium_tag }
}
