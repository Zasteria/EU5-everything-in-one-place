# Age 1: Steppe horse archer (levy cavalry, fast raider with ranged power)
a_nd_mge_steppe_horse_archer = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Base: max_strength=0.2, combat_power=2.6
	# A horde unit (owner's ruling of 2026-10-02): three times the
	# template's riders, as vanilla's own horde levy unit has (a_steppe_horde
	# writes max_strength = 0.4 on the same template), with its first
	# strike. The levy then stands a tenth above levy_steppe_horde (1.752),
	# which every pop of a steppe horde can raise, and is raised where the
	# empire is formed by a horde.
	max_strength = 0.4
	movement_speed = 1
	initiative = 2
	morale_damage_done = 0.1

	country_potential = {
		has_or_had_tag = MGE
	}

	gfx_tags = { medium_tag }
}
