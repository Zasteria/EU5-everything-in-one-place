# Westphalian Freischöffen Guard: urban militia drawn from the free judges (Freischöffen)

	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_2_renaissance_heavy_infantry
	buildable = no
	levy = yes

	# Base template: max_strength=1.0, combat_power=1
	# Signature: the dread of the Vehmic court (morale damage). Initiative
	# sits at the best of vanilla's age 2 heavy infantry (gallowglass, 2).

	combat_power = 0.17	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = WES
	}

	gfx_tags = { medium_tag }
