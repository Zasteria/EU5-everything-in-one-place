# Men of Poljica: infantry (levy), Age 3
# The militia of the twelve villages of Poljica, the peasant commune in the
# hills behind Split (statute of 1440; Ottoman armies turned back in 1530 and
# 1686). The key keeps its first name: until 2026-10-02 this was a Morlach
# levy, which vanilla 1.4 tells itself in dal_morlach_frontier_levies.
a_nd_dal_morlach = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_3_discovery_heavy_infantry
	levy = yes
	buildable = no
	# A small commune (fewer men) that holds its own hills: hard to break,
	# quick to close a pass behind an enemy. Placed by hand at about 1.22
	# times vanilla's matchlock levy of age 3.
	max_strength = -0.1
	movement_speed = 0.2
	flanking_ability = 0.15
	morale_damage_taken = -0.15
	combat = {
		mountains = 0.15
		hills = 0.1
	}
	age = "age_3_discovery"
}

# Argosy: heavy merchant ship (buildable), Age 3
# The legendary Ragusan argosy: large, heavily built cargo vessels that doubled as warships
n_nd_dal_argosy = {
	is_special = yes
	category = navy_heavy_ship
	copy_from = n_age_3_discovery_heavy_ship
	upgrades_to = n_galleon
	use_ship_names = yes
	buildable = yes
	# Base: movement_speed=6.5, hull_size=25, cannons=50, crew_size=0.300, blockade_capacity=0.75
	# The crew saving stops at the most vanilla gives a ship (-0.03, owner's
	# ruling 2026-10-02); the rest of the edge is the hull.
	movement_speed = -0.5
	hull_size = 4
	cannons = -5
	crew_size = -0.03
	blockade_capacity = 0.05
	maritime_presence = ship_medium_maritime
	age = "age_3_discovery"
}
