# Pfumo Spearmen: Elite Shona spearmen forming the core of Mutapa royal armies
# Heavy infantry specializing in close-order defense with large shields
# Levy unit: raised through levy system
a_nd_mtp_pfumo_spearmen = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Base: max_strength=0.5, combat_power=1
	morale_damage_taken = -0.1
	strength_damage_taken = -0.1

	country_potential = {
		has_or_had_tag = MTP
	}

	gfx_tags = { medium_tag }
}

# Nhare Scouts: Fast-moving skirmishers and ambush troops
# Named for the eland (nhare), swift and elusive
# Buildable: recruited directly
a_nd_mtp_nhare_scouts = {
	is_special = yes
	category = army_auxiliary
	copy_from = a_age_2_renaissance_auxiliary
	upgrades_to = a_supply_convoy
	buildable = yes

	# Deltas on the age 2 auxiliary template (0.1 men, combat power 0.25),
	# each at or under vanilla's largest for an auxiliary: speed +1
	# (a_a_urughs), initiative +2 and losses -0.1 (a_jonow_auxiliary),
	# flanking +0.3 (the army's largest, a_tuareg_veiled_cavalry). Scouts
	# are hard to catch and do not hold a line: they break off sooner.
	movement_speed = 1
	initiative = 2
	flanking_ability = 0.3
	strength_damage_taken = -0.1
	morale_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = MTP
	}

	gfx_tags = { medium_tag }
}

# Muzinda Garrison: Provincial fortress troops trained in defensive warfare
# Each muzinda (provincial seat) maintained a permanent garrison
# Buildable: recruited directly
a_nd_mtp_muzinda_garrison = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_3_discovery_light_infantry
	upgrades_to = a_arquebusiers
	buildable = yes

	# Deltas on the age 3 light infantry template (1.5 men, combat power 1).
	# Morale stays at vanilla's largest for light infantry (-0.1,
	# a_lour_lances); the rest of the edge is the stone wall they fight
	# behind (losses, where vanilla goes to -0.2).
	morale_damage_taken = -0.1
	strength_damage_taken = -0.15
	combat_speed = -0.05

	country_potential = {
		has_or_had_tag = MTP
	}

	gfx_tags = { medium_tag }
}
