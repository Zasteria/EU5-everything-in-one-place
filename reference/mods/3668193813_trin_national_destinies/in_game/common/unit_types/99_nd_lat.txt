# Latin Dromon: Age 3 galley, fire-ship tradition inherited from Byzantium
# Base: n_age_3_discovery_galley (speed=4, hull=8, cannons=12, crew=0.075, blockade=0.3)
n_nd_lat_dromon = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_3_discovery_galley
	upgrades_to = n_galleass
	buildable = yes
	use_ship_names = yes

	# Advantage: a smaller crew and the fire siphon, same hull and guns. No
	# vanilla ship changes morale damage (the line that stood here); the fire
	# galleys vanilla gives Byzantium do strength damage, up to 0.2.
	strength_damage_done = 0.05
	crew_size = -0.01

	country_potential = {
		has_or_had_tag = LAT
	}

	gfx_tags = { galley_tag }
}
