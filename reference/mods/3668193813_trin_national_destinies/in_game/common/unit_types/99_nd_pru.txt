# Feldjäger: light rifleman scouts, the Reitende Feldjäger tradition
# Base age 6 infantry: max_strength=3.0, combat_power=3
a_nd_pru_feldjaeger = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_6_revolutions_light_infantry
	buildable = yes

	# Skirmisher: fast, flanking, lower brute strength. Marksmanship is the
	# signature: the corps carried rifles where the line carried smoothbore
	# muskets, so it fires first (initiative) and its fire kills more
	# (strength damage). Flanking at vanilla's best for the template.
	# The corps was recruited from foresters and never numbered more than a
	# few battalions, so a regiment is a sixth smaller than the line's.
	flanking_ability = 0.25
	movement_speed = 0.2
	initiative = 5
	strength_damage_done = 0.05
	max_strength = -0.5

	country_potential = {
		has_or_had_tag = PRU
	}

	gfx_tags = { medium_tag }
}
