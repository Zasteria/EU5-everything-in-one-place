# Punic War Galley: faster galley with ramming capability, age 2 buildable
# Base: hull=7, cannons=9, crew=0.06, speed=4, blockade=0.2
n_nd_cth_war_galley = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_2_renaissance_galley
	upgrades_to = n_war_galley
	buildable = yes
	use_ship_names = yes

	# The ram holes a hull: strength damage, the stat vanilla's own galleys
	# carry (up to 0.2). No vanilla ship changes morale damage, which stood here.
	cannons = 1
	movement_speed = 1
	strength_damage_done = 0.05

	country_potential = {
		has_or_had_tag = CTH
	}

	gfx_tags = { galley_tag }
}
