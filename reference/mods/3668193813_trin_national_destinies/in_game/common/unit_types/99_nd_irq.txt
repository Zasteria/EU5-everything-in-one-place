# === Iraq: custom unit types ===
#
# Mesopotamia's military identity in this period is not a battle line. It is
# terrain and clientage: marsh that swallows an army, desert fringe that
# outruns one, city quarters that arm themselves when the state cannot, and a
# bought household that ends up owning the state. Each unit below is a
# specialist with a real weakness, not an upgrade of the line troop.

# ============================================================
# Madan marshmen (levy, age 1). Reed-skiff skirmishers of the Bataih. They
# fight from the mashhuf and along causeways only they can find; on open
# flatland they are unarmoured men with fish spears.
# ============================================================
a_nd_irq_madan_marshmen = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Base age 1 light infantry: max_strength 0.5, combat_power 1
	morale_damage_done = 0.1
	strength_damage_taken = 0.1
	attrition_loss = -0.1
	max_strength = 0.04	# levy power, scripts/fix_levy_power.py

	combat = {
		wetlands = 0.2
		flatland = -0.1
	}
	impact = {
		wetlands = 0.2
	}

	country_potential = {
		has_or_had_tag = IRQ
	}

	gfx_tags = { spear_tag light_tag }
}

# ============================================================
# Ayyarun (levy, age 2). The armed brotherhoods of Baghdad's quarters, bound
# by futuwwa oath. Ferocious behind a barricade and worth nothing in a field
# formation: they break enemy morale and cannot take a charge.
# ============================================================
a_nd_irq_ayyarun = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_2_renaissance_light_infantry
	buildable = no
	levy = yes

	# Base age 2 light infantry: max_strength 1.0, combat_power 1
	morale_damage_done = 0.2
	strength_damage_done = -0.1
	strength_damage_taken = 0.1
	initiative = 2	# levy power, scripts/fix_levy_power.py
	max_strength = 0.195	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = IRQ
	}

	gfx_tags = { sword_tag light_tag }
}

# ============================================================
# Muntafiq horse (levy, age 3). The confederation's riders on the desert edge
# of the lower Euphrates. They choose whether a fight happens at all, and
# lose it if the enemy gets to choose instead.
# ============================================================
a_nd_irq_muntafiq_horse = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	buildable = no
	levy = yes

	# Base age 3 light cavalry: max_strength 0.6, combat_power 2.6
	movement_speed = 0.2
	flanking_ability = 0.15
	strength_damage_taken = 0.1
	max_strength = 0.695	# levy power, scripts/fix_levy_power.py

	# The template's cavalry terrain entries are restated because an explicit
	# block replaces the copied one wholesale rather than merging into it.
	combat = {
		desert = 0.15
		jungle = -0.10
		wetlands = -0.10
		mountains = -0.10
	}
	impact = {
		desert = 0.2
		jungle = 0.10
		wetlands = 0.10
		mountains = 0.10
	}

	country_potential = {
		has_or_had_tag = IRQ
	}

	gfx_tags = { light_tag }
}

# ============================================================
# Ghurab (buildable, age 3). The Gulf gun-galley, shallow enough for the Shatt
# al-Arab bar and heavily gunned for its size. It trades hull for guns, which
# is the correct trade in a river mouth and the wrong one in open sea.
# ============================================================
n_nd_irq_ghurab = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_3_discovery_galley
	upgrades_to = n_galleass
	buildable = yes

	# Base age 3 galley: speed 4, hull 8, cannons 12, crew 0.075, blockade 0.3
	movement_speed = 0.5
	hull_size = -1
	cannons = 2
	blockade_capacity = 0.05

	country_potential = {
		has_or_had_tag = IRQ
	}

	gfx_tags = { mediterranean_galley_tag medium_tag }
}

# ============================================================
# Kul horsemen (buildable, age 5). The Georgian household of the Baghdad

# they take years to make, which is exactly why they ended up ruling Iraq.

	category = army_heavy_cavalry
	copy_from = a_age_5_absolutism_heavy_cavalry

	morale_damage_done = 0.15
	strength_damage_taken = -0.1
	build_time_modifier = 0.25

	# Restated for the same reason as the Muntafiq block above.
	combat = {
		jungle = -0.10
		wetlands = -0.10
		mountains = -0.10
		river = -0.10
	}
	impact = {
		jungle = 0.10
		wetlands = 0.10
		mountains = 0.10
	}

	country_potential = {
		has_or_had_tag = IRQ
	}

	gfx_tags = { sword_tag heavy_tag }
}
