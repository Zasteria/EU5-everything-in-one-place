# Bundschuh militia: levy infantry from the Swabian League's urban militia tradition
# Disciplined city militia, better morale but lighter than regular infantry
# Unlocked in age 1 (nd_swa_fortified_towns), so it copies the age 1 template.

	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# come in their working clothes, not in harness, and pay for it in men.

	strength_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = SWA
	}

	gfx_tags = { pike_tag medium_tag }
}

# Bundesknechte: the paid foot of the Swabian League (buildable, age 2)
# The League of 1488 fixed a quota of foot for every member and paid them from
# a common chest; the cities sent them out in Augsburg and Ulm harness. These
# were the Swabian "Knechte" from whom Maximilian raised his first
# Landsknechte, which is why the line continues into the pike age.
# Better armoured than the ordinary man-at-arms; a host of many contingents
# under three colleges of captains loses heart sooner when things go badly.

	}
