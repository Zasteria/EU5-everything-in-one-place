# ===== Francia (FRK) Units =====
# Every stat below is a delta added to the copy_from template, never an
# absolute (copy_from ADDS since 1.2.0; restating a template value doubles
# it). No scara, no ordonnance lance, no gendarme: Lotharingia and Greater
# Burgundy already field those, and a former Lotharingia keeps them. No
# Hessian jaeger either: vanilla's a_hesse_jager comes through the German
# culture group advances, which a Lotharingian Francia can hold.

# The Paladins (Age 4, buildable)
# The counts of the palace, the twelve peers of the chansons de geste,
# raised again as the household cavalry of the restored court. Superb in
# the charge and expensive to keep in the field: a paladin travels with his
# own remounts and grooms.
a_nd_frk_paladins = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_4_reformation_heavy_cavalry
	upgrades_to = a_gallop_cavalry
	buildable = yes

	# Template: max_strength 0.8, combat_power 4.
	morale_damage_done = 0.15
	strength_damage_taken = -0.05
	initiative = 1
	supply_weight = 0.25

	country_potential = {
		has_or_had_tag = FRK
	}

	gfx_tags = { heavy_tag }
}

# Basternae (Age 4, buildable auxiliary)
# Chapter 64 of the Capitulare de villis orders the army's carts covered in
# leather sewn so tight that, when a river has to be crossed, they can be
# floated over with the flour still dry. A supply train that treats rivers
# as roads and pays for it with no fighting value at all.
a_nd_frk_basternae = {
	is_special = yes
	category = army_auxiliary
	copy_from = a_age_4_reformation_auxiliary
	upgrades_to = a_wagon_train
	buildable = yes

	# Template: max_strength 0.2, combat_power 0.25, no terrain blocks, so
	# the river line below is new rather than restated.
	combat_power = -0.1
	supply_weight = -0.25
	attrition_loss = -0.1

	impact = {
		river = -0.25
	}

	country_potential = {
		has_or_had_tag = FRK
	}

	gfx_tags = { medium_tag }
}

# The Heerbann (Age 5, levy)
# The ban of the capitulary of 808: every free man with four mansi arms
# himself, poorer men club together to send one of their number, and a man
# with twelve mansi must bring a mail coat. Well armoured landholders who
# hold a line and march slowly, because every one of them has a harvest to
# get home to.
a_nd_frk_heerbann = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_5_absolutism_heavy_infantry
	upgrades_to = a_fusiliers
	buildable = no
	levy = yes

	# Template: max_strength 2.5, combat_power 2.25.
	strength_damage_taken = -0.1
	morale_damage_taken = -0.05
	morale_damage_done = -0.05
	movement_speed = -0.1

	country_potential = {
		has_or_had_tag = FRK
	}

	gfx_tags = { medium_tag }
}
