# Schwarze Reiter: German pistol cavalry (buildable)
# Age 3 heavy cavalry template: max_strength=0.6, combat_power=2.6
a_nd_ger_schwarze_reiter = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_3_discovery_heavy_cavalry
	upgrades_to = a_heavy_lancers
	buildable = yes

	# Pistol caracole: rank after rank fires and wheels away. The volley is
	# the signature, and a pistol reaches the enemy before a lance does.
	morale_damage_done = 0.1
	strength_damage_done = 0.1
	initiative = 1
	# The caracole gave up the charge. Reiters caught reloading by lancers
	# who rode home were broken (Mookerheyde, 1574).
	morale_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = GER
	}

	gfx_tags = { cavalry_tag medium_tag }
}

# Jäger: light infantry skirmishers and sharpshooters (buildable)
# Age 5 infantry template: max_strength=2.5, combat_power=2.25
a_nd_ger_jaeger = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_5_absolutism_light_infantry
	upgrades_to = a_sharpshooters
	buildable = yes

	# Riflemen in open order. Stats are deltas on the template. They work
	# round a flank (0.2; no vanilla unit writes more than 0.3) and their
	# aimed fire kills at a range a musket cannot answer. A rifle takes
	# three times as long to load and carries no bayonet: caught in the open
	# they lose men fast (vanilla's Hessian Jager pays the same 0.1).
	initiative = 2
	flanking_ability = 0.2
	strength_damage_done = 0.1
	strength_damage_taken = 0.1
	combat = {
		forest = 0.1
		hills = 0.05
	}

	country_potential = {
		has_or_had_tag = GER
	}

	gfx_tags = { medium_tag }
}

# Kantonisten: enrolled recruits of the canton system (levy-only)
# Age 5 heavy infantry template: max_strength=2.5, combat_power=2.25
a_nd_ger_kantonisten = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_5_absolutism_heavy_infantry
	buildable = no
	levy = yes

	# Drilled with their regiment every spring: they stand under fire.
	morale_damage_taken = -0.1
	# Ten months of the year at the plough (the Prussian canton regulation
	# of 1733): slower into action than men who drill all year.
	initiative = -0.5

	country_potential = {
		has_or_had_tag = GER
	}

	gfx_tags = { medium_tag }
}

# Deutsche Artillerie: German field artillery (buildable)
# Age 5 artillery template: max_strength=1.0, combat_power=5.5, bombard=0.25, barrage=7
a_nd_ger_feldartillerie = {
	is_special = yes
	category = army_artillery
	copy_from = a_age_5_absolutism_artillery
	upgrades_to = a_flying_battery
	buildable = yes

	# Stats are deltas on the template. Better cast barrels and drilled gun
	# crews: a little more weight of shot and one more round in the barrage

	artillery_barrage = 1
	supply_weight = 0.1

	country_potential = {
		has_or_had_tag = GER
	}

	gfx_tags = { heavy_tag }
}
