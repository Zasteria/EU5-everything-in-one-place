# Rodelero: sword-and-buckler infantry, Reconquista close combat
# Levy unit, age 1 traditions infantry
# Base template: max_strength=0.5, combat_power=1
a_nd_spa_rodelero = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Signature: the shock of sword and buckler men getting in under the
	# pikes (morale damage, above vanilla's almogavars on purpose). Flanking
	# sits at the best of vanilla's age 1 light infantry (caterans, 0.25).
	# Weakness: with no reach against horse or a formed pike line they were
	# ridden down in the open, which is why the tercios dropped them in the
	# 1530s.
	morale_damage_done = 0.15
	flanking_ability = 0.25
	combat_speed = 0.05
	strength_damage_taken = 0.1
	max_strength = 0.07	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = SPA
	}

	gfx_tags = { medium_tag light_tag }
}

# Jinete: Iberian light cavalry, fast raiders and skirmishers
# Levy unit, age 1 traditions cavalry
# Base template: max_strength=0.2, combat_power=2.6
a_nd_spa_jinete = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Levy twin of vanilla's Iberian jinetes and written with that unit's
	# own numbers (initiative 2, flanking 0.2, speed 0.75).
	# Weakness: riders a la jineta carried a leather adarga and light lances
	# and could not stand a charge of men-at-arms.
	initiative = 2
	flanking_ability = 0.2
	movement_speed = 0.75
	combat_speed = 0.05
	strength_damage_taken = 0.1
	max_strength = 0.235	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = SPA
	}

	gfx_tags = { cavalry_tag light_tag }
}

# Infanteria de Marina: Spanish marine infantry (oldest marines in the world)
# Buildable unit, age 5 absolutism infantry
# Base template: max_strength=2.5, combat_power=2.25
a_nd_spa_infanteria_de_marina = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_5_absolutism_light_infantry
	upgrades_to = a_sharpshooters
	buildable = yes

	# Amphibious specialists. Signature: the boarding assault (morale damage
	# and speed between the sections of a battle). Initiative sits at the
	# best of vanilla's age 5 light infantry (2).
	# Weakness: the Cuerpo de Batallones de Marina (1717) was a small corps
	# beside the line infantry, so a regiment musters fewer men.
	morale_damage_done = 0.15
	initiative = 2
	combat_speed = 1.5
	max_strength = -0.25

	country_potential = {
		has_or_had_tag = SPA
	}

	gfx_tags = { medium_tag }
}
