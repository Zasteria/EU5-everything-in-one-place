# ============================================================================
# Leon (LON): custom unit types
# National Destinies mod
# ============================================================================
#
# Vanilla arms every Christian Iberian with the caravel line, the galleons
# and, in age 4, the tercio; Leon's own vanilla advances give it the Order of
# Santiago as a bonus to heavy foot and sieges and, on the released road,
# modern brigades. Nothing here repeats those. This file holds what the
# kingdom itself put in the field: the mountain foot, the knights of
# Alcantara, the town companies of 1520, Menendez de Aviles's oared
# galleons, the Principality's tercio, the Asturian alarm companies of 1808
# and the lancers of the Salamanca oak pasture.
#
# Every stat under copy_from is a DELTA added to the template and then to the
# unit category's own value (common/levies/readme.txt: "EVERY STAT LINE IS
# ADDITIVE, NEVER AN ASSIGNMENT"). Template totals are quoted for the reader
# only; none of them is written in a block.
#
# The three levies are not one line. Each belongs to its own century and has
# no successor on purpose: the mountain foot gives way to vanilla's peasant
# levy in age 2, the Comunidades end at Villalar, and the alarm companies
# are the last age. Between them the pops raise vanilla's ladder.

# ----------------------------------------------------------------------------
# Age 1: Peones de la Montana (levy light infantry)
# ----------------------------------------------------------------------------
# Spear, throwing darts and a hide shield: quick to close and good at
# breaking a line's nerve with a volley from the slope, and cut down fast
# when caught in the open without armour (strength_damage_taken).
# Template a_age_1_traditions_light_infantry: max_strength 0.5, combat_power 1.
a_nd_lon_peones_montaneses = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	initiative = 1
	morale_damage_done = 0.05
	strength_damage_taken = 0.05

	combat = {
		mountains = 0.1
		hills = 0.1
	}

	country_potential = {
		has_or_had_tag = LON
	}

	gfx_tags = { spear_tag light_tag }
}

# ----------------------------------------------------------------------------
# Age 2: Freires de Alcantara (buildable heavy cavalry)
# ----------------------------------------------------------------------------
# Knight brethren under the Cistercian rule and the green cross: they do not
# ransom themselves and do not leave the field without the master's banner,
# which is the morale line. There were never many of them, which is the
# max_strength line: a few hundred lances for a whole frontier.
# Template a_age_2_renaissance_heavy_cavalry: max_strength 0.4, combat_power 2.6.
a_nd_lon_freires_de_alcantara = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_2_renaissance_heavy_cavalry
	upgrades_to = a_lancers
	buildable = yes

	morale_damage_taken = -0.1
	morale_damage_done = 0.05
	max_strength = -0.05

	country_potential = {
		has_or_had_tag = LON
	}

	gfx_tags = { crusader_tag heavy_tag }
}

# ----------------------------------------------------------------------------
# Age 4: Galeoncete (buildable light ship)
# ----------------------------------------------------------------------------

# and sweeps for a calm. Built to carry the king's silver faster than
# anything that can catch it and to fight anything that can keep up. It
# needs rowers as well as sailors (crew_size) and has no hold to spare for
# sitting off a harbour mouth for months (blockade_capacity).
# Template n_age_4_reformation_light_ship: speed 12, hull 10, cannons 12,
