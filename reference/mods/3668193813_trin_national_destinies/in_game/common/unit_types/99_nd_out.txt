# ===== Outremer (OUT) Units =====
# No knights, no sergeants, no turcopoles: those belong to the five crusader
# formables that feed into this one. Outremer's forces are the ones a Latin
# minority actually fields when it has to govern from Cairo to Baghdad: the
# native Christian marksmen of the mountain, the townsmen's brotherhoods,
# the camel logistics that make a four-capital empire possible, and a Red
# Sea navy nobody else in the crusader set has.

# --- Maronite Crossbowmen (Age 1, levy) ---
# William of Tyre calls the Maronites of Mount Lebanon valiant fighters who
# gave the Franks scouts, guides and archers, "good archers using the
# Italian style of cross-bowing". Their patriarch sought union with the
# Latin patriarch of Antioch in 1182: the one native community that came
# over to Rome of its own accord, and it never numbered enough to change the
# realm's arithmetic.
a_nd_out_maronite_crossbow = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Base age 1 light infantry: max_strength = 0.5, combat_power = 1.
	# Heavy bolt volume, loosed first from ground the bowmen chose, but a
	# mountain militia that is brittle once its formation is broken and
	# worth much less on open ground.
	strength_damage_done = 0.1
	initiative = 1
	morale_damage_taken = 0.1
	max_strength = 0.1	# levy power, scripts/fix_levy_power.py
	combat_power = 0.29	# levy power, scripts/fix_levy_power.py

	combat = {
		mountains = 0.10
		hills = 0.10
		grasslands = -0.10
	}

	country_potential = {
		has_or_had_tag = OUT
	}

	gfx_tags = { light_tag }
}

# --- Camel Train (Age 2, buildable auxiliary) ---
# Raynald's shipwrights crossed two hundred kilometres of the Negev on hired
# Bedouin camels. An empire whose capitals are Jerusalem, Cairo, Baghdad and
# Aden runs on the same animal. This is a baggage train that treats the
# desert as a road rather than an obstacle, and it does not fight.
a_nd_out_camel_train = {
	is_special = yes
	category = army_auxiliary
	copy_from = a_age_2_renaissance_auxiliary
	upgrades_to = a_supply_convoy
	buildable = yes

	# Base age 2 auxiliary: max_strength = 0.1, combat_power = 0.25.
	# It fights no worse than any other baggage train. What the animals
	# cost is the pace of a deployed line: a string of loaded camels is slow
	# to form up on a battlefield.
	supply_weight = -0.25
	attrition_loss = -0.15
	movement_speed = 0.15
	combat_speed = -0.1

	impact = {
		desert = -0.5
		arid = -0.5
	}

	country_potential = {
		has_or_had_tag = OUT
	}

	gfx_tags = { light_tag }
}
