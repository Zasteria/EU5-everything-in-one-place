# Aragonese Empire (CDA): Custom unit types
# National Destinies mod
#
# Note: vanilla already grants Catalan crossbowmen (a_catalan_crossbowmen,
# buildable) and Almogavar levies (levy_a_almogavars, levy_a_late_almogavars)
# to all aragonese/catalan-cultured countries via culture_iberia.txt. We do
# NOT duplicate those. Our custom units cover what vanilla doesn't:
# Mediterranean naval specialty (Catalan galley, buildable + levy), royal
# galiot (late galley), and the Tercio de Napoles heavy infantry born of
# the Italian Wars in our alt-history Aragonese Naples.

# Catalan Galley: Age 2 buildable galley, the signature ship of the
# Crown of Aragon's Mediterranean dominance.
# Base n_age_2_renaissance_galley: speed=4, hull=7, cannons=9, crew=0.06,
# blockade=0.2, maritime=ship_small_mid_maritime
n_nd_cda_catalan_galley = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_2_renaissance_galley
	upgrades_to = n_war_galley
	buildable = yes

	movement_speed = 1
	hull_size = 1
	cannons = 1
	blockade_capacity = 0.1

	country_potential = {
		has_or_had_tag = CDA
	}

	gfx_tags = { medium_tag }
}

# Catalan Galley (Levy): non-buildable levy variant, raised from coastal
# burgher pops of the Crown's port cities of Barcelona, Valencia, Palma,
# Naples and Palermo.
n_nd_cda_levy_galley = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_2_renaissance_galley
	buildable = no
	levy = yes

	movement_speed = 1
	hull_size = 1
	blockade_capacity = 0.1

	country_potential = {
		has_or_had_tag = CDA
	}

	gfx_tags = { medium_tag }
}

# Royal Galiot: Age 4 buildable galley, the late evolution of Catalan-
# Aragonese rowed warfare from the Drassanes Reials of Barcelona.
n_nd_cda_royal_galiot = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_4_reformation_galley
	upgrades_to = n_xebec
	buildable = yes

	movement_speed = 1.75
	blockade_capacity = 0.2

	country_potential = {
		has_or_had_tag = CDA
	}

	gfx_tags = { medium_tag }
}

# Tercio de Napoles: Age 4 buildable heavy infantry. Aragonese Naples
# was where the Spanish tercio system was born and tested under Gonzalo
# de Cordoba in the Italian Wars (1495-1504). In this alt-history the
# tactical innovation flowers under Catalan-Aragonese sponsorship.
# Base a_age_4_reformation_heavy_infantry: max_strength=2.0, combat_power=1.5
a_nd_cda_tercio_de_napoles = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_4_reformation_heavy_infantry
	upgrades_to = a_musketeers
	buildable = yes

	# Signature: the steadiness of pike and shot (damage taken) and the shock
	# of its sword and buckler men. A tercio was a large establishment, so
	# it keeps the most men vanilla gives any foot unit (+0.1, Varangians).
	# Weakness: the deep square was slow to redeploy, which is what the
	# thinner Dutch and Swedish lines exploited. Vanilla's legionaries
	# carry the same cut for the same reason.
	max_strength = 0.1
	morale_damage_done = 0.15
	strength_damage_taken = -0.1
	combat_speed = -1

	country_potential = {
		has_or_had_tag = CDA
	}

	gfx_tags = { medium_tag }
}
