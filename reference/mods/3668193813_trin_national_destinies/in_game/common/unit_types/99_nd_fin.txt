### Age 1: Rajasotilaat (Border Soldiers): levy unit ###
# Hardy Finnish border infantry, defending the eastern frontier
a_nd_fin_rajasotilaat = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Defensive frontier troops: resist attrition, endure punishment
	morale_damage_taken = -0.1
	strength_damage_taken = -0.05

	country_potential = {
		has_or_had_tag = FIN
	}

	gfx_tags = { medium_tag }
}

### Age 2: Erämetsästäjät (Wilderness Hunters): levy unit ###
# Finnish woodsmen scouts, skilled at tracking and ambush in forests
a_nd_fin_erametsastajat = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_2_renaissance_light_infantry
	buildable = no
	levy = yes

	# Fast-moving scouts: they strike first from cover and work round a
	# flank, and they do not stand to be charged. Flanking is held at
	# vanilla's largest on light infantry (+0.25, a_caterans).
	movement_speed = 1
	flanking_ability = 0.25
	initiative = 2
	strength_damage_done = 0.1
	morale_damage_taken = 0.05

	combat = {
		forest = 0.15
	}

	country_potential = {
		has_or_had_tag = FIN
	}

	gfx_tags = { light_tag }
}

### Age 4: Talvisoturi (Winter Warriors): buildable ###
# Elite Finnish infantry adapted to harsh winter conditions

	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	upgrades_to = a_hunters
	buildable = yes

	# Winter-hardened elite: excellent attrition resistance, high morale
	morale_damage_done = 0.1
	morale_damage_taken = -0.1

		forest = 0.1
		mountains = 0.1
	}

	country_potential = {
		has_or_had_tag = FIN
	}

### Age 6: Jääkärit (Finnish Jaegers): buildable ###
# Light infantry specialists trained in skirmish tactics
a_nd_fin_jaakari = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_6_revolutions_light_infantry
	buildable = yes

	# Elite light infantry: initiative, flanking, forest combat.
	# Fire is held at vanilla's largest on light infantry (+0.1,
	# a_longbowmen); the marksmen's edge is that they shoot first.
	initiative = 4
	flanking_ability = 0.25
	movement_speed = 0.2
	strength_damage_done = 0.1
	# Open order: small detachments with no closed ranks to absorb a volley
	# or a charge (vanilla's Hessian Jäger and hajduks pay the same price).
	strength_damage_taken = 0.1

	combat = {
		forest = 0.15
	}

	country_potential = {
		has_or_had_tag = FIN
	}

	gfx_tags = { light_tag }
}
