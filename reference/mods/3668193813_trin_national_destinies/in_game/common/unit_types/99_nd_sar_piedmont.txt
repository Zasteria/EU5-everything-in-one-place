# Sardinia-Piedmont units. The kingdom never fielded a large army, so every
# one of these is a specialist: parish militia, valley partisans, tunnel
# fighters, mountain skirmishers, island light horse, and a corsair hunter.
#
# The land units set only stats that are undefined (and therefore zero) on
# their templates, so a copied value is never at risk of being replaced by a
# smaller absolute.
#
# None of the three levy units carries upgrades_to. That matches vanilla:
# a_flintlock_levy, a_militiamen and a_conscripts have none either, because a
# levy is superseded by file order in common/levies, not by an upgrade path.

# Milizia Paesana: the parish militia of the 1566 ordinance, enrolled by
# village and drilled on Sundays. Steady behind a wall, brittle in the open.
a_nd_sar_piedmont_milizia_paesana = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	buildable = no
	levy = yes

	strength_damage_taken = -0.2	# levy power, scripts/fix_levy_power.py
	morale_damage_taken = 0.1
	max_strength = 0.2	# levy power, scripts/fix_levy_power.py

	combat = {
		mountains = 0.1
		hills = 0.1
	}

	country_potential = {
		has_or_had_tag = SRP
	}

	gfx_tags = { medium_tag }
}

# Bande Valdesi: the valley bands Joshua Janavel armed after 1655 and wrote
# a partisan manual for. They know every path above the snow line and cannot
# stand in line of battle for ten minutes.
a_nd_sar_piedmont_bande_valdesi = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_5_absolutism_light_infantry
	buildable = no
	levy = yes

	# Also carried the dead `assault = yes` key (see the Minatori below).
	# The parser aborts the file on the first bad token, so this second
	# instance never reached the log and only surfaced when the validator
	# was taught to reject it. `initiative` is the honest reading for a
	# band that ambushes from the paths rather than storming walls.
	initiative = 2
	movement_speed = 0.15
	attrition_loss = -0.15
	strength_damage_taken = 0.1
	max_strength = 0.5	# levy power, scripts/fix_levy_power.py

	combat = {
		mountains = 0.2
	}

	country_potential = {
		has_or_had_tag = SRP
	}

	gfx_tags = { light_tag }
}

# Minatori: the mine companies of the citadel, who fought underground in the
# countermine galleries and went into the breach first. Slow above ground.
a_nd_sar_piedmont_minatori = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_5_absolutism_heavy_infantry
	buildable = yes
	upgrades_to = a_fusiliers
	# The mine companies go in first. Expressed as `initiative`, NOT as
	# `assault = yes` (2026-08-07): that line was a hard PARSE ERROR that
	# took the whole file down, so every unit defined here silently failed
	# to load. The readme documents `assault = yes/no` at line 12, but the
	# shipped parser rejects it and ZERO vanilla units use it (same for
	# `bombard` and `auxiliary`): another stale readme entry, exactly like
	# the `end_reason` trap. `assault_ability` fails too, because a unit may
	# only carry the modifiers in the readme's own `modifiers:` list, which
	# is accurate (it is why naval `anti_piracy_warfare` does work).
	# `initiative` is on that list; vanilla's land range is -4 to 5 and 2 is
	# the commonest value.
	initiative = 2

	strength_damage_done = 0.15
	morale_damage_taken = -0.1
	movement_speed = -0.1

	country_potential = {
		has_or_had_tag = SRP
	}

	gfx_tags = { medium_tag }
}
