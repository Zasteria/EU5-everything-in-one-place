# ============================================================================
# Aragon (ARA): custom unit types
# National Destinies mod
# ============================================================================
#
# Vanilla already arms the peasants of the Crown in ages 1 and 2 (the
# almogavar levies, levy_a_almogavars and levy_a_late_almogavars), gives it
# the Catalan crossbowmen and the Catalan galley, and gives every Iberian the
# tercio in age 4. Nothing here repeats those or draws the almogavars' pops.
# This file is the inland kingdom's own soldiery: the barons' knight-fees,
# the horsemen of the frontier towns, the Diputacion's road guard, the
# tercios the Cortes paid for, the first light infantry of the Spanish army
# and the parish gunmen of the two sieges.
#
# Every stat under copy_from is a DELTA added to the template and then to the
# unit category's own value (common/levies/readme.txt: "EVERY STAT LINE IS
# ADDITIVE, NEVER AN ASSIGNMENT"). Template totals are quoted for the reader
# only; none of them is written in a block.
#
# The levies do not form one ladder, on purpose. The feudal host of the
# barons and the horse of the towns both faded in the fifteenth century, so
# from age 3 the nobles raise vanilla's cavaliers and the burghers vanilla's
# matchlock levy; the kingdom's own foot returns in age 5 and again in age 6.

# ----------------------------------------------------------------------------
# Age 1: Mesnaderos (levy heavy cavalry)
# ----------------------------------------------------------------------------
# The knights owed for the caballerias de honor, and the king's own household
# knights beside them. Mailed like any knight of the age (the two
# damage_taken lines are vanilla's knight armour, a_mailed_knights), and
# raised on the frontier, where a charge was pressed home.
# Template a_age_1_traditions_heavy_cavalry: max_strength 0.2, combat_power 2.6.
a_nd_ara_mesnaderos = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_1_traditions_heavy_cavalry
	buildable = no
	levy = yes

	strength_damage_taken = -0.25
	morale_damage_taken = -0.25
	morale_damage_done = 0.15

	country_potential = {
		has_or_had_tag = ARA
	}

	gfx_tags = { heavy_tag }
}

# ----------------------------------------------------------------------------
# Age 2: Caballeros Villanos (levy light cavalry)
# ----------------------------------------------------------------------------
# The commoner horsemen of Calatayud, Daroca and Teruel. The frontier fueros
# freed from tax any townsman who kept a horse and arms, so the towns had a
# cavalry of their own: light, fast on the wings and used to raiding. They
# were not knights and did not stand a knights' charge (morale_damage_taken).
# Template a_age_2_renaissance_light_cavalry: max_strength 0.4, combat_power 2.6.
a_nd_ara_caballeros_villanos = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_2_renaissance_light_cavalry
	buildable = no
	levy = yes

	flanking_ability = 0.1
	strength_damage_done = 0.1
	morale_damage_taken = 0.05

	country_potential = {
		has_or_had_tag = ARA
	}

	gfx_tags = { spear_tag light_tag }
}

# ----------------------------------------------------------------------------
# Age 4: Guardia del Reino (buildable light cavalry)
# ----------------------------------------------------------------------------
# The Diputacion's own companies, paid from the kingdom's customs to run down
# bandits on the roads to the passes. Riders who know every track and ford,
# quick to turn a flank and quick to be gone; armed for a pursuit and not for
# a pitched battle (strength_damage_taken).
# Template a_age_4_reformation_light_cavalry: max_strength 0.8, combat_power 2.85.
a_nd_ara_guardia_del_reino = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_4_reformation_light_cavalry
	upgrades_to = a_hussars
	buildable = yes

	initiative = 2
	flanking_ability = 0.2
	movement_speed = 0.5
	strength_damage_taken = 0.05

	country_potential = {
		has_or_had_tag = ARA
	}

	gfx_tags = { pistol_tag medium_tag }
}

# ----------------------------------------------------------------------------
# Age 5: Tercio del Reino (levy heavy infantry)
# ----------------------------------------------------------------------------
# The foot voted by the Cortes of 1645 and 1646: raised by the kingdom,
# officered by its own gentry and sworn to serve on its own border. Men
# defending the next valley to their own do not run.
# Template a_age_5_absolutism_heavy_infantry: max_strength 2.5, combat_power 2.25.
a_nd_ara_tercio_del_reino = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_5_absolutism_heavy_infantry
	buildable = no
	levy = yes

	morale_damage_taken = -0.1

	country_potential = {
		has_or_had_tag = ARA
	}

	gfx_tags = { musket_tag medium_tag }
}

# ----------------------------------------------------------------------------
# Age 6: Voluntarios de Aragon (buildable light infantry)
# ----------------------------------------------------------------------------
# Raised in 1762 from hunters, muleteers and smugglers of the Pyrenean
# valleys: the first regiment of light infantry in the Spanish army. Open
# order, aimed fire and a fast march over broken ground. A light regiment
# mustered fewer men than a line one (max_strength). Age 6 is the last age,
# so there is no upgrade target.
# Template a_age_6_revolutions_light_infantry: max_strength 3, combat_power 3.
a_nd_ara_voluntarios_de_aragon = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_6_revolutions_light_infantry
	buildable = yes

	initiative = 2
	morale_damage_done = 0.1
	max_strength = -0.2

	combat = {
		mountains = 0.15
		hills = 0.1
	}

	country_potential = {
		has_or_had_tag = ARA
	}

	gfx_tags = { musket_tag light_tag }
}
