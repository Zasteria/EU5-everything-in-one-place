# Age 4: Talpasi (the foot of the Kuruc risings, raised from Slovak and Rusyn
# villages of Upper Hungary). Quick to gather and to strike, and unsteady in
# a pitched battle, as at Trencin in 1708.
a_nd_slo_talpasi = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	buildable = no
	levy = yes

	# Placed by hand at about 1.21 times vanilla's flintlock levy of age 4.
	# No stat stands further from the template than vanilla's largest for
	# light infantry (initiative 5, damage done 0.1).
	movement_speed = 0.25
	initiative = 5
	strength_damage_done = 0.1
	morale_damage_done = 0.05
	max_strength = 0.1
	morale_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = SLO
	}

	gfx_tags = { musket_tag light_tag }
}
