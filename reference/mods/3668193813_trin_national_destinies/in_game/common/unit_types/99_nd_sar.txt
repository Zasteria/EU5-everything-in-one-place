# Arborean Militia: defensive island infantry, fight-on-home-soil tradition
# Age 1, levy unit (peasant militia from Giudicato tradition). It copies the
# template of the age its levy unlocks in (nd_sar_arborean_resistance, age 1).
a_nd_sar_arborean_militia = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Stats are deltas on the template. Defensive fighters: they hold their
	# ground and take losses well; they do not hit harder than other levies.
	morale_damage_taken = -0.1
	strength_damage_taken = -0.1	# levy power, scripts/fix_levy_power.py
	max_strength = 0.035	# levy power, scripts/fix_levy_power.py

	combat = {
		hills = 0.05
		mountains = 0.05
	}

	country_potential = {
		has_or_had_tag = SAR
	}

	gfx_tags = { spear_tag medium_tag }
}
