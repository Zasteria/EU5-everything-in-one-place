### Age 1: Silesian Piast Lancers: levy unit ###
# Light Piast dynasty cavalry defending the fragmented duchies
a_nd_sil_piast_lancers = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Fast border cavalry: scouting and raiding. Speed is the signature.
	movement_speed = 1
	morale_damage_done = 0.1
	flanking_ability = 0.2
	# Mail shirt and lance, not a knight's harness: they scout and raid and
	# do not stand in the line.
	strength_damage_taken = 0.1
	max_strength = 0.205	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = SIL
	}

	gfx_tags = { light_tag }
}

### Age 4: Silesian Jaegers: buildable ###
# Light infantry skirmishers from Silesian mountain forests
a_nd_sil_jaegers = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	upgrades_to = a_hunters
	buildable = yes

	# Deltas on a_age_4_reformation_light_infantry. Hunters and foresters
	# of the Sudeten slopes who fight in open order: they shoot first
	# (initiative 3; vanilla's a_bavarian_jager has 2 and its largest is
	# 5), they aim (strength damage done at 0.1, the most vanilla gives
	# light infantry, a_longbowmen), they work round a flank (0.2; vanilla's
	# largest for light infantry is 0.25, a_caterans) and they march light
	# (0.5; vanilla's largest in the army is 1.2). A skirmish line does not
	# stand a charge: they lose morale faster than formed foot.
	initiative = 3
	flanking_ability = 0.2
	movement_speed = 0.5
	strength_damage_done = 0.1
	morale_damage_taken = 0.1

	combat = {
		hills = 0.1
		forest = 0.1
	}

	country_potential = {
		has_or_had_tag = SIL
	}

	gfx_tags = { light_tag }
}
