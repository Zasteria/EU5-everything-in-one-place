# ============================================================================
# Occitania (OCC): custom unit types
# National Destinies mod
# ============================================================================
#
# The Occitan military profile is deliberately not shock. From the bidaus of
# the fourteenth century to the Camisards of the eighteenth, the Midi fought
# the same way: light, fast, at home on broken ground, dangerous in ambush and
# useless in a set-piece line. Every land unit here trades staying power for
# initiative, movement and terrain.
#
# Every stat line under copy_from is a DELTA: the engine adds it to the
# template's value and to the unit category's own value, it never replaces
# them (common/levies/readme.txt: "EVERY STAT LINE IS ADDITIVE, NEVER AN
# ASSIGNMENT"). The template's stats are noted above each unit so the sum can
# be read off. upgrades_to always targets a concrete buildable unit of a LATER
# age, never an a_age_* template (those are buildable = no).

# ----------------------------------------------------------------------------
# Age 1: Bidau (levy light infantry)
# ----------------------------------------------------------------------------
# The bidaus of the Pyrenean slope: javelins, a knife, no armour. Hired by
# both crowns through the fourteenth century to screen, raid and skirmish.
# Base a_age_1_traditions_light_infantry: max_strength 0.5, combat_power 1.
a_nd_occ_bidau = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	initiative = 3
	morale_damage_done = 0.1
	movement_speed = 0.15
	supply_weight = -0.25
	# No armour is the whole point: they close, throw and leave, and they
	# break if made to stand.
	strength_damage_taken = 0.1
	max_strength = 0.09	# levy power, scripts/fix_levy_power.py

	combat = {
		mountains = 0.15
		hills = 0.1
		woods = 0.1
	}
	impact = {
		mountains = 0.15
		hills = 0.1
	}

	country_potential = {
		has_or_had_tag = OCC
	}

	gfx_tags = { light_tag }
}
