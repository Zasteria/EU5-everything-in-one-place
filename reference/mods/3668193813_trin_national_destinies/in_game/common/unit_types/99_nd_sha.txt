# Shan Highland Cavalry: fast raiders who exploit mountain terrain
# Age 1 cavalry with reduced mountain penalties, representing Shan expertise
# in highland warfare. Used by levies.
a_nd_sha_highland_cavalry = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	max_strength = 0.23	# levy power, scripts/fix_levy_power.py

	# Base template: max_strength=0.2, combat_power=2.6
	# Mountain penalties: impact mountains 0.10, combat mountains -0.10
	# The mountain is the signature: we reduce the mountain penalty and add
	# a little flanking.
	combat = {
		mountains = -0.05
	}
	impact = {
		mountains = 0.05
	}
	flanking_ability = 0.2
	# A hill pony of thirteen hands carries no armoured rider and no charge.
	morale_damage_done = -0.1

	country_potential = {
		has_or_had_tag = SHA
	}

	gfx_tags = { medium_tag }
}

# Sawbwa's Elephant Guard: war elephants employed by Shan rulers
# Age 3 auxiliary, powerful shock troops requiring elephant goods
# Buildable.
a_nd_sha_elephant_guard = {
	is_special = yes
	category = army_auxiliary
	copy_from = a_age_3_discovery_auxiliary
	upgrades_to = a_baggage_train
	buildable = yes

	# Base template: max_strength=0.15, combat_power=0.25
	food_storage_per_strength = 11
	impact = {
		jungle = -0.25
	}
	combat = {
		jungle = 0.15
	}

	location_potential = {
		market ?= { is_produced_in_market = goods:elephants }
	}

	country_potential = {
		has_or_had_tag = SHA
	}

	gfx_tags = { elephant_tag auxiliary_tag medium_tag }
	max_strength = 0.015
}
