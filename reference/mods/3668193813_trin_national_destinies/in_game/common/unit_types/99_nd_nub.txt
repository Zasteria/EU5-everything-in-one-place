### Nubian Archers: Age 1 levy unit ###
# The legendary bowmen of Ta-Seti. High morale damage, mobile ranged infantry.
a_nd_nub_nubian_archers = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	morale_damage_done = 0.1
	strength_damage_done = 0.1
	movement_speed = 1
	# Bowmen without armour: deadly at range, cut down quickly once an
	# enemy closes.
	strength_damage_taken = 0.1
	max_strength = 0.095	# levy power, scripts/fix_levy_power.py

	combat = {
		desert = 0.1
	}

	country_potential = {
		has_or_had_tag = NUB
	}

	gfx_tags = { medium_tag }
}

### Jalba: Age 3 buildable transport ###
# The sewn ship of the Red Sea crossing: planks stitched with coir, no nail in
# the hull. Ibn Jubayr sailed in one from Aydhab in 1183, packed with
# pilgrims; Joao de Castro still met them at Suakin and Massawa in 1541.
# Carries more than its template (by vanilla's largest step for a transport,
# 0.1) and rides light and fast, and is weaker for it: a sewn hull bears
# neither a pounding nor many guns.
n_nd_nub_jalba = {
	is_special = yes
	category = navy_transport
	copy_from = n_age_3_discovery_transport
	upgrades_to = n_brig
	buildable = yes

	transport_capacity = 0.1
	movement_speed = 0.5
	hull_size = -3
	cannons = -2

	country_potential = {
		has_or_had_tag = NUB
	}

	gfx_tags = { light_tag }
}

### Black Horse: Age 4 levy heavy cavalry ###
# The mailed cavalry of the Funj sultans, on the horses of the Butana and
# Dongola. James Bruce counted them at Sennar in 1772: coats of mail, a
# broadsword, a quilted cover on the horse. Levy only: nobles raise it.
a_nd_nub_black_horse = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_4_reformation_heavy_cavalry
	buildable = no
	levy = yes

	# Mail and quilting: hard to break, slow to bring up.
	morale_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	strength_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	movement_speed = -0.25
	combat_power = 0.6	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = NUB
	}

	gfx_tags = { heavy_tag }
}
