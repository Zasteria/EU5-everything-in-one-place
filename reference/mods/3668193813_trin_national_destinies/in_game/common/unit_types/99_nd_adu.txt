# Zanata Light Horse (key kept from its first name, Andalusi Jinete): Age 1, levy unit
# Fast Berber-style light horse, hit-and-run javelin skirmishers.
# Vanilla's a_iberian_jinetes is the Christian copy of this rider: buildable,
# behind the Across the Pillars DLC and closed to Andalusi culture. This is
# the levy of the original, named apart for a former who can hold both.
# Flanking sits at vanilla's own a_iberian_jinetes (0.2). The price of the
# speed is armour: a leather adarga and no mail, so a jinete line that is
# caught and made to stand takes more casualties than the template.
# Base age 1 cavalry: max_strength=0.2, combat_power=2.6
a_nd_adu_jinete = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	movement_speed = 1
	flanking_ability = 0.2
	morale_damage_done = 0.1
	strength_damage_taken = 0.1
	max_strength = 0.25	# levy power, scripts/fix_levy_power.py

	combat = {
		hills = 0.05
		desert = 0.05
	}

	country_potential = {
		has_or_had_tag = ADU
	}

	gfx_tags = { spear_tag light_tag }
}
