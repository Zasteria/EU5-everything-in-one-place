# Stratiotai Italici: Age 1 light cavalry of the restored themes.
# Albanian and Greek-Calabrian horsemen drawn from the Mezzogiorno
# Greek substrate and the empire's Adriatic provinces. Faster and
# more aggressive than the standard age 1 levy cavalry.
# Base age 1 light cavalry: max_strength=0.2, combat_power=2.6
a_nd_rom_byz_stratiotai = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Speed and the quick strike are the signature.
	movement_speed = 1
	flanking_ability = 0.2
	initiative = 3
	# Unarmoured skirmishers: they harry and pursue, and do not hold a line
	# against men at arms.
	morale_damage_taken = 0.1
	max_strength = 0.245	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = ROM
		has_variable = nd_rom_byz_formed
	}

	gfx_tags = { light_tag }
}
