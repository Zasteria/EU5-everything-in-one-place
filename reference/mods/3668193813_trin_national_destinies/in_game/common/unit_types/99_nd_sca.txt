# Huovi: the mounted men of the crown's castles in Finland, age 3.

# own Hakkapeliitta in age 5 (country_swe.txt, a_hakkapelitta), and this is
# the same horseman a century and a half earlier. The key is unchanged.
# They raided along the eastern frontier and struck before the enemy had
# formed, which is the initiative. They rode small Finnish horses in buff

		forest = 0.15

		has_or_had_tag = SCA
	}
