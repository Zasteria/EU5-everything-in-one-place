# Lombardy (LMB) custom units
# The Milanese armour industry and the communal-royal military arc:
# Company of Death (Legnano legend, communal levy), Missaglia-plate
# men-at-arms (ducal heavy cavalry), Royal Italian Guard (Napoleonic
# Kingdom of Italy, crowned with the Iron Crown in 1805).

# Company of the Carroccio / Knights of Death: the sworn guard of the
# communal war-wagon in Lombard League tradition (Galvano Fiamma's
# chronicle). Urban militia sworn not to retreat: they hold, whatever
# the cost. Levy unit: raised from the communes, never recruited.
# Age 1 heavy infantry base: max_strength=0.5, combat_power=1.
a_nd_lmb_compagnia_della_morte = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_1_traditions_heavy_infantry
	buildable = no
	levy = yes

	# Oath-bound around the Carroccio: they do not break. Calibrated to
	# the vanilla levy peer cohort (a_reformed_gallowglass).
	morale_damage_taken = -0.1
	strength_damage_taken = -0.05

	country_potential = {
		has_or_had_tag = LMB
	}

	gfx_tags = { medium_tag }
}

# Uomini d'Arme in Missaglia plate: ducal men-at-arms armoured by the
# finest workshops in Europe. The Missaglia dynasty of Milan exported
# harness from Spain to Flanders; their white armour turned aside
# lance and bolt alike. Superbly protected, punishing to field, and
# slower on the march than lighter horse.
# Age 2 heavy cavalry base: max_strength=0.4, combat_power=2.6.
a_nd_lmb_uomini_darme = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_2_renaissance_heavy_cavalry
	upgrades_to = a_lancers
	buildable = yes

	# The best plate in Europe: hard to kill, hard to shake.
	strength_damage_taken = -0.15
	morale_damage_taken = -0.05
	# Full harness for man and horse slows the column.
	movement_speed = -0.1

	construction_demand = expensive_heavy_cavalry_construction
	maintenance_demand = expensive_heavy_cavalry_maintenance

	country_potential = {
		has_or_had_tag = LMB
	}

	gfx_tags = { heavy_tag }
}

# Royal Italian Guard: the guard regiments of the Kingdom of Italy,
# raised in Milan after the Iron Crown coronation of 1805. Elite shock
# infantry in the Napoleonic pattern: ferocious in the assault, but a
# guard is committed where the fire is hottest and bleeds for it.
# Age 6 heavy infantry base: max_strength=3.0, combat_power=3.
a_nd_lmb_guardia_reale = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_6_revolutions_heavy_infantry
	buildable = yes

	# Modelled on vanilla a_grenadiers (+0.2 morale done, +0.1 strength
	# taken): the guard hits harder and stands where others fall back.
	morale_damage_done = 0.2
	morale_damage_taken = -0.1
	strength_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = LMB
	}

	gfx_tags = { grenadier_tag medium_tag }
}
