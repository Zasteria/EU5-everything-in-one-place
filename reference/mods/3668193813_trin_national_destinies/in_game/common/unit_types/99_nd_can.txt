# Northwest Mounted Police: Age 6 cavalry
# Frontier law enforcement mounted force
# Base cav template: max_strength=1.2, combat_power=3.25
a_nd_can_mounted_police = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_6_revolutions_light_cavalry
	buildable = yes

	# Deltas on a_age_6_revolutions_light_cavalry (strength 1.2). The force
	# that made the March West of 1874 (275 men over 1,300 km): it travels
	# far and loses little on the way, and it is a constabulary of a few
	# hundred, so a regiment is a sixth smaller than line cavalry.
	# attrition_loss -0.25 cancels the cavalry category's extra attrition,
	# the same number vanilla gives a_gurkha.
	# With a sixth of the men gone the unit fought worse than the cavalry it
	# copies (total combat power 3.669 against 4.194). What a picked
	# constabulary has is steadiness: it does not break and it breaks
	# others. Morale damage taken -0.1 and done +0.1 put it at 4.271.
	max_strength = -0.2
	movement_speed = 0.15
	attrition_loss = -0.25
	morale_damage_done = 0.1
	morale_damage_taken = -0.1

	country_potential = {
		has_or_had_tag = CAN
	}

	gfx_tags = { medium_tag }
}

# Canot du Maitre: Age 3 transport

	is_special = yes
	category = navy_transport
	copy_from = n_age_3_discovery_transport
	upgrades_to = n_brig
	buildable = yes

	# Trade canoe: faster and roomier than the template's transport, but a
	# bark hull with no guns. Deltas on n_age_3_discovery_transport
	# (speed 6.5, hull 15, cannons 5, transport 0.75, crew 0.06).
	# Load stands at vanilla's largest for a transport (0.1, n_kabang;

	hull_size = -5
	cannons = -5
	crew_size = -0.01

	country_potential = {
		has_or_had_tag = CAN
	}

	gfx_tags = { medium_tag }
}
