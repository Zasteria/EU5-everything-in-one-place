# Shiledar Cavalry: Age 3 self-equipped light cavalry, fast raiders
# Base age 3 cavalry: max_strength=0.6, combat_power=2.6
a_nd_mhr_shiledar = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	upgrades_to = a_pistoleers
	buildable = yes

	movement_speed = 1
	# Raiders who come round the flank and strike before the line has turned:
	# flanking at vanilla's best (0.3) and the first blow as initiative. With
	# 0.2 and no initiative the unit stood below its own template in total
	# combat power (1.645 against 1.677).
	flanking_ability = 0.3
	initiative = 1
	morale_damage_done = 0.05
	# A shiledar's horse was his own capital, and a dead horse ended his
	# service: these riders raid and wheel away, they do not stand.
	morale_damage_taken = 0.1
	combat = {
		hills = 0.1
		desert = 0.05
	}

	country_potential = {
		has_or_had_tag = MHR
	}

	gfx_tags = { medium_tag }
}
