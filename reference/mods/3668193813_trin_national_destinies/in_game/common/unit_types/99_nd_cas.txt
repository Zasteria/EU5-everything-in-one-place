# ============================================================================
# Castile (CAS): custom unit types
# National Destinies mod
# ============================================================================
#
# What is NOT here, because somebody already fields it:
#   vanilla: the Lanzas de Castilla (the men at arms of the Guardas, age 3),
#   the Tercio, the Iberian caravels and galleons, and, reserved for the
#   Iberian DLC, the Iberian Jinetes and the Caballeria de Cuantia
#   (unit_types/1_uniques_for_age_1_traditions.txt, estate privilege
#   cas_regimiento). Neither DLC unit is imitated here.
#   the mod's Spain (99_nd_spa.txt): rodeleros, jinetes, its own tercio, the
#   Galera Real, the galleon and the marine infantry.
#
# What is here is the kingdom's own muster roll from the frontier councils to
# the provincial regiments: one infantry levy per age, the royal vassals as
# the noble levy of age 1, and four units a player recruits.
#
# Every stat under copy_from is a DELTA added to the template and then to the
# unit category's own value (common/levies/readme.txt: "EVERY STAT LINE IS
# ADDITIVE, NEVER AN ASSIGNMENT"). Template totals are quoted for the reader
# only; none of them is written in a block.
#
# The infantry levy is one unit per age because the engine picks the
# strongest levy again every age and upgrades_to does nothing on a levy. The
# noble levy has no successor on purpose: from 1493 the Crown paid standing
# companies instead of stipends, and those are vanilla's Lanzas de Castilla.

# ----------------------------------------------------------------------------
# Age 1: Milicia Concejil (levy heavy infantry)
# ----------------------------------------------------------------------------
# The host of a frontier council: every householder of the town and its
# villages, mustered by parish behind the council's banner, with lance,
# shield and iron cap as the fuero lists them. Steady when it fights for its
# own charter, slow to form up and slower to march away from the harvest.
# Template a_age_1_traditions_heavy_infantry: max_strength 0.5, combat_power 1.
a_nd_cas_milicia_concejil = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_1_traditions_heavy_infantry
	buildable = no
	levy = yes

	morale_damage_taken = -0.1
	strength_damage_taken = -0.05
	combat_speed = -0.5

	country_potential = {
		has_or_had_tag = CAS
	}

	gfx_tags = { medium_tag }
}

# ----------------------------------------------------------------------------
# Age 1: Vasallos del Rey (levy heavy cavalry)
# ----------------------------------------------------------------------------
# Knights and squires who draw a yearly stipend from the royal rents and owe
# a fixed number of lances for it, each lance a man in harness on a covered
# horse. Paid men serve longer and ride better mounts than a baron's
# following, and the king's book says how many must appear. Armoured as
# vanilla's mailed knights are (the two damage lines); the paid man's edge is
# his horse and his practice (combat_power), and his harness is the heavier
# for it (movement_speed).
# Template a_age_1_traditions_heavy_cavalry: max_strength 0.2, combat_power 2.6.
a_nd_cas_vasallos_del_rey = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_1_traditions_heavy_cavalry
	buildable = no
	levy = yes

	strength_damage_taken = -0.25
	morale_damage_taken = -0.25
	combat_power = 0.3
	initiative = 1
	movement_speed = -0.25

	combat = {
		grasslands = 0.1
		wetlands = -0.1
	}

	country_potential = {
		has_or_had_tag = CAS
	}

	gfx_tags = { heavy_tag }
}

# ----------------------------------------------------------------------------
# Age 2: Galera de Castilla (buildable galley)
# ----------------------------------------------------------------------------
# The king's galleys from the yards of Seville, rowed out through the Strait
# and up to the Channel. At La Rochelle in 1372 they waited for the ebb, then
# burned an English fleet whose deeper ships sat fast in the shallows. Built
# to carry crossbowmen and fire, and to cross open water; less handy than a
# Mediterranean galley at closing a harbour.
# Template n_age_2_renaissance_galley: speed 4, hull 7, cannons 9, crew 0.06,
# blockade 0.2.

		has_or_had_tag = CAS
