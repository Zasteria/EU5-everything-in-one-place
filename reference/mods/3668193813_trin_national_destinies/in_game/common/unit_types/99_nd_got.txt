# Regnum Gothorum (GOT): Custom unit types
# National Destinies mod
#
# There are no ships here on purpose. Spain fights the Atlantic, Aragon fights
# the Mediterranean, and this kingdom fights for a mountain corridor. Its
# military identity is the muster clause of its law code (everyone serves,
# regardless of descent or status) and the household horse sworn to the king's
# person. Both survive in modernised form to the last age, because the code
# they come from was never repealed.
#
# Every stat line under copy_from is a DELTA: the engine adds it to the
# template's value and to the unit category's own value (common/levies/
# readme.txt: "EVERY STAT LINE IS ADDITIVE, NEVER AN ASSIGNMENT"). Base
# template stats are noted per unit.

# Thiufa: the levy of "a thousand", the decimal unit of the Gothic host under
# its thiufadus. Ervig's muster law (681) binds general, count and gardingus,
# Goth and Roman, freeman and freedman, to answer the summons alike. A body
# raised by legal compulsion across every rank holds its ground because the
# penalty for breaking it is loss of legal standing, but it is not made of
# soldiers by trade.
# Base template a_age_1_traditions_light_infantry: max_strength=0.5, combat_power=1
a_nd_got_thiufa = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	morale_damage_taken = -0.1
	strength_damage_done = -0.1
	max_strength = 0.1	# levy power, scripts/fix_levy_power.py
	combat_power = 0.03	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = GOT
	}

	gfx_tags = { light_tag }
}

# Gardingi: the king's sworn men, a rank of the royal household named in the
# muster law beside dukes and counts. They hold their standing from a personal
# oath rather than from land, which makes them the crown's counterweight to a
# nobility that elects kings. They charge frontally and are paid for in
# casualties.

a_nd_got_gardingi = {
	is_special = yes

	buildable = no
	levy = yes

	combat_power = 0.15	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = GOT

# Hueste del Fuero: the muster clause restated for an age of firearms. The
# roll is kept by the same city courts that keep the land registers, which is
# why it can be called out faster than a contract army and why it fights like
# a body of townsmen rather than a body of veterans.
# Base template a_age_4_reformation_light_infantry: max_strength=2.0, combat_power=1.5
a_nd_got_hoste_del_fuero = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	buildable = no
	levy = yes

	morale_damage_taken = -0.1
	flanking_ability = 0.25
	# Flanking stands at vanilla's largest for light infantry (a_caterans);
	# the rest of the unit's edge is the speed of its muster.
	initiative = 1
	strength_damage_done = -0.05

	country_potential = {
		has_or_had_tag = GOT
	}

	gfx_tags = { light_tag }
}
