# Armatoloi: Age 1 irregular militia, mountain guerrilla fighters
# Historically recruited by Ottomans to police Greek countryside, they became
# the backbone of Greek resistance. Excellent in rough terrain, high morale.
a_nd_gre_armatoloi = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_1_traditions_heavy_infantry
	buildable = no
	levy = yes

	# Base template: max_strength=0.5, combat_power=1
	morale_damage_done = 0.1
	morale_damage_taken = -0.05

	combat = {
		mountains = 0.15
		hills = 0.1
	}

	country_potential = {
		has_or_had_tag = GRE
	}

	gfx_tags = { medium_tag }
}

# Klephts: Age 2 mountain bandits turned freedom fighters
# Raiding specialists with high initiative and flanking ability
a_nd_gre_klephts = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_2_renaissance_light_infantry
	upgrades_to = a_early_arquebusiers
	buildable = yes

	# Deltas on a_age_2_renaissance_light_infantry. The ambush is the
	# signature: initiative 3 is vanilla's top step for light foot
	# (a_late_longbowmen), flanking 0.25 sits inside vanilla's 0.2 to 0.3,
	# speed 0.2 is the most vanilla gives foot (a_jazayerchi). The price is
	# the one vanilla's own Balkan brigand pays (a_hajduk takes more
	# damage): a klepht band does not stand in the open, it scatters, so it
	# loses morale faster than drilled skirmishers.
	initiative = 3
	flanking_ability = 0.25
	movement_speed = 0.2
	morale_damage_taken = 0.1

	combat = {
		mountains = 0.15
		hills = 0.1
		forest = 0.05
	}

	country_potential = {
		has_or_had_tag = GRE
	}

	gfx_tags = { medium_tag }
}

# Hydriote Fire Ship: Age 4 light ship, devastating but fragile

		has_or_had_tag = GRE
