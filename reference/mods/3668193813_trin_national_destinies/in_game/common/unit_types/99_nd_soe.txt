# Songhai Empire - Custom Unit Types

# Songhai Noble Cavalry (Buildable) - Age 1
# Elite mounted warriors drawn from Songhai nobility, wearing iron breastplates.
# High combat power and mobility, reflecting their shock assault role.
# The charge is the signature. Horses were the weak point: they came across
# the Sahara at great cost and died of sleeping sickness wherever the army
# left the dry Sahel, so the unit loses more to attrition.
# Base age 1 cavalry: max_strength=0.2, combat_power=2.6
a_nd_soe_noble_cavalry = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_1_traditions_heavy_cavalry
	upgrades_to = a_heavy_cavalrymen
	buildable = yes

	morale_damage_done = 0.10
	movement_speed = 0.2
	attrition_loss = 0.25

	country_potential = {
		has_or_had_tag = SOE
	}

	gfx_tags = { heavy_tag }
}

# Songhai Mounted Archers (Levy) - Age 2
# Cavalry archers adapted from Saharan traditions, used for raiding and screening.
# The flanking edge matches vanilla's Tuareg veiled cavalry, the same
# tradition. A screen of archers harasses and gives way: it does not stand
# when charged, so it loses morale faster.
# Base age 2 cavalry: max_strength=0.4, combat_power=2.6
a_nd_soe_mounted_archers = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_2_renaissance_light_cavalry
	buildable = no
	levy = yes

	flanking_ability = 0.3
	movement_speed = 0.25
	morale_damage_taken = 0.1
	max_strength = 0.685	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = SOE
	}

	gfx_tags = { light_tag }
}
