### Age 1: Mazovian Szlachta Levy, petty noble cavalry, light and numerous
### Levy unit: raised through levy system
a_nd_maz_szlachta_cavalry = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	buildable = no
	levy = yes

	# Base template: max_strength=0.2, combat_power=2.6
	# Light noble cavalry: fast, good flankers, and numerous above all
	# (the levy's riders are the signature). The petty nobles of Mazovia
	# farmed their own plots and rode to war with what they could afford,
	# often without mail, so they take more casualties.
	flanking_ability = 0.2
	movement_speed = 1
	morale_damage_done = 0.05
	strength_damage_taken = 0.1
	max_strength = 0.235	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = MAZ
	}

	gfx_tags = { spear_tag light_tag }
}

### Age 2: Szlachta Zagrodowa, yeoman gentry riding as mounted crossbowmen
### Levy unit: the successor of the age 1 szlachta cavalry
a_nd_maz_zagrodowa = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_2_renaissance_light_cavalry
	buildable = no
	levy = yes

	# Deltas on a_age_2_renaissance_light_cavalry. In the Polish lance the
	# poorer rider was the strzelec, a mounted crossbowman; a Mazovian
	# gentleman who ploughed his own field could afford nothing more, so
	# the duchy's banners were mostly strzelcy. They shoot (strength damage
	# done), they ride light (speed), and without mail they take more
	# casualties. The riders are the signature, as in the age 1 levy: 0.69
	# more men puts the levy 1.2 times above vanilla's plated knights,
	# with 2.7 times the template's riders (vanilla's steppe horse has 3).
	strength_damage_done = 0.05
	movement_speed = 0.5
	strength_damage_taken = 0.1
	max_strength = 0.69

	country_potential = {
		has_or_had_tag = MAZ
	}

	gfx_tags = { light_tag }
}

### Age 5: Kurpie Marksmen, hunters and bee keepers of the Green Forest
### Levy unit: raised from the peasants of the forest locations only
a_nd_maz_kurpie = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_5_absolutism_light_infantry
	buildable = no
	levy = yes

	# Deltas on a_age_5_absolutism_light_infantry. The Kurpie lived by the
	# hunt and the forest hive, and in 1708 they fell on Charles XII's
	# column at the causeway of Kopanski Most. Aimed fire (strength damage
	# done at 0.1, the most vanilla gives light infantry, a_longbowmen), the
	# first shot from cover (initiative 3; vanilla's a_bavarian_jager has 2
	# and its largest is 5), hard to hit among the trees (strength damage
	# taken, half of what a_late_longbowmen has) and the jager's forest and
	# woods numbers. They had no drill and no bayonets: caught in the open
	# they scatter (morale damage taken). The levy wins by marksmanship,
	# not by men: these numbers put it 1.16 times above the militia the
	# same peasants would otherwise send.
	strength_damage_done = 0.1
	initiative = 3
	strength_damage_taken = -0.1
	morale_damage_taken = 0.1

	combat = {
		forest = 0.15
		woods = 0.15
	}

	country_potential = {
		has_or_had_tag = MAZ
	}

	gfx_tags = { musket_tag light_tag }
}
