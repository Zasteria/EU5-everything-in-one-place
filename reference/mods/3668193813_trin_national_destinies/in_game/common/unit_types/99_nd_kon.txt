# Shield Warriors: Heavy infantry armed with large shields and swords
# Kongolese heavy troops relied on strong defensive shields and close combat

	buildable = no

		has_or_had_tag = KON
	}

	gfx_tags = { medium_tag }
}

# Pombo Scouts: Elite pursuit troops who could keep up with cavalry
# Used for scouting, pursuit, and special reconnaissance missions
# Buildable: recruited directly
a_nd_kon_pombo_scouts = {
	is_special = yes
	category = army_auxiliary
	copy_from = a_age_1_traditions_auxiliary
	upgrades_to = a_supply_carts
	buildable = yes

	# Runners who keep pace with horses and know the ground: fast, first to
	# strike, and they lose few men on the march. Flanking stays at vanilla's
	# best (0.3). They carry no shield, so they suffer when caught in a line.
	movement_speed = 1
	initiative = 2
	flanking_ability = 0.3
	attrition_loss = -0.25
	strength_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = KON
	}

	gfx_tags = { medium_tag }
}

# Lucanzos: Special detachment troops for raiding and ambush missions
# Used for various asymmetric warfare tasks
# Levy unit: raised through levy system
a_nd_kon_lucanzos = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_4_reformation_light_infantry
	buildable = no
	levy = yes

	# Ambush and speed are the signature. Initiative at vanilla's best for
	# the template (Hessian Jaeger, 2), and the Jaeger's own price: detached
	# parties that take losses when made to stand in a line.
	# Flanking and speed stay at vanilla's best for the category (0.25 of
	# a_caterans, 1.2 of the camelry levy); the rest of the ambush is the
	# shock of it, which breaks nerve more than it kills.
	initiative = 2
	flanking_ability = 0.25
	morale_damage_done = 0.15
	movement_speed = 1.2
	strength_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = KON
	}

	gfx_tags = { medium_tag }
}
