# Hen Ogledd: Custom Unit Types

# Teulu: the king's mounted warband retainers
# Elite heavy cavalry with high morale: shock troops inspired by Y Gododdin
# Age 1 cavalry template: max_strength=0.2, combat_power=2.6
# "Gosgordd" since 2026-08-25 (unit pool probe): vanilla already levies a
# Welsh and Cornish "Teulu" (a_teulu, 05_traditions_levies.txt), and every
# Brythonic-language origin of Hen Ogledd fielded both under one name. The
# Old North's own word for the king's mounted warband is the gosgordd of
# Y Gododdin.
a_nd_hen_teulu = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_1_traditions_heavy_cavalry
	buildable = no
	levy = yes

	morale_damage_done = 0.1
	strength_damage_done = 0.05
	morale_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	strength_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	combat_power = 0.18	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = HEN
	}

	gfx_tags = { medium_tag }
}
