# Bavarian Ritter: heavy armored cavalry, levy unit
# Based on age 3 cavalry: max_strength=0.6, combat_power=2.6
# The levy unlocks in age 3 (nd_bav_landshut_reunification) and from 1.4 has
# to out-power that age's knight levy, so it stands on that age's template.
a_nd_bav_ritter = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_3_discovery_heavy_cavalry
	buildable = no
	levy = yes

	morale_damage_done = 0.1
	strength_damage_done = 0.1
	morale_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	strength_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	combat_power = 0.1	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = BAV
	}

	gfx_tags = { heavy_tag medium_tag }
}

# Bavarian Chevaulegers: light reconnaissance cavalry, buildable
# Based on age 5 cavalry: max_strength=1.0, combat_power=3.25
a_nd_bav_chevaulegers = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_5_absolutism_light_cavalry
	upgrades_to = a_light_dragoons
	buildable = yes

	# Line light horse: from 1811 every Bavarian dragoon regiment was a
	# chevauleger regiment, so the arm scouted and also charged in line
	# (Borodino, 1812). Flanking stands at vanilla's ceiling of 0.3; the
	# charge is the rest of its edge. Light mounts and no cuirass: it loses
	# more men than the template when it is hit.
	movement_speed = 0.1
	flanking_ability = 0.3
	morale_damage_done = 0.1
	strength_damage_taken = 0.05

	country_potential = {
		has_or_had_tag = BAV
	}

	gfx_tags = { medium_tag }
}
