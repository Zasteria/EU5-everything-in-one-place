# Age 4: Misl cavalry: guerrilla horsemen of the confederacy
# Base cavalry template: max_strength=0.8, combat_power=2.85
a_nd_pun_misl_cavalry = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_4_reformation_light_cavalry
	upgrades_to = a_hussars
	buildable = yes

	# Dhai phat, the two and a half strikes: ride in, fire, ride out, come
	# again. Speed is the whole weapon; they are not built to stand in a line.
	# The first volley is theirs (initiative) and the repeated strikes wear a
	# line down (morale damage): without the two the unit stood below its own
	# template in total combat power (2.345 against 2.452).
	flanking_ability = 0.3
	movement_speed = 1
	initiative = 2
	morale_damage_done = 0.05
	morale_damage_taken = 0.1

	country_potential = {
		has_or_had_tag = PUN
	}

	gfx_tags = { medium_tag }
}
