# Thessalonica: custom unit types.
#
# The realm's fighting men come from the three constituencies that
# actually existed here: the mixed-blood waterfront of the port, the
# horse-breeding plain of Thessaly, and the armed mining villages of
# Chalkidiki. Every stat line is a DELTA added to the copy_from template's
# value, never a replacement of it.

# Gasmouloi. Sons of Greek mothers and Latin fathers, enrolled as marines
# from Michael VIII's reign onward and described by contemporaries as
# combining Byzantine caution with Latin recklessness. Both communities
# distrusted them, which is the point: they fight for whoever pays, they
# swarm a beach faster than anyone, and they do not hold when it turns.
# Base template: max_strength 0.5, combat_power 1.
a_nd_the_gasmouloi = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Flanking stops at the most vanilla gives light foot (+0.25, owner's
	# ruling 2026-10-02); the rest of the edge is how fast they are ashore.
	initiative = 3
	flanking_ability = 0.25
	morale_damage_taken = 0.05

	combat = {
		coastal = 0.15
		wetlands = 0.1
	}

	country_potential = {
		has_or_had_tag = THE
	}

	gfx_tags = { light_tag }
}

# The boats of the parathalassioi, the men of the waterfront who made the
# Zealot revolt of 1342. Crowded with oarsmen and short of guns: a hull
# built to keep a lane open under blockade, not to break a battle line.
# Base template: maritime ship_small_mid_maritime, speed 4, hull 7,
# cannons 9, crew 0.06, blockade 0.2.
n_nd_the_parathalassioi_galley = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_2_renaissance_galley
	buildable = no
	levy = yes

	movement_speed = 1
	hull_size = -1
	cannons = -2
	crew_size = 0.02
	# Blockade stops at the most vanilla gives a galley (+0.2, owner's ruling
	# 2026-10-02); 0.15 here on the template's 0.2.
	blockade_capacity = 0.15
	maritime_presence = ship_medium_maritime

	country_potential = {
		has_or_had_tag = THE
	}

	gfx_tags = { mediterranean_galley_tag }
}

# Thessalian lancers. The plain around Larissa has bred cavalry since
# Philip of Macedon and its lords rode for whoever held the town. On the
# flat they are terrifying; the moment the ground rises, which in this
# country is soon, they are worse than useless.
# Base template: max_strength 0.4, combat_power 2.6.
a_nd_the_thessalian_lancers = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_2_renaissance_light_cavalry
	buildable = no
	levy = yes

	initiative = 2
	flanking_ability = 0.2
	# They ride unarmoured and they charge home, which costs them.
	strength_damage_taken = 0.05
	max_strength = 0.54	# levy power, scripts/fix_levy_power.py

	combat = {
		flatland = 0.15
		mountains = -0.1
	}

	country_potential = {
		has_or_had_tag = THE
	}

	gfx_tags = { light_tag }
}

# The Thermaic galliot. The type that ran grain into a city under siege
# from 1422 to 1430 and kept it alive for eight years. Everything is given
# to speed and to holding the lane; the armament is an afterthought,
# because a ship that stops to fight is a ship that does not arrive.
# Base template: maritime ship_medium_maritime, speed 11, hull 8,
# cannons 10, crew 0.075, blockade 1.5.
n_nd_the_thermaic_galliot = {
	is_special = yes
	category = navy_light_ship
	copy_from = n_age_3_discovery_light_ship
	upgrades_to = n_pinnace
	buildable = yes

	movement_speed = 2
	hull_size = -1
	cannons = -4
	blockade_capacity = 0.3
	maritime_presence = ship_large_maritime

	country_potential = {
		has_or_had_tag = THE
	}

	gfx_tags = { light_tag }
}
