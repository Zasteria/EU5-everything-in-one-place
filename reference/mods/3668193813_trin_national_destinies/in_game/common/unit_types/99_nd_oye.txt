# Oyo Empire: Custom Unit Types
# Cavalry-dominant empire with professional infantry guard

# Age 2 levy infantry, disciplined palace guards with high morale

	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_1_traditions_heavy_infantry
	buildable = no
	levy = yes

	# Base template: max_strength=0.5, combat_power=1
	morale_damage_done = 0.15
	morale_damage_taken = -0.1

		has_or_had_tag = OYE
	}

	gfx_tags = { medium_tag }
}

# Savanna Horse Archers (first named Oyo Mounted Archers): light cavalry scouts and skirmishers
# Age 4 buildable cavalry, fast and mobile with ranged capability. Age 3
# until 2026-10-02: vanilla's Esho Cavalry (a_esho_cavalry_oyo) is the
# age 3 light cavalry of the same country.
a_nd_oye_mounted_archers = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_4_reformation_light_cavalry
	upgrades_to = a_hussars
	buildable = yes

	# Deltas on a_age_4_reformation_light_cavalry. Initiative 3 is what
	# vanilla gives its horse archers (a_steppe_horse_archers,
	# a_tuareg_veiled_cavalry); flanking and speed are what vanilla gives
	# its own raiders (a_akinji 0.2, a_light_camelry_3 0.2). The price is the
	# horse itself: Oyo could not breed horses under the tsetse fly and
	# bought every mount from the north, so the regiment wastes away on
	# campaign twice as fast as other cavalry.
	initiative = 3
	movement_speed = 0.2
	flanking_ability = 0.2
	attrition_loss = 0.25

	country_potential = {
		has_or_had_tag = OYE
	}

	gfx_tags = { light_tag }
}
