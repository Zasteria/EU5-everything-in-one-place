# Turkmen Raider: fast nomad light cavalry, buildable
a_nd_rum_turkmen_raider = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	upgrades_to = a_cavalrymen
	buildable = yes

	# Base: max_strength=0.2, combat_power=2.6
	# Signature: nomad speed (+1, vanilla's own steppe and tribal horse) and
	# the horse archer's readiness to engage (initiative, as vanilla's steppe
	# horse archers). Flanking sits at the best of vanilla's age 1 light
	# cavalry (0.2).
	# The arrows loosed from the saddle take back half of what the light
	# cavalry category gives up in strength damage.
	# Weakness: the Turkmen fought for plunder under their own beys and
	# left the field when a battle turned against them.
	movement_speed = 1
	initiative = 3
	flanking_ability = 0.2
	strength_damage_done = 0.05
	morale_damage_taken = 0.1

	country_potential = {
		has_variable = nd_rum_formed
	}

	gfx_tags = { medium_tag }
}

# Composite Bow Archer: Turkic archery tradition, ranged infantry
# Light infantry since 2026-10-01: an unarmoured foot archer is what vanilla
# calls light infantry, and the unit had been given the initiative of that
# category by hand (4.5 on a heavy template, above any vanilla unit).
a_nd_rum_composite_archer = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_3_discovery_light_infantry
	upgrades_to = a_arquebusiers
	buildable = yes

	# Base: max_strength=1.5, combat_power=1
	# Signature: the draw weight of the composite bow (strength damage, as
	# vanilla's Andalusi arquebusiers) and its rate of fire (initiative).
	# Weakness: archers without armour or pikes are cut down once the enemy
	# closes, the reason the bow gave way to pike and shot in this age.
	strength_damage_done = 0.1
	initiative = 2
	strength_damage_taken = 0.1

	country_potential = {
		has_variable = nd_rum_formed
	}

	gfx_tags = { medium_tag }
}
