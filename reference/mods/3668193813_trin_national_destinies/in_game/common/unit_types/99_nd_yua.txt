# Khorchin Scouts: fast light cavalry from the eastern Mongol plains, age 3
# Khorchin were allied tribesmen who served as scouts and raiders
# Copies the age 3 template since 2026-10-01: the levy is unlocked in age 3
# (nd_yua_dayan_khan_revival) and had been carrying age 4 numbers.
a_nd_yua_khorchin_scout = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	buildable = no
	levy = yes

	# Base: max_strength=0.6, combat_power=2.6
	# Flanking and shock sit at the best of vanilla's age 3 light cavalry
	# (akinji 0.2, camelry 0.05).
	# Weakness: scouts ride light to ride far, and unarmoured men cannot
	# hold a line against lancers.
	movement_speed = 0.15
	flanking_ability = 0.2
	morale_damage_done = 0.05
	strength_damage_taken = 0.1
	max_strength = 0.62	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = YUA
	}

	gfx_tags = { steppe_tag }
}

# Tumen Horse: the herdsmen of the six tumen, called out by wing and banner, age 3
a_nd_yua_tumen_horse = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_3_discovery_light_cavalry
	buildable = no
	levy = yes

	# Base: max_strength=0.6, combat_power=2.6
	# Signature: steppe speed and the horse archer's readiness to engage,
	# the numbers vanilla gives its own steppe horse (+1 and +3).
	# Weakness: commoners ride to war in felt and leather, without the mail
	# of the khan's guard.
	movement_speed = 1
	initiative = 3
	strength_damage_taken = 0.1
	# Riders: every pop of a steppe horde raises vanilla's horde levy
	# (a_steppe_horde, total combat power 1.752). This many riders put the
	# unit 1.19 times over it. The levy script measures only the
	# unconditional tribesmen levy (0.469), so this number is set by hand.
	max_strength = 0.18

	country_potential = {
		has_or_had_tag = YUA
	}

	gfx_tags = { steppe_tag }
}
