# Coptic Egypt: custom units.
#
# A Coptic state has a military problem no other Egyptian formable has. The
# Mamluk army was a slave-soldier aristocracy held by land grant, religious
# bond and a household culture no Christian treasury can reproduce, and there
# was no Coptic soldiery in Mamluk Egypt at all. So every unit below is one of
# only three honest answers: a militia raised on a story, archers hired from
# the Christian kingdom upriver, or cavalry bought outright for cash.
#
# No ships. Egyptian naval identity belongs to EGY (felucca, Nile galley) and
# AEG (Bahriyya marines); a riverine minority regime that has to purchase its
# horse has no business also owning the sea.

# The Bashmurite. Levy only, never recruitable, because this is the countryside
# rising rather than a regiment the state raises. Bashmur was narrow sandy
# banks between reed thickets, supplied on fish and fowl, and it cost the
# caliph al-Ma'mun a personal expedition under al-Afshin to end it in 831.
# Five hundred years later the tradition is being invented rather than
# continued, and it shows: they own the marsh, they are worthless in the open
# desert, and they break.
a_nd_cpt_bashmurite = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry
	buildable = no
	levy = yes

	# Base age 1 light infantry: max_strength=0.5, combat_power=1
	supply_weight = -0.2
	attrition_loss = -0.15
	morale_damage_taken = 0.1
	max_strength = 0.1	# levy power, scripts/fix_levy_power.py

	combat = {
		wetlands = 0.25
		river = 0.15
		desert = -0.15
	}

	country_potential = {
		has_or_had_tag = CPT
	}

	gfx_tags = { light_tag }
}

# Nubian bowmen, the rumat al-hadaq, "shooters at the pupils of the eye". They
# broke Arab invasions in 641 and 652 and Egypt bought them for centuries after
# under the Baqt. Recruitable only on the southern frontier and in Nubia
# itself, because that is where the contingents came through: they are an
# allied levy, not your own men, and they will withdraw sooner than yours will.
a_nd_cpt_nubian_bowmen = {
	is_special = yes
	category = army_light_infantry
	copy_from = a_age_2_renaissance_light_infantry
	upgrades_to = a_early_arquebusiers
	buildable = yes

	# Base age 2 light infantry: max_strength=1.0, combat_power=1
	# Archery at vanilla's largest for light infantry (a_longbowmen); the
	# rest of what the shooters at the eye did is counted as fear.
	strength_damage_done = 0.1
	morale_damage_done = 0.1
	flanking_ability = 0.15
	morale_damage_taken = 0.1

	combat = {
		desert = 0.15
	}

	country_potential = {
		has_or_had_tag = CPT
	}

	# The Baqt contingents came up the river through Aswan and Qus. Restricting
	# recruitment to the southern frontier and to Nubia keeps them what they
	# historically were, a bought southern auxiliary, instead of a national arm
	# raisable in the Delta.
	location_potential = {
		OR = {
			area = area:upper_egypt_area
			region = region:nubia_region
		}
	}

	gfx_tags = { light_tag }
}

# Qasr wardens. Not fighting monks: the armed servants, tenants and dependent
# villagers a great monastery could put on its walls, which is exactly who the
# tower over the well and storehouse was built to shelter. They hold ground and
# outlast a siege. They do not march.
a_nd_cpt_qasr_warden = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_3_discovery_heavy_infantry
	buildable = no
	levy = yes

	# Base age 3 heavy infantry: max_strength=1.5, combat_power=1
	# Toughness at vanilla's largest for heavy infantry
	# (a_pontifical_swiss_guard); men behind their own wall hold their nerve.
	strength_damage_taken = -0.15
	morale_damage_taken = -0.25	# levy power, scripts/fix_levy_power.py
	movement_speed = -0.15
	combat_power = 0.05	# levy power, scripts/fix_levy_power.py

	country_potential = {
		has_or_had_tag = CPT
	}

	gfx_tags = { heavy_tag }
}

# The Paid Corps. Mamluk cavalry retained by a Christian treasury has no iqta'
# and no confessional bond with its paymaster: it is bought in coin from the
# diwan and it serves while the coin lasts. Excellent horse, and the first
# formation to come apart when the state looks poor.
a_nd_cpt_paid_corps = {
	is_special = yes
	category = army_heavy_cavalry
	copy_from = a_age_4_reformation_heavy_cavalry
	upgrades_to = a_gallop_cavalry
	buildable = yes

	# Base age 4 heavy cavalry: max_strength=0.8, combat_power=2.85
	# Killing power at vanilla's largest for heavy cavalry (a_banner_cavalry);
	# the shock of a mamluk charge carries the rest.
	strength_damage_done = 0.1
	morale_damage_done = 0.1
	initiative = 2
	morale_damage_taken = 0.15

	country_potential = {
		has_or_had_tag = CPT
	}

	gfx_tags = { heavy_tag }
}
