# Gap audit against vanilla, done 2026-09-10 on 1.3.11:
#   advances/country_FRA.txt, 16 advances: diplomacy and noble levies (age 1),
#     court language and elan (age 2), Estates-General, gendarmes and
#     integration (age 3), privateers, tax reform and Fontainebleau (age 4),
#     Vauban, the philosophes and expansion (age 5), revolutionary tolerance,
#     Code civil and discipline (age 6).
#   government_reforms/country_specific.txt: taxation, feudal nobility,
#     centralized monarchy, appanages, gabelle, Code Louis.
#   DHE/flavor_FRA.txt, 208 events: routiers, Du Guesclin, Joan of Arc,
#     Jean Bureau, Jacques Coeur, Richelieu, Colbertism, Versailles, Louvois,
#     Vauban, Ecole militaire, Mezieres, John Law, Jansenism, Fleury's roads,
#     the Encyclopedists, Turgot, Necker; all as timed modifiers.
# Vanilla has NO France unit, building, production method or levy (the
# gendarmes are a modifier advance; Versailles and the Louvre museum are
# event-only). Every entry below fills a gap in those systems or a theme
# vanilla never touches. The Rhine and the Empire, and the rivalry with
# England across the Channel, are left to the destiny tree.
#
# Heritage potential is has_or_had_tag = FRA only: nd_fra_formed is the
# destiny discriminator and must not appear here.

# ===== DESTINY TREE =====
# Two roads, both earned by deeds and both open for the whole campaign (user
# ruling 2026-09-10, PLAN.md section 10, the France package, item 3). nd_fra.1
# sets no path variable and nothing here reads one: every stage opens on what
# France takes or builds.
#   The Rhine and the Empire: every location lies in the areas Francia (FRK)
#     requires, so this road is also the road to the tier 4. The end of the
#     Empire stays Francia's own gate and is not asked for here.
#   The Channel and England: the English out of France, a navy to match
#     theirs, and the colonial wars of 1689 to 1815.
# Both roads can be completed, so they are balanced as a sum: no modifier key
# appears twice in the tree or its events, values sit at or under the vanilla
# advance median, and the roads feed different outcomes (the land frontier
# against the navy and the colonies).
# Location choices: Metz, Toul and Verdun start as imperial states of their
# own (MTZ, TUL, VDU), not French subjects. Calais starts with the French
# appanage of Artois but can pass to England in the Hundred Years' War
# (hundred_years_war.205.b), and Jersey starts English. Lille and Tournai
# start French and Dunkirk starts with a French vassal (Flanders), so none of
# them is an objective.

nd_fra_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = FRA has_variable = nd_fra_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = fra_expansion_of_france
	monthly_prestige = 0.1
}

# Stage 2: Alsace and the Comte (Westphalia 1648, Nijmegen 1678, Strasbourg
# 1681). Strasbourg and Colmar lie in the Empire's south_german_region.
# The two shares measure the stage's own land (user ruling 2026-09-10). They
# replace total_development >= 7000, which measured the realm instead: the
# three cities add 74 development and France with its subjects already has
# about 7,540 at the 1337 start. Subject land counts exactly as in the other
# stages: each location is handed to owns_or_non_sovereign_subject_owns
# through prev, the vanilla idiom of DHE/flavor_ser.txt:650.
#   Alsace 75% (9 of 11): nine rulers share it at start, and France held all
#     of it but Mulhouse, a Swiss ally until 1798.
#   Franche-Comte 80% (12 of 14): small and inside the France region, so a
#     formed France may hold much of it already; France took the whole
#     county in 1678 except Montbeliard, which stayed with Wurttemberg.
nd_fra_dest_alsace = {
	age = age_1_traditions
	requires = nd_fra_dest_trois_eveches
	potential = { has_or_had_tag = FRA has_variable = nd_fra_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:strasbourg
		owns_or_non_sovereign_subject_owns = location:colmar
		owns_or_non_sovereign_subject_owns = location:besancon
		custom_tooltip = {
			text = nd_fra_dest_alsace_share_tt
			area:alsace_area = {
				any_location_in_area = {
					percent >= 0.75
					root = { owns_or_non_sovereign_subject_owns = prev }
				}
			}
		}
		custom_tooltip = {
			text = nd_fra_dest_franche_comte_share_tt
			area:franche_comte_area = {
				any_location_in_area = {
					percent >= 0.8
					root = { owns_or_non_sovereign_subject_owns = prev }
				}
			}
		}
	}
	icon = pal_castles_on_rhine
	cultures_capacity = 1
	diplomatic_annexation_efficiency = 0.2
}

# === THE CHANNEL AND ENGLAND ===

# Stage 1: the English driven out (Formigny 1450, Castillon 1453, Jersey
# 1461, Calais 1558). The region test copies vanilla's own measure of English
# presence in France (hundred_years_war_unified_requirements,
# scripted_triggers/situation_triggers.txt:70) and adds Great Britain, which
# replaces the ENG tag when it forms.
nd_fra_dest_calais = {
	age = age_1_traditions
	requires = nd_fra_destiny
	potential = { has_or_had_tag = FRA has_variable = nd_fra_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:calais
		owns_or_non_sovereign_subject_owns = location:jersey
		custom_tooltip = {
			text = nd_fra_dest_no_english_in_france_tt
			region:france_region = {
				NOT = {
					any_ownable_location_in_region = {
						owner ?= {
							OR = {
								top_overlord_or_this ?= c:ENG
								top_overlord_or_this ?= c:GBR
							}
						}
					}
				}
			}
		}
	}
	icon = maritime_advance_age_2
	global_maritime_presence_modifier = 0.1
	global_sailors_modifier = 0.1
}
