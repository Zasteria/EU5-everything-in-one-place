# ==========================================================================
# DANUBIAN MONARCHY: advance tree.
#
# Covers the Habsburg/Danubian identity vanilla leaves untold:
#  - the composite-monarchy administration (Hofkammer, Hofkriegsrat,
#    Pragmatic Sanction, Theresian-Josephine reforms)
#  - the Military Frontier / Grenzer soldier-colony system
#  - the Danube riverine trade spine (Vienna staple, salt and iron)
#  - dynastic diplomacy ("bella gerant alii, tu felix Austria nube")
#  - the multi-ethnic estate balance that defined the realm
#
# Vanilla already provides (and is NOT duplicated here): the Catholic
# religion advances (tolerance, conversion, devotion, prestige, literacy),
# the Balkan/Carpathia region advances (poklisars diplomats, balkan_hajduks
# = a_hajduk, cro_pandurs_recruitment = a_pandur, expansionist_ambitions
# integration), and the HUN/CRO country advances. Story begins in the
# Renaissance: the 1526 dual inheritance of Bohemia and Hungary is when
# the composite monarchy is born, so no age 1 advances.
#
# requires: same-age vanilla attachment points only (cross-age requires
# silently makes advances invisible). All attach to always-available
# 0_age_of_* trees, never the for=adm/dip/mil specialization trees.
# ==========================================================================

# ==========================================================================
# ===== DESTINY TREE =====
#
# Root + 2 paths x 3 stages + a converged age-6 capstone fork. Gated on
# nd_dnm_formed (set in the formation event's hidden_effect) so the
# cluster is invisible until the Danubian Monarchy is actually forged.
# The two branches are the formation-event choices, NOT origin-based:
#   path_a = Antemurale Christianitatis: the bulwark of Christendom,
#            the Militaergrenze identity, the Ottoman war pressed south.
#   path_b = Gesamtmonarchie: the centralized total monarchy, the three
#            crowns welded into one administrative body from Vienna.
#
# REPACED 2026-07-31 (Item 3, dnm_proposal): the tree runs ages 2-5 with
# the capstone fork at age 6, matching the heritage cadence (the union is
# a 1526+ event; nothing sits at age_1 anymore). Cross-age progression
# lives in `allow = { has_advance = ... }`: NEVER in `requires`, which
# stays same-age only (the kor_geobukseon idiom, country_KOR.txt:125-135;
# cross-age requires silently hides the advance). Tree attachment per
# stage is a same-age advance instead of the old cross-age chain.
# Banned stat-count gates (army_size, total_development, gold piles, bare
# prestige) are replaced with deeds: real conquests, heritage doctrines
# actually adopted, named bureaucracies actually running, the Ottoman
# state actually broken. Each gate's least-legible condition is wrapped
# in its ambition line (the requirement is the advertisement).
#
# Save-compat P3: age reassignment keeps taken-state in live saves; every
# read of these advances is has_advance (state), never age-derived.
#
# Per the spec (resolution #2): the Gesamtmonarchie branch THEMATICALLY
# builds centralization/compromise capacity but does NOT create an
# "Ausgleich" / "Austria-Hungary" capstone: that beat is reserved for
# the Nationalities Question disaster's compromise exit. The age-6 fork
# instead reads that disaster's OUTCOME (nd_dnm_nq_outcome_fragment).
# ==========================================================================

nd_dnm_destiny = {
	age = age_2_renaissance
	depth = 0
	potential = { has_or_had_tag = DNM has_variable = nd_dnm_formed has_game_rule = nd_destinies_enabled }
	# Not a count: stability decays and is continuously re-bought: the
	# closest thing the engine has to an ongoing competence test.
	allow = {
		custom_tooltip = {
			text = nd_dnm_amb_root_tt
			stability >= 75
		}
	}
	icon = crown_power_advance_renaissance
	monthly_legitimacy = 0.05
	monthly_devotion = 0.05
	monthly_republican_tradition = 0.05
}

# ----- PATH A : ANTEMURALE CHRISTIANITATIS -----------------------------

# A1 : The Long Turkish War: recover the lower Danube line. The Sava-
# Danube confluence: Belgrade, Smederevo, the Vidin gate. The frontier
# pressed back to where Eugene of Savoy would draw it. Attached to the
# Militaergrenze heritage advance (same age): the Frontier system must
# actually be built before the line can be pressed: the requires IS the
# gate the proposal's table names.
nd_dnm_dest_sava_line = {
	age = age_3_discovery
	requires = nd_dnm_militaergrenze
	potential = { has_or_had_tag = DNM has_variable = nd_dnm_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_dnm_path_a_tt
			has_variable = nd_dnm_path_a
		}
		custom_tooltip = {
			text = nd_dnm_amb_sava_tt
			has_advance = nd_dnm_destiny
		}
		# Hungary holds Belgrade on day one, so holding it asks nothing of a
		# monarchy formed from Hungary. The deed is what the Habsburgs did
		# with it after 1717: the fortress rebuilt and the town resettled.
		# 25.5 in 1337, asked to grow by a quarter
		# (scripts/development_estimate.py).
		custom_tooltip = {
			text = nd_dnm_dest_belgrad_developed_tt
			location:belgrad = {
				owner ?= root
				development >= 32
			}
		}
		owns_or_non_sovereign_subject_owns = location:smederevo
	}
	icon = feudalism_advance
	army_heavy_infantry_power = 0.1
	land_unit_attrition = -0.1
}

# ----- THE RECKONING : AGE-6 CONVERGED CAPSTONE FORK -------------------
# The late fork the tree lacked (Item 3): either path's third stage opens
# both capstones; the running bureaucratic apparatus and the Nationalities
# Question's outcome decide which one the realm can honestly claim. The
# reforms they unlock are major (exclusive): the engine itself enforces
# that only one of the two ever governs. Rests on facts #1/#22/#23/#29:
# the centuries-long centralization-vs-estate-autonomy axis, from the
# Golden Bull's ius resistendi (1222) through its abolition at Pressburg
# (1687) to Joseph II's contested patents (1781-85).

# Path X : Josephine Triumph: one patent, one code, the uniform
# administrative state completed where the historical version was fought
# to a draw. A crown that already lost a constituent kingdom cannot
# credibly claim to have finished unifying the rest.
nd_dnm_dest_josephine_triumph = {
	age = age_6_revolutions
	requires = modern_bureaucracy
	potential = { has_or_had_tag = DNM has_variable = nd_dnm_formed has_game_rule = nd_destinies_enabled }
	allow = {
		OR = {
			has_advance = nd_dnm_dest_eastern_crusade
			has_advance = nd_dnm_dest_total_monarchy
		}
		custom_tooltip = {
			text = nd_dnm_amb_patent_tt
			has_bureaucracy_of_type = bureaucracy_type:nd_dnm_directorium_bureaucracy
			NOT = { has_variable = nd_dnm_nq_outcome_fragment }
		}
	}
	# fra_code_civil (imperial eagle holding the scales): the codified
	# empire. NOT modern_bureaucracy: that advance has no dds of its own
	# (it declares icon = scholasticism), so referencing it renders broken.
	icon = fra_code_civil
	unlock_government_reform = nd_dnm_uniform_patent_reform
}
