# Mallorca (MLL): advance tree
# National Destinies mod
#
# Read before designing (EU5 1.4):
#   vanilla tag advances, one per age (advances/culture_catalan.txt): the
#     Carta de Franquesa, the cartographic school of the Cresques, the
#     Mallorcan corsairs, the coastal watchtowers, the stone of Santanyi and
#     the Escola Mallorquina of poets.
#   vanilla Catalan culture advances, one per age (same file): the maritime
#     realm, fleet ordinances, the Valencian golden age, chartered merchant
#     companies, constellation cities, the Mossos d'Esquadra.
#   vanilla Iberian advances (advances/region_iberia.txt): almogavars and
#     late almogavars, Catalan crossbowmen, the Catalan galley, the tercio,
#     the caravels and galleons.
#   vanilla elsewhere: the Leges Palatinae are a law policy
#     (laws/01_legal_system.txt:496), the Pearl of the Mediterranean a
#     government reform behind a decision, Sa Llotja de Palma a monument,
#     the quarrel with Aragon, Montpellier and its university, the walls of
#     Alcudia and the Barbary raids fifteen flavour events
#     (events/DHE/flavor_mll.txt). The Consolat de Mar is an estate privilege
#     of a DLC not yet released (mll_consolat_de_mar): neither used nor
#     imitated here.
#   the mod: the Aragonese Empire (99_nd_cda.txt) has the Catalan Atlas, the
#     Drassanes and the Llotja de Mar; Catalonia (99_nd_cat.txt) has the
#     brandy still, the paper mill, the glass furnace and the woollen
#     workshop, and its destiny tree conquers Palma, Ciutadella and Ibiza;
#     Valencia (99_nd_val.txt) has the watchtower building, the snow well and
#     the sugar mill.
#
# None of that is repeated. What is left is the kingdom seen from inside:
# Llull's college of Arabic on the cliff at Miramar, the villages' own
# syndicate against the City, the armed horses owed since the Repartiment,
# the salt of Ibiza, the coinage and new towns of 1300, a galley that rowed
# for the River of Gold, the signal tower of Portopi, the contract that
# pawned the kingdom's taxes to its creditors, the lustreware Italy named
# after the island, two risings over the same debt, the Lullian school, the
# masons who built Naples a hall, the parish militia of 1561, the oil of the
# Serra, the bastioned walls, the peace between Canamunt and Canavall, the
# corsair xebecs, the harbour of Mahon, two brothers who ruled Malta, the
# last redoubt of 1715, Barcelo's gunboats, gin, the cloth of tongues, a
# friar from Petra in California and the society of 1778.
#
# No key below repeats, in the same age, a key that a vanilla MLL, Catalan
# or Iberian identity advance grants (the per-age blocklist printed by
# scripts/creator_context.py MLL). No modifier key is granted twice in the
# tree.
#
# Stepping stone toward the Aragonese Empire and Spain: 4 or 5 per age.
# 26 advances, 15 of them unlocks (58 percent).

# Ramon Llull of Ciutat de Mallorca (c. 1232 to c. 1316) and his Art, a
# method meant to prove the faith to an unbeliever by reason alone. In 1276
# the Infant James endowed a college at Miramar where thirteen Franciscans
# learned Arabic before sailing for Barbary. Unlocks the Col.legi de
# Llengues.
nd_mll_art_de_ramon_llull = {
	age = age_1_traditions
	potential = { has_or_had_tag = MLL }
	requires = scholasticism
	icon = missionary_schools

	unlock_building = nd_mll_collegi_de_llengues
	global_clergy_max_literacy = 5
}

# ===== DESTINY TREE =====
#
# Two roads out of the formation choice (nd_mll.1).
# Path A, the City: the merchants' own court (1326), a held port on the
# Barbary coast where the wax came from, and the island the Mallorcan
# voyages of 1342 to 1386 sailed for. Two conquests overseas, and Palma
# grown from its 1337 value (23.2: scripts/development_estimate.py).
# Path B, the Part Forana: nothing is conquered. The market villages, the
# two lesser islands and at last a kingdom that is nobody's subject. Every
# place it names is one Mallorca holds in 1337, so each is asked for with
# growth, a rank or a building, never by ownership alone (owner's ruling,
# 2026-10-02). 1337 values: Manacor 23.2, Pollensa 18, Ibiza 19.2,
# Ciutadella 27; raised by a quarter, a half and double by stage.
# No stage reads an institution or a late advance: both paths are open from
# the first age (scripts/destiny/stage_floors.py prints no gap).
# Not repeated from vanilla's decision mll_a_hub_of_mediterranean_trade,
# which asks for the whole Balearics at 25 development with twelve market
# levels and three cities: no stage counts cities or market levels.

nd_mll_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = MLL has_variable = nd_mll_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = town_rights_advance
	diplomatic_reputation = diplomatic_reputation_weak_bonus
}

# ----- Path A: the City -----

# Stage 1. The Consolat de Mar of Mallorca, 1326: consuls chosen by the
# merchants each Saint John's Eve, judging by the sea customs of Barcelona.
nd_mll_dest_consolat_de_mar = {
	age = age_1_traditions
	requires = nd_mll_destiny
	potential = { has_or_had_tag = MLL has_variable = nd_mll_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_mll_path_a_tt
			has_variable = nd_mll_path_a
		}
		custom_tooltip = {
			text = nd_mll_dest_palma_ciutat_tt
			location:palma = {
				owner ?= root
				development >= 29
				OR = {
					location_rank ?= location_rank:city
					location_rank ?= location_rank:megalopolis
				}
			}
		}
		gold >= 1000
		prestige >= 50
	}
	icon = port_customs
	global_trades_per_burgher = 0.15
	global_port_build_buildings_efficiency = 0.2
}
