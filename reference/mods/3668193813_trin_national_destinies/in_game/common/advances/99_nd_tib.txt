# ===== DESTINY TREE =====
# Tibet's two post-formation futures, both real historical projects.
#
# Path A (Ganden Phodrang) is the road actually taken: the Fifth Dalai Lama's
# 1642 settlement made the Gelug hierarchy the state, and Lhasa governed the
# plateau through monks, dzong magistrates and the Monlam Chenmo rather than
# through a king. Its rewards are administrative and religious reach.
#
# Path B (Dharma Kings) revives the Yarlung tsenpo: the 7th-9th century
# empire that held the Hexi corridor, sacked Chang'an in 763 and treated with
# the Tang as an equal. Its rewards are military and territorial. Note this
# is Tibet's own imperial revival: the separate TBT formable is the
# pre-Gelug Bon-Buddhist restoration and does not overlap these advances.

nd_tib_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = TIB has_variable = nd_tib_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = mandinka_warrior_spirit
	global_estate_target_satisfaction = 0.025
}

# === Path A: The Ganden Phodrang ===

nd_tib_dest_lhasa_settlement = {
	age = age_1_traditions
	requires = nd_tib_destiny
	potential = { has_or_had_tag = TIB has_variable = nd_tib_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_tib_path_a_tt
			has_variable = nd_tib_path_a
		}
		# In 1337 the government sits at Sakya (start setup, `capital =
		# sakya`) and holds Lhasa and Shigatse already, so owning them asks
		# nothing. The settlement of 1642 made Lhasa the seat of the state.
		custom_tooltip = {
			text = nd_tib_dest_lhasa_capital_tt
			capital ?= location:lhasa
		}
		# Shigatse grew around Tashilhunpo (founded 1447), the seat of the
		# second hierarch. 11.5 in 1337, raised by a quarter for a first
		# stage.
		custom_tooltip = {
			text = nd_tib_dest_shigatse_developed_tt
			location:shigatse = {
				owner ?= root
				development >= 14
			}
		}
		stability >= 75
	}
	icon = organized_religion
	# Lhasa and Shigatse under one administration is administrative reach,
	# not religious influence: that pool exists only for a religion carrying
	# `has_religious_influence` (the Christian family plus mayan and
	# hellenism), and a Tibetan-culture Tibet is Buddhist. Reach is written
	# as proximity, and the end of three centuries of sectarian war as
	# cheaper stability.
	global_distance_from_capital_speed_propagation = 0.1
	stability_investment = 0.02
}

nd_tib_dest_dzong_magistracy = {

	age = age_1_traditions

# === Path B: The Dharma Kings ===

nd_tib_dest_tsenpo_restored = {
	age = age_1_traditions
	requires = nd_tib_destiny
	potential = { has_or_had_tag = TIB has_variable = nd_tib_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_tib_path_b_tt
			has_variable = nd_tib_path_b
		}
		# Yarlung, not Lhasa: Lhasa anchors the Ganden Phodrang branch, and
		# a location shared by both makes the branch choice partly free.
		# The tsenpo line came out of the Yarlung valley, so the imperial
		# restoration starts where the dynasty did: Nedong, the seat in
		# the Yarlung valley. The main former holds Nedong on day one, so
		# the stage asks for the seat to be built: a town or larger
		# (vanilla's own shape, advances/country_yem.txt, port_of_aden).
		custom_tooltip = {
			text = nd_tib_dest_nedong_town_tt
			location:nedong = {
				owner ?= root
				OR = {
					location_rank ?= location_rank:town
					location_rank ?= location_rank:city
					location_rank ?= location_rank:megalopolis
				}
			}
		}
		owns = location:dartsedo
		# Tibet starts as a vassal of the Yuan (12_diplomacy.txt). A tsenpo
		# has no overlord. No army_size here: the plateau's army is a levy.
		is_subject = no
	}
	icon = horse_riding_advance
	land_morale_recovery = 0.04
	regiment_recruit_speed = 0.1
}
