# Path A: The Golden Road (trade/scholarship, extends "Golden Throne of Niani")

nd_mal_dest_crossroads = {
	age = age_1_traditions
	requires = nd_mal_destiny
	potential = { has_or_had_tag = MAL has_variable = nd_mal_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_mal_path_a_tt
			has_variable = nd_mal_path_a
		}
		# Both cities are held by vassals on day one (Timbuktu by GHA, Djenne
		# by BMR), so plain ownership asks nothing. The stage asks for the
		# Mansa's own rule and for the growth history records: Mansa Musa took
		# Timbuktu in 1324 and built the Djinguereber there; Djenne is the
		# river market that fed it. 1337 values 10 and 12
		# (development_estimate.py), each +25%.
		custom_tooltip = {
			text = nd_mal_dest_timbuktu_developed_tt
			location:timbuktu = {
				owner ?= root
				development >= 13
			}
		}
		custom_tooltip = {
			text = nd_mal_dest_djenne_developed_tt
			location:djenne = {
				owner ?= root
				development >= 15
			}
		}
		gold >= 1000
	}
	icon = merchants_and_trade
	prestige_decay = -0.001
	global_build_buildings_efficiency = 0.1
}

nd_mal_dest_salt_gold = {
	age = age_1_traditions
	requires = nd_mal_dest_crossroads
	potential = { has_or_had_tag = MAL has_variable = nd_mal_formed has_game_rule = nd_destinies_enabled }
	allow = {
		# Taghaza is a vassal's on day one (GHA): direct ownership is the deed.
		owns = location:taghaza
		has_embraced_institution = institution:banking
		gold >= 2000
	}
	icon = global_trade_advance
	bank_interest = -0.01
	spy_network_construction = 0.15
}

nd_mal_dest_empire_gold = {
	age = age_1_traditions
	requires = nd_mal_dest_salt_gold
	potential = { has_or_had_tag = MAL has_variable = nd_mal_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:sijilmasa
		prestige >= 75
		# Niani, the capital Mansa Musa rebuilt after the pilgrimage (the
		# audience hall of al-Sahili). 1337 value 18 (development_estimate.py),
		# +100%.
		custom_tooltip = {
			text = nd_mal_dest_niani_developed_tt
			location:niani = {
				owner ?= root
				development >= 36
			}
		}
		gold >= 3000
	}
	icon = imperial_ambitions
	unlock_building = nd_mal_house_of_gold
	global_construction_speed = 0.1
	# The golden chain itself: what the trade pays and what the gold dust
	# mints. Vanilla's advances grant 0.1 of each.
	trade_income = 0.1
	minting_income_factor = 0.1
}
