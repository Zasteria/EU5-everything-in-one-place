# duchy, the watered plain, Borromean Milan, Austrian reform, and the
# Napoleonic Kingdom of Italy crowned with the Iron Crown.
# Vanilla already covers (do not duplicate): lombard culture advances
# (milanese_tradition, patron_of_music, age_of_condottieri,
# merchant_princes, lombard_silk_masters, lombard_the_manifattura),
# region italy (great_families, italian_condottieri), catholic set.
# Silk and cloth output are therefore deliberately absent here.

# ===== DESTINY TREE =====
# Two memories of the regnum Italicum, extended into destinies.
# Path A, the Iron Crown: the Visconti road completed, from the
# Scaliger wars through Gian Galeazzo's encirclement of Florence to
# the coronation no Visconti lived to hold.
# Path B, the Ambrosian Commune: the League's road, civic finance,
# a gate to the sea at Genoa, and the golden republic made to last.

nd_lmb_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = LMB has_variable = nd_lmb_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	country_cabinet_efficiency = 0.05
}

# Path A stage 1: break the della Scala encirclement, as Milan did in
# the wars of the 1330s and again in 1387.
nd_lmb_dest_eredita_scaligera = {
	age = age_1_traditions
	requires = nd_lmb_destiny
	potential = { has_or_had_tag = LMB has_variable = nd_lmb_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_lmb_path_a_tt
			has_variable = nd_lmb_path_a
		}
		stability >= 75
		owns_or_non_sovereign_subject_owns = location:verona
		owns_or_non_sovereign_subject_owns = location:parma
		army_size >= 30
	}
	icon = imperial_ambitions
	military_tactics = 0.1
	global_manpower_modifier = 0.1
}

# Path B capstone: the Aurea Repubblica Ambrosiana of 1447 earned its
# name for three desperate years; this one earns it for good.
nd_lmb_dest_repubblica_aurea = {
	age = age_1_traditions
	requires = nd_lmb_dest_porta_del_mare
	potential = { has_or_had_tag = LMB has_variable = nd_lmb_formed has_game_rule = nd_destinies_enabled }
	allow = {
		# Milan itself, the city the republic of 1447 was proclaimed in and
		# the Sforza then built up (the Ospedale Maggiore of 1456, the
		# Naviglio della Martesana): 36 in 1337, asked to double
		# (scripts/development_estimate.py).
		custom_tooltip = {
			text = nd_lmb_dest_milano_developed_tt
			location:milano = {
				owner ?= root
				development >= 72
			}
		}
		# The republic of 1447 was proclaimed by jurists of the University of
		# Pavia: a humanist commonwealth. Global trade stood here and held
		# this path three ages behind the other.
		has_embraced_institution = institution:renaissance
		gold >= 3000
	}
	icon = burghers_rights_laws_advance
	unlock_building = nd_lmb_broletto
	global_production_efficiency = 0.05
	global_monthly_literacy = 0.01
	global_estate_target_satisfaction = 0.025
}
