nd_hab_dest_burgundia = {
	age = age_1_traditions
	requires = nd_hab_dest_corona
	potential = { has_or_had_tag = HAB has_variable = nd_hab_formed has_game_rule = nd_destinies_enabled }
	# Antwerp is the Habsburg Netherlands' own boom: about 40,000 people when
	# Philip the Fair inherited, about 100,000 under Charles V, the staple of
	# Portuguese spice and English cloth. Its 1337 value is 32.8
	# (scripts/development_estimate.py --explain antwerp); stage two asks
	# for half as much again. Ghent, where Charles V was born in 1500, stands
	# for the cities of Flanders the stage's text names.
	allow = {
		custom_tooltip = {
			text = nd_hab_dest_antwerp_developed_tt
			location:antwerp = {
				owner ?= root
				development >= 49
			}
		}
		owns = location:ghent
		gold >= 2000
	}
	icon = merchants_and_trade
	trade_land_efficiency = 0.1

	trade_sea_efficiency = 0.1

	global_monthly_prosperity = 0.001
}

nd_hab_dest_monarchia = {
