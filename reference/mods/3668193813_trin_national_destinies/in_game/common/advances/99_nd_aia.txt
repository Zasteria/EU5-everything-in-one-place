nd_aia_dest_monsoon = {
	age = age_1_traditions
	requires = nd_aia_dest_gulf
	potential = { has_or_had_tag = AIA has_variable = nd_aia_formed has_game_rule = nd_destinies_enabled }
	allow = {
		has_embraced_institution = institution:banking
		any_market_present_in_country = {
			OR = {
				is_produced_in_market = goods:incense
				is_traded_in_market = goods:incense
			}
		}
		gold >= 2000
		# From the 1420s the India fleets came in on the monsoon to Jeddah
		# instead of Aden, and the town grew on their customs. 7.2 in 1337,
		# grown by half.
		custom_tooltip = {
			text = nd_aia_dest_jeddah_developed_tt
			location:jeddah = {
				owner ?= root
				development >= 11
			}
		}
	}
	icon = global_trade_advance
	global_institution_growth_modifier = 0.15
	global_monthly_development = 0.001
}

nd_aia_dest_crossroads = {
	age = age_1_traditions
	requires = nd_aia_dest_monsoon
	potential = { has_or_had_tag = AIA has_variable = nd_aia_formed has_game_rule = nd_destinies_enabled }
	allow = {
		# Aden is the Yemeni former's own on day one, so the stage asks for
		# what is done there: 14 in 1337, doubled.
		custom_tooltip = {
			text = nd_aia_dest_adan_developed_tt
			location:adan = {
				owner ?= root
				development >= 28
			}
		}
		# The pepper of Malabar, landed at Aden and carried on to Egypt, is
		# what made Arabia the hinge of the trade. This stands where the
		# institution global_trade stood: that one waits for the fourth age,
		# while the other path can be finished in the first (owner's ruling
		# 2026-10-02, scripts/destiny/stage_floors.py).
		any_market_present_in_country = {
			OR = {
				is_produced_in_market = goods:pepper
				is_traded_in_market = goods:pepper
			}
		}
		gold >= 3000
		prestige >= 75
	}
	icon = maritime_advance_age_5
	trade_income = 0.1
	global_population_growth = 0.001
	navy_light_ship_power = 0.1
	unlock_building = nd_aia_trade_exchange
}
