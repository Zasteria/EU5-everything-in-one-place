# Thessalonica: heritage advances.
#
# Vanilla already ships six THE advances in frankokratia.txt (the crown's
# tax right, the Fair of Saint Demetrios, the Lombard Guard, the Via
# Egnatia, the Macedonian fur trade, the University). None of them is
# repeated here and none of their modifier keys is reused in the same age.
#
# What vanilla leaves untouched is everything that made this city famous:
# it governed itself. The Zealots held Thessalonica as a commune from 1342
# to 1350, expelled its aristocracy, and shut its own archbishop out of his
# see. A Latin crown over a Greek city is the whole design problem, and the
# tree is built around the three bodies that claimed the place: the
# Lombard baronage of Boniface's knights, the guilds and dockhands of the
# harbour, and the Orthodox church with the Holy Mountain behind it. The
# hinterland supplies the rest: the gold and silver of Chalkidiki, the
# wool and grain of the Macedonian plain, the horse country of Thessaly.
#
# Chain position: Thessalonica is a stepping stone (Greece at level 2,
# the Latin Empire at 3, Byzantium at 4 all sit above it), so density is
# held at the low end, 4 to 6 per age.

# ===== DESTINY TREE =====
#
# Two of the three formation choices carry a destiny. The barons and the
# commune are the two ends of the same argument, who holds the city and on
# what terms, so they take the two branches. A crown that ruled through the
# church has its concordat and its heritage tree and takes neither branch:
# that choice was about the altar, not about the state's direction.
#
# Scope. This is a kingdom of six provinces and it stays one. Path A reaches
# for the frontier that killed the first Kingdom of Thessalonica in 1207 and
# 1224, and then for the homage its barons once did at Ravennika. Path B
# reaches for three islands and the money that passes through them. Neither
# is a Balkan empire and neither should be.

# The Zealot decade is over. Whichever body the crown ruled through, it is
# still here, and a state that survived its own capital now has a direction
# to choose.
nd_the_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = THE has_variable = nd_the_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	# A realm that came through the Zealot decade is harder to shake a second
	# time. 0.1 is vanilla's typical advance value for this key.
	stability_cost_efficiency = 0.1
}

# === Path A: the fief and the sword ===
# The Lombard programme. Enfeoff the marches, hold the passes that killed
# Boniface, answer Epirus, and make the lordships of Frankish Greece do
# homage again for what they hold.

# Prosek on its cliff above the Vardar and Melnik in its ravine in the
# Pirin: the two rocks that hemmed the first kingdom in, held by Dobromir
# Strez and by Alexius Slav, and neither of them reachable from the sea.
nd_the_dest_marches = {
	age = age_1_traditions
	requires = nd_the_destiny
	potential = { has_or_had_tag = THE has_variable = nd_the_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_the_path_a_tt
			has_variable = nd_the_path_a
		}
		stability >= 75
		owns_or_non_sovereign_subject_owns = location:prosek
		owns_or_non_sovereign_subject_owns = location:melnik
		# Strumica lies in the valley between the two rocks. (The army size
		# objectives of this path went 2026-10-02: the kingdom fights with a
		# feudal levy, and no file says whether raised levies count.)
		owns_or_non_sovereign_subject_owns = location:strumica
	}
	icon = hai_frontier_march
	global_manpower_modifier = 0.1
	army_heavy_cavalry_power = 0.1
	monthly_towards_aristocracy = 0.05
}

nd_the_dest_epirus = {

# === Path B: the harbour and the ledger ===
# The commune's programme. The crown's writ in its own capital is a
# negotiated thing, so the men of the waterfront spend the difference on
# what a harbour is actually for: reach, credit and a share of every cargo
# that crosses the Aegean.

# The dye vats of the wool halls run on alum, and the nearest alum in the
# world is the Lemnian earth dug on that island every August under the eye
# of a priest. The fair has to draw the eastern goods too, or it is only a
# regional market with a saint's day.
nd_the_dest_lemnos = {
	age = age_1_traditions
	requires = nd_the_destiny
	potential = { has_or_had_tag = THE has_variable = nd_the_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_the_path_b_tt
			has_variable = nd_the_path_b
		}
		stability >= 75
		owns_or_non_sovereign_subject_owns = location:lemnos
		gold >= 1000
		capital = {
			market = {
				OR = {
					is_produced_in_market = goods:silk
					is_traded_in_market = goods:silk
				}
			}
		}
	}
	icon = overseas_trade
	# Lemnos is an alum location on the map (location_templates.txt).
	global_alum_output_modifier = 0.1
	monthly_towards_plutocracy = 0.05
}
