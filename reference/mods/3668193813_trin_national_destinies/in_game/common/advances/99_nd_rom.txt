# ===== DESTINY TREE =====

nd_rom_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = ROM has_variable = nd_rom_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	prestige_decay = -0.001
}

# === PATH A: IMPERIUM ROMANUM RESTITUTUM (Iron Eagles) ===

nd_rom_dest_pax_imposita = {
	age = age_1_traditions
	requires = nd_rom_destiny
	potential = { has_or_had_tag = ROM has_variable = nd_rom_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_rom_path_a_tt
			has_variable = nd_rom_path_a
		}
		owns_or_non_sovereign_subject_owns = location:mecca
		owns_or_non_sovereign_subject_owns = location:cologne
		army_size >= 30
	}
	icon = feudalism_advance
	army_light_infantry_power = 0.1

	army_heavy_infantry_power = 0.1

	global_supply_limit_modifier = 0.075
}

nd_rom_dest_limes_ultimus = {
	age = age_1_traditions
	requires = nd_rom_dest_pax_imposita
	potential = { has_or_had_tag = ROM has_variable = nd_rom_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:moscow
		owns_or_non_sovereign_subject_owns = location:isfahan
		army_size >= 50
		has_embraced_institution = institution:professional_armies
	}
	icon = imperial_ambitions
	military_tactics = 0.1
	land_unit_attrition = -0.1
}

nd_rom_dest_orbis_terrarum = {
	age = age_1_traditions
	requires = nd_rom_dest_limes_ultimus
	potential = { has_or_had_tag = ROM has_variable = nd_rom_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:samarkand
		owns_or_non_sovereign_subject_owns = location:dublin
		owns_or_non_sovereign_subject_owns = location:novgorod
		army_size >= 75
		prestige >= 75
		# The legion drilled anew in pike and shot (an institution of the
		# third age). Without it this path could be finished in the second age
		# while the other waits for the fifth (ruling of 2026-10-02: both
		# paths end within two ages of each other, scripts/destiny/stage_floors.py).
		has_embraced_institution = institution:pike_and_shot
	}
	icon = imperial_ambitions
	unlock_building = nd_rom_castra_aeterna
	expected_army_size_modifier = 0.15
	army_movement_speed = 0.1
	global_hostile_attrition = 0.5
}

# === PATH B: LUX AETERNA (The Eternal Light) ===

nd_rom_dest_renovatio_studii = {
	age = age_1_traditions
	requires = nd_rom_destiny
	potential = { has_or_had_tag = ROM has_variable = nd_rom_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_rom_path_b_tt
			has_variable = nd_rom_path_b
		}
		has_embraced_institution = institution:banking
		gold >= 1000
		any_market_present_in_country = {
			OR = {
				is_produced_in_market = goods:silk
				is_traded_in_market = goods:silk
			}
		}
	}
	icon = organized_religion
	global_monthly_literacy = 0.01
	global_construction_speed = 0.1
}

nd_rom_dest_mare_mercator = {
	age = age_1_traditions
	requires = nd_rom_dest_renovatio_studii
	potential = { has_or_had_tag = ROM has_variable = nd_rom_formed has_game_rule = nd_destinies_enabled }
	allow = {
		has_embraced_institution = institution:printing_press
		# The two great harbours of the Roman sea, each grown by half from its
		# 1337 value (scripts/development_estimate.py --explain): Alexandria, the
		# emporium of the East and port of the grain fleet (39), and Carthage,
		# which Augustus refounded and which became the second city of the Latin
		# West (the location is Tunis, 31).
		custom_tooltip = {
			text = nd_rom_dest_alexandria_developed_tt
			location:alexandria = {
				owner ?= root
				development >= 59
			}
		}
		custom_tooltip = {
			text = nd_rom_dest_carthage_developed_tt
			location:tunis = {
				owner ?= root
				development >= 47
			}
		}
		gold >= 2000
	}
	icon = merchants_and_trade
	trade_land_efficiency = 0.2

	trade_sea_efficiency = 0.2

	global_population_growth = 0.00025
}

nd_rom_dest_caput_mundi = {
	age = age_1_traditions
	requires = nd_rom_dest_mare_mercator
	potential = { has_or_had_tag = ROM has_variable = nd_rom_formed has_game_rule = nd_destinies_enabled }
	allow = {
		has_embraced_institution = institution:scientific_revolution
		# Rome itself: a town among ruins in 1337, the popes gone to Avignon
		# (31.5 by the estimate), to be doubled.
		custom_tooltip = {
			text = nd_rom_dest_rome_developed_tt
			location:rome = {
				owner ?= root
				development >= 63
			}
		}
		gold >= 3000
		prestige >= 75
	}
	icon = global_trade_advance
	unlock_building = nd_rom_forum_universale
	diplomatic_capacity = 1
	global_raw_material_output = 0.1
	monthly_legitimacy = 0.1
	monthly_devotion = 0.1
	monthly_republican_tradition = 0.1
}
