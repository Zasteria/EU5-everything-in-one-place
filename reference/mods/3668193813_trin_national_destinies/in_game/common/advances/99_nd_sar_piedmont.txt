# Sardinia-Piedmont: a crown assembled out of an Alpine duchy and a
# Mediterranean island, which never became one country and governed as two.
# The tree follows that seam: the island's own law and grain banks on one
# side, the passes, the citadel and the silk mills on the other, and the
# Waldensian valleys that neither half could digest running through both.
#
# SAR (Sardinia) and ITA (Italy) are separate tags with separate trees. This
# one deliberately avoids their ground: no Carabinieri, no Bersaglieri, no
# Risorgimento, no coral, no Nuraghi.

# ===== DESTINY TREE =====

# What a crown assembled by treaty rests on. Path A takes the ground as the
# instrument: hold the mouths of the passes from the far side, buy a quay, and
# end up holding the whole Alpine arc from Geneva to Genoa. Path B takes the
# law as the instrument: measure the realm, register a third people, and put
# every subject under one bench.
#
# The third formation choice, the Palaiologos inheritance, deliberately has no
# destiny branch. It is a flavour route for a Montferrat-descended crown, not
# a national programme.

nd_sar_piedmont_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = SRP has_variable = nd_sar_piedmont_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	prestige_decay = -0.001
}

# --- Path A: Hold the Doors of Italy ---

# the Montgenevre from the Dauphine slope, Martigny the Great St Bernard from

	global_supply_limit_modifier = 0.1

# --- Path B: Two Codes, One Registry, One Crown ---

# From 1697 the crown measured Piedmont field by field, the first general
# cadastre any European state completed, and then carried the measure out over
# the plain of Pavia, where the Lombard law had been written down first.
nd_sar_piedmont_dest_perequazione = {
	age = age_1_traditions
	requires = nd_sar_piedmont_destiny
	potential = { has_or_had_tag = SRP has_variable = nd_sar_piedmont_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_sar_piedmont_path_b_tt
			has_variable = nd_sar_piedmont_path_b
		}
		owns_or_non_sovereign_subject_owns = location:pavia
		has_embraced_institution = institution:printing_press
		# Turin, which the dukes who ordered the measure also enlarged three
		# times by plan (1620, 1673, 1714): 33 in 1337 (development_estimate.py
		# --explain turin), a quarter more for a first stage.
		custom_tooltip = {
			text = nd_sar_piedmont_dest_turin_developed_tt
			location:turin = {
				owner ?= root
				development >= 41
			}
		}
	}
	icon = court_accounting
	tax_income_efficiency = 0.03
	global_max_rural_control = 0.03
}

# Bologna taught Roman law to Europe and Padua taught it to Venice. A crown
# that wants one bench over many peoples has to hire the men who train the
# judges, which means holding the schools that train them.
nd_sar_piedmont_dest_one_bench = {
	age = age_1_traditions
	requires = nd_sar_piedmont_dest_third_custom
	potential = { has_or_had_tag = SRP has_variable = nd_sar_piedmont_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:bologna
		owns_or_non_sovereign_subject_owns = location:padova
		# Four sovereign courts sitting, as Savoy's own four did (Turin,
		# Chambery, Nice and the Reale Udienza at Cagliari). The Senato comes
		# with nd_sar_piedmont_statuta_sabaudiae (age 2). The Enlightenment
		# stood here and kept this path three ages behind the other.
		any_owned_location = {
			count >= 4
			has_building = building_type:nd_sar_piedmont_senato
		}
		# Cagliari, seat of the island's own bench: the Reale Udienza sat
		# there, and the university was refounded there in 1764. 23 in 1337
		# (development_estimate.py --explain cagliari), doubled for a capstone.
		custom_tooltip = {
			text = nd_sar_piedmont_dest_cagliari_developed_tt
			location:cagliari = {
				owner ?= root
				development >= 46
			}
		}
	}
	icon = vid_the_codex_shishman
	unlock_building = nd_sar_piedmont_archivio
	research_speed_modifier = 0.05
	government_size = 1
	global_estate_target_satisfaction = 0.025
}
