nd_rus_dest_siberian_wealth = {
	age = age_1_traditions
	requires = nd_rus_dest_eastern_road
	potential = { has_or_had_tag = RUS has_variable = nd_rus_formed has_game_rule = nd_destinies_enabled }
	allow = {
		has_embraced_institution = institution:printing_press
		# Tobolsk, the capital of Siberia from 1590, grown from a stockade
		# into a town. A rank and not a development figure: the land beyond the
		# Urals starts at zero, so a share of its 1337 value means nothing.
		# Vanilla's own form (advances/country_yem.txt, port_of_aden).
		custom_tooltip = {
			text = nd_rus_dest_tobolsk_town_tt
			location:tobolsk = {
				owner ?= root
				OR = {
					location_rank ?= location_rank:town
					location_rank ?= location_rank:city
					location_rank ?= location_rank:megalopolis
				}
			}
		}
		any_market_present_in_country = {
			OR = {
				is_produced_in_market = goods:fur
				is_traded_in_market = goods:fur
			}
		}
	}
	icon = global_trade_advance
	global_merchant_capacity_modifier = 0.1
	colonial_maintenance_efficiency = 0.1
}
