	age = age_1_traditions
	requires = nd_spa_dest_pacific_link
	potential = { has_or_had_tag = SPA has_variable = nd_spa_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:qusqu
		custom_tooltip = {
			text = nd_spa_dest_sevilla_developed_tt
			location:sevilla = {
				owner ?= root
				development >= 75
			}
		}
		# The mints are an age 4 building of the country's own. The stage asked
		# for manufactories (age 5) while the other path ends in age 2.
		any_owned_location = {
			count >= 3
			has_building = building_type:nd_spa_silver_minting_house
		}
	}
	icon = maritime_advance_age_5
	unlock_building = nd_spa_real_audiencia
	cultural_influence_modifier = 0.1
	prestige_decay = -0.001
	global_population_growth = 0.00025
}
