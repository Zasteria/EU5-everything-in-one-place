# === Path A: Hussite Crusade ===
# The Hussites defeated five crusades, then launched the "Beautiful Rides":
# offensive campaigns into Saxony, Brandenburg, and Silesia. This path extends
# that military-religious expansion into a permanent conquest of Bohemia's neighbors.

nd_boh_dest_spanile_jizdy = {
	age = age_1_traditions
	requires = nd_boh_destiny
	potential = { has_or_had_tag = BOH has_variable = nd_boh_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_boh_path_a_tt
			has_variable = nd_boh_path_a
		}
		owns = location:wroclaw
		# Zittau, not Gorlitz: Bohemia holds Gorlitz on day one
		# (validate_destiny.py, owned_at_start), while Zittau, the richest
		# town of the Lusatian League, stands under the Duke of Jawor.
		owns = location:zittau
		stability >= 75
	}
	icon = feudalism_advance
	# The rides lived off what they carried home, and the wagon columns
	# covered ground no feudal host could (0.33 and 0.1 are vanilla's
	# commonest advance values for these keys).
	amount_looted_modifier = 0.33
	army_movement_speed = 0.1
}

nd_boh_dest_bohemian_hegemony = {
	age = age_1_traditions
	requires = nd_boh_dest_spanile_jizdy
	potential = { has_or_had_tag = BOH has_variable = nd_boh_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns = location:dresden
		owns = location:meissen
		# The ride of 1429 and 1430 went down the Elbe and across the whole
		# Wettin land to the gates of Leipzig and through the Vogtland:
		# half of Upper Saxony (49 locations) and not its two capitals alone.
		custom_tooltip = {
			text = nd_boh_dest_upper_saxony_share_tt
			area:upper_saxony_area = {
				any_location_in_area = {
					percent >= 0.5
					owner ?= root
				}
			}
		}
		gold >= 2000
	}
	icon = imperial_ambitions
	# The wagon fort as a way of fighting, carried abroad.
	military_tactics = 0.1
	siege_ability = 0.1
}

nd_boh_dest_warriors_of_god = {
	age = age_1_traditions
	requires = nd_boh_dest_bohemian_hegemony
	potential = { has_or_had_tag = BOH has_variable = nd_boh_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns = location:nuremberg
		owns = location:berlin
		owns = location:krakow

		# bought the field armies off: half of Franconia (38 locations).
		custom_tooltip = {
			text = nd_boh_dest_franconia_share_tt
			area:franconia_area = {
				any_location_in_area = {
					percent >= 0.5
					owner ?= root
				}
			}
		}
		prestige >= 75
	}
	icon = pike_square
	unlock_building = nd_boh_hussite_arsenal

	# The field armies of Tabor and the Orphans stayed under arms for
	# years, so what they had learned was not lost between campaigns, and

	# value for the decay key. Not monthly_army_tradition: every Catholic
	# country's crusading_knights grants that in the same age.)
	army_tradition_decay = -0.001
	global_war_score_efficiency = 0.1
}

# === Path B: Crown of the Empire ===
# Charles IV made Prague the seat of the Holy Roman Empire and a cultural beacon.
# This path extends Bohemia's imperial-cultural role through institutional embrace,
# scholarly prestige, economic development, and control of key imperial cities.

nd_boh_dest_carolinum = {
	age = age_1_traditions
	requires = nd_boh_destiny
	potential = { has_or_had_tag = BOH has_variable = nd_boh_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_boh_path_b_tt
			has_variable = nd_boh_path_b
		}
		has_embraced_institution = institution:banking
		gold >= 1000
		stability >= 75
	}
	icon = patron_of_art
	global_institution_growth_modifier = 0.15
	# The masters of the university were clerics (5 is vanilla's commonest
	# advance value for this key).
	global_clergy_max_literacy = 5
}

nd_boh_dest_imperial_capital = {
	age = age_1_traditions
	requires = nd_boh_dest_carolinum
	potential = { has_or_had_tag = BOH has_variable = nd_boh_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns = location:vienna
		prestige >= 50
		# Kutna Hora's silver paid for the court at Prague: the mint in the
		# Italian Court struck the Prague groschen from 1300, and the town
		# grew into the second city of the kingdom. 1337 value 21
		# (scripts/development_estimate.py --explain kutna_hora), raised by
		# half for a second stage.
		custom_tooltip = {
			text = nd_boh_dest_kutna_hora_developed_tt
			location:kutna_hora = {
				owner ?= root
				development >= 32
			}
		}
		gold >= 2000
	}
	icon = vibrant_court_advance
	global_monthly_development = 0.001
	# An imperial court draws painters, builders and stargazers, as
	# Charles IV's and Rudolf II's did (1 is the one value vanilla's
	# advances write for this key).
	num_possible_artists = 1
}

nd_boh_dest_praga_caput = {

	allow = {

	skill_of_new_artists = 0.1
	monthly_diplomats = 0.1
}
