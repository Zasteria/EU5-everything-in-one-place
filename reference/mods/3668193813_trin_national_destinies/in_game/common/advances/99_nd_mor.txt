nd_mor_dest_two_oceans = {
	age = age_1_traditions
	requires = nd_mor_dest_atlantic
	potential = { has_or_had_tag = MOR has_variable = nd_mor_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:funchal
		# Mogador, the Atlantic port Sidi Mohammed ben Abdallah laid out in
		# 1764 as the Sultan's own harbour. 1337 value 16.2
		# (development_estimate.py), +100%.
		custom_tooltip = {
			text = nd_mor_dest_mogador_developed_tt
			location:mogador = {
				owner ?= root
				development >= 32
			}
		}
		# The ocean routes themselves: the institution of age 3. It stood at
		# global_trade (age 4), three ages after the other path could end
		# (owner's ruling 2026-10-02).
		has_embraced_institution = institution:new_world
	}
	icon = maritime_advance_age_5
	unlock_building = nd_mor_dar_al_bahr
	ship_build_speed = -0.1
	colonial_maintenance_efficiency = 0.1
	navy_heavy_ship_power = 0.1
}
