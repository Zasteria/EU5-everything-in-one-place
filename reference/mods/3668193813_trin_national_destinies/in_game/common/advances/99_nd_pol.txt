	potential = { has_or_had_tag = POL has_variable = nd_pol_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:vilnius
		has_embraced_institution = institution:global_trade
		# Warsaw: incorporated with Mazovia in 1526, seat of the Sejm from 1569,
		# of the royal elections from 1573, of the court from 1596. 21 in 1337
		# (scripts/development_estimate.py --explain warsaw), doubled for a capstone.
		custom_tooltip = {
			text = nd_pol_dest_warsaw_developed_tt
			location:warsaw = {
				owner ?= root
				development >= 42
			}
		}
		prestige >= 75
	}
	icon = global_trade_advance
	cultures_capacity = 1
	research_speed_modifier = 0.05
	diplomatic_capacity_modifier = 0.1
	unlock_building = nd_pol_trybunaly
}
