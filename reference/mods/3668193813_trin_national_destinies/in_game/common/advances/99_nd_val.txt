# Valencia (VAL): advance tree
# National Destinies mod
#
# Read before designing (EU5 1.4):
#   vanilla tag advances, one per age (advances/culture_catalan.txt): the
#     Llibre del Repartiment, the first printing press, the Germanies as
#     brotherhoods, Valencian silk and its Col.legi de l'Art Major, the Real
#     Societat Economica, the Renaixenca.
#   vanilla Catalan culture advances, one per age (same file): the maritime
#     realm, fleet ordinances, the Valencian golden age (Ausias March, Tirant
#     lo Blanch), chartered merchant companies, constellation cities, the
#     Mossos d'Esquadra.
#   vanilla Iberian advances (advances/region_iberia.txt): almogavars and
#     late almogavars, Catalan crossbowmen, the Catalan galley, the tercio.
#   vanilla elsewhere: the Furs of Valencia are a town right
#     (town_rights/15_iberia.txt:216), the Consulate of the Sea a law and a
#     town right, the Llotja de la Seda an event building (flavor_ara.106),
#     the University of Valencia, Vicent Ferrer, Ausias March and Tirant lo
#     Blanch are Aragon flavour events, the War of the Union a disaster.
#   the mod: the Aragonese Empire (99_nd_cda.txt) has the Almudin of
#     Valencia, the Consolat de Mar, the Drassanes and the Llotja; Catalonia
#     (99_nd_cat.txt) has the municipal Taula de Canvi, the paper mill and
#     the glass furnace, and its destiny tree swears the Furs; Al-Andalus has
#     the acequia and a lustreware kiln of its own.
#
# None of that is repeated. What is left is the kingdom at ground level: the
# irrigators who judge one another at the cathedral door, the order that
# inherited the Temple's castles, the Muslim villages the lords would not
# give up, the paper of Xativa, the board that kept the capital's walls, the
# hundred crossbowmen of the banner, the gold lustre of Manises, a madhouse
# founded by merchants, two popes from one town, the sugar of Gandia, the
# man who paid for Columbus, the towers and riders of the coast, the charters
# that refilled the emptied villages, the painters, the snow wells, the
# peasants who rose for the Archduke, the faience of l'Alcora, the hemp of
# the Plana, the rice and its fevers, the mathematicians, the paper on a cane
# that declared war on Napoleon, and the mills of Alcoi.
#
# No key below repeats, in the same age, a key that a vanilla VAL, Catalan,
# Andorran or Iberian identity advance grants (the per-age blocklist printed
# by scripts/creator_context.py VAL).
#
# Stepping stone toward the Aragonese Empire and Spain: 4 or 5 per age.
# 26 advances, 17 of them unlocks (65 percent).

# Every Thursday the syndics of the canals of the Horta sit outside the
# Apostles' Door of the cathedral and judge water disputes by word of mouth,

# "as was of old the custom in the time of the Saracens" (1239). Unlocks the

# negative value is the benefit.

	age = age_1_traditions

	requires = agriculture_advance
	icon = valley_irrigation

	expand_rgo_farming_cost_modifier = -0.1
}

# ===== DESTINY TREE =====
#
# Root, then two paths of three stages, chosen in the formation event.
# Path A (the king's towns under the Furs) is the capital's road: its supply
# from Castile, the galleys it armed against Barbary in 1398, and the silk
# that made it the largest city of the Crown. One conquest per stage at most;
# the rest is the growth of named towns, institutions and workshops.
# Path B (the Alfonsine jurisdiction) is the lords' road south: the
# marquisate of Villena, the kingdom of Murcia James II took in 1296, and the
# sixth of Granada promised to the Crown in 1308.
# No stage names a place of the Valencia area by ownership alone: a Valencia
# that was released holds all of it from the first day.
# Floors (scripts/destiny/stage_floors.py): A ends in age 3 (printing press),
# B ends in age 3 (the trapig is an age 3 building).

nd_val_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = VAL has_variable = nd_val_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = estates_general

	parliament_base_support = 0.05
}

# --- Path A: the king's towns ---

# Requena was the Castilian dry port on the road to the capital (customs on
# livestock, wool and wheat granted in 1264), and stayed Castilian until 1851.
# Valencia 36.5 in 1337 (development_estimate.py), raised by a quarter.
nd_val_dest_cami_de_requena = {
	age = age_1_traditions
	requires = nd_val_destiny
	potential = { has_or_had_tag = VAL has_variable = nd_val_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_val_path_a_tt
			has_variable = nd_val_path_a
		}
		owns_or_non_sovereign_subject_owns = location:requena
		custom_tooltip = {
			text = nd_val_dest_valencia_46_tt
			location:valencia = {
				owner ?= root
				development >= 46
			}
		}
		gold >= 1000
	}
	icon = port_customs

	global_urban_build_buildings_efficiency = 0.1
	max_tariff = 0.1
}

# The crusade of 1398: after corsairs sacked Torreblanca and carried off the
# monstrance, a fleet armed in Valencia and Mallorca took Tedelis (Dellys).
# The city paid for its galleys with annuities sold on its own credit, which
# is what the banking institution stands for here.
# Valencia raised by half of its 1337 value.
nd_val_dest_croada_de_barbaria = {

		custom_tooltip = {
			text = nd_val_dest_valencia_55_tt
			location:valencia = {
				owner ?= root
				development >= 55
			}
		}
		gold >= 2000
	}
	icon = scourge_of_the_corsairs

	anti_piracy_warfare_modifier = 0.25
	navy_galley_maintenance_cost_modifier = -0.1
}

# The velvet weavers' guild (ordinances of 1479), raised to a college in 1686.
# Silk needs dye, which the kingdom does not grow: the market condition asks
# for it. The workshops are the heritage Obrador de Manises (age 2), whose
# lustreware went out through the same merchants. Alicante was given the
# title of city in 1490 and became the kingdom's second port; it starts as a
# rural settlement with no development of its own worth doubling, so the test
# is its rank.
nd_val_dest_art_major_de_la_seda = {
	age = age_1_traditions
	requires = nd_val_dest_croada_de_barbaria
	potential = { has_or_had_tag = VAL has_variable = nd_val_formed has_game_rule = nd_destinies_enabled }
	allow = {
		has_embraced_institution = institution:printing_press
		capital = {
			market = {
				OR = {
					is_produced_in_market = goods:dyes
					is_traded_in_market = goods:dyes
				}
			}
		}
		custom_tooltip = {
			text = nd_val_dest_obradors_tt
			any_owned_location = {
				count >= 3
				has_building_with_at_least_one_level = nd_val_obrador_de_manises
			}
		}
		custom_tooltip = {
			text = nd_val_dest_alacant_ciutat_tt
			location:alicante = {
				owner ?= root
				OR = {
					location_rank ?= location_rank:city
					location_rank ?= location_rank:megalopolis
				}
			}
		}
		gold >= 3000
		prestige >= 75
	}
	icon = muc_silk_looms

	unlock_building = nd_val_dest_collegi_de_la_seda
	global_merchant_power = 0.1
	embrace_institution_cost_modifier = -0.1
	non_rural_migration_attraction = 0.1
}
