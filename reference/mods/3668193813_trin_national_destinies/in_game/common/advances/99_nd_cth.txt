# ==========================================
# Carthage (CTH): Advance Tree
# 6 vanilla tag advances since 1.4 (one per age, country_cth.txt), plus
# the Amazigh group and Maghreb advances every country that can form
# Carthage holds (country_tle.txt, region_maghreb.txt). Level 4 since 1.4,
# still a step below Rome (level 5).
# 4 per age, fill gaps vanilla misses:
# religion/culture, fortifications, units,
# industrial buildings, governance
#
# No key below repeats a key of vanilla's six Carthage advances, and none
# repeats a key the Amazigh group or Maghreb advances grant in the same age.
# The validator's overlap check cannot see vanilla values written as named
# constants (medium_trade_land_efficiency_bonus), so check those by hand.
# ==========================================

	potential = { has_or_had_tag = CTH has_variable = nd_cth_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:genoa
		owns_or_non_sovereign_subject_owns = location:alexandria
		owns_or_non_sovereign_subject_owns = location:candia
		prestige >= 75
		# Carthage itself (the location is Tunis): 31 in 1337
		# (scripts/development_estimate.py --explain tunis), to be doubled. The
		# ancient city was the largest of the western Mediterranean.
		custom_tooltip = {
			text = nd_cth_dest_carthage_developed_tt
			location:tunis = {
				owner ?= root
				development >= 62
			}
		}
	}
	icon = maritime_advance_age_5
	tax_income_efficiency = 0.03
	# A sea ringed by Punic harbours is home water everywhere: a bonus to
	# fleets fighting next to owned shores. Colonial range stood here, the
	# key vanilla's cth_punic_global_trade grants.
	own_coast_naval_combat_bonus = 0.1
	prestige_decay = -0.001
	unlock_building = nd_cth_dest_cothon_emporium
}
