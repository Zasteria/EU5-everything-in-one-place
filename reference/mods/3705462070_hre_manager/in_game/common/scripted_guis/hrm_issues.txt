# Build list of heretic princes (religion != imperial religion)
hrm_sgui_build_heretics = {
	scope = country
	effect = {
		clear_variable_list = hrm_heretics
		save_scope_as = hrm_actor

		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:hre
			}
			save_scope_as = hrm_hre
			# Only check if HRE has a leader
			if = {
				limit = { international_organization_has_leader = yes }
				ordered_international_organization_member = {
					limit = {
						NOT = { religion = scope:hrm_hre.leader_country.religion }
					}
					order_by = {
						value = 0
						subtract = this_opinion_of_prev
					}
					max = 999
					scope:hrm_actor = {
						add_to_variable_list = {
							name = hrm_heretics
							target = prev
						}
					}
				}
			}
		}
	}
}

# Build list of subjugated HRE members (subject of non-HRE overlord)
hrm_sgui_build_subjugated = {
	scope = country
	effect = {
		clear_variable_list = hrm_subjugated
		save_scope_as = hrm_actor

		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:hre
			}
			save_scope_as = hrm_hre
			every_international_organization_member = {
				limit = {
					is_subject = yes
					top_overlord ?= {
						NOT = { is_member_of_international_organization = scope:hrm_hre }
					}
				}
				scope:hrm_actor = {
					add_to_variable_list = {
						name = hrm_subjugated
						target = prev
					}
				}
			}
		}
	}
}

# Build list of German locations owned by non-HRE countries
hrm_sgui_build_foreign_german = {
	scope = country
	effect = {
		clear_variable_list = hrm_foreign_german
		save_scope_as = hrm_actor

		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:hre
			}
			save_scope_as = hrm_hre
			every_international_organization_owned_location = {
				limit = {
					has_owner = yes
					owner = {
						NOT = { is_member_of_international_organization = scope:hrm_hre }
					}
					OR = {
						region = region:north_german_region
						region = region:south_german_region
					}
				}
				scope:hrm_actor = {
					add_to_variable_list = {
						name = hrm_foreign_german
						target = prev
					}
				}
			}
		}
	}
}
