crossing_the_pillars = {
	possible = {
		has_dlc = "d009_across_the_pillars"
		OR = {
			has_or_had_tag = CAS
			has_or_had_tag = MOR
			has_or_had_tag = GRA
		}
	}

	happened = {
		custom_tooltip = {
			text = crossing_the_pillars_battle_for_the_strait_tt
			global_var:battle_for_the_strait_winner_country ?= ROOT
		}
		custom_tooltip = {
			text = crossing_the_pillars_golden_century_tt
			has_variable = unlock_achievement_crossing_the_pillars
		}
	}
}

tinto_monta = {
	possible = {
		has_dlc = "d009_across_the_pillars"
		OR = {
			has_or_had_tag = CAS
			has_or_had_tag = ARA
		}
	}

	happened = {
		exists = union
		union = { has_variable = iberian_union_active }
		has_special_status_in_international_organization = {
			type = special_status:senior_partner
			international_organization = union
		}
	}
}

endured_atlas = {
	possible = {
		has_dlc = "d009_across_the_pillars"
		has_or_had_tag = MOR
	}

	happened = {
		tag = MOR
		area:sus_area = {
			any_location_in_area = {
				percent = 1
				trigger_if = {
					limit = { is_ownable = yes }
					top_owner ?= root
				}
			}
		}
		area:morocco_area = {
			any_location_in_area = {
				percent = 1
				trigger_if = {
					limit = { is_ownable = yes }
					top_owner ?= root
				}
			}
		}
		OR = {
			AND = {
				has_policy = centralized_makhzen_policy
				societal_value:centralization_vs_decentralization >= 25
				"estate_satisfaction:crown_estate" > 0.55
			}
			AND = {
				has_policy = negotiated_authority_policy
				societal_value:centralization_vs_decentralization <= -25
				"estate_satisfaction:tribes_estate" > 0.55
			}
		}
	}
}
