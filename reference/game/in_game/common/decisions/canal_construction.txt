build_suez_canal = {
	decision_category = infrastructure_decisions
	hide_repeatable_tag = yes

	potential = {
		current_age_or_later = {
			age = age_4_reformation
		}
		has_discovered = location:suez
		location:suez = {
			is_canal_open = no
		}
	}

	allow = {
		gold >= 200000

		owns_or_non_sovereign_subject_owns = location:tinnis
		owns_or_non_sovereign_subject_owns = location:suez

		has_advance = suez_canal_advance
		location:suez = {
			NOT = { has_canal_construction = yes }
		}

	}

	ai_will_do = {
		value = -1
		if = {
			limit = {
				gold >= 205000
				last_months_balance > 0
			}
			add = 100
		}
	}

	option = {
		name = build_suez_canal.a
		ai_chance = { value = 100 }
		effect = {
			location:suez = {
				start_canal_construction = { }
			}
			add_gold = -200000
		}
	}
}
