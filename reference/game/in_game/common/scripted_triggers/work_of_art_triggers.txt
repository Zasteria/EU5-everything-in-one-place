work_of_art_exists = {
	exists = $work_of_art$
	$work_of_art$ = { is_art_destroyed = no }
}

imperial_work_of_art_can_be_moved = {
	exists = $work_of_art$
	$work_of_art$ = {
		is_art_destroyed = no
		owner ?= {
			this != $country$
			is_member_of_international_organization = $international_organization$
		}
	}
}

work_of_art_is_imperial_regalia = {
	OR = {
		this = work_of_art:imperial_crown_hre
		this = work_of_art:imperial_orb_hre
		this = work_of_art:saint_stephans_purse
		this = work_of_art:coronation_gospel_hre
		this = work_of_art:kusanagi_no_tsurugi
		this = work_of_art:yata_no_kagami
		this = work_of_art:yasakani_no_magatama
		this = work_of_art:preah_khan_reach
		this = work_of_art:preah_lompeng_chey
	}
}
