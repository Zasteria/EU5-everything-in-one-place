papal_election_regency = {
	
	
	allow = {
		succession_law = heir_selection:papal_election
	}
	
	modifier = {
		stability_investment = stability_investment_penalty
	}
	
	
	start_effect = {
		random_character = {
			limit = {
				religious_figure_type = cardinal
				is_alive = yes
				is_adult = yes
				is_female = no
				is_regent = no
				has_exploration = no
				has_exploration_construction = no
				modifier:blocked_from_cabinet = no
				modifier:blocked_from_being_ruler = no
				religion = root.religion
			}
			save_scope_as = new_leader
		}
		if = {
			limit = { exists = scope:new_leader }
			set_regent = scope:new_leader
		}
		else_if = {
			limit = {
				any_character = {
					can_become_a_regent = yes
					is_female = no
					has_estate = estate_type:clergy_estate
					religion = root.religion
				}
			} 
			random_character = {
				limit = {
					can_become_a_regent = yes
					is_female = no
					has_estate = estate_type:clergy_estate
					religion = root.religion
				}
				save_scope_as = new_leader
			}
			set_regent = scope:new_leader
		}
		else = {
			hidden_effect = {
				create_character = {
					religion = root.religion
					estate = estate_type:clergy_estate
					age = 35
					female = no
					save_scope_as = new_leader
				}
				set_regent = scope:new_leader
			}
		}
		religion:catholic = {
			add_religion_modifier = {
				modifier = papal_conclave_modifier
				days = -1
			}
		}

		
		
		trigger_event_silently = { id = papal_conclave_events.11  days = { 1 650 } }
	}
	end_effect = {
		religion:catholic = {
			remove_religion_modifier = papal_conclave_modifier
		}
	}	
}
