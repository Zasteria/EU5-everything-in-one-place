# cm_dev_perf: CMF's is_host under CM's own name. The game's 1.4 trigger
# is_host shadows CMF's, and cmf_is_host exists only in cmf_dev_beta.
# True when this country is controlled by the host player, or in singleplayer.
# Scope: country
cm_is_host = {
	trigger_if = {
		limit = { has_multiple_players = no }
		is_human = yes
	}
	trigger_else = {
		trigger_if = {
			limit = { has_global_variable = cmf_core_host_country }
			this ?= global_var:cmf_core_host_country
		}
		trigger_else = {
			always = no
		}
	}
}
