# assimilate_primary: the council's hidden «Действия совета» automation
# (cabinetactions) back on wherever the panel's «Совет» (cabinet) is on. Off,
# seats are filled and never given an action (his run 05.10). The engine merges
# on_actions lists, so this adds to monthly_country_pulse, it does not replace it.
# Rebuilt by tools/generate.py.
monthly_country_pulse = {
	on_actions = {
		assimilate_primary_council_pulse
	}
}

assimilate_primary_council_pulse = {
	trigger = {
		is_ai = no
		is_system_automated = cabinet
		NOT = { is_system_automated = cabinetactions }
	}
	effect = {
		set_automated_system = { system = cabinetactions activate = yes }
		debug_log = "assimilate_primary: cabinetactions was off with cabinet on, switched back on"
	}
}
