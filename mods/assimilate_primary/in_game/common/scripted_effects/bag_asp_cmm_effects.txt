# Scope: country. var:cmf_callback names what was clicked; switch as in
# war_sliders.
bag_asp_handle_callback = {
	switch = {
		trigger = var:cmf_callback
		flag:bag_asp__auto = {
			bag_asp_sync = yes
			if = {
				limit = { has_variable = bag_asp_auto_on }
				# Ours and the game's «Совет» would fight over the same seats.
				set_automated_system = { system = cabinet activate = no }
				debug_log = "bag_asp: own council automation on, the game's «Совет» switched off"
				set_variable = { name = bag_asp_months value = 0 }
				set_variable = { name = bag_asp_review_months value = 0 }
				bag_asp_plan_all = yes
			}
			else = {
				debug_log = "bag_asp: own council automation off"
			}
		}
		flag:bag_asp__period = { bag_asp_sync = yes }
		flag:bag_asp__review = { bag_asp_sync = yes }
		flag:bag_asp__culture_target = { bag_asp_sync = yes }
		flag:bag_asp__culture_speedup = { bag_asp_sync = yes }
		flag:bag_asp__religion_target = { bag_asp_sync = yes }
		flag:bag_asp__assim_control = { bag_asp_sync = yes }
		flag:bag_asp__develop_target = { bag_asp_sync = yes }
		flag:bag_asp__develop_control = { bag_asp_sync = yes }
		flag:bag_asp__develop_seats = { bag_asp_sync = yes }
		flag:bag_asp__log = { bag_asp_sync = yes }
	}
	# His ask 13.10: what he changes goes into the log, with every setting after it.
	if = {
		limit = { has_variable = bag_asp_log_on }
		debug_log = "bag_asp: you changed a setting"
		bag_asp_log_settings = yes
	}
}
