# assimilate_primary: the mod's own council automation, the monthly pass.
# Rebuilt by tools/generate.py (tools/council.py).

# Scope: country. Every month for a player with the checkbox on, from CMF's
# monthly pulse (the one CM's auto-build runs on). Every `review` months
# (0 = never) all seats are handed out again; otherwise every `period` months
# only seats without an action are filled.
bag_asp_tick = {
	if = {
		limit = { NOT = { has_variable = bag_asp_period } }
		bag_asp_sync = yes
	}
	if = {
		limit = { NOT = { has_variable = bag_asp_months } }
		set_variable = { name = bag_asp_months value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = bag_asp_review_months } }
		set_variable = { name = bag_asp_review_months value = 0 }
	}
	change_variable = { name = bag_asp_months add = 1 }
	change_variable = { name = bag_asp_review_months add = 1 }
	debug_log = "bag_asp: month"
	if = {
		limit = {
			var:bag_asp_review > 0
			var:bag_asp_review_months >= var:bag_asp_review
		}
		set_variable = { name = bag_asp_review_months value = 0 }
		set_variable = { name = bag_asp_months value = 0 }
		bag_asp_plan_all = yes
	}
	else_if = {
		limit = { var:bag_asp_months >= var:bag_asp_period }
		set_variable = { name = bag_asp_months value = 0 }
		bag_asp_plan_fill = yes
	}
}

# Scope: country. Every seat handed out down «Приоритет».
bag_asp_plan_all = {
	debug_log = "bag_asp: handing out every seat"
	bag_asp_plan = yes
}

# Scope: country. Only seats without an action: the others are taken as chosen.
bag_asp_plan_fill = {
	debug_log = "bag_asp: filling empty seats"
	set_variable = { name = bag_asp_fill_only value = yes }
	bag_asp_plan = yes
	remove_variable = bag_asp_fill_only
}

# Scope: country. Down «Приоритет», each action in use first keeps the seats
# already on it, then takes free seats, the advisor best in its skill first,
# while it has a target and a free seat is left. A seat is free until this pass
# chose it; at the end, seats nothing could take keep what they had.
bag_asp_plan = {
	save_scope_as = actor
	# The lists may have moved without their callback (a typed place goes
	# through the console), so the order is read afresh every pass.
	bag_asp_rebuild = yes
	if = {
		limit = { NOT = { any_in_list = { variable = bag_asp_used } } }
		debug_log = "bag_asp: no action is ticked in use"
	}
	bag_asp_mark_busy = yes
	if = {
		limit = { has_variable = bag_asp_fill_only }
		every_cabinet = {
			limit = {
				has_cabinet_action = yes
				exists = cabinet_member
			}
			cabinet_member = { set_variable = { name = bag_asp_claimed value = yes } }
		}
	}
	every_in_list = {
		variable = bag_asp_order
		save_temporary_scope_as = bag_asp_item
		if = {
			limit = {
				scope:actor = {
					is_target_in_variable_list = { name = bag_asp_used target = scope:bag_asp_item }
					any_cabinet = { bag_asp_free_seat = yes }
				}
			}
			scope:actor = {
				switch = {
					trigger = scope:bag_asp_item
					flag:bag_asp_integrate_province = { bag_asp_place_integrate_province = yes }
					flag:bag_asp_integrate_area = { bag_asp_place_integrate_area = yes }
					flag:bag_asp_reduce_war_exhaustion = { bag_asp_place_reduce_war_exhaustion = yes }
					flag:bag_asp_stabilize_country = { bag_asp_place_stabilize_country = yes }
					flag:bag_asp_increase_control_province = { bag_asp_place_increase_control_province = yes }
					flag:bag_asp_increase_control_area = { bag_asp_place_increase_control_area = yes }
					flag:bag_asp_send_secret_royal_inspectors = { bag_asp_place_send_secret_royal_inspectors = yes }
					flag:bag_asp_promote_culture = { bag_asp_place_promote_culture = yes }
					flag:bag_asp_promote_religion = { bag_asp_place_promote_religion = yes }
					flag:bag_asp_pacify_population = { bag_asp_place_pacify_population = yes }
					flag:bag_asp_recovery_effort = { bag_asp_place_recovery_effort = yes }
					flag:bag_asp_appease_estate = { bag_asp_place_appease_estate = yes }
					flag:bag_asp_alleviate_concerns = { bag_asp_place_alleviate_concerns = yes }
					flag:bag_asp_reduce_inflation = { bag_asp_place_reduce_inflation = yes }
					flag:bag_asp_develop_province = { bag_asp_place_develop_province = yes }
					flag:bag_asp_study_institutions = { bag_asp_place_study_institutions = yes }
					flag:bag_asp_pop_promote_action = { bag_asp_place_pop_promote_action = yes }
					flag:bag_asp_strengthen_government = { bag_asp_place_strengthen_government = yes }
					flag:bag_asp_diplomatic_endeavors = { bag_asp_place_diplomatic_endeavors = yes }
					flag:bag_asp_counter_espionage_action = { bag_asp_place_counter_espionage_action = yes }
				}
			}
		}
	}
	if = {
		limit = { any_cabinet = { bag_asp_free_seat = yes } }
		debug_log = "bag_asp: a seat is left as it was, nothing in use fits it"
	}
	every_cabinet = {
		limit = { exists = cabinet_member }
		cabinet_member = {
			if = {
				limit = { has_variable = bag_asp_claimed }
				remove_variable = bag_asp_claimed
			}
		}
	}
	bag_asp_clear_busy = yes
}

# Scope: country (scope:actor); the seat is scope:bag_asp_seat.
bag_asp_try_integrate_province = {
	if = {
		limit = {
			cabinet_action:integrate_province = { is_available_for = scope:actor }
		}
		scope:actor = {
			ordered_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_integrate_province }
					NOT = { has_variable = bag_asp_busy_integrate_province }
					any_location_in_province = {
						integration_level = conquered
					}
				}
				order_by = {
					value = 0
					every_location_in_province = {
						limit = {
							integration_level = conquered
							culture_percentage = { culture = scope:actor.culture value >= 0.5 }
						}
						add = 10000
					}
					every_location_in_province = {
						limit = {
							integration_level = conquered
							NOT = { culture_percentage = { culture = scope:actor.culture value >= 0.5 } }
							dominant_culture ?= { is_primary_or_accepted_in = scope:actor }
							culture_percentage = { culture = dominant_culture value >= 0.5 }
						}
						add = 100
					}
					add = {
						value = population
						max = 99
					}
				}
				save_temporary_scope_as = bag_asp_pick
				scope:bag_asp_seat = {
					if = {
						limit = { has_cabinet_action = yes }
						stop_cabinet_action = {}
					}
					set_cabinet_action = { type = cabinet_action:integrate_province target = scope:bag_asp_pick }
				}
				scope:bag_asp_pick = { set_variable = { name = bag_asp_busy_integrate_province value = yes } }
				scope:bag_asp_seat = { cabinet_member = { set_variable = { name = bag_asp_claimed value = yes } } }
				scope:actor = { set_variable = { name = bag_asp_placed value = yes } }
				debug_log = "bag_asp: seat set to integrate_province"
			}
		}
	}
}

# Scope: country (scope:actor); the seat is scope:bag_asp_seat.
bag_asp_try_promote_culture = {
	if = {
		limit = {
			cabinet_action:promote_culture = { is_available_for = scope:actor }
		}
		scope:actor.culture = { save_scope_as = target }
		scope:actor = {
			ordered_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_promote_culture }
					any_location_in_province = {
						OR = {
							NOT = { dominant_culture = scope:target }
							local_cultural_unity < 0.95
						}
					}
					NOT = { has_variable = bag_asp_busy_promote_culture }
					OR = {
						scope:actor = { is_ai = no }
						AND = {
							trigger_if = {
								limit = { this = scope:actor.capital.province }
								scope:target = scope:actor.culture
							}
							any_location_in_province = { naturally_assimilating_culture = scope:target }
							trigger_if = {
								limit = { exists = dominant_culture }
								OR = {
									AND = {
										NOT = { dominant_culture = scope:target }
										dominant_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
									}
									AND = {
										dominant_culture = scope:target
										any_location_in_province = {
											exists = secondary_culture

										}
									}
								}

					}
				}
				order_by = population
				save_temporary_scope_as = bag_asp_pick
				scope:bag_asp_seat = {
					if = {
						limit = { has_cabinet_action = yes }
						stop_cabinet_action = {}

# Scope: country. While the game's «devastated locations» alert shows, every
# couple of seconds (bag_asp_devastation_watch.gui): one province below zero
# prosperity without recovery gets it now: an empty seat first, else the seat
# whose action ranks lowest in «Приоритет» — only among actions ranked below
# recovery (unticked ones count as below). Actions outside the pool stay.
bag_asp_emergency = {
	if = {
		limit = { has_variable = bag_asp_auto_on }
		save_scope_as = actor
		bag_asp_rebuild = yes
		if = {
			limit = { is_target_in_variable_list = { name = bag_asp_used target = flag:bag_asp_recovery_effort } }
			bag_asp_mark_busy = yes
			ordered_cabinet = {
				limit = {
					exists = cabinet_member
					OR = {
						has_cabinet_action = no
						AND = { cabinet_action = cabinet_action:integrate_province scope:actor.var:bag_asp_rank_integrate_province > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:integrate_area scope:actor.var:bag_asp_rank_integrate_area > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:reduce_war_exhaustion scope:actor.var:bag_asp_rank_reduce_war_exhaustion > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:stabilize_country scope:actor.var:bag_asp_rank_stabilize_country > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:increase_control_province scope:actor.var:bag_asp_rank_increase_control_province > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:increase_control_area scope:actor.var:bag_asp_rank_increase_control_area > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:send_secret_royal_inspectors scope:actor.var:bag_asp_rank_send_secret_royal_inspectors > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:promote_culture scope:actor.var:bag_asp_rank_promote_culture > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:promote_religion scope:actor.var:bag_asp_rank_promote_religion > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:pacify_population scope:actor.var:bag_asp_rank_pacify_population > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:appease_estate scope:actor.var:bag_asp_rank_appease_estate > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:alleviate_concerns scope:actor.var:bag_asp_rank_alleviate_concerns > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:reduce_inflation scope:actor.var:bag_asp_rank_reduce_inflation > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:develop_province scope:actor.var:bag_asp_rank_develop_province > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:study_institutions scope:actor.var:bag_asp_rank_study_institutions > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:pop_promote_action scope:actor.var:bag_asp_rank_pop_promote_action > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:strengthen_government scope:actor.var:bag_asp_rank_strengthen_government > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:diplomatic_endeavors scope:actor.var:bag_asp_rank_diplomatic_endeavors > scope:actor.var:bag_asp_rank_recovery_effort }
						AND = { cabinet_action = cabinet_action:counter_espionage_action scope:actor.var:bag_asp_rank_counter_espionage_action > scope:actor.var:bag_asp_rank_recovery_effort }
					}
				}
				order_by = {
					value = 0
					if = { limit = { has_cabinet_action = no } add = 1000 }
					else_if = { limit = { cabinet_action = cabinet_action:integrate_province } add = scope:actor.var:bag_asp_rank_integrate_province }
					else_if = { limit = { cabinet_action = cabinet_action:integrate_area } add = scope:actor.var:bag_asp_rank_integrate_area }
					else_if = { limit = { cabinet_action = cabinet_action:reduce_war_exhaustion } add = scope:actor.var:bag_asp_rank_reduce_war_exhaustion }
					else_if = { limit = { cabinet_action = cabinet_action:stabilize_country } add = scope:actor.var:bag_asp_rank_stabilize_country }
					else_if = { limit = { cabinet_action = cabinet_action:increase_control_province } add = scope:actor.var:bag_asp_rank_increase_control_province }
					else_if = { limit = { cabinet_action = cabinet_action:increase_control_area } add = scope:actor.var:bag_asp_rank_increase_control_area }
					else_if = { limit = { cabinet_action = cabinet_action:send_secret_royal_inspectors } add = scope:actor.var:bag_asp_rank_send_secret_royal_inspectors }
					else_if = { limit = { cabinet_action = cabinet_action:promote_culture } add = scope:actor.var:bag_asp_rank_promote_culture }
					else_if = { limit = { cabinet_action = cabinet_action:promote_religion } add = scope:actor.var:bag_asp_rank_promote_religion }
					else_if = { limit = { cabinet_action = cabinet_action:pacify_population } add = scope:actor.var:bag_asp_rank_pacify_population }
					else_if = { limit = { cabinet_action = cabinet_action:appease_estate } add = scope:actor.var:bag_asp_rank_appease_estate }
					else_if = { limit = { cabinet_action = cabinet_action:alleviate_concerns } add = scope:actor.var:bag_asp_rank_alleviate_concerns }
					else_if = { limit = { cabinet_action = cabinet_action:reduce_inflation } add = scope:actor.var:bag_asp_rank_reduce_inflation }
					else_if = { limit = { cabinet_action = cabinet_action:develop_province } add = scope:actor.var:bag_asp_rank_develop_province }
					else_if = { limit = { cabinet_action = cabinet_action:study_institutions } add = scope:actor.var:bag_asp_rank_study_institutions }
					else_if = { limit = { cabinet_action = cabinet_action:pop_promote_action } add = scope:actor.var:bag_asp_rank_pop_promote_action }
					else_if = { limit = { cabinet_action = cabinet_action:strengthen_government } add = scope:actor.var:bag_asp_rank_strengthen_government }
					else_if = { limit = { cabinet_action = cabinet_action:diplomatic_endeavors } add = scope:actor.var:bag_asp_rank_diplomatic_endeavors }
					else_if = { limit = { cabinet_action = cabinet_action:counter_espionage_action } add = scope:actor.var:bag_asp_rank_counter_espionage_action }
					add = {
						value = cabinet_member.dip
						divide = 100
					}
				}
				save_scope_as = bag_asp_seat
				bag_asp_try_recovery_effort = yes
				if = {
					limit = { exists = cabinet_member }
					cabinet_member = {
						if = {
							limit = { has_variable = bag_asp_claimed }
							remove_variable = bag_asp_claimed
						}
					}
				}
			}
			if = {
				limit = { has_variable = bag_asp_placed }
				debug_log = "bag_asp: emergency, a seat moved to recovery"
				remove_variable = bag_asp_placed
			}
			bag_asp_clear_busy = yes
		}
	}
}

	while = {
		limit = {
