# assimilate_primary: the mod's own council automation, the monthly pass.
# Rebuilt by tools/generate.py (tools/council.py).

# Scope: country. Every month for a player with the checkbox on, from CMF's
# monthly pulse (the one CM's auto-build runs on). Every `review` months
# (0 = never) all seats are handed out again; otherwise every `period` months
# only seats without an action are filled.
bag_asp_tick = {
	if = {
		limit = { NOT = { has_variable = bag_asp_period } }
		bag_asp_sync = yes
	}
	if = {
		limit = { NOT = { has_variable = bag_asp_months } }
		set_variable = { name = bag_asp_months value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = bag_asp_review_months } }
		set_variable = { name = bag_asp_review_months value = 0 }
	}
	change_variable = { name = bag_asp_months add = 1 }
	change_variable = { name = bag_asp_review_months add = 1 }
	debug_log = "bag_asp: month"
	bag_asp_assim_check = yes
	if = {
		limit = {
			var:bag_asp_review > 0
			var:bag_asp_review_months >= var:bag_asp_review
		}
		set_variable = { name = bag_asp_review_months value = 0 }
		set_variable = { name = bag_asp_months value = 0 }
		bag_asp_plan_all = yes
	}
	else_if = {
		limit = { var:bag_asp_months >= var:bag_asp_period }
		set_variable = { name = bag_asp_months value = 0 }
		bag_asp_plan_fill = yes
	}
}

# Scope: country. Every seat handed out down «Приоритет».
bag_asp_plan_all = {
	debug_log = "bag_asp: handing out every seat"
	bag_asp_plan = yes
}

# Scope: country. Only seats without an action: the others are taken as chosen.
bag_asp_plan_fill = {
	debug_log = "bag_asp: filling empty seats"
	set_variable = { name = bag_asp_fill_only value = yes }
	bag_asp_plan = yes
	remove_variable = bag_asp_fill_only
}

# Scope: country. Down «Приоритет», each action in use first keeps the seats
# already on it, then takes free seats, the advisor best in its skill first,
# while it has a target and a free seat is left. A seat is free until this pass
# chose it; at the end, seats nothing could take keep what they had.
bag_asp_plan = {
	save_scope_as = actor
	# The lists may have moved without their callback (a typed place goes
	# through the console), so the order is read afresh every pass.
	bag_asp_rebuild = yes
	# His ask 13.10: the seats as the pass finds them, so what he changed by
	# hand between two passes shows against the last pass's «after».
	if = {
		limit = { has_variable = bag_asp_log_on }
		debug_log = "bag_asp: the seats before the pass:"
		bag_asp_probe_seats = yes
	}
	if = {
		limit = { NOT = { any_in_list = { variable = bag_asp_used } } }
		debug_log = "bag_asp: no action is ticked in use"
	}
	# His rule 13.10: an assimilation whose province has every location at
	# half is freed, unless the full row is in use; before the marking, so its
	# control seat goes with it.
	if = {
		limit = { NOT = { is_target_in_variable_list = { name = bag_asp_used target = flag:bag_asp_promote_culture_full } } }
		every_cabinet = {
			limit = {
				has_cabinet_action = yes
				cabinet_action = cabinet_action:promote_culture
				interaction_target:target_1 ?= { NOT = { bag_asp_needs_half = yes } }
			}
			stop_cabinet_action = {}
			debug_log = "bag_asp: an assimilation freed, every location of its province is at half"
		}
	}
	bag_asp_mark_busy = yes
	bag_asp_unpair = yes
	if = {
		limit = {
			has_variable = bag_asp_log_on
			NOT = { has_variable = bag_asp_fill_only }
		}
		bag_asp_probe_order = yes
		bag_asp_probe_half = yes
	}
	if = {
		limit = { has_variable = bag_asp_fill_only }
		every_cabinet = {
			limit = {
				has_cabinet_action = yes
				exists = cabinet_member
			}
			cabinet_member = { set_variable = { name = bag_asp_claimed value = yes } }
		}
	}
	every_in_list = {
		variable = bag_asp_order
		save_temporary_scope_as = bag_asp_item
		if = {
			limit = {
				scope:actor = {
					is_target_in_variable_list = { name = bag_asp_used target = scope:bag_asp_item }
					any_cabinet = { bag_asp_free_seat = yes }
				}
			}
			scope:actor = {
				switch = {
					trigger = scope:bag_asp_item
					flag:bag_asp_integrate_province = { bag_asp_place_integrate_province = yes }
					flag:bag_asp_integrate_area = { bag_asp_place_integrate_area = yes }
					flag:bag_asp_reduce_war_exhaustion = { bag_asp_place_reduce_war_exhaustion = yes }
					flag:bag_asp_stabilize_country = { bag_asp_place_stabilize_country = yes }
					flag:bag_asp_increase_control_province = { bag_asp_place_increase_control_province = yes }
					flag:bag_asp_increase_control_area = { bag_asp_place_increase_control_area = yes }
					flag:bag_asp_send_secret_royal_inspectors = { bag_asp_place_send_secret_royal_inspectors = yes }
					flag:bag_asp_promote_culture = { bag_asp_place_promote_culture = yes }
					flag:bag_asp_promote_religion = { bag_asp_place_promote_religion = yes }
					flag:bag_asp_pacify_population = { bag_asp_place_pacify_population = yes }
					flag:bag_asp_recovery_effort = { bag_asp_place_recovery_effort = yes }
					flag:bag_asp_appease_estate = { bag_asp_place_appease_estate = yes }
					flag:bag_asp_alleviate_concerns = { bag_asp_place_alleviate_concerns = yes }
					flag:bag_asp_reduce_inflation = { bag_asp_place_reduce_inflation = yes }
					flag:bag_asp_develop_province = { bag_asp_place_develop_province = yes }
					flag:bag_asp_study_institutions = { bag_asp_place_study_institutions = yes }
					flag:bag_asp_pop_promote_action = { bag_asp_place_pop_promote_action = yes }
					flag:bag_asp_strengthen_government = { bag_asp_place_strengthen_government = yes }
					flag:bag_asp_diplomatic_endeavors = { bag_asp_place_diplomatic_endeavors = yes }
					flag:bag_asp_counter_espionage_action = { bag_asp_place_counter_espionage_action = yes }
					flag:bag_asp_noblesse_oblige_cabinet_action = { bag_asp_place_noblesse_oblige_cabinet_action = yes }
					flag:bag_asp_diplomatic_antagonism = { bag_asp_place_diplomatic_antagonism = yes }
					flag:bag_asp_promote_culture_full = { bag_asp_place_promote_culture_full = yes }
					flag:bag_asp_change_societal_values = { bag_asp_place_change_societal_values = yes }
				}
			}
		}
	}
	if = {
		limit = { any_cabinet = { bag_asp_free_seat = yes } }
		debug_log = "bag_asp: a seat is left as it was, nothing in use fits it"
	}
	if = {
		limit = { has_variable = bag_asp_log_on }
		debug_log = "bag_asp: the seats after the pass:"
		bag_asp_probe_seats = yes
		bag_asp_probe_promote = yes
	}
	every_cabinet = {
		limit = { exists = cabinet_member }
		cabinet_member = {
			if = {
				limit = { has_variable = bag_asp_claimed }
				remove_variable = bag_asp_claimed
			}
		}
	}
	bag_asp_clear_busy = yes
}

# Scope: country (scope:actor); the seat is scope:bag_asp_seat.
bag_asp_try_integrate_province = {
	if = {
		limit = {
			cabinet_action:integrate_province = { is_available_for = scope:actor }
		}
		scope:actor = {
			ordered_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_integrate_province }
					NOT = { has_variable = bag_asp_busy_integrate_province }
					any_location_in_province = {
						integration_level = conquered
					}
				}
				order_by = {
					value = 0
					every_location_in_province = {
						limit = {
							integration_level = conquered
							culture_percentage = { culture = scope:actor.culture value >= 0.5 }
						}
						add = 10000
					}
					every_location_in_province = {
						limit = {

bag_asp_try_promote_culture = {
	if = {
		limit = {
			cabinet_action:promote_culture = { is_available_for = scope:actor }
		}
		if = {
			limit = {
				scope:actor = {
					var:bag_asp_culture_target = 2
					any_accepted_culture = { always = yes }
				}
			}
			scope:actor = {
				ordered_accepted_culture = {
					order_by = "scope:actor.culture_percentage_in_country(this)"
					save_scope_as = target
				}
			}
		}
		else = {
			scope:actor.culture = { save_scope_as = target }
		}
		scope:actor = {
			ordered_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_promote_culture }
					any_location_in_province = { bag_asp_assimilable_to_target = yes }
					any_location_in_province = {
						OR = {
							NOT = { dominant_culture = scope:target }
							local_cultural_unity < 0.95
						}
					}
					OR = {
						"culture_percentage(scope:target)" < 0.95
						any_location_in_province = { NOT = { dominant_culture = scope:target } }
					}
					NOT = { has_variable = bag_asp_busy_promote_culture }
					OR = {
						scope:actor = { is_ai = no }
						AND = {
							trigger_if = {
								limit = { this = scope:actor.capital.province }
								scope:target = scope:actor.culture
							}
							any_location_in_province = { naturally_assimilating_culture = scope:target }
							trigger_if = {
								limit = { exists = dominant_culture }
								OR = {
									AND = {
										NOT = { dominant_culture = scope:target }
										dominant_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
									}
									AND = {
										dominant_culture = scope:target
										any_location_in_province = {
											exists = secondary_culture
											secondary_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
										}
									}
								}
							}
							trigger_else = { always = no }
						}
					}
					# His rule 05.10: where an accepted culture leads the primary by
					# 30 points, do not push the primary.
					trigger_if = {
						limit = { scope:target = scope:actor.culture }
						NAND = {
							exists = dominant_culture
							NOT = { dominant_culture = scope:actor.culture }
							dominant_culture = { is_primary_or_accepted_in = scope:actor }
							"culture_percentage(dominant_culture)" >= {
								value = "culture_percentage(scope:actor.culture)"
								add = 0.3
							}
						}
					}
					bag_asp_needs_half = yes bag_asp_control_reachable = yes
				}
				order_by = bag_asp_half_score
				save_temporary_scope_as = bag_asp_pick
				scope:bag_asp_pick = { save_scope_as = bag_asp_last }
				scope:bag_asp_seat = {
					bag_asp_vacate = yes
					set_cabinet_action = { type = cabinet_action:promote_culture target = scope:target target_1 = scope:bag_asp_pick }
				}
				scope:bag_asp_pick = { set_variable = { name = bag_asp_busy_promote_culture value = yes } }
				scope:bag_asp_seat = { cabinet_member = { set_variable = { name = bag_asp_claimed value = yes } } }
				scope:actor = { set_variable = { name = bag_asp_placed value = yes } }
				debug_log = "bag_asp: seat set to promote_culture"
			}
		}
	}
}

bag_asp_try_promote_culture_full = {
	if = {
		limit = {
			cabinet_action:promote_culture = { is_available_for = scope:actor }
		}
		if = {
			limit = {
				scope:actor = {
					var:bag_asp_culture_target = 2
					any_accepted_culture = { always = yes }
				}
			}
			scope:actor = {
				ordered_accepted_culture = {
					order_by = "scope:actor.culture_percentage_in_country(this)"
					save_scope_as = target
				}
			}
		}
		else = {
			scope:actor.culture = { save_scope_as = target }
		}
		scope:actor = {
			ordered_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_promote_culture }
					any_location_in_province = { bag_asp_assimilable_to_target = yes }
					any_location_in_province = {
						OR = {
							NOT = { dominant_culture = scope:target }
							local_cultural_unity < 0.95
						}
					}
					OR = {
						"culture_percentage(scope:target)" < 0.95
						any_location_in_province = { NOT = { dominant_culture = scope:target } }
					}
					NOT = { has_variable = bag_asp_busy_promote_culture }
					OR = {
						scope:actor = { is_ai = no }
						AND = {
							trigger_if = {
								limit = { this = scope:actor.capital.province }
								scope:target = scope:actor.culture
							}
							any_location_in_province = { naturally_assimilating_culture = scope:target }
							trigger_if = {
								limit = { exists = dominant_culture }
								OR = {
									AND = {
										NOT = { dominant_culture = scope:target }
										dominant_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
									}
									AND = {
										dominant_culture = scope:target
										any_location_in_province = {
											exists = secondary_culture
											secondary_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
										}
									}
								}
							}
							trigger_else = { always = no }
						}
					}
					# His rule 05.10: where an accepted culture leads the primary by
					# 30 points, do not push the primary.
					trigger_if = {
						limit = { scope:target = scope:actor.culture }
						NAND = {
							exists = dominant_culture
							NOT = { dominant_culture = scope:actor.culture }
							dominant_culture = { is_primary_or_accepted_in = scope:actor }
							"culture_percentage(dominant_culture)" >= {
								value = "culture_percentage(scope:actor.culture)"
								add = 0.3
							}
						}
					}
				}
				order_by = population
				save_temporary_scope_as = bag_asp_pick
				scope:bag_asp_pick = { save_scope_as = bag_asp_last }
				scope:bag_asp_seat = {
					bag_asp_vacate = yes
					set_cabinet_action = { type = cabinet_action:promote_culture target = scope:target target_1 = scope:bag_asp_pick }
				}
				scope:bag_asp_pick = { set_variable = { name = bag_asp_busy_promote_culture value = yes } }
				scope:bag_asp_seat = { cabinet_member = { set_variable = { name = bag_asp_claimed value = yes } } }
				scope:actor = { set_variable = { name = bag_asp_placed value = yes } }
				debug_log = "bag_asp: seat set to promote_culture_full"
			}
		}
	}
}

# Scope: country (scope:actor). The «Приоритет» row of change_societal_values.
bag_asp_place_change_societal_values = {
	# A pass that only fills empty seats leaves a seat already on values alone.
	if = {
		limit = {
			any_cabinet = {
				has_cabinet_action = yes cabinet_action = cabinet_action:change_societal_values
				NOT = { bag_asp_free_seat = yes }
			}
		}
		debug_log = "bag_asp: values, the seat on values stays: this pass fills empty seats only"
	}
	else = {
		# What the mod asked for last holds only while a seat is still on values.
		# His log 14.10 10:02: the game ends the action itself once its pair
		# cannot move (serfdom at 63.30, stop 62.85), and that month's modifier
		# still carries the seat's push (0.6285 read with the seat empty). So the
		# pair just left is tested this pass as if the seat were still on it,
		# and the request is forgotten only after the test.
		if = {
			limit = {
				has_variable = bag_asp_sv_cur
				NOT = { any_cabinet = { has_cabinet_action = yes cabinet_action = cabinet_action:change_societal_values } }
			}
			set_local_variable = { name = bag_asp_sv_gone value = yes }
		}
		ordered_cabinet = {
			limit = { bag_asp_free_seat = yes }
			order_by = cabinet_member.dip
			save_scope_as = bag_asp_seat
		}
		if = {
			limit = {
				exists = scope:bag_asp_seat
				cabinet_action:change_societal_values = { is_available_for = scope:actor }
			}
			set_local_variable = { name = bag_asp_sv_p value = bag_asp_sv_push }
			if = {
				limit = { has_variable = bag_asp_log_on }
				set_global_variable = { name = bag_asp_sv_n3 value = local_var:bag_asp_sv_p }
				debug_log = "bag_asp: values, the best free diplomat would push [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_sv_n3_value')|4] a month"
			}
			bag_asp_sv_choose = yes
			if = {
				limit = { has_local_variable = bag_asp_sv_gone }
				remove_variable = bag_asp_sv_cur
				remove_variable = bag_asp_sv_cur_side
			}
			if = {
				limit = {
					has_variable = bag_asp_sv_want
					has_variable = bag_asp_sv_cur
					var:bag_asp_sv_want = var:bag_asp_sv_cur
					var:bag_asp_sv_want_side = var:bag_asp_sv_cur_side
				}
				every_cabinet = {
					limit = { has_cabinet_action = yes cabinet_action = cabinet_action:change_societal_values }
					cabinet_member ?= { set_variable = { name = bag_asp_claimed value = yes } }
				}
				debug_log = "bag_asp: values, the seat keeps its pair"
			}
			else_if = {
				limit = { has_variable = bag_asp_sv_want }
				# One seat on values: another one still on it gives it up.
				every_cabinet = {
					limit = {
						has_cabinet_action = yes cabinet_action = cabinet_action:change_societal_values
						NOT = { this = scope:bag_asp_seat }
					}
					bag_asp_vacate = yes
				}
				scope:bag_asp_seat = {
					bag_asp_vacate = yes
					set_cabinet_action = {
						type = cabinet_action:change_societal_values
						# Any pair: the target sets none (his run 12.10); the window picks it.
						target = societal_value_type:traditionalist_vs_innovative
					}
					cabinet_member = { set_variable = { name = bag_asp_claimed value = yes } }
				}
				if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
				if = { limit = { has_variable = bag_asp_sv_left } remove_variable = bag_asp_sv_left }
				if = { limit = { has_variable = bag_asp_sv_right } remove_variable = bag_asp_sv_right }
				if = { limit = { has_variable = bag_asp_sv_pair } remove_variable = bag_asp_sv_pair }
				set_variable = { name = bag_asp_sv_cur value = var:bag_asp_sv_want }
				set_variable = { name = bag_asp_sv_cur_side value = var:bag_asp_sv_want_side }
				set_variable = { name = bag_asp_sv_cab value = scope:bag_asp_seat }
				bag_asp_sv_ask = yes
			}
			else = {
				debug_log = "bag_asp: values, no ticked pair would move a whole point"
			}
			if = { limit = { has_variable = bag_asp_sv_want } remove_variable = bag_asp_sv_want }
			if = { limit = { has_variable = bag_asp_sv_want_side } remove_variable = bag_asp_sv_want_side }
		}
	}
}

# Scope: country. One pair, set up by bag_asp_sv_choose: local
# bag_asp_sv_i (its row), _s (side), _pos (where it stands toward the side),
# _net (the monthly net push toward it without this seat), _p (the seat's push).
bag_asp_sv_test = {
	if = {
		limit = {
			has_variable = bag_asp_sv_cur
			var:bag_asp_sv_cur = local_var:bag_asp_sv_i
		}
		# The seat pushes this pair: its push is in the modifier already. Pushed
		# the other way, the pair is passed over: the window would not list it.
		if = {
			limit = { NOT = { var:bag_asp_sv_cur_side = local_var:bag_asp_sv_s } }
			set_local_variable = { name = bag_asp_sv_net value = -10 }
		}
	}
	else = {
		change_local_variable = { name = bag_asp_sv_net add = local_var:bag_asp_sv_p }
	}
	set_local_variable = {
		name = bag_asp_sv_stop
		value = {
			value = local_var:bag_asp_sv_net
			multiply = 100
			max = 100
		}
	}
	if = {
		limit = { has_variable = bag_asp_log_on }
		set_global_variable = { name = bag_asp_sv_n1 value = local_var:bag_asp_sv_pos }
		set_global_variable = { name = bag_asp_sv_n2 value = local_var:bag_asp_sv_stop }
		set_global_variable = { name = bag_asp_sv_n3 value = local_var:bag_asp_sv_net }
	}
	set_local_variable = {
		name = bag_asp_sv_stop
		value = {
			value = local_var:bag_asp_sv_stop
			floor = yes
		}
	}
	set_local_variable = {
		name = bag_asp_sv_at
		value = {
			value = local_var:bag_asp_sv_pos
			floor = yes
		}
	}
	if = {
		limit = {
			has_variable = bag_asp_log_on
			local_var:bag_asp_sv_r = 1
		}

			# side is still being implemented, its stop keeps creeping up; the
			# pair is not taken until it is in full (a seat already on it stays).

				AND = {
					has_variable = bag_asp_sv_cur
					var:bag_asp_sv_cur = local_var:bag_asp_sv_i

# Scope: country. From bag_asp_tick, monthly: a seat on promote_culture whose province

		limit = {
			has_cabinet_action = yes

			exists = interaction_target:target
			exists = interaction_target:target_1
		}
		interaction_target:target = { save_temporary_scope_as = target }
		if = {
			limit = {
				interaction_target:target_1 = {
					NOT = { any_location_in_province = { bag_asp_assimilable_to_target = yes } }
				}
			}
			debug_log = "bag_asp: assimilation freed: no pop left in its province that can turn (tribesmen, a blocked class, or a faith-tied culture)"

				limit = { scope:actor = { has_variable = bag_asp_log_on } }
				interaction_target:target_1 = { debug_log_scopes = no }
			}
			bag_asp_vacate = yes
		}
	}
}

						else_if = { limit = { cabinet_action = cabinet_action:promote_culture interaction_target:target_1 ?= { bag_asp_needs_half = yes OR = { bag_asp_half_control_ok = yes has_variable = bag_asp_busy_noblesse_oblige_cabinet_action has_variable = bag_asp_busy_increase_control_province } } } add = scope:actor.var:bag_asp_rank_promote_culture }
						else_if = { limit = { cabinet_action = cabinet_action:diplomatic_endeavors scope:actor = { OR = { has_countries_with_near_coalition_grade_antagonism = yes has_countries_with_coalition_grade_antagonism = yes } } } add = scope:actor.var:bag_asp_rank_diplomatic_antagonism }
						else_if = { limit = { cabinet_action = cabinet_action:integrate_province } add = scope:actor.var:bag_asp_rank_integrate_province }
						else_if = { limit = { cabinet_action = cabinet_action:integrate_area } add = scope:actor.var:bag_asp_rank_integrate_area }
						else_if = { limit = { cabinet_action = cabinet_action:reduce_war_exhaustion } add = scope:actor.var:bag_asp_rank_reduce_war_exhaustion }
						else_if = { limit = { cabinet_action = cabinet_action:stabilize_country } add = scope:actor.var:bag_asp_rank_stabilize_country }
						else_if = { limit = { cabinet_action = cabinet_action:increase_control_province NOT = { bag_asp_paired_seat = yes } } add = scope:actor.var:bag_asp_rank_increase_control_province }
						else_if = { limit = { cabinet_action = cabinet_action:increase_control_area } add = scope:actor.var:bag_asp_rank_increase_control_area }
						else_if = { limit = { cabinet_action = cabinet_action:send_secret_royal_inspectors } add = scope:actor.var:bag_asp_rank_send_secret_royal_inspectors }
						else_if = { limit = { cabinet_action = cabinet_action:promote_religion } add = scope:actor.var:bag_asp_rank_promote_religion }
						else_if = { limit = { cabinet_action = cabinet_action:pacify_population } add = scope:actor.var:bag_asp_rank_pacify_population }
						else_if = { limit = { cabinet_action = cabinet_action:appease_estate } add = scope:actor.var:bag_asp_rank_appease_estate }
						else_if = { limit = { cabinet_action = cabinet_action:alleviate_concerns } add = scope:actor.var:bag_asp_rank_alleviate_concerns }
						else_if = { limit = { cabinet_action = cabinet_action:reduce_inflation } add = scope:actor.var:bag_asp_rank_reduce_inflation }
						else_if = { limit = { cabinet_action = cabinet_action:develop_province } add = scope:actor.var:bag_asp_rank_develop_province }
						else_if = { limit = { cabinet_action = cabinet_action:study_institutions } add = scope:actor.var:bag_asp_rank_study_institutions }

						limit = { exists = cabinet_member }

								limit = { has_variable = bag_asp_claimed }

						}

					limit = { has_variable = bag_asp_placed }

			}
			if = {
				limit = { has_variable = bag_asp_em_go }
				remove_variable = bag_asp_em_go
			}
		}
		bag_asp_clear_busy = yes
	}

bag_asp_place_integrate_province = {
	every_cabinet = {
		limit = {
			bag_asp_free_seat = yes
			has_cabinet_action = yes
			cabinet_action = cabinet_action:integrate_province
		}
		cabinet_member = {
			set_variable = { name = bag_asp_claimed value = yes }
			set_variable = { name = bag_asp_kept value = yes }
		}
	}
	# His rule 13.10: in a full hand-out the row takes the best advisors for it,
	# even from a lower row: while a free seat beats its weakest kept one in the
	# skill, the action moves there with its targets.
	if = {
		limit = { NOT = { has_variable = bag_asp_fill_only } }
		set_variable = { name = bag_asp_moving value = yes }
		while = {
			limit = {
				has_variable = bag_asp_moving
				any_cabinet = { cabinet_member ?= { has_variable = bag_asp_kept } }
			}
			remove_variable = bag_asp_moving
			ordered_cabinet = {
				limit = { cabinet_member ?= { has_variable = bag_asp_kept } }
				order_by = {
					value = 0
					subtract = cabinet_member.mil
				}
				save_scope_as = bag_asp_from
			}
			set_variable = { name = bag_asp_low value = scope:bag_asp_from.cabinet_member.mil }
			ordered_cabinet = {
				limit = {
					bag_asp_free_seat = yes
					cabinet_member = { mil > scope:actor.var:bag_asp_low }
				}
				order_by = cabinet_member.mil
				save_scope_as = bag_asp_seat
				scope:actor = {
					bag_asp_move_integrate_province = yes
					set_variable = { name = bag_asp_moving value = yes }
				}
			}
		}
		if = {
			limit = { has_variable = bag_asp_moving }
			remove_variable = bag_asp_moving
		}
	}
	every_cabinet = {
		limit = { cabinet_member ?= { has_variable = bag_asp_kept } }
		cabinet_member = { remove_variable = bag_asp_kept }
	}
	set_variable = { name = bag_asp_placed value = yes }
	while = {
		limit = {
			has_variable = bag_asp_placed
			any_cabinet = { bag_asp_free_seat = yes }
		}
		remove_variable = bag_asp_placed
		ordered_cabinet = {
			limit = {
				bag_asp_free_seat = yes
			}
			order_by = cabinet_member.mil
			save_scope_as = bag_asp_seat
			bag_asp_try_integrate_province = yes
		}
	}
	if = {
		limit = { has_variable = bag_asp_placed }
		remove_variable = bag_asp_placed
	}
}

# Scope: country (scope:actor).
bag_asp_place_promote_culture = {
	every_cabinet = {
		limit = {
			bag_asp_free_seat = yes
			has_cabinet_action = yes
			cabinet_action = cabinet_action:promote_culture
			interaction_target:target_1 ?= { bag_asp_needs_half = yes OR = { bag_asp_half_control_ok = yes has_variable = bag_asp_busy_noblesse_oblige_cabinet_action has_variable = bag_asp_busy_increase_control_province } }
		}
		cabinet_member = {
			set_variable = { name = bag_asp_claimed value = yes }
			set_variable = { name = bag_asp_kept value = yes }
		}
	}
	# His rule 13.10: locations assimilate each on their own, so a kept
	# assimilation moves to a province with more locations to bring to half.
	if = {
		limit = { NOT = { has_variable = bag_asp_fill_only } }
		every_cabinet = {
			limit = {
				cabinet_member ?= { has_variable = bag_asp_kept }
				exists = interaction_target:target
				exists = interaction_target:target_1
			}
			save_scope_as = bag_asp_seat
			interaction_target:target = { save_scope_as = target }
			interaction_target:target_1 = {
				save_scope_as = bag_asp_cur
				save_scope_value_as = { name = bag_asp_cur_rank value = bag_asp_half_rank }
			}
			scope:actor = {
				ordered_province = {
					limit = {
						NOT = { has_variable = bag_asp_busy_promote_culture }
						any_location_in_province = { bag_asp_assimilable_to_target = yes }
						any_location_in_province = {
							OR = {
								NOT = { dominant_culture = scope:target }
								local_cultural_unity < 0.95
							}
						}
						OR = {
							"culture_percentage(scope:target)" < 0.95
							any_location_in_province = { NOT = { dominant_culture = scope:target } }
						}
						NOT = { has_variable = bag_asp_busy_promote_culture }
						OR = {
							scope:actor = { is_ai = no }
							AND = {
								trigger_if = {
									limit = { this = scope:actor.capital.province }
									scope:target = scope:actor.culture
								}
								any_location_in_province = { naturally_assimilating_culture = scope:target }
								trigger_if = {
									limit = { exists = dominant_culture }
									OR = {
										AND = {
											NOT = { dominant_culture = scope:target }
											dominant_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
										}
										AND = {
											dominant_culture = scope:target
											any_location_in_province = {
												exists = secondary_culture
												secondary_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
											}
										}
									}
								}
								trigger_else = { always = no }
							}
						}
						bag_asp_needs_half = yes bag_asp_control_reachable = yes
						bag_asp_half_rank > scope:bag_asp_cur_rank
					}
					order_by = bag_asp_half_score
					save_scope_as = bag_asp_new
					scope:bag_asp_cur = { remove_variable = bag_asp_busy_promote_culture }
					scope:bag_asp_new = { set_variable = { name = bag_asp_busy_promote_culture value = yes } }
					scope:bag_asp_seat = {
						stop_cabinet_action = {}
						set_cabinet_action = { type = cabinet_action:promote_culture target = scope:target target_1 = scope:bag_asp_new }
					}
					debug_log = "bag_asp: assimilation moved to a province with more locations to bring to half"
				}
			}
		}
		# The control seat of a province left behind goes now.
		bag_asp_unpair = yes
	}
	# His rule 13.10: in a full hand-out the row takes the best advisors for it,
	# even from a lower row: while a free seat beats its weakest kept one in the
	# skill, the action moves there with its targets.
	if = {
		limit = { NOT = { has_variable = bag_asp_fill_only } }
		set_variable = { name = bag_asp_moving value = yes }
		while = {
			limit = {
				has_variable = bag_asp_moving
				any_cabinet = { cabinet_member ?= { has_variable = bag_asp_kept } }
			}
			remove_variable = bag_asp_moving
			ordered_cabinet = {
				limit = { cabinet_member ?= { has_variable = bag_asp_kept } }
				order_by = {
					value = 0
					subtract = cabinet_member.dip
				}
				save_scope_as = bag_asp_from
			}
			set_variable = { name = bag_asp_low value = scope:bag_asp_from.cabinet_member.dip }
			ordered_cabinet = {
				limit = {
					bag_asp_free_seat = yes
					NOT = { bag_asp_paired_seat = yes }
					cabinet_member = { dip > scope:actor.var:bag_asp_low }
				}
				order_by = cabinet_member.dip
				save_scope_as = bag_asp_seat
				scope:actor = {
					bag_asp_move_promote_culture = yes
					set_variable = { name = bag_asp_moving value = yes }
				}
			}
		}
		if = {
			limit = { has_variable = bag_asp_moving }
			remove_variable = bag_asp_moving
		}
	}
	every_cabinet = {
		limit = { cabinet_member ?= { has_variable = bag_asp_kept } }
		cabinet_member = { remove_variable = bag_asp_kept }
	}
	if = {
		limit = { has_variable = bag_asp_speedup_on }
		bag_asp_keep_pairs = yes
	}
	set_variable = { name = bag_asp_placed value = yes }
	while = {
		limit = {
			has_variable = bag_asp_placed
			any_cabinet = { bag_asp_free_seat = yes }
		}
		remove_variable = bag_asp_placed
		ordered_cabinet = {
			limit = {
				bag_asp_free_seat = yes
			}
			order_by = cabinet_member.dip
			save_scope_as = bag_asp_seat
			bag_asp_try_promote_culture = yes
		}
		if = {
			limit = {
				has_variable = bag_asp_placed
				has_variable = bag_asp_speedup_on
			}
			bag_asp_pair = yes
		}
	}
	if = {
		limit = { has_variable = bag_asp_placed }
		remove_variable = bag_asp_placed
	}
}

	every_cabinet = {
		limit = {
			bag_asp_free_seat = yes
			has_cabinet_action = yes
			cabinet_action = cabinet_action:promote_culture
		}
		cabinet_member = {
			set_variable = { name = bag_asp_claimed value = yes }
			set_variable = { name = bag_asp_kept value = yes }
		}
	}
	# His rule 13.10: in a full hand-out the row takes the best advisors for it,
	# even from a lower row: while a free seat beats its weakest kept one in the
	# skill, the action moves there with its targets.
	if = {
		limit = { NOT = { has_variable = bag_asp_fill_only } }
		set_variable = { name = bag_asp_moving value = yes }
		while = {
			limit = {
				has_variable = bag_asp_moving
				any_cabinet = { cabinet_member ?= { has_variable = bag_asp_kept } }
			}
			remove_variable = bag_asp_moving
			ordered_cabinet = {
				limit = { cabinet_member ?= { has_variable = bag_asp_kept } }
				order_by = {
					value = 0
					subtract = cabinet_member.dip
				}
				save_scope_as = bag_asp_from
			}
			set_variable = { name = bag_asp_low value = scope:bag_asp_from.cabinet_member.dip }
			ordered_cabinet = {
				limit = {
					bag_asp_free_seat = yes
					NOT = { bag_asp_paired_seat = yes }
					cabinet_member = { dip > scope:actor.var:bag_asp_low }
				}
				order_by = cabinet_member.dip
				save_scope_as = bag_asp_seat
				scope:actor = {
					bag_asp_move_promote_culture_full = yes
					set_variable = { name = bag_asp_moving value = yes }
				}
			}
		}
		if = {
			limit = { has_variable = bag_asp_moving }
			remove_variable = bag_asp_moving
		}
	}
	every_cabinet = {
		limit = { cabinet_member ?= { has_variable = bag_asp_kept } }
		cabinet_member = { remove_variable = bag_asp_kept }
	}
	if = {
		limit = { has_variable = bag_asp_speedup_on }
		bag_asp_keep_pairs = yes
	}
	set_variable = { name = bag_asp_placed value = yes }
	while = {
		limit = {
			has_variable = bag_asp_placed
			any_cabinet = { bag_asp_free_seat = yes }
		}
		remove_variable = bag_asp_placed
		ordered_cabinet = {
			limit = {
				bag_asp_free_seat = yes
			}
			order_by = cabinet_member.dip
			save_scope_as = bag_asp_seat
			bag_asp_try_promote_culture_full = yes
		}
		if = {
			limit = {
				has_variable = bag_asp_placed
				has_variable = bag_asp_speedup_on
			}
			bag_asp_pair = yes
		}
	}
	if = {
		limit = { has_variable = bag_asp_placed }
		remove_variable = bag_asp_placed
	}
}

			set_variable = { name = bag_asp_kept value = yes }
		}
	}
	debug_log = "bag_asp: integrate_province moved to an advisor better at it"

# «Ускорять контролем», his rule 13.10: a control action (noblesse_oblige_cabinet_action, increase_control_province,
# the first that fits; the game forbids both on one province) is bound to an
# assimilation. It comes with it, whatever its own row's place, from a free
# seat (in a full hand-out a seat a lower row would have taken; in a fill, an
# empty one), and is released with it. The seat's advisor carries
# bag_asp_paired while it lasts.

# Scope: country (scope:actor); scope:bag_asp_last is the province just given
# to an assimilation. Leaves bag_asp_placed set, as the assimilation left it.
bag_asp_pair = {
	scope:bag_asp_last = { save_scope_as = bag_asp_at }
	remove_variable = bag_asp_placed
	ordered_cabinet = {
		limit = {
			bag_asp_free_seat = yes
		}
		order_by = cabinet_member.adm
		save_scope_as = bag_asp_seat
		bag_asp_try_at_noblesse_oblige_cabinet_action = yes
		if = {
			limit = { scope:actor = { NOT = { has_variable = bag_asp_placed } } }
			bag_asp_try_at_increase_control_province = yes
		}
		if = {
			limit = { scope:actor = { has_variable = bag_asp_placed } }
			cabinet_member = { set_variable = { name = bag_asp_paired value = yes } }
			debug_log = "bag_asp: assimilation sped up with a control action"
		}
	}
	set_variable = { name = bag_asp_placed value = yes }
}

# Scope: country (scope:actor). For the assimilations this pass keeps, the
# best province first: a control seat already with one is kept with it; one
# without gets one now. His rule 13.10 is assimilation, control, assimilation,
# control: where no seat is free (his run 13.10 07:12: two integrations above,
# two assimilations, no control for a month), the weakest assimilation still
# waiting gives its seat to the control of a better one, in a full hand-out
# and only where a control action fits that province.
bag_asp_keep_pairs = {
	every_cabinet = {
		limit = {
			has_cabinet_action = yes
			cabinet_action = cabinet_action:promote_culture
			exists = interaction_target:target_1
			cabinet_member ?= { has_variable = bag_asp_claimed }
		}
		cabinet_member = { set_variable = { name = bag_asp_pending value = yes } }
	}
	while = {
		limit = { any_cabinet = { cabinet_member ?= { has_variable = bag_asp_pending } } }
		ordered_cabinet = {
			limit = { cabinet_member ?= { has_variable = bag_asp_pending } }
			order_by = bag_asp_seat_rank
			cabinet_member = { remove_variable = bag_asp_pending }
			interaction_target:target_1 = { save_scope_as = bag_asp_last }
		}
		if = {
			limit = {
				any_cabinet = {
					bag_asp_paired_seat = yes
					interaction_target:selected_province ?= { this = scope:bag_asp_last }
				}
			}
			every_cabinet = {
				limit = {
					bag_asp_paired_seat = yes
					interaction_target:selected_province ?= { this = scope:bag_asp_last }
				}
				cabinet_member = { set_variable = { name = bag_asp_claimed value = yes } }
			}
		}
		else = {
			if = {
				limit = {
					NOT = { has_variable = bag_asp_fill_only }
					NOT = { any_cabinet = { bag_asp_free_seat = yes } }
					any_cabinet = { cabinet_member ?= { has_variable = bag_asp_pending } }
					scope:bag_asp_last = { bag_asp_control_fits = yes }
				}
				ordered_cabinet = {
					limit = { cabinet_member ?= { has_variable = bag_asp_pending } }
					order_by = {
						value = 0
						subtract = bag_asp_seat_rank
					}
					bag_asp_vacate = yes
					cabinet_member = {
						remove_variable = bag_asp_pending
						remove_variable = bag_asp_claimed
					}
					debug_log = "bag_asp: an assimilation gave its seat to the control of a better one"
				}
			}
			bag_asp_pair = yes
			remove_variable = bag_asp_placed
		}
	}
}

# Scope: country (scope:actor), after bag_asp_mark_busy. A control seat whose
# assimilation is gone is stopped, so this pass can fill it; with the setting
# off, or the seat given something else, it is only no longer bound.
bag_asp_unpair = {
	every_cabinet = {
		limit = { cabinet_member ?= { has_variable = bag_asp_paired } }
		if = {
			limit = {
				scope:actor = { has_variable = bag_asp_speedup_on }
				bag_asp_paired_seat = yes
				interaction_target:selected_province ?= { NOT = { has_variable = bag_asp_busy_promote_culture } }
			}
			bag_asp_vacate = yes
			debug_log = "bag_asp: a control seat released with its assimilation"
		}
		else_if = {
			limit = {
				OR = {
					scope:actor = { NOT = { has_variable = bag_asp_speedup_on } }
					NOT = { bag_asp_paired_seat = yes }
				}
			}
			cabinet_member = { remove_variable = bag_asp_paired }
		}
	}
}

# not taken, and the 40 % control floor here hid it; the floor is gone and a
# failed status test now says which half of it failed.
bag_asp_probe_half = {
	if = {
		limit = { has_variable = bag_asp_auto_on }
		if = {
			limit = {
				scope:actor = {
					var:bag_asp_culture_target = 2
					any_accepted_culture = { always = yes }
				}
			}
			scope:actor = {
				ordered_accepted_culture = {
					order_by = "scope:actor.culture_percentage_in_country(this)"
					save_scope_as = target
				}
			}
		}
		else = {
			scope:actor.culture = { save_scope_as = target }
		}
		scope:actor = {
			every_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_promote_culture }
					any_location_in_province = {
						owner ?= scope:actor
						OR = {
							integration_level = integrated
							integration_level = conquered
						}
					}
				}
				set_global_variable = { name = bag_asp_probe value = bag_asp_half_rank }
				if = {
					limit = { NOT = { bag_asp_needs_half = yes } }
					if = {
						limit = {
							any_location_in_province = {
								owner ?= scope:actor
								OR = {
									integration_level = integrated
									AND = {
										integration_level = conquered
										bag_asp_short_of_half = yes
									}
								}
								any_pop = { culture = { NOT = { is_primary_or_accepted_in = scope:actor } } }
							}
						}
						debug_log = "bag_asp: probe: not taken, its foreign pops are all locked (tribesmen, a culture-locked faith, a blocked class); rank [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_probe_value')|0]:"
						debug_log_scopes = no
						bag_asp_probe_pops = yes
					}
					else_if = {
						limit = {
							any_location_in_province = {
								owner ?= scope:actor
								integration_level = integrated
							}
						}
						debug_log = "bag_asp: probe: not taken, an integrated location here has no pop of a culture neither primary nor accepted; rank [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_probe_value')|0]:"
					}
					else = {
						debug_log = "bag_asp: probe: not taken, no location integrated, and the conquered ones are at half; rank [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_probe_value')|0]:"
					}
				}
				else_if = {
					limit = { NOT = { bag_asp_half_control_ok = yes } }
					debug_log = "bag_asp: probe: needs a control seat: no location to assimilate has control at the floor; rank [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_probe_value')|0]:"
				}
				else_if = {
					limit = {
						NAND = {
							NOT = { has_variable = bag_asp_busy_promote_culture }
							any_location_in_province = { bag_asp_assimilable_to_target = yes }
							any_location_in_province = {
								OR = {
									NOT = { dominant_culture = scope:target }
									local_cultural_unity < 0.95
								}
							}
							OR = {
								"culture_percentage(scope:target)" < 0.95
								any_location_in_province = { NOT = { dominant_culture = scope:target } }
							}
							NOT = { has_variable = bag_asp_busy_promote_culture }
							OR = {
								scope:actor = { is_ai = no }
								AND = {
									trigger_if = {
										limit = { this = scope:actor.capital.province }
										scope:target = scope:actor.culture
									}
									any_location_in_province = { naturally_assimilating_culture = scope:target }
									trigger_if = {
										limit = { exists = dominant_culture }
										OR = {
											AND = {
												NOT = { dominant_culture = scope:target }
												dominant_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
											}
											AND = {
												dominant_culture = scope:target
												any_location_in_province = {
													exists = secondary_culture
													secondary_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
												}
											}
										}
									}
									trigger_else = { always = no }
								}
							}
						}
					}
					debug_log = "bag_asp: probe: not taken, the game's own province test fails; rank [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_probe_value')|0]:"
				}
				else_if = {
					limit = {
						NAND = {
							# His rule 05.10: where an accepted culture leads the primary by
							# 30 points, do not push the primary.
							trigger_if = {
								limit = { scope:target = scope:actor.culture }
								NAND = {
									exists = dominant_culture
									NOT = { dominant_culture = scope:actor.culture }
									dominant_culture = { is_primary_or_accepted_in = scope:actor }
									"culture_percentage(dominant_culture)" >= {
										value = "culture_percentage(scope:actor.culture)"
										add = 0.3
									}
								}
							}
						}
					}
					debug_log = "bag_asp: probe: not taken, an accepted culture leads the primary by 30 points; rank [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_probe_value')|0]:"
				}
				else = {
					debug_log = "bag_asp: probe: passes every test; rank [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_probe_value')|0]:"
				}
				debug_log_scopes = no
			}
		}
	}
	remove_global_variable = bag_asp_probe
}
