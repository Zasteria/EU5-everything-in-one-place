# assimilate_primary: the mod's own council automation, the monthly pass.
# Rebuilt by tools/generate.py (tools/council.py).

# Scope: country. Every month for a player with the checkbox on; the work runs
# every `period` months, and every `review` months (0 = never) the standing
# actions, which never finish on their own, are stopped and re-chosen.
bag_asp_tick = {
	if = {
		limit = { NOT = { has_variable = bag_asp_period } }
		bag_asp_sync = yes
	}
	change_variable = { name = bag_asp_months add = 1 }
	change_variable = { name = bag_asp_review_months add = 1 }
	if = {
		limit = {
			var:bag_asp_review > 0
			var:bag_asp_review_months >= var:bag_asp_review
		}
		set_variable = { name = bag_asp_review_months value = 0 }
		bag_asp_review_seats = yes
		set_variable = { name = bag_asp_months value = var:bag_asp_period }
	}
	if = {
		limit = { var:bag_asp_months >= var:bag_asp_period }
		set_variable = { name = bag_asp_months value = 0 }
		bag_asp_run = yes
	}
}

# Scope: country. Stop the standing actions of the pool so the next pass
# re-chooses those seats down the priority list.
bag_asp_review_seats = {
	every_cabinet = {
		limit = {
			has_cabinet_action = yes
			OR = {
				cabinet_action = cabinet_action:stabilize_country
				cabinet_action = cabinet_action:increase_control_province
				cabinet_action = cabinet_action:increase_control_area
				cabinet_action = cabinet_action:send_secret_royal_inspectors
				cabinet_action = cabinet_action:pacify_population
				cabinet_action = cabinet_action:reduce_inflation
				cabinet_action = cabinet_action:develop_province
				cabinet_action = cabinet_action:pop_promote_action
				cabinet_action = cabinet_action:strengthen_government
				cabinet_action = cabinet_action:diplomatic_endeavors
				cabinet_action = cabinet_action:counter_espionage_action
			}
		}
		stop_cabinet_action = {}
		debug_log = "bag_asp: review, a standing action stopped to be re-chosen"
	}
}

# Scope: country. Every seat with a member and no action gets the first action
# down «Приоритет» that is in use, allowed now and has a target.
bag_asp_run = {
	save_scope_as = actor
	if = {
		limit = { NOT = { has_variable_list = bag_asp_order } }
		bag_asp_rebuild = yes
	}
	bag_asp_mark_busy = yes
	every_cabinet = {
		limit = {
			has_cabinet_action = no
			exists = cabinet_member
		}
		save_scope_as = bag_asp_seat
		scope:actor = {
			every_in_list = {
				variable = bag_asp_order
				save_temporary_scope_as = bag_asp_item
				if = {
					limit = {
						scope:actor = {
							NOT = { has_variable = bag_asp_filled }
							is_target_in_variable_list = { name = bag_asp_used target = scope:bag_asp_item }
						}
					}
					scope:bag_asp_seat = {
						switch = {
							trigger = scope:bag_asp_item
							flag:bag_asp_integrate_province = { bag_asp_try_integrate_province = yes }
							flag:bag_asp_integrate_area = { bag_asp_try_integrate_area = yes }
							flag:bag_asp_reduce_war_exhaustion = { bag_asp_try_reduce_war_exhaustion = yes }
							flag:bag_asp_stabilize_country = { bag_asp_try_stabilize_country = yes }
							flag:bag_asp_increase_control_province = { bag_asp_try_increase_control_province = yes }
							flag:bag_asp_increase_control_area = { bag_asp_try_increase_control_area = yes }
							flag:bag_asp_send_secret_royal_inspectors = { bag_asp_try_send_secret_royal_inspectors = yes }
							flag:bag_asp_promote_culture = { bag_asp_try_promote_culture = yes }
							flag:bag_asp_promote_religion = { bag_asp_try_promote_religion = yes }
							flag:bag_asp_pacify_population = { bag_asp_try_pacify_population = yes }
							flag:bag_asp_recovery_effort = { bag_asp_try_recovery_effort = yes }
							flag:bag_asp_appease_estate = { bag_asp_try_appease_estate = yes }
							flag:bag_asp_alleviate_concerns = { bag_asp_try_alleviate_concerns = yes }
							flag:bag_asp_reduce_inflation = { bag_asp_try_reduce_inflation = yes }
							flag:bag_asp_develop_province = { bag_asp_try_develop_province = yes }
							flag:bag_asp_study_institutions = { bag_asp_try_study_institutions = yes }
							flag:bag_asp_pop_promote_action = { bag_asp_try_pop_promote_action = yes }
							flag:bag_asp_strengthen_government = { bag_asp_try_strengthen_government = yes }
							flag:bag_asp_diplomatic_endeavors = { bag_asp_try_diplomatic_endeavors = yes }
							flag:bag_asp_counter_espionage_action = { bag_asp_try_counter_espionage_action = yes }
						}
					}
				}
			}
			if = {
				limit = { has_variable = bag_asp_filled }
				remove_variable = bag_asp_filled
			}
			else = {
				debug_log = "bag_asp: a seat stays empty, nothing in use fits"
			}
		}
	}
	bag_asp_clear_busy = yes
}

			interaction_target:target ?= { set_variable = { name = bag_asp_busy_develop_province value = yes } }

# Scope: cabinet (a seat), saved as scope:bag_asp_seat; scope:actor = country.
bag_asp_try_integrate_province = {
	if = {
		limit = {
			scope:actor = { NOT = { has_variable = bag_asp_filled } }
			cabinet_action:integrate_province = { is_available_for = scope:actor }
		}
		scope:actor = {
			ordered_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_integrate_province }
					NOT = { has_variable = bag_asp_busy_integrate_province }
					any_location_in_province = {
						integration_level = conquered
					}
				}
				order_by = population
				save_temporary_scope_as = bag_asp_pick
				scope:bag_asp_seat = { set_cabinet_action = { type = cabinet_action:integrate_province target = scope:bag_asp_pick } }
				scope:bag_asp_pick = { set_variable = { name = bag_asp_busy_integrate_province value = yes } }
				scope:actor = { set_variable = { name = bag_asp_filled value = yes } }
				debug_log = "bag_asp: seat set to integrate_province"
			}
		}
	}
}

# Scope: cabinet (a seat), saved as scope:bag_asp_seat; scope:actor = country.
bag_asp_try_promote_culture = {
	if = {
		limit = {
			scope:actor = { NOT = { has_variable = bag_asp_filled } }
			cabinet_action:promote_culture = { is_available_for = scope:actor }
		}
		scope:actor.culture = { save_scope_as = target }
		scope:actor = {
			ordered_province = {
				limit = {
					NOT = { has_variable = bag_asp_busy_promote_culture }
					any_location_in_province = {
						OR = {
							NOT = { dominant_culture = scope:target }
							local_cultural_unity < 0.95
						}
					}
					NOT = { has_variable = bag_asp_busy_promote_culture }
					OR = {
						scope:actor = { is_ai = no }
						AND = {
							trigger_if = {
								limit = { this = scope:actor.capital.province }
								scope:target = scope:actor.culture
							}
							any_location_in_province = { naturally_assimilating_culture = scope:target }
							trigger_if = {
								limit = { exists = dominant_culture }
								OR = {
									AND = {
										NOT = { dominant_culture = scope:target }
										dominant_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
									}
									AND = {
										dominant_culture = scope:target
										any_location_in_province = {
											exists = secondary_culture
											secondary_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
										}
									}
								}
							}
							trigger_else = { always = no }
						}
					}
					# His rule 05.10: where an accepted culture leads the primary by
					# 30 points, do not push the primary.
					NAND = {
						exists = dominant_culture
						NOT = { dominant_culture = scope:actor.culture }
						dominant_culture = { is_primary_or_accepted_in = scope:actor }
						"culture_percentage(dominant_culture)" >= {
							value = "culture_percentage(scope:actor.culture)"
							add = 0.3
						}
					}
				}
				order_by = population
				save_temporary_scope_as = bag_asp_pick
				scope:bag_asp_seat = { set_cabinet_action = { type = cabinet_action:promote_culture target = scope:target target_1 = scope:bag_asp_pick } }
				scope:bag_asp_pick = { set_variable = { name = bag_asp_busy_promote_culture value = yes } }
				scope:actor = { set_variable = { name = bag_asp_filled value = yes } }
				debug_log = "bag_asp: seat set to promote_culture"
			}
		}
	}
}
