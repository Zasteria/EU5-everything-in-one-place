# Scope: cabinet (a seat). It has an advisor whom this pass has not chosen yet.
bag_asp_free_seat = {
	exists = cabinet_member
	cabinet_member = { NOT = { has_variable = bag_asp_claimed } }
}

# Scope: location. No primary or accepted culture holds half of it.
bag_asp_short_of_half = {
	NAND = {
		exists = dominant_culture
		dominant_culture = { is_primary_or_accepted_in = scope:actor }
		"culture_percentage(dominant_culture)" >= 0.5
	}
}

# Scope: province; scope:actor is the country. His rule 13.10, by status: a
# location of it is integrated but not core yet, or conquered and short of half
# (conquered at half needs integration, not assimilation).
bag_asp_needs_half = {
	any_location_in_province = {
		owner ?= scope:actor
		# His run 13.10: only while a pop here can still turn.
		bag_asp_assimilable_any = yes
		OR = {
			integration_level = integrated
			AND = {
				integration_level = conquered
				bag_asp_short_of_half = yes
			}
		}
	}
}

# Scope: any target. A busy mark of the pass is on it.
bag_asp_busy_any = {
	OR = { has_variable = bag_asp_busy_appease_estate has_variable = bag_asp_busy_develop_province has_variable = bag_asp_busy_increase_control_area has_variable = bag_asp_busy_increase_control_province has_variable = bag_asp_busy_integrate_area has_variable = bag_asp_busy_integrate_province has_variable = bag_asp_busy_noblesse_oblige_cabinet_action has_variable = bag_asp_busy_pacify_population has_variable = bag_asp_busy_pop_promote_action has_variable = bag_asp_busy_promote_culture has_variable = bag_asp_busy_promote_religion has_variable = bag_asp_busy_recovery_effort has_variable = bag_asp_busy_send_secret_royal_inspectors has_variable = bag_asp_busy_study_institutions }
}

# Scope: cabinet. A control action bound to an assimilation («Ускорять контролем»).
bag_asp_paired_seat = {
	has_cabinet_action = yes
	OR = {
		cabinet_action = cabinet_action:noblesse_oblige_cabinet_action
		cabinet_action = cabinet_action:increase_control_province
	}
	cabinet_member ?= { has_variable = bag_asp_paired }
}

# Scope: province; scope:actor is the country. A control action of the
# speedup can be put on it now: available, not on a seat yet (one seat each),
# and its target test holds here.
bag_asp_control_fits = {
	OR = {
		AND = {
			cabinet_action:noblesse_oblige_cabinet_action = { is_available_for = scope:actor }
			scope:actor = { NOT = { any_cabinet = { has_cabinet_action = yes cabinet_action = cabinet_action:noblesse_oblige_cabinet_action } } }
			NOT = { has_variable = bag_asp_busy_noblesse_oblige_cabinet_action }
			province_average_control < 0.95
			NOT = { has_variable = bag_asp_busy_noblesse_oblige_cabinet_action }
			NOT = { has_variable = bag_asp_busy_increase_control_province }
			NOT = { area ?= { has_variable = bag_asp_busy_increase_control_area } }
		}
		AND = {
			cabinet_action:increase_control_province = { is_available_for = scope:actor }
			scope:actor = { NOT = { any_cabinet = { has_cabinet_action = yes cabinet_action = cabinet_action:increase_control_province } } }
			NOT = { has_variable = bag_asp_busy_increase_control_province }
			province_average_control < 0.95
			NOT = { has_variable = bag_asp_busy_increase_control_province }
			NOT = { has_variable = bag_asp_busy_noblesse_oblige_cabinet_action }
			NOT = { area ?= { has_variable = bag_asp_busy_increase_control_area } }
		}
	}
}

# Scope: province; scope:actor is the country. His rule 14.10: a location the
# half row would assimilate has control at its floor or above (Sauerland: the
# province at 20 %, Siegen, its only integrated location, at 0.12 %, and the
# assimilation there stood still). Without the setting, above zero.
bag_asp_half_control_ok = {
	any_location_in_province = {
		owner ?= scope:actor
		OR = {
			integration_level = integrated
			AND = {
				integration_level = conquered
				bag_asp_short_of_half = yes
			}
		}
		bag_asp_assimilable_any = yes
		local_control > 0
		OR = {
			scope:actor = { NOT = { has_variable = bag_asp_assim_control } }
			local_control >= {
				value = scope:actor.var:bag_asp_assim_control
				divide = 100
			}
		}
	}
}

# Scope: province; scope:actor is the country, scope:bag_asp_seat the seat
# about to assimilate. His rules 13.10 and 14.10: control at the floor
# (bag_asp_half_control_ok), or «Ускорять контролем» on, another free seat,
# and a control action that fits here. Neither: skipped until one holds.
bag_asp_control_reachable = {
	OR = {
		bag_asp_half_control_ok = yes
		AND = {
			scope:actor = {
				has_variable = bag_asp_speedup_on
				any_cabinet = {
					bag_asp_free_seat = yes
					NOT = { this = scope:bag_asp_seat }
				}
			}
			bag_asp_control_fits = yes
		}
	}
}
