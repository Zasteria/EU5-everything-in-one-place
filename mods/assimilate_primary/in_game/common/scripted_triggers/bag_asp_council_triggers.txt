# assimilate_primary: triggers of the council pass. Rebuilt by tools/generate.py.

# Scope: cabinet (a seat). It has an advisor whom this pass has not chosen yet.
bag_asp_free_seat = {
	exists = cabinet_member
	cabinet_member = { NOT = { has_variable = bag_asp_claimed } }
}

# Scope: location. No primary or accepted culture holds half of it.
bag_asp_short_of_half = {
	NAND = {
		exists = dominant_culture
		dominant_culture = { is_primary_or_accepted_in = scope:actor }
		"culture_percentage(dominant_culture)" >= 0.5
	}
}

# Scope: province; scope:actor is the country. His rule 13.10, by status: a
# location of it is integrated but not core yet, or conquered and short of half
# (conquered at half needs integration, not assimilation).
bag_asp_needs_half = {
	any_location_in_province = {
		owner ?= scope:actor
		OR = {
			integration_level = integrated
			AND = {
				integration_level = conquered
				bag_asp_short_of_half = yes
			}
		}
	}
}

# Scope: cabinet. A control action bound to an assimilation («Ускорять контролем»).
bag_asp_paired_seat = {
	has_cabinet_action = yes
	OR = {
		cabinet_action = cabinet_action:noblesse_oblige_cabinet_action
		cabinet_action = cabinet_action:increase_control_province
	}
	cabinet_member ?= { has_variable = bag_asp_paired }
}

# Scope: province; scope:actor is the country, scope:bag_asp_seat the seat
# about to assimilate. His rule 13.10: control above zero, or «Ускорять
# контролем» on, another free seat, and a control action that fits here.
bag_asp_control_reachable = {
	OR = {
		province_average_control > 0
		AND = {
			scope:actor = {
				has_variable = bag_asp_speedup_on
				any_cabinet = {
					bag_asp_free_seat = yes
					NOT = { this = scope:bag_asp_seat }
				}
			}
			OR = {
				AND = {
					cabinet_action:noblesse_oblige_cabinet_action = { is_available_for = scope:actor }
					scope:actor = { NOT = { any_cabinet = { has_cabinet_action = yes cabinet_action = cabinet_action:noblesse_oblige_cabinet_action } } }
					NOT = { has_variable = bag_asp_busy_noblesse_oblige_cabinet_action }
					province_average_control < 0.95
					NOT = { has_variable = bag_asp_busy_noblesse_oblige_cabinet_action }
					NOT = { has_variable = bag_asp_busy_increase_control_province }
					NOT = { area ?= { has_variable = bag_asp_busy_increase_control_area } }
				}
				AND = {
					cabinet_action:increase_control_province = { is_available_for = scope:actor }
					scope:actor = { NOT = { any_cabinet = { has_cabinet_action = yes cabinet_action = cabinet_action:increase_control_province } } }
					NOT = { has_variable = bag_asp_busy_increase_control_province }
					province_average_control < 0.95
					NOT = { has_variable = bag_asp_busy_increase_control_province }
					NOT = { has_variable = bag_asp_busy_noblesse_oblige_cabinet_action }
					NOT = { area ?= { has_variable = bag_asp_busy_increase_control_area } }
				}
			}
		}
	}
}
