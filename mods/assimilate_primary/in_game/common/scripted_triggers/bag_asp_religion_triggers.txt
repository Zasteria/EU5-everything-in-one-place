# Scope: province. Use the same population share on selection and on review.
bag_asp_religion_needs_half = {
	any_location_in_province = {
		owner ?= scope:actor
		population > 0
		religion_percentage = { religion = $FAITH$ value < 0.5 }
	}
}

# Scope: cabinet. Read the faith selected for this particular assignment.
bag_asp_running_religion_needs_half = {
	exists = interaction_target:target_religion
	exists = interaction_target:target_province
	interaction_target:target_religion = { save_temporary_scope_as = bag_asp_running_faith }
	interaction_target:target_province = {
		bag_asp_religion_needs_half = { FAITH = scope:bag_asp_running_faith }
	}
}
