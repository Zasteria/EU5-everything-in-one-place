# The list draws only while this exists, and a list change is committed here.
bag_asp__available_on_changed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		cmm_apply_list_change = { setting = bag_asp__available }
		bag_asp_rebuild = yes
		if = {
			limit = { has_variable = bag_asp_log_on }
			debug_log = "bag_asp: you changed the list available"
			save_scope_as = actor
			bag_asp_probe_order = yes
		}
		# His rule 13.10: a change of the lists hands every seat out at once.
		if = {
			limit = { has_variable = bag_asp_auto_on }
			bag_asp_plan_all = yes
		}
	}
}

# Called by the hidden window right after it asked the game's value window to
# open for the seat in var:bag_asp_sv_cab.
bag_asp_sv_opening = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: values, the hidden window asked the game's value window to open"
		if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
	}
}

# Called by a row of the value window right after it pressed a side.
bag_asp_sv_pressed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: values, the value window pressed the asked side"
		if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
		if = { limit = { has_variable = bag_asp_sv_left } remove_variable = bag_asp_sv_left }
		if = { limit = { has_variable = bag_asp_sv_right } remove_variable = bag_asp_sv_right }
		if = { limit = { has_variable = bag_asp_sv_pair } remove_variable = bag_asp_sv_pair }
	}
}

# Called by the hidden window as it closes the panels after a press.
bag_asp_sv_closed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: values, the panels close after the press"
		# The pair the bounce was for (bag_asp_sv_bounce): asked now, before the
		# panels close in this same call, so the window opens after them.
		if = {
			limit = { has_variable = bag_asp_sv_then }
			set_variable = { name = bag_asp_sv_want value = var:bag_asp_sv_then }
			set_variable = { name = bag_asp_sv_want_side value = var:bag_asp_sv_then_side }
			remove_variable = bag_asp_sv_then
			remove_variable = bag_asp_sv_then_side
			debug_log = "bag_asp: values, now the pair itself"
			bag_asp_sv_ask = yes
			remove_variable = bag_asp_sv_want
			remove_variable = bag_asp_sv_want_side
		}
		else_if = {
			limit = { has_variable = bag_asp_sv_bounced }
			remove_variable = bag_asp_sv_bounced
		}
	}
}

# Called by the hidden window two seconds after it opened the value window.
# A side still asked means no row pressed: the window left the pair out. His
# run 13.10 12:50:46: he had moved the advisor off centralization by hand, the
# mod put him back on values, and the window listed 11 rows without it. So
# the seat is pressed onto another pair first and then onto this one, as a
# second press switches pairs (his run 13.10 07:04). A second miss gives up:
# the next pass takes another pair.
bag_asp_sv_timeout = {
	scope = country
	is_shown = { always = yes }
	effect = {
		if = {
			limit = {
				OR = {
					has_variable = bag_asp_sv_left
					has_variable = bag_asp_sv_right
				}
			}
			# Probe: the pair the seat holds in the game's eyes when its row is missing.
			set_global_variable = { name = bag_asp_sv_seat value = var:bag_asp_sv_cab }
			debug_log = "bag_asp: values, no row pressed in two seconds; the seat holds [GetGlobalVariable('bag_asp_sv_seat').GetCabinet.GetSocietalValueName]"
			remove_global_variable = bag_asp_sv_seat
			if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
			if = { limit = { has_variable = bag_asp_sv_left } remove_variable = bag_asp_sv_left }
			if = { limit = { has_variable = bag_asp_sv_right } remove_variable = bag_asp_sv_right }
			if = { limit = { has_variable = bag_asp_sv_pair } remove_variable = bag_asp_sv_pair }
			# The window is closed by bag_asp_sv_closer, and the bounce waits
			# for it (bag_asp_sv_after_close): his run 13.10 18:30.
			set_variable = { name = bag_asp_sv_close value = yes }
			if = {
				limit = {
					has_variable = bag_asp_sv_cur
					NOT = { has_variable = bag_asp_sv_bounced }
				}
				set_variable = { name = bag_asp_sv_bounce_due value = yes }
			}
			else_if = {
				limit = { has_variable = bag_asp_sv_cur }
				debug_log = "bag_asp: values, missed again after another pair; the next pass takes another pair"
				set_variable = { name = bag_asp_sv_cur_side value = 0 }
				if = { limit = { has_variable = bag_asp_sv_bounced } remove_variable = bag_asp_sv_bounced }
			}
		}
	}
}

# Called by bag_asp_sv_closer once it has closed the window after a miss.
bag_asp_sv_after_close = {
	scope = country
	is_shown = { always = yes }
	effect = {
		if = { limit = { has_variable = bag_asp_sv_close } remove_variable = bag_asp_sv_close }
		if = {
			limit = { has_variable = bag_asp_sv_bounce_due }
			remove_variable = bag_asp_sv_bounce_due
			bag_asp_sv_bounce = yes
		}
	}
}
