# assimilate_primary: the lists' callbacks. Rebuilt by tools/generate.py.

# The list draws only while this exists, and a list change is committed here.
bag_asp__available_on_changed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		cmm_apply_list_change = { setting = bag_asp__available }
		bag_asp_rebuild = yes
		if = {
			limit = { has_variable = bag_asp_log_on }
			debug_log = "bag_asp: you changed the list available"
			save_scope_as = actor
			bag_asp_probe_order = yes
		}
		# His rule 13.10: a change of the lists hands every seat out at once.
		if = {
			limit = { has_variable = bag_asp_auto_on }
			bag_asp_plan_all = yes
		}
	}
}

# Called by the hidden window (bag_asp_hidden.gui) with the chosen seat's
# limit as the window reads it (Cabinet.GetMaxSocietalValueReachable): the
# number the monthly rule compares against, read where only the interface can.
bag_asp_sv_limit = {
	scope = country
	is_shown = { always = yes }
	effect = {
		set_global_variable = { name = bag_asp_sv_limit value = scope:limit }
		debug_log = "bag_asp: sv the window read the seat's limit [GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('bag_asp_sv_limit_value')|2]"
	}
}

# Called by the hidden window right after it asked the game's value window to
# open for the seat in var:bag_asp_sv_cab.
bag_asp_sv_opening = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: sv the hidden window asked the game's value window to open"
		if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
	}
}

# Called by a row of the value window right after it pressed a side; the
# window then closes itself (bag_asp_sv_close).
bag_asp_sv_pressed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: sv the value window pressed the asked side"
		if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
		if = { limit = { has_variable = bag_asp_sv_right } remove_variable = bag_asp_sv_right }
		if = { limit = { has_variable = bag_asp_sv_pair } remove_variable = bag_asp_sv_pair }
		if = { limit = { has_variable = bag_asp_sv_close } remove_variable = bag_asp_sv_close }
		set_variable = { name = bag_asp_sv_close value = yes }
	}
}

# Called by the hidden window as it shuts the value window. His ask 13.10.
bag_asp_sv_closed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: sv the value window closes"
		if = { limit = { has_variable = bag_asp_sv_close } remove_variable = bag_asp_sv_close }
	}
}

# Called once by each row of the value window while a press is awaited: names
# the row's pair, so a run without a press still says what the window showed.
bag_asp_sv_row = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: sv the value window shows a row"
		scope:sv ?= { debug_log_scopes = no }
	}
}
