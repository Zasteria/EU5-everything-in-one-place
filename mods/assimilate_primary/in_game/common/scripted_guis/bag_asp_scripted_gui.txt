# assimilate_primary: the lists' callbacks. Rebuilt by tools/generate.py.

# The list draws only while this exists, and a list change is committed here.
bag_asp__available_on_changed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		cmm_apply_list_change = { setting = bag_asp__available }
		bag_asp_rebuild = yes
		if = {
			limit = { has_variable = bag_asp_log_on }
			debug_log = "bag_asp: you changed the list available"
			save_scope_as = actor
			bag_asp_probe_order = yes
		}
		# His rule 13.10: a change of the lists hands every seat out at once.
		if = {
			limit = { has_variable = bag_asp_auto_on }
			bag_asp_plan_all = yes
		}
	}
}

# Called by the hidden window right after it asked the game's value window to
# open for the seat in var:bag_asp_sv_cab.
bag_asp_sv_opening = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: values, the hidden window asked the game's value window to open"
		if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
	}
}

# Called by a row of the value window right after it pressed a side.
bag_asp_sv_pressed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: values, the value window pressed the asked side"
		if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
		if = { limit = { has_variable = bag_asp_sv_left } remove_variable = bag_asp_sv_left }
		if = { limit = { has_variable = bag_asp_sv_right } remove_variable = bag_asp_sv_right }
		if = { limit = { has_variable = bag_asp_sv_pair } remove_variable = bag_asp_sv_pair }
	}
}

# Called by the hidden window as it closes the panels after a press.
bag_asp_sv_closed = {

	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: values, the panels close after the press"
	}
}

# Called by the hidden window two seconds after it opened the value window.
# A side still asked means no row pressed: the window leaves out the pair the
# seat already pushes. The next pass then takes another pair, and comes back
# to this one when the seat is off it.
bag_asp_sv_timeout = {
	scope = country
	is_shown = { always = yes }
	effect = {
		if = {
			limit = {
				OR = {
					has_variable = bag_asp_sv_left
					has_variable = bag_asp_sv_right
				}
			}
			debug_log = "bag_asp: values, no row pressed in two seconds; the panels close, the next pass takes another pair"
			if = { limit = { has_variable = bag_asp_sv_open } remove_variable = bag_asp_sv_open }
			if = { limit = { has_variable = bag_asp_sv_left } remove_variable = bag_asp_sv_left }
			if = { limit = { has_variable = bag_asp_sv_right } remove_variable = bag_asp_sv_right }
			if = { limit = { has_variable = bag_asp_sv_pair } remove_variable = bag_asp_sv_pair }
			if = {
				limit = { has_variable = bag_asp_sv_cur }
				set_variable = { name = bag_asp_sv_cur_side value = 0 }
			}
		}
	}
}

# Called once by each row of the value window while a press is awaited: names
# the row's pair, so a run without a press still says what the window showed.
bag_asp_sv_row = {
	scope = country
	is_shown = { always = yes }
	effect = {
		debug_log = "bag_asp: values, the value window shows a row"
		scope:sv ?= { debug_log_scopes = no }
	}
}
