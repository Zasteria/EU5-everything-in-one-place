# 1 nobles, 2 clergy, 3 burghers, 4 laborers, 5 soldiers, 6 peasants, 7 slaves, 8 tribesmen; -1 no pop type known. Root location, scope:cm_building_type.
cm_perf_bp_pop_code = {
	value = -1
	if = { limit = { NOT = { is_key_in_global_variable_map = { name = cm_building_type_to_pop_type target = scope:cm_building_type } } } value = -1 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:nobles } value = 1 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:clergy } value = 2 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:burghers } value = 3 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:laborers } value = 4 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:soldiers } value = 5 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:peasants } value = 6 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:slaves } value = 7 }
	else_if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:tribesmen } value = 8 }
}
