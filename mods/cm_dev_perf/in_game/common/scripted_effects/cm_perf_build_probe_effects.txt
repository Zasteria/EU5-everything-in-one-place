# cm_dev_perf +perf51: one debug.log line per build CM approves, with the numbers its
# worker gates read, so a run can be checked location by location (his ask 10-26: shortage
# limits that still build where there are no peasants, sliders that seem not to apply).
# `feature` is the pass that approved it (cm_perf_bp_feat, set in cm_q_approval_pass);
# the location and the building type follow on lines of their own (debug_log_scopes).
# Units are the game's: pop counts in thousands. Root is the location.
# Expects scope:cm_country, scope:cm_building_type.
cm_perf_build_probe = {
	# The type's own numbers only where CM knows its pop type: a lookup into a
	# key the map lacks is an error, not a zero. -1 everywhere else.
	if = {
		limit = { is_key_in_global_variable_map = { name = cm_building_type_to_pop_type target = scope:cm_building_type } }
		set_global_variable = { name = cm_perf_bp_v1 value = cm_perf_bp_pop_code }
		set_global_variable = { name = cm_perf_bp_v2 value = cm_location_unemployed_for_building_type }
		set_global_variable = { name = cm_perf_bp_v3 value = cm_perf_bp_free }
		set_global_variable = { name = cm_perf_bp_v4 value = cm_perf_bp_inflow }
		set_global_variable = { name = cm_perf_bp_v5 value = cm_perf_bp_measured }
		set_global_variable = { name = cm_perf_bp_v8 value = cm_perf_ab_prod_slack }
	} else = {
		set_global_variable = { name = cm_perf_bp_v1 value = -1 }
		set_global_variable = { name = cm_perf_bp_v2 value = -1 }
		set_global_variable = { name = cm_perf_bp_v3 value = -1 }
		set_global_variable = { name = cm_perf_bp_v4 value = -1 }
		set_global_variable = { name = cm_perf_bp_v5 value = -1 }
		set_global_variable = { name = cm_perf_bp_v8 value = -1 }
	}
	set_global_variable = { name = cm_perf_bp_v6 value = cm_perf_ab_prod_horizon_months }
	set_global_variable = { name = cm_perf_bp_v7 value = cm_perf_ab_food_horizon_months }
	set_global_variable = { name = cm_perf_bp_v9 value = cm_location_laborers_or_slaves_available }
	set_global_variable = { name = cm_perf_bp_v10 value = cm_location_fill_rate_for_laborers_or_slaves }
	set_global_variable = { name = cm_perf_bp_v11 value = cm_perf_ab_food_rgo_slack }
	set_global_variable = { name = cm_perf_bp_v12 value = num_pop_type:peasants }
	set_global_variable = { name = cm_perf_bp_v13 value = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)" }
	set_global_variable = { name = cm_perf_bp_v14 value = cm_location_fill_rate_for_peasants }
	debug_log = "CM build probe: feature [GetGlobalVariable('cm_perf_bp_feat').GetFlagName] pop_code=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v1')|2] unemployed=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v2')|2] free=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v3')|2] inflow=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v4')|2] measured=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v5')|2] prod_months=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v6')|2] food_months=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v7')|2] prod_slack=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v8')|2] rgo_free=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v9')|2] rgo_inflow=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v10')|2] food_rgo_slack=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v11')|2] peasants=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v12')|2] peasants_idle=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v13')|2] peasants_inflow=[GuiScope.SetRoot(GetPlayer.MakeScope).ScriptValue('cm_perf_bp_v14')|2]"
	debug_log_scopes = no
	scope:cm_building_type = { debug_log_scopes = no }
}
