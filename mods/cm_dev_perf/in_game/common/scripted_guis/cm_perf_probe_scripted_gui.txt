# cm_dev_perf probe (+perf4, 10-01). Root is the player. True while the classification
# has not processed scope:cm_probe_bt — the probe lists those types by name.
cm_perf_probe_unprocessed = {
	saved_scopes = { cm_probe_bt }
	is_shown = {
		OR = {
			NOT = { has_global_variable_map = cm_building_type_processed }
			NOT = { is_key_in_global_variable_map = { name = cm_building_type_processed target = scope:cm_probe_bt } }
		}
	}
}
