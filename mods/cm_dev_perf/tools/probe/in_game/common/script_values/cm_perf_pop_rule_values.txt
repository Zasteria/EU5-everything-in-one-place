# cm_dev_perf +perf57: what the peasants of a location already owe, by his rule 10-26.
# Every type that promotes out of peasants and is short now (burghers, clergy, soldiers,
# laborers with slaves and the RGO) will draw on the same free peasants; they are summed,
# and the 500 peasants that must stay are added. Below 0, the free peasants cover all of
# it with room to spare; a new level that needs E more pops fits while this plus E stays
# below 0 (his screenshot 10-26: a weapons workshop built on 1 080 free peasants while
# burghers stood 320 short and laborers 2 300). Pop counts are the game's thousands.
# Root is the location.
cm_perf_peasant_debt = {
	value = 0.5
	subtract = cm_location_peasants_available
	add = { value = 0 subtract = cm_location_burghers_available min = 0 }
	add = { value = 0 subtract = cm_location_clergy_available min = 0 }
	add = { value = 0 subtract = cm_location_soldiers_available min = 0 }
	add = { value = 0 subtract = cm_location_laborers_or_slaves_available min = 0 }
}

# Months until the debt is paid off by the peasants' own growth, as the sampler measured
# it (births and migration; promotion moves peasants on and pays nothing), halved: his
# rule is to look again at half the time it would take; never under 1 nor over 24 (+perf59:
# at 60 his age-6 save held almost every short location for five years). Where
# that cannot be counted he wants a look «между 3 и 6 месяцами» (10-26): 3 while the
# sampler has no reading yet (it has one the month after), 6 where the peasants do not
# grow. Root is the location.
cm_perf_pop_hold_months = {
	value = 3
	if = {
		limit = { exists = var:cm_pop_rate_peasants }
		value = 6
	}
	if = {
		limit = {
			exists = var:cm_pop_rate_peasants
			var:cm_pop_rate_peasants > 0
		}
		value = cm_perf_peasant_debt
		divide = var:cm_pop_rate_peasants
		multiply = 0.5
		min = 1
		max = 24
	}
}

# The hiring horizon in months for the pass running now (+perf59, his ask 10-26: a slider of
# its own for auto-expand and for auto-food on the auto-expand tab, not only Auto-Build's).
# cm_perf_bp_feat is the pass, set where staging and the approval drain enter each feature
# and category. Auto-Build food and production keep their tabs' sliders. Root is the location.
cm_perf_ae_horizon_months = {
	value = 72
	if = {
		limit = { owner = { exists = var:cm_ae_horizon } }
		value = owner.var:cm_ae_horizon
		multiply = 12
	}
}
cm_perf_af_horizon_months = {
	value = 72
	if = {
		limit = { owner = { exists = var:cm_af_horizon } }
		value = owner.var:cm_af_horizon
		multiply = 12
	}
}
cm_perf_horizon_months = {
	value = cm_perf_ab_prod_horizon_months
	if = {
		limit = { exists = global_var:cm_perf_bp_feat }
		if = {
			limit = {
				OR = {
					global_var:cm_perf_bp_feat = flag:cm_feature_auto_expand_buildings
					global_var:cm_perf_bp_feat = flag:cm_feature_auto_expand_rgos
					global_var:cm_perf_bp_feat = flag:cm_feature_upgrade_all_buildings
				}
			}
			value = cm_perf_ae_horizon_months
		} else_if = {
			limit = { global_var:cm_perf_bp_feat = flag:cm_feature_auto_food }
			value = cm_perf_af_horizon_months
		} else_if = {
			limit = { global_var:cm_perf_bp_feat = flag:cm_ab_cat_food }
			value = cm_perf_ab_food_horizon_months
		}
	}
}
