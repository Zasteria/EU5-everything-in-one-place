# cm_dev_perf +perf53: his rule for the pops a new level may count on, 2026-10-26. It
# replaces CM's projection (the type's own inflow over a hiring horizon), which his age-6
# save showed building burgher workshops where laborers were thousands short and peasants
# held 430-670 people for thousands of jobs: the inflow it counted had nowhere to come from.
#
#   - 500 peasants or fewer in the location: no level of anything but nobles.
#   - nobles: always.
#   - peasants: more free peasants than the level plus 500.
#   - laborers, burghers, clergy, soldiers: a free pop of the type, or more free peasants
#     than the shortfall plus 500 - free meaning after every peasant job, so a promotion
#     never empties a peasant building (his correction 10-26: «иначе он выгребет на
#     повышение людей из крестьянских домиков»; «я вообще никогда не хочу видеть …
#     крестьян в локации -300»).
#   - slaves, tribesmen: a free pop of the type.
#
# Pop counts are the game's thousands; a level's jobs are its type's employment_size.
# Every count here is read live, levels under construction included, so a loop building
# several levels sees each one it has started.

# Root is the location. $T$ = the building type.
cm_perf_pops_ok = {
	$T$ = { save_temporary_scope_as = cm_perf_pt }
	has_global_variable_map = cm_building_type_to_pop_type
	is_key_in_global_variable_map = { name = cm_building_type_to_pop_type target = scope:cm_perf_pt }
	switch = {
		trigger = "global_variable_map(cm_building_type_to_pop_type|scope:cm_perf_pt)"
		pop_type:nobles = { always = yes }
		pop_type:peasants = {
			cm_location_peasants_available > { value = scope:cm_perf_pt.employment_size add = 0.5 }
		}
		pop_type:laborers = { cm_perf_pops_promote = { FREE = cm_location_laborers_available } }
		pop_type:burghers = { cm_perf_pops_promote = { FREE = cm_location_burghers_available } }
		pop_type:clergy = { cm_perf_pops_promote = { FREE = cm_location_clergy_available } }
		pop_type:soldiers = { cm_perf_pops_promote = { FREE = cm_location_soldiers_available } }
		pop_type:slaves = {
			num_pop_type:peasants > 0.5
			"unemployed_pops_of_pop_type_in_location(pop_type:slaves)" > 0
		}
		pop_type:tribesmen = {
			num_pop_type:peasants > 0.5
			"unemployed_pops_of_pop_type_in_location(pop_type:tribesmen)" > 0
		}
		fallback = { always = no }
	}
}

# Root is the location, expects scope:cm_perf_pt. $FREE$ = the script value counting the
# type's free places (its pops less every job of the type, built or building).
cm_perf_pops_promote = {
	num_pop_type:peasants > 0.5
	OR = {
		$FREE$ > 0
		AND = {
			cm_location_peasants_available > { value = scope:cm_perf_pt.employment_size subtract = $FREE$ add = 0.5 }
		}
	}
}

# Root is the location. The next RGO level: a thousand laborers or slaves, by the same rule
# (CM counts an RGO level as one of its laborer jobs, cm_location_laborers_available).
cm_perf_rgo_level_ok = {
	num_pop_type:peasants > 0.5
	OR = {
		cm_location_laborers_or_slaves_available > 0
		AND = {
			cm_location_peasants_available > { value = 1.5 subtract = cm_location_laborers_or_slaves_available }
		}
	}
}
