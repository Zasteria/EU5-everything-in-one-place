# A concept of our own, so the marker tooltip can have a hoverable term that explains OUR meaning.
#
# Linking [production_method|E] gave the affordance but showed the GAME's Production Method concept,
# which answers a different question. A registered concept keeps the native look and behaviour -- the
# term renders highlighted and hoverable exactly like vanilla's -- while the text under it is ours,
# from game_concept_urqol_input_share / _desc.
#
# Concepts live under main_menu/, where vanilla's 00_game_concepts.txt is. Mods can add to this
# database; CMF ships its own (common/game_concepts/cmf_core_banner.txt).
#
# A generic goods icon rather than the production-method default, which read as one specific
# commodity and so implied the concept was about that commodity.

urqol_input_share = {
	texture = "flat_icons/common_goods"
	shown_in_encyclopedia = no
}
