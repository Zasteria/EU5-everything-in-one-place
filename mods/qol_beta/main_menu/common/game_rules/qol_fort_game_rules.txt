# Restrict the location types in which the automation may raise fortifications,
# when Community Mod Framework (CMF) is absent.
# The Community Mod Menu overrides this rule mid-game when CMF is present.
# Each option forbids that location rank and every rank below it; "all" forbids
# every location regardless of rank.
qol_fort_build_forbidden_rule = {
	default = qol_fort_build_forbidden_off

	qol_fort_build_forbidden_off = {
		flag = general_rule
	}

	qol_fort_build_forbidden_rural = {
		flag = general_rule
	}

	qol_fort_build_forbidden_town = {
		flag = general_rule
	}

	qol_fort_build_forbidden_city = {
		flag = general_rule
	}

	qol_fort_build_forbidden_all = {
		flag = general_rule
	}
}
