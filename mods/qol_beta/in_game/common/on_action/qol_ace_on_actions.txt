# Auto child education - monthly pulse.
# Scope: country

monthly_country_pulse = {
	on_actions = {
		qol_ace_monthly_check
	}
}

qol_ace_monthly_check = {
	trigger = {
		is_ai = no
	}
	effect = {
		# The Community Mod Menu governs when its settings are registered;
		# otherwise the game rule enables the feature with the default
		# behaviour (no exclusions, no expensive education).
		if = {
			limit = { is_key_in_variable_map = { name = cmm target = flag:qol__auto_ce_educate_heirs } }
			qol_ace_process_children = yes
		}
		else_if = {
			limit = { has_game_rule = qol_ace_auto_educate_enabled }
			qol_ace_process_children = yes
		}
	}
}
