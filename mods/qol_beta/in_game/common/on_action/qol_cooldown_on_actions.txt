# Calidad de vida EU5 - cooldown alerts - monthly pulse hook.
# This ADDS a new on_actions entry to monthly_country_pulse; it does not overwrite it.
# Scope: country (inherited from monthly_country_pulse).
monthly_country_pulse = {
	on_actions = {
		qol_cd_on_monthly
	}
}

# Our monthly dispatcher. Runs for every country; we only act for human players
# (AI does not need notifications, and gating here keeps it cheap).
# Scope: country (inherited from monthly_country_pulse).
qol_cd_on_monthly = {
	trigger = {
		is_ai = no
	}
	effect = {
		qol_cd_check_all = yes
		qol_cd_check_order_of_chivalry = yes
	}
}
