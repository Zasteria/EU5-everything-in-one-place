# List _on_changed for the fort cleanup buildings list.
# Applies the pending field change and clears CMM list markers.
# Scope: country
qol__fort_cleanup_buildings_on_changed = {
	scope = country

	effect = {
		cmm_apply_list_change = {
			setting = qol__fort_cleanup_buildings
		}
	}
}

# List _on_changed for the papal bulls auto-vote list.
# Applies the pending dropdown change and rebuilds the yes/no vote lists.
# Scope: country
qol__auto_vote_on_changed = {
	scope = country

	effect = {
		cmm_apply_list_change = {
			setting = qol__auto_vote
		}
		qol_papal_bulls_sync_vote_lists = yes
	}
}
