# Effective auto-pause mode for parliament sessions.
# If CMF is present, the CMM dropdown aliases (qol__parliament_auto_pause_*)
# are set and take precedence. Without CMF, the game rule governs.
# Scope: country

qol_parliament_mode_off = {
	OR = {
		has_variable = qol__parliament_auto_pause_off
		AND = {
			NOT = { has_variable = qol__parliament_auto_pause_always }
			NOT = { has_variable = qol__parliament_auto_pause_if_fail }
			has_game_rule = qol_parliament_auto_pause_off
		}
	}
}

qol_parliament_mode_always = {
	OR = {
		has_variable = qol__parliament_auto_pause_always
		AND = {
			NOT = { has_variable = qol__parliament_auto_pause_off }
			NOT = { has_variable = qol__parliament_auto_pause_if_fail }
			has_game_rule = qol_parliament_auto_pause_always
		}
	}
}

qol_parliament_mode_if_fail = {
	OR = {
		has_variable = qol__parliament_auto_pause_if_fail
		AND = {
			NOT = { has_variable = qol__parliament_auto_pause_off }
			NOT = { has_variable = qol__parliament_auto_pause_always }
			has_game_rule = qol_parliament_auto_pause_if_fail
		}
	}
}

# Effective state of the parliament availability reminder.
# Scope: country
qol_parliament_availability_enabled_effective = {
	OR = {
		has_variable = qol__parliament_availability_reminder_enabled
		AND = {
			NOT = { has_variable = qol__parliament_availability_reminder_disabled }
			has_game_rule = qol_parliament_availability_enabled
		}
	}
}
