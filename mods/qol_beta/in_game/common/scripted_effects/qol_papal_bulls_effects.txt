# Auto-vote on papal resolutions for the Catholic Church.
# Rebuilds the two variable lists used by qol_papal_bulls_auto_vote from the
# CMF dropdown field "vote" (1 = disabled, 2 = vote yes, 3 = vote no).
# Scope: country
qol_papal_bulls_sync_vote_lists = {
	clear_variable_list = qol_papal_auto_vote_yes
	clear_variable_list = qol_papal_auto_vote_no
	cmm_for_each_list_item = {
		setting = qol__auto_vote
		effect = qol_papal_bulls_sync_vote_item
	}
	cmm_for_each_list_item = {
		setting = qol__auto_vote_schism
		effect = qol_papal_bulls_sync_vote_item
	}
}

# Per-item: add the resolution to the yes/no list depending on its field value.
# Called by cmm_for_each_list_item.
# Scope: country
# Args: setting (key), i (number)
qol_papal_bulls_sync_vote_item = {
	set_local_variable = { name = qol_tmp_pv value = flag:$setting$_i$i$_f1 }
	if = {
		limit = { "variable_map(cmm|local_var:qol_tmp_pv)" = 2 }
		add_to_variable_list = {
			name = qol_papal_auto_vote_yes
			target = scope:cmm_list_current_item_value
		}
	}
	else_if = {
		limit = { "variable_map(cmm|local_var:qol_tmp_pv)" = 3 }
		add_to_variable_list = {
			name = qol_papal_auto_vote_no
			target = scope:cmm_list_current_item_value
		}
	}
}
