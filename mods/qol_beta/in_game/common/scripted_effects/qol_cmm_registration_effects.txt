# Root scope: country.
# Registers Calidad de vida EU5 settings with the Community Mod Menu (CMM).
qol_register_cmf_mod = {
	# Tab "Calidad de vida EU5" (qol). Two group headers in the same tab:
	# "Prohibition of fortifications" (fort_build_forbidden) with the
	# prohibition toggles, then "Fortification demolition" (fort_cleanup) with
	# the cleanup settings and the buildings list. CMF auto-creates a group per
	# list, so the buildings list is moved into the cleanup group after
	# registration (see the merge effect below).

	# Prohibit fortifications: one list item per location criterion; "enabled" =
	# automation may not raise a fortification in that location type. Read from
	# the fort buildings' allow block via qol_fort_build_forbidden_at_location
	# (scripted_triggers/qol_fort_build_triggers.txt). Rows are fixed (item_count
	# = 10) and loc-named qol__fort_build_forbidden_i<N>_name, the fixed-count
	# pattern; see the cooldown watch lists. All unchecked by default = vanilla.
	cmm_register_settings_list = {
		mod_id = qol
		setting_id = fort_build_forbidden
		tab_id = qol
		item_count = 10
		is_ordered = 0
	}
	cmm_register_list_bool_field = {
		mod_id = qol
		setting_id = fort_build_forbidden
		field_id = enabled
		default_value = 0
	}

	# CMF shows list rows in the order of the item list, which is initialized in
	# index order. Rebuild it so "Every location" (i8) sits at the very top; the
	# rest keep their relative order. Values stay keyed to the item flags, so
	# toggles keep working regardless of display position. Runs every
	# registration cycle; clearing and re-adding keeps it idempotent.
	clear_variable_list = cmm_list_items_qol__fort_build_forbidden
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i8 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i1 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i2 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i3 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i4 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i5 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i6 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i7 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i9 }
	add_to_variable_list = { name = cmm_list_items_qol__fort_build_forbidden target = flag:qol__fort_build_forbidden_i10 }

	# Cleanup scope: 1 = my country only, 2 = my country and dependencies,
	# 3 = dependencies only, 4 = all countries.
	cmm_register_dropdown_setting = {
		mod_id = qol
		setting_id = fort_cleanup_scope
		tab_id = qol
		group_id = fort_cleanup
		default_index = 2
		option_count = 4
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_scope
		index = 1
		alias = qol__fort_cleanup_scope_own
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_scope
		index = 2
		alias = qol__fort_cleanup_scope_dependencies
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_scope
		index = 3
		alias = qol__fort_cleanup_scope_dependencies_only
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_scope
		index = 4
		alias = qol__fort_cleanup_scope_all
	}

	# Location rank filter: 1 = all, 2 = rural settlement, 3 = town, 4 = city, 5 = metropolis, 6 = not province capital.
	cmm_register_dropdown_setting = {
		mod_id = qol
		setting_id = fort_cleanup_rank
		tab_id = qol
		group_id = fort_cleanup
		default_index = 1
		option_count = 6
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_rank
		index = 1
		alias = qol__fort_cleanup_rank_all
	}
	cmm_sync_dropdown_option_alias = {

		alias = qol__fort_cleanup_rank_settlement

	cmm_sync_dropdown_option_alias = {

		index = 3
		alias = qol__fort_cleanup_rank_town
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_rank
		index = 4
		alias = qol__fort_cleanup_rank_city
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_rank
		index = 5
		alias = qol__fort_cleanup_rank_metropolis
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__fort_cleanup_rank
		index = 6
		alias = qol__fort_cleanup_rank_not_province_capital
	}

	# Demolish-now button.
	cmm_register_button_setting = {
		mod_id = qol
		setting_id = fort_cleanup_now
		tab_id = qol
		group_id = fort_cleanup
	}

	# Building types to demolish, one list item each.
	cmm_begin_settings_list = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		tab_id = qol
		is_ordered = 0
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:stockade
		name = qol__fort_cleanup_buildings_stockade_name
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:castle
		name = qol__fort_cleanup_buildings_castle_name
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:bastion
		name = qol__fort_cleanup_buildings_bastion_name
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:star_fort
		name = qol__fort_cleanup_buildings_star_fort_name
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:fortress
		name = qol__fort_cleanup_buildings_fortress_name
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:city_walls
		name = qol__fort_cleanup_buildings_city_walls_name
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:coastal_fort
		name = qol__fort_cleanup_buildings_coastal_fort_name
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		value = building_type:naval_battery
		name = qol__fort_cleanup_buildings_naval_battery_name
	}
	cmm_finish_settings_list = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
	}
	cmm_register_list_bool_field = {
		mod_id = qol
		setting_id = fort_cleanup_buildings
		field_id = enabled
		default_value = 1
	}

	# CMF auto-creates a group per list; merge the buildings list into the
	# "Fortification demolition" group (fort_cleanup) so it renders under the
	# demolition header. Runs every registration cycle; the guards keep it
	# idempotent. The orphaned auto group stays registered but empty, and CMF
	# hides empty groups.
	if = {
		limit = {
			is_target_in_variable_list = {
				name = cmm_group_setting_keys_qol__qol__fort_cleanup_buildings
				target = flag:qol__fort_cleanup_buildings
			}
		}
		remove_list_variable = {
			name = cmm_group_setting_keys_qol__qol__fort_cleanup_buildings
			target = flag:qol__fort_cleanup_buildings
		}
	}
	if = {
		limit = {
			NOT = {
				is_target_in_variable_list = {
					name = cmm_group_setting_keys_qol__qol__fort_cleanup
					target = flag:qol__fort_cleanup_buildings
				}
			}
		}
		add_to_variable_list = {
			name = cmm_group_setting_keys_qol__qol__fort_cleanup
			target = flag:qol__fort_cleanup_buildings
		}
	}
	remove_from_variable_map = {
		name = cmm_setting_owner_group
		key = flag:qol__fort_cleanup_buildings
	}
	add_to_variable_map = {
		name = cmm_setting_owner_group
		key = flag:qol__fort_cleanup_buildings
		value = flag:qol__qol__fort_cleanup
	}
	set_variable = {
		name = cmm_setting_owner_group_key_qol__fort_cleanup_buildings
		value = flag:qol__qol__fort_cleanup
	}

	# Tab "Bulas papales" (papal_bulls): auto-vote on papal resolutions.
	# One row per resolution; the dropdown chooses 1 = disabled (default),
	# 2 = vote yes, 3 = vote no.
	# Force the list to rebuild on every registration so rows removed from the
	# registration (e.g. the Western Schism, moved to its own list) also
	# disappear from saves created before that change.
	if = {
		limit = { has_variable = cmm_list_initialized_qol__auto_vote }
		remove_variable = cmm_list_initialized_qol__auto_vote
	}
	cmm_begin_settings_list = {
		mod_id = qol
		setting_id = auto_vote
		tab_id = papal_bulls
		is_ordered = 0
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:benedictus_deus
		name = benedictus_deus
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:christiana_pietas
		name = christiana_pietas
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:in_coena_domini
		name = in_coena_domini
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:apostolicae_servitutis
		name = apostolicae_servitutis
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:immensa_aeterni_dei
		name = immensa_aeterni_dei
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:inter_gravissimas
		name = inter_gravissimas
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:libertas_ecclesiae
		name = libertas_ecclesiae
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:dei_gratia_rex
		name = dei_gratia_rex
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:illius_qui_se_pro_divini
		name = illius_qui_se_pro_divini
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:excommunicate_resolution
		name = excommunicate_resolution
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote
		value = resolution:call_crusade
		name = call_crusade
	}
	cmm_finish_settings_list = {
		mod_id = qol
		setting_id = auto_vote
	}
	cmm_register_list_dropdown_field = {
		mod_id = qol
		setting_id = auto_vote
		field_id = vote
		default_index = 1
		option_count = 3
	}

	# Western Schism vote: the resolution is directed (for Rome or for the
	# antipope's backer), so it gets its own single-row list with dedicated
	# options (1 = disabled, 2 = for Rome, 3 = against Rome).
	cmm_begin_settings_list = {
		mod_id = qol
		setting_id = auto_vote_schism
		tab_id = papal_bulls
		is_ordered = 0
	}
	cmm_add_settings_list_item = {
		mod_id = qol
		setting_id = auto_vote_schism
		value = resolution:western_schism_resolution
		name = qol__auto_vote_schism_western_schism_resolution_name
	}
	cmm_finish_settings_list = {
		mod_id = qol
		setting_id = auto_vote_schism
	}
	cmm_register_list_dropdown_field = {
		mod_id = qol
		setting_id = auto_vote_schism
		field_id = vote
		default_index = 1
		option_count = 3
	}
	qol_papal_bulls_sync_vote_lists = yes
	qol_papal_bulls_setup_item_tooltips = yes

	# Tab "Parlamento" (parliament), group "Pausa automática" (auto_pause).

	# Auto-pause mode: 1 = off, 2 = always, 3 = only if it will fail.
	# The game rule qol_parliament_auto_pause_rule governs when CMF is absent.
	cmm_register_dropdown_setting = {
		mod_id = qol
		setting_id = parliament_auto_pause
		tab_id = parliament
		group_id = auto_pause
		default_index = 3
		option_count = 3
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__parliament_auto_pause
		index = 1
		alias = qol__parliament_auto_pause_off
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__parliament_auto_pause
		index = 2
		alias = qol__parliament_auto_pause_always
	}
	cmm_sync_dropdown_option_alias = {
		setting = qol__parliament_auto_pause
		index = 3
		alias = qol__parliament_auto_pause_if_fail
	}

	# Availability reminder: 1 = enabled, 2 = disabled.
	# The game rule qol_parliament_availability_rule governs when CMF is absent.
	cmm_register_dropdown_setting = {

		tab_id = parliament

	cmm_sync_dropdown_option_alias = {

		index = 2

	# Tab "Educación" (education), group "Educación automática" (auto_ce).
	# Every month each eligible child is auto-assigned the education matching
	# its highest aptitude. The toggles choose which children are educated and

	# qol_ace_auto_educate_rule governs.
	cmm_register_bool_setting = {
		mod_id = qol
		setting_id = auto_ce_educate_heirs

# Callback handler for cmf_on_callback (setting toggles, alert clicks, button clicks).
# var:cmf_callback is the flag key of the interacted element.
# Scope: country
qol_handle_cmf_callback = {
	if = {
		limit = { var:cmf_callback = flag:qol__fort_cleanup_scope }
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_scope
			index = 1
			alias = qol__fort_cleanup_scope_own
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_scope
			index = 2
			alias = qol__fort_cleanup_scope_dependencies
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_scope
			index = 3
			alias = qol__fort_cleanup_scope_dependencies_only
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_scope
			index = 4
			alias = qol__fort_cleanup_scope_all
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__fort_cleanup_rank }
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_rank
			index = 1
			alias = qol__fort_cleanup_rank_all
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_rank
			index = 2
			alias = qol__fort_cleanup_rank_settlement
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_rank
			index = 3
			alias = qol__fort_cleanup_rank_town
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_rank
			index = 4
			alias = qol__fort_cleanup_rank_city
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_rank
			index = 5
			alias = qol__fort_cleanup_rank_metropolis
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__fort_cleanup_rank
			index = 6
			alias = qol__fort_cleanup_rank_not_province_capital
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__parliament_auto_pause }
		cmm_sync_dropdown_option_alias = {
			setting = qol__parliament_auto_pause
			index = 1
			alias = qol__parliament_auto_pause_off
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__parliament_auto_pause
			index = 2
			alias = qol__parliament_auto_pause_always
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__parliament_auto_pause
			index = 3
			alias = qol__parliament_auto_pause_if_fail
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__parliament_availability_reminder }
		cmm_sync_dropdown_option_alias = {
			setting = qol__parliament_availability_reminder
			index = 1
			alias = qol__parliament_availability_reminder_enabled
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__parliament_availability_reminder
			index = 2
			alias = qol__parliament_availability_reminder_disabled
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__fort_cleanup_now }
		qol_fort_cleanup = yes
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__auto_ce_educate_heirs }
		cmm_sync_bool_alias = {
			setting = qol__auto_ce_educate_heirs
			alias = qol_ace_educate_heirs
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__auto_ce_educate_crown }
		cmm_sync_bool_alias = {
			setting = qol__auto_ce_educate_crown
			alias = qol_ace_educate_crown
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__auto_ce_educate_prodigy }
		cmm_sync_bool_alias = {
			setting = qol__auto_ce_educate_prodigy
			alias = qol_ace_educate_prodigy
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__auto_ce_expensive_scope }
		cmm_sync_dropdown_option_alias = {
			setting = qol__auto_ce_expensive_scope
			index = 2
			alias = qol_ace_exp_heirs
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__auto_ce_expensive_scope
			index = 3
			alias = qol_ace_exp_heirs_crown
		}
		cmm_sync_dropdown_option_alias = {
			setting = qol__auto_ce_expensive_scope
			index = 4
			alias = qol_ace_exp_all
		}
	}
	else_if = {
		limit = { var:cmf_callback = flag:qol__auto_ce_run_now }
		qol_ace_run_now = yes
	}

	# Cooldown handler: handles both CMM setting sync AND alert left-click dismiss.
	# Called unconditionally — its switch no-ops for unrecognized flags.
	qol_cd_handle_cmf_callback = yes

	# Cabinet action alerts: same pattern (setting sync + alert left-click dismiss).
	qol_cab_handle_cmf_callback = yes
}
