# Карта Construction Manager «Потенциал продовольствия за уровень РГО»,
# перенесённая целиком.
#
# **Скопировано файлом, а не переписано формулой.** Его слово, 2026-09-12, когда
# он снял CM с плейсета: «Достань от туда карту и ту функцию, которая вычисляет
# пригодность локации для производства еды. Карта "житницы". Перенеси её в наш
# мод в том виде, в котором она была в CM.» Правило «чужой инструмент, который
# просят взять, — копировать файлами, а не воспроизводить формулой» — из
# `docs/archive/wtp2_failed.md`, и оно стоило отдельной неудачной попытки.
#
# Взято из `cm_food_map_modes.txt` (Construction Manager dev) на сборке. Все
# цвета, пороги, градиенты и комментарии — его; переименованы только имена,
# которых без CM в игре нет:
#
#   cm_location_food_potential (значение)  -> bag_wtp_fp_value
#   cm_location_is_auto_food_active        -> bag_wtp_is_granary (штриховка)
#   cm_location_food_potential (режим)     -> bag_wtp_food_potential_map
#
# `bag_wtp_fp_value` — то же самое число: его формула уже списана генератором в
# `bag_wtp_generated_foodpot.txt`, поэтому карта без CM красит ровно тем, чем
# красила у него.
#
# **Скрытый дубликат `cm_location_food_potential_hidden` не перенесён.** Он
# существует затем, чтобы CM мог на миг переключить режим и заставить карту
# перекраситься; наш мод режимами не переключает, а перекраску решает
# `color_refresh_counters` ниже.
#
# **«Житницы» перенесены и убраны обратно 2026-09-12**, в тот же день: «режим
# карты "житницы" мне не нужен в целом, меня интересовала только карта
# "Потенциал продовольствия"». Штриховка помеченных житницей осталась здесь —
# она часть этой карты у него же.

bag_wtp_food_potential_map = {
	map_color = {
		if = {
			limit = { is_land = no }
			value = hsv { 0.58 0.50 0.52 }
		}
		else_if = {
			# 45 = [food_value * raw_out_mult + livestock_addback] * food_mult
			# food_value = rice = 10
			# raw_out_mult = 1 + [16 FV_levels * 0.025 FV_raw_mat_out] = 1.40
			# livestock_addback = 0.5 livestock_per_FV * 8 livestock_food_val * 0.75 FV/RGO_ratio * 1.15 econ_of_scale = 3.45
			# food_mult = 1 + 0.33 farmland + 0.10 assumed_rural_rank + 0.25 river_5 + 0.10 windmill + 0.80 irrigation = 2.58
			limit = { bag_wtp_fp_value >= 45 }
			value = rgb { 0 255 0 }
		}
		else_if = {
			limit = { bag_wtp_fp_value >= 27.5 }  # 45 * 22/36
			lerp = {
				max_color = rgb { 0 255 0 }
				min_color = rgb { 0 180 0 }
				factor = {
					value = bag_wtp_fp_value
					subtract = 27.5
					divide = 17.5  # 45 - 27.5
				}
			}
		}
		else_if = {
			limit = { bag_wtp_fp_value >= 20 }  # 45 * 16/36
			lerp = {
				max_color = rgb { 0 180 0 }
				min_color = rgb { 255 255 0 }
				factor = {
					value = bag_wtp_fp_value
					subtract = 20
					divide = 7.5  # 27.5 - 20
				}
			}
		}
		else_if = {
			limit = { bag_wtp_fp_value >= 12.5 }  # 45 * 10/36
			lerp = {
				max_color = rgb { 255 255 0 }
				min_color = rgb { 255 128 0 }
				factor = {
					value = bag_wtp_fp_value
					subtract = 12.5
					divide = 7.5  # 20 - 12.5
				}
			}
		}
		else_if = {
			limit = { bag_wtp_fp_value >= 3.75 }  # 45 * 3/36
			lerp = {
				max_color = rgb { 255 128 0 }
				min_color = rgb { 255 0 0 }
				factor = {
					value = bag_wtp_fp_value
					subtract = 3.75
					divide = 8.75  # 12.5 - 3.75
				}
			}
		}
		else = {
			lerp = {
				max_color = rgb { 255 0 0 }
				min_color = rgb { 0 0 0 }
				factor = {
					value = bag_wtp_fp_value
					divide = 3.75  # 3.75 - 0
				}
			}
		}
	}

	secondary_map_color = {
		if = {
			limit = {
				owner ?= scope:actor
				bag_wtp_is_granary = yes
			}
			value = rgb { 255 0 255 }
		}
		else_if = {
			limit = {
				exists = scope:actor
				is_land = yes
				NOT = { is_owned_or_owned_by_subjects_or_below_of = scope:actor }
			}
			value = rgb { 100 100 100 }
		}
	}

	legend_key = {
		desc = "bag_wtp_fp_auto_food_legend_key"
		color = rgb { 255 0 255 }
	}
	legend_key = {
		desc = "bag_wtp_fp_bright_green_legend_key"
		color = rgb { 0 255 0 }
	}
	legend_key = {

		color = rgb { 0 180 0 }
	}
	legend_key = {

		color = rgb { 255 255 0 }
	}
	legend_key = {
		desc = "bag_wtp_fp_orange_legend_key"
		color = rgb { 255 128 0 }
	}
	legend_key = {

		color = rgb { 255 0 0 }
	}
	legend_key = {
		desc = "bag_wtp_fp_black_legend_key"
		color = rgb { 0 0 0 }
	}

	tooltip_key = {
		if = {
			limit = { is_land = yes }
