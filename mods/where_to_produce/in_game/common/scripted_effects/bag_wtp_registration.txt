# Settings, registered on CMF's mod registration hook.
#
# **Three tabs and four buttons, and that is the whole of the page.** His words,
# 2026-09-06: «мод внутри стал довольно громоздким и неудобным», and the rule he
# gave with them -- у функции на странице мода одна кнопка «открыть», всё
# остальное живёт внутри её окна. So the goods, the rights, the two «Считать»
# buttons, the two «План» buttons, the caps, the rights tick and the town/village
# reset are all controls inside the windows they belong to now, and what is left
# here opens those windows. `docs/investigations/wtp_menu_rebuild.md` is the
# whole move, with what went where.
#
# Everything that depends on the game's own data -- which goods exist, which
# buildings make them, what each method is worth -- is generated beside this, so
# a patch that adds any of those is picked up by a rebuild rather than by an
# edit.
#
# **A tab exists because a setting names it**, in the order the settings are
# registered (`_cmm_register_tab`, CMF 2.4.1): the zone lists come first and
# «Земля» is the tab the page opens on.
#
# Scope: country
bag_wtp_register = {
	cmf_register_mod = { mod_id = bag_wtp }

	# **Что стране по зубам сейчас** -- по числу на здание, для фишки «Из плана —
	# сюда»: план «на конец» ставит верхнюю ступень лестницы, а показать надо ту,
	# которую можно построить. Здесь, а не только в «Пересчитать», потому что
	# `cmf_on_mod_registration` срабатывает и на загрузке сохранения: иначе после
	# загрузки фишка считала бы недоступным вообще всё, пока не пересчитан план.
	bag_wtp_set_bavail = yes

	# ---- «Земля» -----------------------------------------------------------
	#
	# Where to look. Continents rather than the world's seventy-seven regions:
	# the fine choosing happens on the map afterwards, and a region list six
	# groups deep was a worse way to say "Europe". **This tab is untouched by
	# the rebuild** -- it was already one job on one page.
	bag_wtp_register_zone_list = yes
	bag_wtp_register_region_lists = yes

	# ---- «Техническая»: моды, из которых взяты здания ----------------------
	#
	# Его слово, 2026-09-09: «список модов из которых наш мод подхватил
	# какие-либо сооружения… мы можем нажать галочку включить или выключить
	# использование этих зданий в наших планах». Снятая галочка не убирает
	# здание из файлов — скрипт написан на сборке, — а закрывает ему ворота
	# `_avail_` и `_reach_`, то есть и «на сейчас», и «на конец».
	bag_wtp_register_source_list = yes

	# ---- «Расчёты» — three functions, three doors --------------------------
	#
	# **Each of these only opens a window.** «как "расчёт" отдельно стоит от
	# "общий план", так и "редактор" должен иметь свою область» -- so the editor
	# stands beside the plan here rather than being a button inside it.

	# The ranking for one good or one urban right: which of the chosen ground
	# ranks best for it. The good, the right, «на сейчас»/«на конец» and the
	# buildable filter are all in the window, and «Искать локации» there is what
	# runs the pass. This button opens whichever of the two ranking windows
	# matches what is chosen -- they are two windows because a row draws one
	# method for a good and one per good of a right's bundle.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = show
		tab_id = plan
		group_id = open
	}

	# The whole-map plan: every good at once over the same ground, one building
	# at a time, under a cap per location. Its window carries the caps, the
	# «на конец» switch, the urban-rights tick, the town/village reset and
	# «Пересчитать», which is what runs it.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = plan_show
		tab_id = plan
		group_id = open
	}

	# **The plan editor is a window and this is the door to it.** It was nine
	# settings on this page -- three dropdowns and five buttons -- and he refused
	# to use them, 2026-09-03: «Сейчас окно настроек мода - стало засраным и
	# неудобным. Более того это именно что окно настроек и основных функций, я не
	# хочу делать подобные вещи там.» One button, and everything else lives in
	# `bag_wtp_edit_window.gui`: the three slots, the goods as icons, «+1» and
	# «−1» beside each, and the changes list in a window of its own.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = edit_open
		tab_id = plan
		group_id = open
	}

	# ---- «Техническая» -----------------------------------------------------
	#
	# «Диагностика». One press, everything the mod knows, into `debug.log` and
	# the headline of it into `error.log` besides.
	#
	# **It reads and writes nothing of the plan's**, so it is safe at any moment:
	# pressed before a plan it prints zeros throughout, which is itself the
	# answer to «did the pass run at all». The expensive half of it -- one walk
	# over the ground working out where each good was lost -- is why it is a
	# button and not the tail of the plan: the plan stays exactly as fast as it
	# was, and this is pressed when there is something to look at.
	#
	# `docs/pitfalls/diagnosis.md` has what it prints and the four things about
	# `debug_log` that were measured rather than assumed.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = diag
		tab_id = tech
		group_id = tech
	}

	# «Пересчитать карту прав». Проход карты городских прав идёт сам на первой
	# загрузке и больше не идёт никогда: он читает сырьё провинций и модификаторы
	# вывода, а они меняются по ходу партии. Эта кнопка сбрасывает клеймо и
	# считает заново -- нажимать, когда карта отстала от игры.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = trmm
		tab_id = tech
		group_id = tech
	}

	# Every picker folded shut, once, the first time this player sees the page.
	# Five region lists and the continent list all open is a page to scroll
	# rather than a page to read; after this the player's own folding is his.
	bag_wtp_fold_pickers = yes

	# The editor's counters, created before anything can read them: a script
	# value reading a global that is not there is the failure that logs nothing,
	# and these two are printed on buttons from the moment the page opens.
	if = {
		limit = { NOT = { has_global_variable = bag_wtp_edit_moved } }
		set_global_variable = { name = bag_wtp_edit_moved value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = bag_wtp_save_n } }
		set_global_variable = { name = bag_wtp_save_n value = 0 }
	}
	# And one pair per slot. `_sl<n>_n` is what «загрузить» tests before it throws
	# the plan on the map away, so a slot whose counter does not exist would read
	# as empty for ever -- or, worse, clear a good plan for nothing.
	bag_wtp_edit_init_slots = yes

	bag_wtp_init_counters = yes
	# Типы зданий деревень в глобалках: без них ячейка редактора рисует значок
	# со скоупа, которого нет, и вместо значка встаёт пустота.
	bag_wtp_init_vbt = yes
	bag_wtp_rebuild_zone = yes
	bag_wtp_rebuild_regions = yes
	bag_wtp_rebuild_sources = yes
	# Which urban rights this country may ever hold, as one variable per right:
	# a `visible` in a window cannot ask a trigger, and a unique right nobody
	# can take is a cell that must not be there to click.
	bag_wtp_refresh_rights = yes
	# And the circle under the good or right that is already chosen. `_sel<n>` is
	# newer than the answer it draws: a save made before 2026-09-06 carries the
	# index and no circle, and a chosen good the player cannot see is the fault
	# this mod has paid for twice.
	bag_wtp_sync_sel = yes

	# Карта «Лучшее городское право», перенесённая из CM: один проход по всем
	# определениям провинций, за клеймом, ровно как у него. Регистрация CMF
	# срабатывает и при каждом открытии страницы мода, поэтому без клейма это был
	# бы полный проход по карте на каждое открытие; с клеймом второй раз не делает
	# ничего. Пересчёт вручную -- кнопка «Пересчитать карту прав» на «Технической».
	if = {
		limit = { NOT = { bag_wtp_trmm_data_is_current = yes } }
		bag_wtp_trmm_recompute_all = yes
		# Проход синхронный, так что недосчитанного состояния, которое клеймо
		# заверило бы, не бывает.
		set_global_variable = { name = bag_wtp_trmm_stamp value = bag_wtp_trmm_version_value }
	}

	# **Nothing here throws the answer away.** This runs on CMF's registration
	# hook, and the fifteenth run showed that hook firing again every time the
	# mod page is opened: it used to end with `bag_wtp_drop_browse` and
	# `bag_wtp_clear_rows`, so opening the menu to press «Открыть» wiped the
	# result that button exists to reopen, zeroed the counters beside it, and
	# took the rank off every row. On a new game there is nothing to clear --
	# `bag_wtp_init_counters` makes the numbers exist and the lists start empty.
}

# **`bag_wtp_sync` is gone with the settings it copied.** It carried
# `buildable_only`, the two caps and the urban-rights tick out of CMM's variable
# map into globals a location-scoped trigger can read; every one of those is a
# control in a window now and writes its own global directly, so there is
# nothing left to copy and nothing that can be one press out of date.
