namespace = bag_wsl

# **Пустышка для старых сохранений.** До 0.6.0 это был дневной дозор: событие
# перезапускало себя на завтра. В сохранении, сделанном тогда, одно такое уже
# стоит в очереди; без этого определения игра ругнулась бы на него. Себя оно
# больше не перезапускает, так что срабатывает самое большее один раз.
bag_wsl.1 = {
	type = country_event
	hidden = yes
}

# Месяц правки инфляции прошёл: дальше править или отдать автоматике.
# В очереди стоит, только пока правка идёт (`bag_wsl_mint_schedule`).
bag_wsl.2 = {
	type = country_event
	hidden = yes

	immediate = {
		bag_wsl_mint_month = yes
	}
}

# День после конца войны (`bag_wsl_on_war_end`): если войн больше нет и
# автоматика не выключена галочкой — то же, что кнопка «Выправить».
bag_wsl.3 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				at_war = no
				has_variable = bag_wsl_auto_on
			}
			bag_wsl_manual_run = yes
		}
	}
}
