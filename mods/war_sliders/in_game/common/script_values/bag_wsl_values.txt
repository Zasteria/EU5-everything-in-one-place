# Настройки CMM приходят в процентах — интерфейсу нужна доля 0..1.
# Считается в момент срабатывания, а не каждый кадр.

# Scope: country
bag_wsl_army_fraction = {
	value = 0
	if = { limit = { has_variable = bag_wsl_army_pct } add = var:bag_wsl_army_pct }
	else = { add = 50 }
	divide = 100
}

# Целевая инфляция задаётся в сотых процента (5 = 0.05%), а игра держит её
# долей: 0.05% — это 0.0005.
# Scope: country
bag_wsl_inflation_goal = {
	value = 0
	if = { limit = { has_variable = bag_wsl_infl_pct } add = var:bag_wsl_infl_pct }
	divide = 10000
}

# Порог, ниже которого правка считается доведённой: сама цель. В 0.4.1 к ней
# добавлялась полсотая процента, чтобы правка не висела на невидимом остатке
# (прогон 2026-09-27); от зависания теперь спасает минимальный шаг
# (`bag_wsl_mint_aim`), а остаток он хочет гасить любой, 2026-09-29: «он всё
# ещё не исправляет 0.0001 инфляцию… инфляция ниже нуля стать не может».
# Scope: country
bag_wsl_inflation_stop = {
	value = bag_wsl_inflation_goal
}

# Число, которым взводится привод: цель плюс сотая, у верха шкалы — минус.
# Смена числа под показанным ползунком и рождает событие; см. `bag_wsl_on_peace`.

# Scope: country
bag_wsl_army_arm = {
	value = bag_wsl_army_fraction
	if = { limit = { bag_wsl_army_fraction > 0.98 } subtract = 0.01 }
	else = { add = 0.01 }
}
