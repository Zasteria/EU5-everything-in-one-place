# Считается в момент срабатывания, а не каждый кадр.

# Scope: country
bag_wsl_army_fraction = {
	value = 0
	if = { limit = { has_variable = bag_wsl_army_pct } add = var:bag_wsl_army_pct }
	else = { add = 50 }
	divide = 100
}

# Целевая инфляция задаётся в сотых процента (5 = 0.05%), а игра держит её
# долей: 0.05% — это 0.0005.
# Scope: country
bag_wsl_inflation_goal = {
	value = 0
	if = { limit = { has_variable = bag_wsl_infl_pct } add = var:bag_wsl_infl_pct }
	divide = 10000
}

# Порог, ниже которого правка считается доведённой: цель плюс полсотой
# процента. Панель показывает инфляцию до сотых, и остаток меньше 0.005% она
# рисует как 0.00% — прогон 2026-09-27 (скриншот): инфляция «0.00%», прирост
# «0.00%», авто-чеканка выключена, и мод месяцами дожидался невидимого остатка.
# Scope: country
bag_wsl_inflation_stop = {
	value = bag_wsl_inflation_goal
	add = 0.00005
}
