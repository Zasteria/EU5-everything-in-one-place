# Обратная дорога: интерфейс сделал своё — скрипт гасит флаг.
#
# **Гасит именно интерфейс, а не такт.** Флаг живёт ровно от месячного такта до
# того кадра, в котором ползунок встал и `Post` ушёл движку; иначе наш ползунок
# держал бы содержание постоянно, а мод должен просыпаться на переключении и
# засыпать обратно.
#
# `bag_wsl_runs` — счётчик сработавших приводов, его показывает окно
# диагностики. Пустой счётчик при непустом флаге — это «привод не дошёл», и
# отличить одно от другого без него нельзя.

# Scope: country
bag_wsl_clear_army_go = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		remove_variable = bag_wsl_army_go
		bag_wsl_count_run = yes
	}
}

# Снять всё разом: и невыполненные приводы, и правку чеканки, и счётчик.
# Scope: country
bag_wsl_reset = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		remove_variable = bag_wsl_army_go
		remove_variable = bag_wsl_navy_go
		remove_variable = bag_wsl_fort_go
		remove_variable = bag_wsl_runs
		if = {
			limit = { has_variable = bag_wsl_mint_phase }
			bag_wsl_mint_stand_down = yes
		}
	}
}

# Первый месяц правки — не вслепую. Ползунок уже стоит на полу, и игра сама
# говорит, сколько инфляции он снимет за месяц (`EconomyView.GetInflationChange`,
# то же число, что в скобках рядом с инфляцией на панели). Раньше первый месяц
# всегда шёл на полу, потому что темп был ещё не измерен, и остаток 0.05 при
# темпе 0.15 проскакивал на полную (его жалоба 2026-09-27).
#
# Прогноз приходит через `AddScope` + `MakeScopeValue` — форма
# `autonomous_diplomats` (`atd_automation_lateralview.gui:882`); без
# `saved_scopes` скоуп не доходит.
# Scope: country
bag_wsl_mint_measured = {
	scope = country
	saved_scopes = { bag_wsl_change }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { has_variable = bag_wsl_mint_phase }
			if = {
				limit = { NOT = { inflation > bag_wsl_inflation_stop } }
				bag_wsl_mint_stand_down = yes
			}
			else = {
				set_variable = { name = bag_wsl_rem value = inflation }
				change_variable = { name = bag_wsl_rem subtract = bag_wsl_inflation_goal }
				# Прогноз со знаком: снятие инфляции — минус.
				set_variable = { name = bag_wsl_drain value = 0 }
				change_variable = { name = bag_wsl_drain subtract = scope:bag_wsl_change }
				set_variable = { name = bag_wsl_held value = var:bag_wsl_mint_floor }
				bag_wsl_mint_aim = yes
			}
		}
	}
}
