# Мод висит на двух крючках CMF (регистрация страницы и обратный звонок с неё) и
# на одном крючке игры — конце войны.
#
# **Опроса нет.** До 0.6 здесь висел `cmf_monthly_human_country_pulse`, а дневной
# дозор перезапускал скрытое событие каждый день — «жрёт тики». С 0.7 мод зовёт
# сама игра, один раз на закрытую войну (`on_ending_war`); между войнами он не
# делает ничего.

cmf_on_mod_registration = {
	on_actions = {
		bag_wsl_on_register
	}
}

# Scope: country
bag_wsl_on_register = {
	effect = {
		bag_wsl_register = yes
	}
}

cmf_on_callback = {
	on_actions = {
		bag_wsl_on_callback
	}
}

# Scope: country
bag_wsl_on_callback = {
	effect = {
		bag_wsl_handle_callback = yes
	}
}

# Конец войны. Игра зовёт его «победителю и проигравшему» (комментарий
# `_hardcoded.txt`), root — страна; дойдёт ли он до союзника, который войну не
# вёл, и до сепаратного мира — не проверено, на это остаётся кнопка. Ванилла
# цепляется сюда так же (`religious_order_conquest.txt`), список дописывается.
on_ending_war = {
	on_actions = {
		bag_wsl_on_war_end
	}
}

# Scope: country. Через день — мирный договор к тому времени исполнен, и
# `at_war` честно говорит, не осталось ли второй войны.
bag_wsl_on_war_end = {
	effect = {
		bag_wsl_queue_war_end = yes
	}
}

# **Восстание и гражданская война** (0.7.2). Его прогон 10-06: подавил восстание —
# мод не сработал. У игры для них свои крючки (`_hardcoded.txt`): конец —
# `on_civil_war_won`/`_lost` (root, scope:winner, scope:loser), поглощение
# мятежной страны — `on_civil_war_annex`/`_annexed` (root поглощает,
# scope:target поглощена). Какой из них зовётся при подавленном восстании и
# зовётся ли `on_ending_war` — не проверено, поэтому висим на всех четырёх;
# `bag_wsl_end_queued` не даёт поставить в очередь второй раз.
on_civil_war_won = {
	on_actions = {
		bag_wsl_on_civil_war_result
	}
}

on_civil_war_lost = {
	on_actions = {
		bag_wsl_on_civil_war_result
	}
}

on_civil_war_annex = {
	on_actions = {
		bag_wsl_on_civil_war_annex
	}
}

on_civil_war_annexed = {
	on_actions = {
		bag_wsl_on_civil_war_annex
	}
}

# root и scope:winner — победитель, scope:loser — проигравший.
bag_wsl_on_civil_war_result = {
	effect = {
		bag_wsl_queue_war_end = yes
		scope:winner ?= { bag_wsl_queue_war_end = yes }
		scope:loser ?= { bag_wsl_queue_war_end = yes }
	}
}

# root — кто поглощает, scope:target — кого.
bag_wsl_on_civil_war_annex = {
	effect = {
		bag_wsl_queue_war_end = yes
		scope:target ?= { bag_wsl_queue_war_end = yes }
	}
}
