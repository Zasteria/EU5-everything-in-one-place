# Мод висит на двух крючках CMF (регистрация страницы и обратный звонок с неё) и
# на одном крючке игры — конце войны.
#
# **Опроса нет.** До 0.6 здесь висел `cmf_monthly_human_country_pulse`, а дневной
# дозор перезапускал скрытое событие каждый день — «жрёт тики». С 0.7 мод зовёт
# сама игра, один раз на закрытую войну (`on_ending_war`); между войнами он не
# делает ничего.

cmf_on_mod_registration = {
	on_actions = {
		bag_wsl_on_register
	}
}

# Scope: country
bag_wsl_on_register = {
	effect = {
		bag_wsl_register = yes
	}
}

cmf_on_callback = {
	on_actions = {
		bag_wsl_on_callback
	}
}

# Scope: country
bag_wsl_on_callback = {
	effect = {
		bag_wsl_handle_callback = yes
	}
}

# Конец войны. Игра зовёт его «победителю и проигравшему» (комментарий
# `_hardcoded.txt`), root — страна; дойдёт ли он до союзника, который войну не
# вёл, и до сепаратного мира — не проверено, на это остаётся кнопка. Ванилла
# цепляется сюда так же (`religious_order_conquest.txt`), список дописывается.
on_ending_war = {
	on_actions = {
		bag_wsl_on_war_end
	}
}

# Scope: country. Через день — мирный договор к тому времени исполнен, и
# `at_war` честно говорит, не осталось ли второй войны.
bag_wsl_on_war_end = {
	trigger = {
		is_human = yes
		has_variable = bag_wsl_auto_on
	}
	effect = {
		trigger_event_silently = { id = bag_wsl.3 days = 1 }
	}
}
