# Всё, чем мод касается игры, висит на трёх крючках CMF.
#
# **Такт — месячный и человеческий.** `cmf_monthly_human_country_pulse` бьёт
# только по странам игроков; в мирный месяц такт читает один флаг и выходит.
# Отдельного on_action «страна вышла из всех войн» у игры нет: `on_pre_ending_war`
# бьёт до того, как война кончилась, и при двух войнах сразу соврал бы. Поэтому
# состояние державы хранится само: `at_war` истинно, пока идёт хоть одна война,
# и мод ловит именно переход «было да — стало нет».

cmf_on_mod_registration = {
	on_actions = {
		bag_wsl_on_register
	}
}

# Scope: country
bag_wsl_on_register = {
	effect = {
		bag_wsl_register = yes
	}
}

cmf_on_callback = {
	on_actions = {
		bag_wsl_on_callback
	}
}

# Scope: country
bag_wsl_on_callback = {
	effect = {
		bag_wsl_handle_callback = yes
	}
}

cmf_monthly_human_country_pulse = {
	on_actions = {
		bag_wsl_on_monthly
	}
}

# Scope: country
bag_wsl_on_monthly = {
	effect = {
		bag_wsl_tick = yes
	}
}
