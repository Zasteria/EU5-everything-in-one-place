# Мод висит на двух крючках CMF: регистрация страницы и обратный звонок с неё.
#
# **Месячного такта нет** (0.6.0). До 0.6 здесь висел
# `cmf_monthly_human_country_pulse`, а дневной дозор перезапускал скрытое событие
# каждый день; теперь мод работает, только когда нажата его кнопка.

cmf_on_mod_registration = {
	on_actions = {
		bag_wsl_on_register
	}
}

# Scope: country
bag_wsl_on_register = {
	effect = {
		bag_wsl_register = yes
	}
}

cmf_on_callback = {
	on_actions = {
		bag_wsl_on_callback
	}
}

# Scope: country
bag_wsl_on_callback = {
	effect = {
		bag_wsl_handle_callback = yes
	}
}
