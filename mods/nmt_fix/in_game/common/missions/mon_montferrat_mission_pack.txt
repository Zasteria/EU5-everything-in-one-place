mon_montferrat_mission_pack = {
	icon = nmt_italy
	repeatable = no
	player_playstyle = military

		game_has_missions_enabled = yes
		has_or_had_tag = MFA
		NOR = { country_type = pop country_type = building }
	}

	on_completion = { remove_variable = mon_mission_started }
	on_abort = { remove_variable = mon_mission_started }

	# ==================== A: ギリシャ侵攻 ====================

	mon_claim_thessaloniki = {
		icon = glorious_arms
		requires = { mon_palaiologos_ambition }
		enabled = {
			OR = {
				has_fired_unique_event = flavor_mfa.3
				AND = {
					country_exists = c:BYZ
					NOT = { location:thessaloniki = { owner = c:BYZ } }
				}
				NOT = { country_exists = c:BYZ }
			}
		}
		on_completion = {
			if = {
				limit = { location:thessaloniki = { has_owner = yes } }
				add_casus_belli = {
					type = casus_belli:cb_conquer_province
					target = location:thessaloniki.top_owner
					years = 30
				}
			}
			add_country_modifier = { modifier = mon_thessaloniki_claim_mod years = 30 mode = add }
		}
	}

	mon_seize_thessaloniki = {
		icon = crown_power_advance_renaissance

			owns_or_non_sovereign_subject_owns = location:thessaloniki
