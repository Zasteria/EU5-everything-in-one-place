ayy_ayyubid_mission_pack = {
	icon = nmt_ayyubid
	repeatable = no
	player_playstyle = military
	chance = 10000000

	visible = {
		game_has_missions_enabled = yes
		has_or_had_tag = HSN
		NOR = { country_type = pop country_type = building }
	}

	enabled = { always = yes }

	on_start = { set_variable = ayy_mission_started }
	on_completion = { remove_variable = ayy_mission_started }
	on_abort = { remove_variable = ayy_mission_started }

	# ==================== ROOT ====================

	ayy_rise_of_ayyubids = {
		icon = legacy_of_great_empires
		requires = { }

			trigger_event_non_silently = ayy_events.1
		}
	}

	# ==================== A: エジプト ====================

	ayy_mamluk_rivalry = {
		icon = crusader_tradition
		requires = { ayy_rise_of_ayyubids }
		enabled = {
			OR = {
				NOT = { country_exists = c:MAM }
				c:MAM = { is_rival_of = root }
				opinion = { target = c:MAM value <= 0 }
			}
		}
		on_completion = {
			if = {
				limit = {
					country_exists = c:MAM
					c:MAM = { has_owner = location:aleppo }
				}
				add_casus_belli = {
					type = casus_belli:cb_conquer_province
					target = location:aleppo.top_owner
					years = 60
				}
			}
			if = {
				limit = { location:aleppo = { has_owner = yes } }
				add_casus_belli = {
					type = casus_belli:cb_conquer_province
					target = location:aleppo.top_owner
					years = 60
				}
			}
		}
	}

	ayy_secure_sinai = {
		icon = expansionism
		requires = { ayy_mamluk_rivalry }
		enabled = {
			any_owned_location = {
				area = area:sinai_area

		}

			add_country_modifier = { modifier = ayy_sinai_mod years = -1 mode = add }
