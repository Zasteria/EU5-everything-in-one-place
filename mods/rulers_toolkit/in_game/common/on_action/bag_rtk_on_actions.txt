# Сборник мелкой автоматики. Каждая функция — своя группа на странице CMF и своя
# строка в `bag_rtk_monthly` ниже; новая функция дописывается туда же.

cmf_on_mod_registration = {
	on_actions = {
		bag_rtk_on_register
	}
}

# Scope: country
bag_rtk_on_register = {
	effect = {
		bag_rtk_register = yes
	}
}

cmf_on_callback = {
	on_actions = {
		bag_rtk_on_callback
	}
}

# Scope: country
bag_rtk_on_callback = {
	effect = {
		bag_rtk_handle_callback = yes
	}
}

# Раз в месяц, только у игрока (CMF сам отсеивает ИИ).
cmf_monthly_human_country_pulse = {
	on_actions = {
		bag_rtk_monthly
	}
}

# Scope: country
bag_rtk_monthly = {
	effect = {
		bag_rtk_commanders_monthly = yes
		bag_rtk_prisoners_monthly = yes
		bag_rtk_art_monthly = yes
		bag_rtk_faith_check_all = yes
	}
}

# Ванильный хук победы; списки `on_actions` движок сливает, ванильные события и
# эффект остаются. Пленные, отделённые к этому мигу, уходят в тыл сразу; кто
# отделится позже — 1-го числа.
on_battle_won = {
	on_actions = {
		bag_rtk_on_battle_won
	}
}

# Scope: country
bag_rtk_on_battle_won = {
	effect = {
		bag_rtk_prisoners_after_battle = yes
	}
}

# root — отряд пленных (комментарий ванили, `_hardcoded.txt`).
on_overrun_imprisoned = {
	on_actions = {
		bag_rtk_on_overrun_imprisoned
	}
}

# Scope: unit
bag_rtk_on_overrun_imprisoned = {
	effect = {
		bag_rtk_safety_unit = yes
	}
}
