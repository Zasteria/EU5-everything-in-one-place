# «Помощник правителя» 0.12.x: своя «Продажа творения» (его слово 10-26:
# «своё действие продажи, а не пинай ванилу»). Действие державы из
# `country_interactions` скрипт мода не запускает ни в одном режиме (0.11.0–0.11.3,
# его прогоны 10-26/27, `docs/TESTLOG.md`), а своё общее действие
# (`generic_actions`) — запускает, через такт после вызова (0.12.0, прогон 10-27):
# так Construction Manager и Playmaker Cabinet Queue зовут свои `fire_generic_action`.
# Что выяснилось по дороге (его прогоны 10-27):
# - `type = diplomacy` общему действию нельзя (ошибка разбора), а с `recipient`
#   «Targetted action shouldn't be looking for a recipient» — тип `owncountry`;
# - `target_1` вызов не передаёт (0.12.2: всё ушло одной стране);
# - творение переменных не держит (0.12.3).
# Поэтому покупатель держит творение в своей переменной `bag_rtk_art_incoming`,
# эффект находит его по ней. Цена, престиж −2 % и перенос творения — как у
# ванильного `sell_work_of_art` (1 золото × цена творения, платит покупатель);
# платит сам эффект: плательщика движку назвать нечем. Дипломата берёт вызов
# (`bag_rtk_art_fire_own`). В списках игры не показывается, ИИ им не пользуется.
bag_rtk_sell_art = {
	type = owncountry
	show_in_gui_list = no
	ai_tick = never
	automation_tick = never

	# Только для страны, где включена продажа мода (не is_ai = no: в hands-off
	# страна игрока под ИИ, а проверять там).
	potential = {
		scope:actor = { has_variable = bag_rtk_art_on }
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = work_of_art
		source = actor
		target_flag = target
		name = "sell_work_of_art_select_art"
		none_available_msg_key = "sell_work_of_art_no_art"
		column = {
			data = name
		}
		visible = {
			art_is_priceless = no
			is_art_destroyed = no
			work_of_art_is_imperial_regalia = no
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			random_country = {
				limit = {
					has_variable = bag_rtk_art_incoming
					var:bag_rtk_art_incoming = { this = scope:target }
				}
				save_scope_as = bag_rtk_art_to
			}
		}
		if = {
			limit = { exists = scope:bag_rtk_art_to }
			scope:actor = {
				add_gold = scope:target.art_price
				add_prestige = {
					value = scope:target.art_price
					multiply = -0.02
				}
			}
			scope:bag_rtk_art_to = {
				add_gold = {
					value = scope:target.art_price
					multiply = -1
				}
				remove_variable = bag_rtk_art_incoming
			}
			scope:target = {
				move_art_and_owner = { location = scope:bag_rtk_art_to.capital country = scope:bag_rtk_art_to }
			}
			# Учёт мода: сколько продаж дошло до эффекта, история продаж.
			scope:actor = {
				if = { limit = { has_variable = bag_rtk_art_done } change_variable = { name = bag_rtk_art_done add = 1 } } else = { set_variable = { name = bag_rtk_art_done value = 1 } }
				save_scope_as = bag_rtk_seller
			}
			scope:target = { save_scope_as = bag_rtk_art }
			scope:bag_rtk_art_to = { save_scope_as = bag_rtk_art_buyer }
			scope:actor = {
				set_variable = { name = bag_rtk_art_log_price value = scope:target.art_price }
				set_variable = { name = bag_rtk_art_log_q value = scope:target.art_quality }
				set_variable = { name = bag_rtk_art_log_gold value = scope:bag_rtk_art_to.gold }
				bag_rtk_art_history_add = yes
				# Следующая продажа месяца — следующим тактом (одно общее действие за такт).
				bag_rtk_art_chain = yes
			}
		}
	}
}
