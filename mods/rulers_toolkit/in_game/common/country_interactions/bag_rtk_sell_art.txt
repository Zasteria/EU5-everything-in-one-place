# «Помощник правителя» 0.11.0: своя «Продажа творения» (его слово 10-26).
# Копия ванильного `sell_work_of_art.txt`, но цель — только держава-покупатель:
# творение мод кладёт в переменную продавца `bag_rtk_art_na` и зовёт действие
# `perform_diplomatic_action` с `mode = accept` (`bag_rtk_art_fire_own`), как
# ваниль зовёт `force_embargo`. Цена, плательщик, получатель — как у ванильного.
# Игроку в меню не видно (нет творения в переменной), ИИ им не пользуется.
bag_rtk_sell_art = {
	category = CATEGORY_ECONOMY_ACTIONS
	type = diplomacy

	price = price:sell_work_of_art
	price_modifier = {
		if = {
			limit = { scope:actor ?= { has_variable = bag_rtk_art_na } }
			add = {
				desc = "ART_SALE_VALUE"
				value = scope:actor.var:bag_rtk_art_na.art_price
			}
		}
		else = {
			value = 1
		}
	}
	payer = scope:recipient
	payee = scope:actor

	potential = {
		scope:actor = {
			has_variable = bag_rtk_art_na
		}
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_trade
		}
		visible = {
			is_real_country = yes
			NOT = { this = scope:actor }
			NOT = { country_type = building }
		}
		enabled = {
			NOT = { is_enemy_of = scope:actor }
		}
	}

	effect = {
		# Зонд 0.11.1: дошёл ли движок до эффекта и было ли творение в переменной.
		if = {
			limit = { has_global_variable = bag_rtk_log }
			if = {
				limit = { scope:actor = { has_variable = bag_rtk_art_na } }
				debug_log = "bag_rtk: own sell action EFFECT ran, art in variable"
			}
			else = {
				debug_log = "bag_rtk: own sell action EFFECT ran, NO art in variable"
			}
		}
		if = {
			limit = { scope:actor = { has_variable = bag_rtk_art_na } }
			scope:actor = {
				add_prestige = {
					value = scope:actor.var:bag_rtk_art_na.art_price
					multiply = -0.02
				}
			}
			custom_tooltip = {
				text = sell_work_of_art_move_art_generic_tt
				hidden_effect = {
					scope:actor.var:bag_rtk_art_na = {
						move_art_and_owner = { location = scope:recipient.capital country = scope:recipient }
					}
				}
			}
		}
	}

	ai_will_do = {
		value = 0
	}

	diplo_chance = {
		base = -30
	}

	accept = {
		if = {
			limit = { exists = scope:recipient }
			add = {
				desc = "THEIR_OPINION_TOOLTIP"
				value = "scope:recipient.opinion(scope:actor)"
				divide = 3
			}
		}
		if = {
			limit = { scope:recipient ?= { gold < { value = monthly_income_total multiply = 6 } } }
			add = {
				desc = "LOW_TREASURY"
				value = -60
			}
		}
		if = {
			limit = { scope:recipient ?= { gold > { value = monthly_income_total multiply = 24 } } }
			add = {
				desc = "WEALTHY_BUYER"
				value = 30
			}
		}
		if = {
			limit = { scope:recipient ?= { is_rival_of = scope:actor } }
			add = {
				desc = "game_concept_rival"
				value = -100
			}
		}
	}
}
