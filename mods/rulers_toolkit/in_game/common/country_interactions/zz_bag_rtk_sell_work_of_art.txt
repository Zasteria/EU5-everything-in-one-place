# 0.4.5 (10-18): ванильный `country_interactions/sell_work_of_art.txt`
# целиком, плюс запасной путь для творения. Вручную продажа идёт (его скрины
# 10-18); из `perform_diplomatic_action` — нет: 0.4.2 (`propose`) тратил
# дипломата без продажи, 0.4.3 (`accept`) не делал ничего. Догадка: творение
# (`target`) из скрипта до действия не доходит, и оно идёт веткой «без творения»
# — цена 1 золото, ничего не переезжает. Поэтому мод кладёт творение ещё и в
# переменную продавца `bag_rtk_art_pending`, а здесь цена, согласие и эффект
# берут его оттуда, когда `scope:target` нет. Для ручной продажи ничего не
# меняется: там `scope:target` есть. Зонды в `effect` говорят, какой путь шёл.
REPLACE:sell_work_of_art = {
	category = CATEGORY_ECONOMY_ACTIONS
	type = diplomacy

	ai_tick = daily
	ai_tick_frequency = 360

	ai_prerequisite = {
		exists = scope:actor
		scope:actor = {
			gold < { value = monthly_income_total multiply = 6 }
		}
	}

	price = price:sell_work_of_art
	price_modifier = {
		if = {
			limit = { exists = scope:target }
			add = {
				desc = "ART_SALE_VALUE"
				value = scope:target.art_price
			}
		}
		else_if = {
			limit = { scope:actor ?= { has_variable = bag_rtk_art_pending } }
			add = {
				desc = "ART_SALE_VALUE"
				value = scope:actor.var:bag_rtk_art_pending.art_price
			}
		}
		else = {
			value = 1
		}
	}
	payer = scope:recipient
	payee = scope:actor

	potential = {
		scope:actor = {
			has_sellable_art = yes
		}
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		check_cheapest_price = yes
		column = {
			data = country_diplomacy_info_trade
		}
		visible = {
			is_real_country = yes
			NOT = { this = scope:actor }
			NOT = { country_type = building }
		}
		enabled = {
			NOT = { is_enemy_of = scope:actor }
		}
	}

	select_trigger = {
		looking_for_a = work_of_art
		source = actor
		cache_targets = yes
		target_flag = target
		name = "sell_work_of_art_select_art"
		none_available_msg_key = "sell_work_of_art_no_art"
		column = {
			data = name
		}
		column = {
			data = select_target_acceptance_frame
		}
		visible = {
			art_is_priceless = no
			is_art_destroyed = no
			work_of_art_is_imperial_regalia = no
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			debug_log = "bag_rtk: probe, sell_work_of_art effect ran with the art"
			scope:actor = {
				add_prestige = {
					value = scope:target.art_price
					multiply = -0.02
				}
			}
			custom_tooltip = {
				text = sell_work_of_art_move_art_tt
				hidden_effect = {
					scope:target = {
						move_art_and_owner = { location = scope:recipient.capital country = scope:recipient }
					}
				}
			}
		}
		else_if = {
			limit = { scope:actor ?= { has_variable = bag_rtk_art_pending } }
			debug_log = "bag_rtk: probe, sell_work_of_art effect ran with the art from the variable"
			scope:actor = {
				add_prestige = {
					value = var:bag_rtk_art_pending.art_price
					multiply = -0.02
				}
				var:bag_rtk_art_pending = {
					move_art_and_owner = { location = scope:recipient.capital country = scope:recipient }
				}
				remove_variable = bag_rtk_art_pending
			}
		}
		else = {
			debug_log = "bag_rtk: probe, sell_work_of_art effect ran without the art"
			custom_tooltip = sell_work_of_art_move_art_generic_tt
			custom_tooltip = woa_trade_price_scales_with_value_tt
		}
	}

	ai_will_do = {
		add = -100
		# sell when treasury is below 6 months of income
		if = {
			limit = {
				scope:actor = {
					gold < { value = monthly_income_total multiply = 6 }
				}
			}
			add = 100
		}
		if = {
			limit = { scope:actor = { is_enemy_of = scope:recipient } }
			multiply = 0
		}
	}

	diplo_chance = {
		base = -30
	}

	accept = {
		if = {
			limit = { exists = scope:recipient }
			add = {
				desc = "THEIR_OPINION_TOOLTIP"
				value = "scope:recipient.opinion(scope:actor)"
				divide = 3
			}
		}
		if = {
			limit = { scope:recipient ?= { gold < { value = monthly_income_total multiply = 6 } } }
			add = {
				desc = "LOW_TREASURY"
				value = -60
			}
		}
		if = {
			limit = { scope:recipient ?= { gold > { value = monthly_income_total multiply = 24 } } }
			add = {
				desc = "WEALTHY_BUYER"
				value = 30
			}
		}
		if = {
			limit = { exists = scope:target }
			scope:target ?= {
				if = {
					limit = { art_quality > 50 }
					add = {
						desc = "HIGH_QUALITY_ART"
						value = art_quality
						subtract = 50
					}
				}
				if = {
					limit = { art_quality < 50 }
					add = {
						desc = "LOW_QUALITY_ART"
						value = art_quality
						subtract = 50
					}
				}
			}
		}
		else_if = {
			limit = { scope:actor ?= { has_variable = bag_rtk_art_pending } }
			scope:actor.var:bag_rtk_art_pending ?= {
				if = {
					limit = { art_quality > 50 }
					add = {
						desc = "HIGH_QUALITY_ART"
						value = art_quality
						subtract = 50
					}
				}
				if = {
					limit = { art_quality < 50 }
					add = {
						desc = "LOW_QUALITY_ART"
						value = art_quality
						subtract = 50
					}
				}
			}
		}
		if = {
			limit = { scope:recipient ?= { is_rival_of = scope:actor } }
			add = {
				desc = "game_concept_rival"
				value = -100
			}
		}
	}
}
