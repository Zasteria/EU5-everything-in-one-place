# Зонд 0.11.3 (10-27): запускает ли `perform_diplomatic_action` своё действие
# державы вообще. Самое простое действие: без цены, без условий, эффект — строка
# журнала. Мод зовёт его дважды: от игрока к покупателю и от покупателя к игроку
# (ИИ — действующая сторона). Видно, пока стоит глобальная `bag_rtk_probe_pa`.
bag_rtk_probe_action = {
	category = CATEGORY_FRIENDLY_ACTIONS
	type = diplomacy

	potential = {
		has_global_variable = bag_rtk_probe_pa
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = country_diplomacy_info_trade
		}
		visible = {
			NOT = { this = scope:actor }
		}
	}

	effect = {
		if = {
			limit = { scope:actor = { is_ai = no } }
			debug_log = "bag_rtk: PROBE ACTION ran, actor is the player"
		}
		else = {
			debug_log = "bag_rtk: PROBE ACTION ran, actor is AI"
		}
	}

	ai_will_do = {
		value = 0
	}

	diplo_chance = {
		base = 1000
	}

	accept = {
		add = {
			desc = "bag_rtk_art_agreed_desc"
			value = 1000
		}
	}
}
