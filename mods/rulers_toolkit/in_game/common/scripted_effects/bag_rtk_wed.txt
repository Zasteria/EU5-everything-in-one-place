# Женитьба династии (0.7.0–0.8.1, его просьба 10-19): каждый член правящей семьи
# игрока без пары с 16 лет — 1-го числа женится на самом молодом подходящем
# члене правящей семьи другой страны (16 лет — первым).
#
# 0.8.1 — по прогонам игры без игрока на его румынском сейве (10-20,
# `-loadsavehandsoff`, восемь запусков, TESTLOG): `perform_diplomatic_action`
# с `type = royal_marriage` молча ничего не делает ни в одном виде аргументов, ни
# с `mode = accept`, ни через месяц, даже с невестой Блуа, которую окно игры
# считало согласной; `create_relation` его тоже не создаёт. Женит только
# `marry_character`, и отношения «династический брак» между странами он не
# даёт. Поэтому брак — прямой, а невеста — как в окне игры: правитель или его
# близкая родственница, монархия, в дипломатической дистанции, одна группа вер.
# 0.8.0 брал любого члена короны и женил Вука на Фуггерах.

# Scope: country (игрок).
bag_rtk_wed_monthly = {
	if = {
		limit = { has_variable = bag_rtk_wed_on }
		save_scope_as = bag_rtk_wed_home
		every_character = {
			limit = {
				bag_rtk_wed_single = yes
				OR = {
					is_female = no
					root = { has_variable = bag_rtk_wed_women_on }
				}
			}
			save_scope_as = bag_rtk_wed_one
			save_scope_as = recipient
			bag_rtk_wed_try = yes
		}
	}
}

# Один наш член семьи — `scope:bag_rtk_wed_one`. Scope: character.
bag_rtk_wed_try = {
	clear_variable_list = bag_rtk_wed_pool
	every_country = {
		limit = {
			bag_rtk_wed_land_ok = yes
			any_character = { bag_rtk_wed_match = yes }
		}
		every_character = {
			limit = { bag_rtk_wed_match = yes }
			save_scope_as = bag_rtk_wed_cand
			scope:bag_rtk_wed_one = {
				add_to_variable_list = { name = bag_rtk_wed_pool target = scope:bag_rtk_wed_cand }
			}
		}
	}
	if = {
		limit = { NOT = { variable_list_size = { name = bag_rtk_wed_pool value >= 1 } } }
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: wedding, no match for:" }
		if = { limit = { has_global_variable = bag_rtk_log } debug_log_scopes = no }
	}
	ordered_in_list = {
		variable = bag_rtk_wed_pool
		order_by = bag_rtk_wed_youth
		position = 0
		check_range_bounds = no
		save_scope_as = bag_rtk_wed_cand
		scope:bag_rtk_wed_one = { marry_character = scope:bag_rtk_wed_cand }
		if = {
			limit = { scope:bag_rtk_wed_one = { is_married = yes } }
			if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: wedding done, to:" }
		}
		else = {
			if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: wedding failed, to:" }
		}
		if = { limit = { has_global_variable = bag_rtk_log } debug_log_scopes = no }
		owner = { if = { limit = { has_global_variable = bag_rtk_log } debug_log_scopes = no } }
	}
	clear_variable_list = bag_rtk_wed_pool
}
