# Женитьба династии (0.7.0–0.8.0, его просьба 10-19): каждый член правящей семьи
# игрока без пары с 16 лет — 1-го числа женится на самом молодом подходящем
# члене правящей семьи любой страны (16 лет — первым).
#
# 0.8.0 — как `auto_marry` (`mods/auto_marry/own/in_game/events/auto_marry_royal.txt`),
# который у него работал (его слова 10-19 13:55): брак — игровым `royal_marriage`
# с вынужденным согласием (`mode = accept`), без дипломатов; если после него
# человек не женат — прямым `marry_character`. Предложения с ожиданием согласия
# (0.7.x) не женили никого: шесть отказов за шесть месяцев (TESTLOG 10-19 13:41).
# Аргументы пробуются в обоих видах: `target_1`/`target_2`, как scope у
# `on_royal_marriage` в `_hardcoded.txt`, затем `target`/`target_1`, как у
# `auto_marry`. Не мерено, какой из путей женит: зонд `bag_rtk: wedding done`
# пишет путь и появился ли династический брак между странами.

# Scope: country (игрок).
bag_rtk_wed_monthly = {
	if = {
		limit = { has_variable = bag_rtk_wed_on }
		save_scope_as = bag_rtk_wed_home
		every_character = {
			limit = {
				bag_rtk_wed_single = yes
				OR = {
					is_female = no
					root = { has_variable = bag_rtk_wed_women_on }
				}
			}
			save_scope_as = bag_rtk_wed_one
			save_scope_as = recipient
			bag_rtk_wed_try = yes
		}
	}
}

# Один наш член семьи — `scope:bag_rtk_wed_one`. Scope: character.
bag_rtk_wed_try = {
	clear_variable_list = bag_rtk_wed_pool
	every_country = {
		limit = {
			NOT = { this = scope:bag_rtk_wed_home }
			any_character = { bag_rtk_wed_match = yes }
		}
		every_character = {
			limit = { bag_rtk_wed_match = yes }
			save_scope_as = bag_rtk_wed_cand
			scope:bag_rtk_wed_one = {
				add_to_variable_list = { name = bag_rtk_wed_pool target = scope:bag_rtk_wed_cand }
			}
		}
	}
	ordered_in_list = {
		variable = bag_rtk_wed_pool
		order_by = bag_rtk_wed_youth
		position = 0
		check_range_bounds = no
		save_scope_as = bag_rtk_wed_cand
		owner = { save_scope_as = bag_rtk_wed_land }
		scope:bag_rtk_wed_home = {
			perform_diplomatic_action = {
				type = royal_marriage
				actor = scope:bag_rtk_wed_home
				recipient = scope:bag_rtk_wed_land
				target_1 = scope:bag_rtk_wed_one
				target_2 = scope:bag_rtk_wed_cand
				mode = accept
				hidden = yes
			}
		}
		if = {
			limit = { scope:bag_rtk_wed_one = { is_married = yes } }
			debug_log = "bag_rtk: wedding done, path target_1/target_2, to:"
		}
		else = {
			scope:bag_rtk_wed_home = {
				perform_diplomatic_action = {
					type = royal_marriage
					actor = scope:bag_rtk_wed_home
					recipient = scope:bag_rtk_wed_land
					target = scope:bag_rtk_wed_one
					target_1 = scope:bag_rtk_wed_cand
					mode = accept
					hidden = yes
				}
			}
			if = {
				limit = { scope:bag_rtk_wed_one = { is_married = yes } }
				debug_log = "bag_rtk: wedding done, path target/target_1, to:"
			}
			else = {
				scope:bag_rtk_wed_one = { marry_character = scope:bag_rtk_wed_cand }
				if = {
					limit = { scope:bag_rtk_wed_one = { is_married = yes } }
					debug_log = "bag_rtk: wedding done, path marry_character, to:"
				}
				else = {
					debug_log = "bag_rtk: wedding failed, all three paths, to:"
				}
			}
		}
		debug_log_scopes = no
		if = {
			limit = { scope:bag_rtk_wed_home = { has_royal_marriage_with = scope:bag_rtk_wed_land } }
			debug_log = "bag_rtk: wedding, royal marriage between countries yes"
		}
		else = {
			debug_log = "bag_rtk: wedding, royal marriage between countries no"
		}
	}
	clear_variable_list = bag_rtk_wed_pool
}
