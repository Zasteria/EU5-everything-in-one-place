# Напоминалка о религиозных действиях (0.5.0, его просьба 10-18): как
# напоминалки об откатах в Calidad de vida (`qol_cooldown_effects.txt`), только
# по действиям, отмеченным на вкладке «Религия». Таблица, галочки и тексты —
# `bag_rtk_faith_generated.txt` из `tools/gen_faith.py`.
#
# 1-го числа: откат идёт — запоминаем (`bag_rtk_rel_cd_<тип>`) и убираем старое
# напоминание; отката нет, а был — напоминание CMF `bag_rtk_rel_ready_<тип>`.
# Действие, которым ни разу не пользовались при галочке, не напомнит: откат
# надо сначала увидеть.

# Scope: country. Args: type — тип отката (`has_cooldown`).
bag_rtk_faith_check_one = {
	# Галочка — в карте `cmm` державы, как читает её `cmm_sync_bool_alias`.
	set_local_variable = { name = bag_rtk_faith_flag value = flag:bag_rtk__rel_$type$ }
	if = {
		limit = { "variable_map(cmm|local_var:bag_rtk_faith_flag)" >= 1 }
		if = {
			limit = { has_cooldown = $type$ }
			if = {
				limit = { NOT = { has_variable = bag_rtk_rel_cd_$type$ } }
				set_variable = { name = bag_rtk_rel_cd_$type$ value = yes }
				cmf_remove_alert = { alert = bag_rtk_rel_ready_$type$ }
			}
		}
		else_if = {
			limit = { has_variable = bag_rtk_rel_cd_$type$ }
			remove_variable = bag_rtk_rel_cd_$type$
			cmf_show_alert = { alert = bag_rtk_rel_ready_$type$ }
			# Клик открывает вкладку религии — как ванильные религиозные алерты
			# (`alertmanager.gui`: `OpenLateralView('country_religion')`).
			cmf_set_alert_open_view = { alert = bag_rtk_rel_ready_$type$ view = country_religion }
			debug_log = "bag_rtk: faith reminder shown, $type$"
		}
	}
	else_if = {
		limit = { has_variable = bag_rtk_rel_cd_$type$ }
		remove_variable = bag_rtk_rel_cd_$type$
	}
}
