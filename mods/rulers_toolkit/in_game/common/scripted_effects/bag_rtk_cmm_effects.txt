# Страница мода в меню CMF. Одна вкладка, по группе на функцию.
#
# **Макрос CMM, позванный с аргументом, которого CMF не объявляет, молча не
# делает ничего и уносит с собой весь остаток эффекта.** После правки —
# `python3 tools/check_cmm.py mods/rulers_toolkit/in_game/common`.

# Scope: country
bag_rtk_register = {
	# Коронные командиры (commanders)
	cmm_register_bool_setting = {
		mod_id = bag_rtk
		setting_id = cmd_enabled
		tab_id = main
		group_id = commanders
		default_value = 1
	}
	cmm_sync_bool_alias = {
		setting = bag_rtk__cmd_enabled
		alias = bag_rtk_cmd_on
	}

	cmm_register_bool_setting = {
		mod_id = bag_rtk
		setting_id = cmd_navy
		tab_id = main
		group_id = commanders
		default_value = 0
	}
	cmm_sync_bool_alias = {
		setting = bag_rtk__cmd_navy
		alias = bag_rtk_cmd_navy_on
	}

	cmm_register_bool_setting = {
		mod_id = bag_rtk
		setting_id = cmd_war_ruler
		tab_id = main
		group_id = commanders
		default_value = 1
	}
	cmm_sync_bool_alias = {
		setting = bag_rtk__cmd_war_ruler
		alias = bag_rtk_cmd_war_ruler_on
	}
}

# Обратный звонок CMF: `var:cmf_callback` — флаг того, что тронули.
# Scope: country
bag_rtk_handle_callback = {
	switch = {
		trigger = var:cmf_callback
		flag:bag_rtk__cmd_enabled = {
			cmm_sync_bool_alias = {
				setting = bag_rtk__cmd_enabled
				alias = bag_rtk_cmd_on
			}
		}
		flag:bag_rtk__cmd_navy = {
			cmm_sync_bool_alias = {
				setting = bag_rtk__cmd_navy
				alias = bag_rtk_cmd_navy_on
			}
		}
		flag:bag_rtk__cmd_war_ruler = {
			cmm_sync_bool_alias = {
				setting = bag_rtk__cmd_war_ruler
				alias = bag_rtk_cmd_war_ruler_on
			}
		}
	}
}
