# Продажа творений (0.4.0, его просьба 10-18).
#
# Раз в месяц: пока свободных дипломатов больше, чем велено оставить, самое
# дорогое творение отмеченного качества уходит той державе, что согласна охотнее
# всех. Продажу мод проводит сам по правилам действия «Продажа творения»:
# действие, вызванное из скрипта (`fire_generic_action`,
# `perform_diplomatic_action` с `propose` и с `accept`), до своего эффекта не
# доходило ни разу (0.4.0–0.4.5). Если владелец после продажи прежний, зонд
# пишет `not sold`, и месяц на этом кончается.

# Scope: country
bag_rtk_art_monthly = {
	if = {
		limit = { has_variable = bag_rtk_art_on }
		save_scope_as = bag_rtk_seller
		while = {
			count = 5
			limit = {
				NOT = { has_variable = bag_rtk_art_stop }
				num_of_diplomats > bag_rtk_art_keep
				any_work_of_art_in_country = { bag_rtk_art_for_sale = yes }
			}
			bag_rtk_art_sell_one = yes
		}
		remove_variable = bag_rtk_art_stop
	}
}

# Одна продажа: творения от дорогих к дешёвым, первое, у которого есть
# покупатель. `bag_rtk_art_stop` снимается только продажей — иначе месяц кончен.
# Scope: country (scope:bag_rtk_seller)
bag_rtk_art_sell_one = {
	set_variable = { name = bag_rtk_art_stop value = yes }
	remove_variable = bag_rtk_art_tried
	ordered_work_of_art_in_country = {
		limit = { bag_rtk_art_for_sale = yes }
		order_by = art_price
		max = 100
		check_range_bounds = no
		if = {
			limit = { scope:bag_rtk_seller = { NOT = { has_variable = bag_rtk_art_tried } } }
			save_scope_as = bag_rtk_art
			if = {
				limit = {
					scope:bag_rtk_seller = {
						any_country_in_diplomatic_range = { bag_rtk_art_buyer = yes }
					}
				}
				bag_rtk_art_fire = yes
			}
		}
	}
	if = {
		limit = { NOT = { has_variable = bag_rtk_art_tried } }
		debug_log = "bag_rtk: no buyer for any work of art"
	}
	remove_variable = bag_rtk_art_tried
}

# Scope: work_of_art (scope:bag_rtk_art), покупатель есть.
bag_rtk_art_fire = {
	scope:bag_rtk_seller = {
		set_variable = { name = bag_rtk_art_tried value = yes }
		ordered_country_in_diplomatic_range = {
			limit = { bag_rtk_art_buyer = yes }
			order_by = bag_rtk_art_accept
			max = 1
			save_scope_as = bag_rtk_art_buyer
		}
		# Продажа — тело ванильного `sell_work_of_art` и его цена
		# (`price:sell_work_of_art` = 1 золото × `art_price`, платит
		# покупатель; престиж −2 % цены; дипломат). Само действие из скрипта до
		# эффекта не доходит ни одним путём (0.4.0–0.4.5, его прогоны 10-18,
		# `docs/TESTLOG.md`), хотя вручную продаётся; согласие покупателя
		# проверено выше его же формулой.
		add_gold = scope:bag_rtk_art.art_price
		add_prestige = {
			value = scope:bag_rtk_art.art_price
			multiply = -0.02
		}
		add_diplomats = -1
	}
	scope:bag_rtk_art_buyer = {
		add_gold = {
			value = scope:bag_rtk_art.art_price
			multiply = -1
		}
	}
	move_art_and_owner = {
		location = scope:bag_rtk_art_buyer.capital
		country = scope:bag_rtk_art_buyer
	}
	if = {
		limit = { owner ?= scope:bag_rtk_art_buyer }
		debug_log = "bag_rtk: work of art sold"
		scope:bag_rtk_seller = { remove_variable = bag_rtk_art_stop }
	}
	else = {
		debug_log = "bag_rtk: work of art not sold"
	}
}
