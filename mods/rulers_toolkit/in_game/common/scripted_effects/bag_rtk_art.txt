# Продажа творений (0.4.0, его просьба 10-18).
#
# Раз в месяц: пока свободных дипломатов больше, чем велено оставить, самое
# дорогое творение отмеченного качества уходит той державе, что согласна охотнее
# всех. Продажу мод проводит сам по правилам действия «Продажа творения»:
# действие, вызванное из скрипта (`fire_generic_action`,
# `perform_diplomatic_action` с `propose` и с `accept`), до своего эффекта не
# доходило ни разу (0.4.0–0.4.5). Если владелец после продажи прежний, зонд
# пишет `not sold`, и месяц на этом кончается.

# Scope: country
bag_rtk_art_monthly = {
	if = {
		limit = { has_variable = bag_rtk_art_on }
		save_scope_as = bag_rtk_seller
		while = {
			count = 5
			limit = {
				NOT = { has_variable = bag_rtk_art_stop }
				num_of_diplomats >= bag_rtk_art_dipl_floor
				any_work_of_art_in_country = { bag_rtk_art_for_sale = yes }
			}
			bag_rtk_art_sell_one = yes
		}
		remove_variable = bag_rtk_art_stop
	}
}

# Одна продажа: творения от дорогих к дешёвым, первое, у которого есть
# покупатель. `bag_rtk_art_stop` снимается только продажей — иначе месяц кончен.
# Scope: country (scope:bag_rtk_seller)
bag_rtk_art_sell_one = {
	set_variable = { name = bag_rtk_art_stop value = yes }
	remove_variable = bag_rtk_art_tried
	ordered_work_of_art_in_country = {
		limit = { bag_rtk_art_for_sale = yes }
		order_by = art_price
		max = 100
		check_range_bounds = no
		if = {
			limit = { scope:bag_rtk_seller = { NOT = { has_variable = bag_rtk_art_tried } } }
			save_scope_as = bag_rtk_art
			if = {
				limit = {
					scope:bag_rtk_seller = {
						any_country_in_diplomatic_range = { bag_rtk_art_buyer = yes }
					}
				}
				bag_rtk_art_fire = yes
			}
		}
	}
	if = {
		limit = { NOT = { has_variable = bag_rtk_art_tried } }
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: no buyer for any work of art" }
	}
	remove_variable = bag_rtk_art_tried
}

# Scope: work_of_art (scope:bag_rtk_art), покупатель есть.
bag_rtk_art_fire = {
	scope:bag_rtk_seller = {
		set_variable = { name = bag_rtk_art_tried value = yes }
		ordered_country_in_diplomatic_range = {
			limit = { bag_rtk_art_buyer = yes }
			order_by = bag_rtk_art_accept
			max = 1
			save_scope_as = bag_rtk_art_buyer
		}
	}
	# Для строки в журнал: согласие покупателя до оплаты.
	scope:bag_rtk_art_buyer = {
		set_variable = { name = bag_rtk_art_log_acc value = bag_rtk_art_accept }
	}
	scope:bag_rtk_seller = {
		set_variable = { name = bag_rtk_art_log_acc value = scope:bag_rtk_art_buyer.var:bag_rtk_art_log_acc }
	}
	scope:bag_rtk_art_buyer = { remove_variable = bag_rtk_art_log_acc }
	if = {
		limit = { scope:bag_rtk_seller = { has_variable = bag_rtk_art_native_on } }
		bag_rtk_art_fire_own = yes
	}
	else = {
		bag_rtk_art_pay_and_move = yes
	}
	if = {
		limit = { owner ?= scope:bag_rtk_art_buyer }
		scope:bag_rtk_seller = {
			set_variable = { name = bag_rtk_art_log_price value = scope:bag_rtk_art.art_price }
			set_variable = { name = bag_rtk_art_log_q value = scope:bag_rtk_art.art_quality }
			set_variable = { name = bag_rtk_art_log_gold value = scope:bag_rtk_art_buyer.gold }
		}
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: work of art sold, buyer:" }
		scope:bag_rtk_art_buyer = { if = { limit = { has_global_variable = bag_rtk_log } debug_log_scopes = no } }
		scope:bag_rtk_seller = { bag_rtk_art_history_add = yes }
		scope:bag_rtk_seller = { remove_variable = bag_rtk_art_stop }
	}
	else = {
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: work of art not sold" }
	}
}

# Продажа скриптом — тело ванильного `sell_work_of_art` и его цена
# (`price:sell_work_of_art` = 1 золото × `art_price`, платит покупатель;
# престиж −2 % цены; дипломат). Ванильное действие из скрипта до эффекта не
# доходит ни одним путём (0.4.0–0.4.5, его прогоны 10-18, `docs/TESTLOG.md`),
# хотя вручную продаётся; согласие покупателя проверено выше его же формулой.
# Scope: work_of_art (scope:bag_rtk_art), покупатель выбран.
bag_rtk_art_pay_and_move = {
	scope:bag_rtk_seller = {
		add_gold = scope:bag_rtk_art.art_price
		add_prestige = {
			value = scope:bag_rtk_art.art_price
			multiply = -0.02
		}
		add_diplomats = -1
	}
	scope:bag_rtk_art_buyer = {
		add_gold = {
			value = scope:bag_rtk_art.art_price
			multiply = -1
		}
	}
	move_art_and_owner = {
		location = scope:bag_rtk_art_buyer.capital
		country = scope:bag_rtk_art_buyer
	}
}

# 0.11.0 (его слово 10-26: «своё действие продажи, а не пинай ванилу»): своё
# действие державы `bag_rtk_sell_art` (`country_interactions/bag_rtk_sell_art.txt`)
# — цель только держава, творение в переменной продавца `bag_rtk_art_na`;
# так ваниль зовёт `force_embargo` из `hegemon_demands.txt` (`mode = accept`).
# Платит ли движок цену и берёт ли дипломата при `accept` — не мерено: зонд
# сверяет золото и дипломатов и доплачивает сам, если нет. Не сработало действие
# вовсе — продаёт скриптом, как раньше.
# Scope: work_of_art (scope:bag_rtk_art), покупатель выбран.
bag_rtk_art_fire_own = {
	scope:bag_rtk_seller = {
		set_variable = { name = bag_rtk_art_na value = scope:bag_rtk_art }
		set_variable = { name = bag_rtk_art_gold0 value = gold }
		set_variable = { name = bag_rtk_art_dipl0 value = num_of_diplomats }
		perform_diplomatic_action = {
			type = bag_rtk_sell_art
			actor = scope:bag_rtk_seller
			recipient = scope:bag_rtk_art_buyer
			mode = accept
		}
		remove_variable = bag_rtk_art_na
	}
	if = {
		limit = { owner ?= scope:bag_rtk_art_buyer }
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: own sell action ran" }
		scope:bag_rtk_seller = {
			if = {
				limit = { gold > { value = var:bag_rtk_art_gold0 add = { value = scope:bag_rtk_art.art_price multiply = 0.5 } } }
				if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: own sell action, the game paid" }
			}
			else = {
				if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: own sell action, the game did NOT pay, paying by script" }
				add_gold = scope:bag_rtk_art.art_price
				scope:bag_rtk_art_buyer = {
					add_gold = {
						value = scope:bag_rtk_art.art_price
						multiply = -1
					}
				}
			}
			if = {
				limit = { num_of_diplomats < { value = var:bag_rtk_art_dipl0 subtract = 0.5 } }
				if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: own sell action, the game took a diplomat" }
			}
			else = {
				if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: own sell action, no diplomat taken, taking by script" }
				add_diplomats = -1
			}
		}
	}
	else = {
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: own sell action did NOT run, selling by script" }
		bag_rtk_art_pay_and_move = yes
	}
	scope:bag_rtk_seller = {
		remove_variable = bag_rtk_art_gold0
		remove_variable = bag_rtk_art_dipl0
	}
}

		}
		set_variable = { name = bag_rtk_art_h1_art value = scope:bag_rtk_art }
		set_variable = { name = bag_rtk_art_h1_buyer value = scope:bag_rtk_art_buyer }
		if = { limit = { has_variable = bag_rtk_art_log_price } set_variable = { name = bag_rtk_art_h1_price value = var:bag_rtk_art_log_price } }
		if = { limit = { has_variable = bag_rtk_art_log_q } set_variable = { name = bag_rtk_art_h1_q value = var:bag_rtk_art_log_q } }
		if = { limit = { has_variable = bag_rtk_art_log_acc } set_variable = { name = bag_rtk_art_h1_acc value = var:bag_rtk_art_log_acc } }
		if = { limit = { has_variable = bag_rtk_art_log_gold } set_variable = { name = bag_rtk_art_h1_gold value = var:bag_rtk_art_log_gold } }
		set_variable = { name = bag_rtk_art_h1_year value = current_year }
		change_variable = { name = bag_rtk_art_h_count add = 1 }
	}
}
