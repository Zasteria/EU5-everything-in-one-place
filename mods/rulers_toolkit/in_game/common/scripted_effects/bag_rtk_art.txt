# Продажа творений (0.4.0, его просьба 10-18).
#
# Раз в месяц: пока свободных дипломатов больше, чем велено оставить, самое
# дорогое творение отмеченного качества уходит той державе, что согласна охотнее
# всех. Продажу мод проводит сам по правилам действия «Продажа творения»:
# действие, вызванное из скрипта (`fire_generic_action`,
# `perform_diplomatic_action` с `propose` и с `accept`), до своего эффекта не
# доходило ни разу (0.4.0–0.4.5). Если владелец после продажи прежний, зонд
# пишет `not sold`, и месяц на этом кончается.

# Scope: country
bag_rtk_art_monthly = {
	if = {
		limit = { has_global_variable = bag_rtk_log has_variable = bag_rtk_art_native_on NOT = { has_variable = bag_rtk_art_on } }
		debug_log = "bag_rtk: art native ticked, but selling is off"
	}
	if = {
		limit = { has_variable = bag_rtk_art_on has_variable = bag_rtk_art_native_on }
		bag_rtk_art_native_pick = yes
	}
	else_if = {
		limit = { has_variable = bag_rtk_art_on }
		save_scope_as = bag_rtk_seller
		while = {
			count = 5
			limit = {
				NOT = { has_variable = bag_rtk_art_stop }
				num_of_diplomats >= bag_rtk_art_dipl_floor
				any_work_of_art_in_country = { bag_rtk_art_for_sale = yes }
			}
			bag_rtk_art_sell_one = yes
		}
		remove_variable = bag_rtk_art_stop
	}
}

# Одна продажа: творения от дорогих к дешёвым, первое, у которого есть
# покупатель. `bag_rtk_art_stop` снимается только продажей — иначе месяц кончен.
# Scope: country (scope:bag_rtk_seller)
bag_rtk_art_sell_one = {
	set_variable = { name = bag_rtk_art_stop value = yes }
	remove_variable = bag_rtk_art_tried
	ordered_work_of_art_in_country = {
		limit = { bag_rtk_art_for_sale = yes }
		order_by = art_price
		max = 100
		check_range_bounds = no
		if = {
			limit = { scope:bag_rtk_seller = { NOT = { has_variable = bag_rtk_art_tried } } }
			save_scope_as = bag_rtk_art
			if = {
				limit = {
					scope:bag_rtk_seller = {
						any_country_in_diplomatic_range = { bag_rtk_art_buyer = yes }
					}
				}
				bag_rtk_art_fire = yes
			}
		}
	}
	if = {
		limit = { NOT = { has_variable = bag_rtk_art_tried } }
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: no buyer for any work of art" }
	}
	remove_variable = bag_rtk_art_tried
}

# Scope: work_of_art (scope:bag_rtk_art), покупатель есть.
bag_rtk_art_fire = {
	scope:bag_rtk_seller = {
		set_variable = { name = bag_rtk_art_tried value = yes }
		ordered_country_in_diplomatic_range = {
			limit = { bag_rtk_art_buyer = yes }
			order_by = bag_rtk_art_accept
			max = 1
			save_scope_as = bag_rtk_art_buyer
		}
		# Продажа — тело ванильного `sell_work_of_art` и его цена
		# (`price:sell_work_of_art` = 1 золото × `art_price`, платит
		# покупатель; престиж −2 % цены; дипломат). Само действие из скрипта до
		# эффекта не доходит ни одним путём (0.4.0–0.4.5, его прогоны 10-18,
		# `docs/TESTLOG.md`), хотя вручную продаётся; согласие покупателя
		# проверено выше его же формулой.
		add_gold = scope:bag_rtk_art.art_price
		add_prestige = {
			value = scope:bag_rtk_art.art_price
			multiply = -0.02
		}
		add_diplomats = -1
	}
	# Для строки в журнал: согласие покупателя до оплаты.
	scope:bag_rtk_art_buyer = {
		set_variable = { name = bag_rtk_art_log_acc value = bag_rtk_art_accept }
	}
	scope:bag_rtk_seller = {
		set_variable = { name = bag_rtk_art_log_acc value = scope:bag_rtk_art_buyer.var:bag_rtk_art_log_acc }
	}
	scope:bag_rtk_art_buyer = {
		remove_variable = bag_rtk_art_log_acc
		add_gold = {
			value = scope:bag_rtk_art.art_price
			multiply = -1
		}
	}
	move_art_and_owner = {
		location = scope:bag_rtk_art_buyer.capital
		country = scope:bag_rtk_art_buyer
	}
	if = {
		limit = { owner ?= scope:bag_rtk_art_buyer }
		scope:bag_rtk_seller = {
			set_variable = { name = bag_rtk_art_log_price value = scope:bag_rtk_art.art_price }
			set_variable = { name = bag_rtk_art_log_q value = scope:bag_rtk_art.art_quality }
			set_variable = { name = bag_rtk_art_log_gold value = scope:bag_rtk_art_buyer.gold }
		}
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: work of art sold, buyer:" }
		scope:bag_rtk_art_buyer = { if = { limit = { has_global_variable = bag_rtk_log } debug_log_scopes = no } }
		scope:bag_rtk_seller = { bag_rtk_art_history_add = yes }
		scope:bag_rtk_seller = { remove_variable = bag_rtk_art_stop }
	}
	else = {
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: work of art not sold" }
	}
}

# Проба 0.10.0 (его мысль 10-26): продать ванильным действием, как вручную.
# Мод выбирает творение и покупателя по тем же правилам и кладёт их в
# переменные; привод `gui/bag_rtk_art_driver.gui` жмёт у покупателя
# «Продать творение» (`DiplomaticActionItem.OnClick`, как ПКМ по стране), а
# копия `gui/select_interaction_target.gui` жмёт в списке строку покупателя и
# строку творения (клик с параметрами, как у Playmaker Cabinet Queue). Одна
# продажа в месяц. Scope: country (игрок).
bag_rtk_art_native_pick = {
	# `bag_rtk_art_for_sale` читает `scope:bag_rtk_seller` — сохранить до первой проверки
	# (0.10.0–0.10.1 сохраняли после: «нет творений» на каждом месяце, его прогоны 10-26).
	save_scope_as = bag_rtk_seller
	# 0.10.1: его прогон 10-26 — три месяца ни строки; какое условие не прошло.
	if = {
		limit = { has_global_variable = bag_rtk_log }
		if = { limit = { has_variable = bag_rtk_art_native_due } debug_log = "bag_rtk: art native, skip, last sale still pending" }
		if = { limit = { num_of_diplomats < bag_rtk_art_dipl_floor } debug_log = "bag_rtk: art native, skip, diplomats below keep+1" }
		if = { limit = { NOT = { any_work_of_art_in_country = { bag_rtk_art_for_sale = yes } } } debug_log = "bag_rtk: art native, skip, no work of art of a ticked quality" }
	}
	if = {
		limit = {
			NOT = { has_variable = bag_rtk_art_native_due }
			num_of_diplomats >= bag_rtk_art_dipl_floor
			any_work_of_art_in_country = { bag_rtk_art_for_sale = yes }
		}
		save_scope_as = bag_rtk_seller
		remove_variable = bag_rtk_art_tried
		ordered_work_of_art_in_country = {
			limit = { bag_rtk_art_for_sale = yes }
			order_by = art_price
			max = 100
			check_range_bounds = no
			if = {
				limit = {
					scope:bag_rtk_seller = {
						NOT = { has_variable = bag_rtk_art_tried }
						any_country_in_diplomatic_range = { bag_rtk_art_buyer = yes }
					}
				}
				save_scope_as = bag_rtk_art
				scope:bag_rtk_seller = {
					set_variable = { name = bag_rtk_art_tried value = yes }
					ordered_country_in_diplomatic_range = {
						limit = { bag_rtk_art_buyer = yes }
						order_by = bag_rtk_art_accept
						max = 1
						save_scope_as = bag_rtk_art_buyer
					}
					set_variable = { name = bag_rtk_art_na value = scope:bag_rtk_art }
					set_variable = { name = bag_rtk_art_nb value = scope:bag_rtk_art_buyer }
					set_variable = { name = bag_rtk_art_native_due value = yes }
				}
				if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: art native, picked, buyer:" }
				scope:bag_rtk_art_buyer = { if = { limit = { has_global_variable = bag_rtk_log } debug_log_scopes = no } }
			}
		}
		if = {
			limit = { NOT = { has_variable = bag_rtk_art_tried } }
			if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: no buyer for any work of art" }
		}
		remove_variable = bag_rtk_art_tried
	}
}

# Через секунду после нажатия: ушло ли творение. Scope: country (игрок).
bag_rtk_art_native_check = {
	remove_variable = bag_rtk_art_native_due
	if = {
		limit = { has_variable = bag_rtk_art_na has_variable = bag_rtk_art_nb }
		var:bag_rtk_art_nb = { save_scope_as = bag_rtk_art_buyer }
		var:bag_rtk_art_na = { save_scope_as = bag_rtk_art }
		save_scope_as = bag_rtk_seller
		var:bag_rtk_art_na = {
			if = {
				limit = { owner ?= scope:bag_rtk_art_buyer }
				if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: art native, SOLD" }
				# В историю продаж, как своя продажа (согласия тут нет — его считала игра).
				scope:bag_rtk_seller = {
					remove_variable = bag_rtk_art_log_acc
					set_variable = { name = bag_rtk_art_log_price value = scope:bag_rtk_art.art_price }
					set_variable = { name = bag_rtk_art_log_q value = scope:bag_rtk_art.art_quality }
					set_variable = { name = bag_rtk_art_log_gold value = scope:bag_rtk_art_buyer.gold }
					bag_rtk_art_history_add = yes
				}
			}
			else = {
				if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: art native, not sold" }
			}
		}
	}
	remove_variable = bag_rtk_art_na
	remove_variable = bag_rtk_art_nb
}
