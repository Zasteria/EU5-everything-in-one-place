# Всё, что мод делает, делается по одной кнопке на его странице CMF.
#
# **Своего такта у мода нет** (0.6.0). До 0.6 он сам следил за войной и миром:
# скрытое событие каждый день, месячная сверка ползунков, правка инфляции
# месяцами. Его слово 2026-09-30: «вар слайдерс 100% сильно жрёт тики… просто
# вызывается только кнопкой». Теперь нажатие — один проход привода, после
# которого `bag_wsl_busy` гаснет и интерфейс мода снова читает одну переменную
# за кадр.

# **Кнопка «Выправить».** Его условие 2026-09-30: «снизить инфляцию до 0,
# включить автоматику, выправить ползунки содержания военные и уснуть только
# после этого до следующего нажатия». Содержание — на цель сразу, без сверки.
# Чеканка: инфляция выше цели — автоматика выключается и начинается правка,
# которую ведёт месячное событие `bag_wsl.2`, пока инфляция не сойдёт к цели;
# тогда чеканка отдаётся автоматике, и мод спит. Не выше цели — автоматика
# возвращается сразу.
# Конец войны любого рода (`bag_wsl_on_actions.txt`): через день `bag_wsl.3`.
# Крючков несколько и зовутся они разным странам войны; повтор за те же два дня
# не ставится. Scope: country
bag_wsl_queue_war_end = {
	if = {
		limit = {
			is_human = yes
			has_variable = bag_wsl_auto_on
			NOT = { has_variable = bag_wsl_end_queued }
		}
		set_variable = { name = bag_wsl_end_queued days = 2 }
		trigger_event_silently = { id = bag_wsl.3 days = 1 }
	}
}

# Scope: country
bag_wsl_manual_run = {
	# Хвосты автоматики до 0.6 в старом сохранении: сверки, дозор, память о войне.
	remove_variable = bag_wsl_army_check
	remove_variable = bag_wsl_navy_check
	remove_variable = bag_wsl_fort_check
	remove_variable = bag_wsl_watch
	remove_variable = bag_wsl_at_war
	bag_wsl_maint_arm = yes
	if = {
		limit = { has_variable = bag_wsl_mint_on }
		if = {
			limit = { inflation > bag_wsl_inflation_stop }
			if = {
				limit = { has_variable = bag_wsl_mint_phase }
				# Правка уже идёт — её месячное событие стоит в очереди;
				# пересчитать положение сейчас, вторую цепочку не заводить.
				bag_wsl_mint_rearm = yes
			}
			else = {
				bag_wsl_mint_arm = yes
				bag_wsl_mint_schedule = yes
			}
		}
		else_if = {
			limit = { has_variable = bag_wsl_mint_phase }
			bag_wsl_mint_stand_down = yes
		}
		else = {
			set_variable = { name = bag_wsl_mint_release value = 1 }
		}
	}
	bag_wsl_update_busy = yes
}

# **Правка живёт только пока идёт.** Одно скрытое событие через месяц; оно
# либо заканчивает правку, либо ставит себя ещё на месяц. Пока правки нет, в
# очереди ничего не стоит.
#
# `bag_wsl_mint_wait` отсекает событие старой цепочки: нажали, правка
# кончилась, нажали снова в том же месяце — первое событие ещё в очереди и
# придёт раньше, пока `wait` новой цепочки жив, и ничего не сделает.
# Scope: country
bag_wsl_mint_schedule = {
	set_variable = { name = bag_wsl_mint_wait value = 1 days = 29 }
	trigger_event_silently = { id = bag_wsl.2 days = 30 }
}

# Месяц правки прошёл. Scope: country
bag_wsl_mint_month = {
	if = {
		limit = {
			has_variable = bag_wsl_mint_phase
			NOT = { has_variable = bag_wsl_mint_wait }
		}
		if = {
			limit = { inflation > bag_wsl_inflation_stop }
			bag_wsl_mint_rearm = yes
			bag_wsl_mint_schedule = yes
		}
		else = {
			bag_wsl_mint_stand_down = yes
		}
	}
}

# Следующий месяц правки: привод снова ставит пол, спрашивает у игры прогноз и
# встаёт ровно на остаток (`bag_wsl_mint_measured`). Точка нуля `m0` остаётся
# от первого нажатия: автоматика с тех пор выключена, и ползунок стоит не в ней.
# Scope: country
bag_wsl_mint_rearm = {
	set_variable = { name = bag_wsl_mint_target value = bag_wsl_mint_arm }
	set_variable = { name = bag_wsl_mint_go value = 1 }
	bag_wsl_update_busy = yes
}

# **Флаг «привод нужен».** Виджет-привод целиком спит, пока его нет: без
# работы интерфейс мода — одно чтение этой переменной за кадр, без привязки к
# панели экономики (она писала строку в `debug.log` каждый кадр, прогон 09-20).
# Scope: country
bag_wsl_update_busy = {
	if = {
		limit = {
			OR = {
				has_variable = bag_wsl_army_go
				has_variable = bag_wsl_navy_go
				has_variable = bag_wsl_fort_go
				has_variable = bag_wsl_mint_go
				has_variable = bag_wsl_mint_release
				has_variable = bag_wsl_mint_repost
			}
		}
		set_variable = { name = bag_wsl_busy value = 1 }
	}
	else = {
		remove_variable = bag_wsl_busy
	}
}

# **Взводится на сотую выше цели, а не целью** (0.5.3): ползунок рождает
# событие, только когда число под ним меняется. Значения — `bag_wsl_*_arm`.
# Scope: country
bag_wsl_maint_arm = {
	if = {
		limit = { has_variable = bag_wsl_army_on }
		set_variable = { name = bag_wsl_army_target value = bag_wsl_army_arm }
		set_variable = { name = bag_wsl_army_go value = 1 }
	}
	if = {
		limit = { has_variable = bag_wsl_navy_on }
		set_variable = { name = bag_wsl_navy_target value = bag_wsl_navy_arm }
		set_variable = { name = bag_wsl_navy_go value = 1 }
	}
	if = {
		limit = { has_variable = bag_wsl_fort_on }
		set_variable = { name = bag_wsl_fort_target value = bag_wsl_fort_arm }
		set_variable = { name = bag_wsl_fort_go value = 1 }
	}
}

# Scope: country
bag_wsl_mint_arm = {
	if = {
		limit = {
			has_variable = bag_wsl_mint_on
			inflation > bag_wsl_inflation_stop
		}
		# Точка нулевого прироста — там, где ползунок стоит **сейчас**: пока
		# работала ванильная автоматика, она держала его ровно в ней
		# («Automated Minting with 0 Monthly Inflation»). Дальше это опора всей
		# арифметики последнего месяца, поэтому снимается до того, как мод
		# тронет ползунок.
		set_variable = { name = bag_wsl_m0 value = slider_minting_value }
		set_variable = { name = bag_wsl_mint_floor value = bag_wsl_mint_fraction }
		# Положение, которое держится ближайший месяц: по нему потом читается
		# измеренный темп. После частичного шага это уже не пол, поэтому
		# записывается отдельно, а не берётся из настройки.
		set_variable = { name = bag_wsl_held value = bag_wsl_mint_fraction }
		set_variable = { name = bag_wsl_infl_prev value = inflation }
		remove_variable = bag_wsl_drain
		set_variable = { name = bag_wsl_mint_target value = bag_wsl_mint_arm }
		set_variable = { name = bag_wsl_mint_phase value = 1 }
		set_variable = { name = bag_wsl_mint_go value = 1 }
		# Привод поставит пол, спросит у игры прогноз на месяц и сразу
		# пересчитает положение — см. `bag_wsl_mint_measured`.
	}
}

# Положение ползунка по остатку `bag_wsl_rem` и прогнозу `bag_wsl_drain`,
# снятому при положении `bag_wsl_held`: m = m0 − R × (m0 − held) / D.
# Прирост инфляции линеен по положению между точкой нуля `m0` и полом.
# Scope: country
bag_wsl_mint_aim = {
	if = {
		limit = {
			var:bag_wsl_drain > 0
			var:bag_wsl_m0 > var:bag_wsl_held
		}
		set_local_variable = { name = bag_wsl_span value = var:bag_wsl_m0 }
		change_local_variable = { name = bag_wsl_span subtract = var:bag_wsl_held }
		set_local_variable = { name = bag_wsl_step value = var:bag_wsl_rem }
		change_local_variable = { name = bag_wsl_step multiply = local_var:bag_wsl_span }
		change_local_variable = { name = bag_wsl_step divide = var:bag_wsl_drain }
		set_local_variable = { name = bag_wsl_aim value = var:bag_wsl_m0 }
		change_local_variable = { name = bag_wsl_aim subtract = local_var:bag_wsl_step }

		# Не ближе сотой к точке нулевого прироста: крошечный остаток давал
		# положение почти в ней, месяц почти ничего не снимал, и правка
		# сдавалась с остатком 0.01% (его жалоба 2026-09-29). Перебор не страшен:
		# ниже нуля инфляция не опускается. Ниже пола незачем — там и так
		# полный темп.
		set_local_variable = { name = bag_wsl_near value = var:bag_wsl_m0 }
		change_local_variable = { name = bag_wsl_near subtract = 0.01 }
		if = {
			limit = { local_var:bag_wsl_aim > local_var:bag_wsl_near }
			set_local_variable = { name = bag_wsl_aim value = local_var:bag_wsl_near }
		}
		if = {
			limit = { local_var:bag_wsl_aim < var:bag_wsl_mint_floor }
			set_local_variable = { name = bag_wsl_aim value = var:bag_wsl_mint_floor }
		}
		set_variable = { name = bag_wsl_mint_target value = local_var:bag_wsl_aim }
	}
	else = {
		# Темпа нет (игра не ответила, месяц ничего не снял) — держим пол, как
		# раньше, и меряем следующий месяц.
		set_variable = { name = bag_wsl_mint_target value = var:bag_wsl_mint_floor }
	}
	set_variable = { name = bag_wsl_mint_repost value = 1 }
	# Чем месяц держался — тем и мерится следующий.
	set_variable = { name = bag_wsl_held value = var:bag_wsl_mint_target }
}

# Вернуть чеканку автоматике: интерфейс увидит `bag_wsl_mint_release` и
# включит галочку обратно.
# Scope: country
bag_wsl_mint_stand_down = {
	remove_variable = bag_wsl_mint_phase
	remove_variable = bag_wsl_mint_go
	remove_variable = bag_wsl_mint_repost
	set_variable = { name = bag_wsl_mint_release value = 1 }
	bag_wsl_update_busy = yes
}

# Счётчик сработавших приводов для окна диагностики. `change_variable` на
# переменной, которой нет, не пишет — заводим нулём.
# Scope: country
bag_wsl_count_run = {
	if = {
		limit = { NOT = { has_variable = bag_wsl_runs } }
		set_variable = { name = bag_wsl_runs value = 0 }
	}
	change_variable = { name = bag_wsl_runs add = 1 }
}
