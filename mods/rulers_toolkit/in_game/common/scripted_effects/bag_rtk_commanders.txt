# Командиры из правителей (сословие «Правители», `crown_estate`).
#
# Мир: раз в месяц свободные правители по одному встают на отряды. Власть от
# командования — два котла по 25 % (`BASE_ESTATE_POWER_FROM_COMMAND`), армии и
# флоты отдельно, каждый делится между командирами своей ветви (его слова
# 10-18). Поэтому каждый следующий правитель идёт туда, где отрядов меньше
# (`bag_rtk_navy_pays_more`): при четырёх армиях и одном флоте первый — на флот.
# В ветви — на самый большой отряд без правителя. Отряд, где уже командует
# правитель, не трогается: никто не скачет между отрядами.
#
# Война: раздача стоит, перестановку делает игра. Одно исключение: при
# «Генерал-правитель» (`ruler_must_be_commander_during_war`) правитель без
# отряда ставится на самую большую армию. Штраф игры
# (`ruler_is_not_a_commander_during_war`, auto_modifiers/country.txt) не
# берётся, когда правитель командует своим отрядом.

# Scope: country
bag_rtk_commanders_monthly = {
	if = {
		limit = {
			has_variable = bag_rtk_cmd_on
			at_war = no
		}
		bag_rtk_count_units = yes
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: month, peace pass" }
		while = {
			count = 20
			if = {
				limit = {
					any_character = { bag_rtk_crown_commander_candidate = yes }
				}
				if = {
					limit = { bag_rtk_navy_pays_more = yes }
					ordered_navy = {
						limit = { bag_rtk_unit_open_to_crown = yes }
						order_by = num_subunits
						max = 1
						check_range_bounds = no
						bag_rtk_give_unit_to_crown = yes
					}
				}
				else = {
					ordered_army = {
						limit = { bag_rtk_unit_open_to_crown = yes }
						order_by = num_subunits
						max = 1
						check_range_bounds = no
						bag_rtk_give_unit_to_crown = yes
					}
				}
			}
		}
	}
	else_if = {
		limit = {
			has_variable = bag_rtk_cmd_war_ruler_on
			at_war = yes
			modifier:ruler_must_be_commander_during_war = yes
		}
		bag_rtk_ruler_to_largest_army = yes
	}
}

# Сколько отрядов делит каждый котёл. Scope: country.
bag_rtk_count_units = {
	set_variable = { name = bag_rtk_n_army value = 0 }
	set_variable = { name = bag_rtk_n_navy value = 0 }
	every_army = {
		limit = { bag_rtk_unit_counts = yes }
		root = { change_variable = { name = bag_rtk_n_army add = 1 } }
	}
	every_navy = {
		limit = { bag_rtk_unit_counts = yes }
		root = { change_variable = { name = bag_rtk_n_navy add = 1 } }
	}
}

# Первый в очереди правитель встаёт на этот отряд. Scope: unit, root — страна.
bag_rtk_give_unit_to_crown = {
	save_scope_as = bag_rtk_unit
	root = {
		ordered_character = {
			limit = { bag_rtk_crown_commander_candidate = yes }
			order_by = bag_rtk_commander_order
			max = 1
			check_range_bounds = no
			scope:bag_rtk_unit = { set_as_commander = prev }
			if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: crown commander set" }
		}
	}
}

# Scope: country
bag_rtk_ruler_to_largest_army = {
	if = {
		limit = {
			ruler ?= {
				is_general = no
				is_admiral = no
				is_eligible_military_leader = root
			}
		}
		ruler = { save_scope_as = bag_rtk_ruler }
		ordered_army = {
			limit = {
				bag_rtk_unit_counts = yes
				in_combat = no
			}
			order_by = num_subunits
			max = 1
			check_range_bounds = no
			set_as_commander = scope:bag_rtk_ruler
			if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: war, ruler put on the largest army" }
		}
	}
}
