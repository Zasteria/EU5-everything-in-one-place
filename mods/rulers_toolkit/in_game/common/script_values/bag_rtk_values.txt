# Очередь коронных командиров: правитель первым, потом по сумме трёх навыков
# (его слова 10-18: «там все навыки важны»). Scope: character.
bag_rtk_commander_order = {
	value = total_abilities
	if = {
		limit = { is_ruler = yes }
		add = 1000
	}
}

# Женитьба династии: моложе — раньше (`ordered_in_list` берёт наибольшее). Scope: character.
bag_rtk_wed_youth = {
	value = age_in_years
	multiply = -1
}

# Согласие на династический брак — числа `royal_marriage` из
# `ai_diplochance/00_ai_diplochance.txt`; игра соглашается от +1 (его скрин 10-19:
# ранг +15, доверие +14, мнение +3, другая культура −5, расстояние −13 = +14,
# сходится). Соперники и другая группа вер (−100, −200) отсеяны раньше.
# Чего игра не раскрывает, посчитано осторожно: плюсы, которых не знаем, не
# прибавляем, минусы вычитаем — лучше пропустить согласного, чем потратить
# дипломата на отказ:
#   `culture_view` 10, `religion_view` 20 — ступень мнения (враг, плохое,
#     нейтральное, хорошее, родственное); вычитаем только плохое и вражду;
#   `union_size` −4 — за страну в униях их династии;
#   `heir` +10 — не прибавляем (у его наследника на скрине его нет);
#   `age_female` −10 — за год жены старше 30 (`AI_MAX_FEMALE_MARRIAGE_AGE`;
#     35 у `PREGNANCY_AGE_THRESHOLD` — берём меньшее) — вывод, не мерено;
#   `marriage_desirability` 1 — модификатор нашего человека.
# Каждое слагаемое — своё значение: зонд печатает их по отдельности, чтобы
# сверить с подсказкой игры построчно (0.7.5).
# Scope: страна-получатель; scope:bag_rtk_wed_home — игрок.
bag_rtk_wed_t_rank = {
	value = 0
	add = {
		value = country_rank_level
		multiply = -15
	}
	add = {
		scope:bag_rtk_wed_home = {
			value = country_rank_level
			multiply = 15
		}
	}
}
bag_rtk_wed_t_opinion = {
	value = "opinion(scope:bag_rtk_wed_home)"
	multiply = 0.1
}
bag_rtk_wed_t_trust = {
	value = "trust(scope:bag_rtk_wed_home)"
	multiply = 2
}
bag_rtk_wed_t_culture = {
	value = 0
	if = {
		limit = { culture = scope:bag_rtk_wed_home.culture }
		add = 5
	}
	else = {
		add = -5
		if = {
			limit = { culture = { cultural_view = { target = scope:bag_rtk_wed_home.culture value = enemy } } }
			add = -20
		}
		else_if = {
			limit = { culture = { cultural_view = { target = scope:bag_rtk_wed_home.culture value <= negative } } }
			add = -10
		}
	}
}
bag_rtk_wed_t_religion = {
	value = 0
	if = {
		limit = { NOT = { religion = scope:bag_rtk_wed_home.religion } }
		add = -40
		if = {
			limit = { religion = { religious_view = { target = scope:bag_rtk_wed_home.religion value = enemy } } }
			add = -40
		}
		else_if = {
			limit = { religion = { religious_view = { target = scope:bag_rtk_wed_home.religion value <= negative } } }
			add = -20
		}
	}
}
bag_rtk_wed_t_language = {
	value = 0
	if = {
		limit = { court_language = scope:bag_rtk_wed_home.court_language }
		add = 5
	}
}
bag_rtk_wed_t_distance = {
	value = "border_distance_to(scope:bag_rtk_wed_home)"
	multiply = -0.025
}
bag_rtk_wed_t_union = {
	value = 0
	if = {
		limit = { exists = ruler.dynasty }
		add = {
			ruler.dynasty = {
				value = num_union_countries
				multiply = -4
			}
		}
	}
}
bag_rtk_wed_accept = {
	value = 0
	add = bag_rtk_wed_t_rank
	add = bag_rtk_wed_t_opinion
	add = bag_rtk_wed_t_trust
	add = bag_rtk_wed_t_culture
	add = bag_rtk_wed_t_religion
	add = bag_rtk_wed_t_language
	add = bag_rtk_wed_t_distance
	add = bag_rtk_wed_t_union
}

# Слагаемые пары, зависящие от людей. Scope: их человек; scope:bag_rtk_wed_one — наш.
bag_rtk_wed_t_desire = {
	value = 0
	add = {
		scope:bag_rtk_wed_one = {
			value = modifier:marriage_desirability
		}
	}
}
bag_rtk_wed_t_age = {
	value = 0
	if = {
		limit = {
			is_female = yes
			age_in_years > 30
		}
		add = {
			value = age_in_years
			subtract = 30
			multiply = -10
		}
	}
	else_if = {
		limit = {
			is_female = no
			scope:bag_rtk_wed_one = {
				is_female = yes
				age_in_years > 30
			}
		}
		add = {
			scope:bag_rtk_wed_one = {
				value = age_in_years
				subtract = 30
				multiply = -10
			}
		}
	}
}
# Согласие на пару: страна плюс люди. Scope: их человек.
bag_rtk_wed_pair_accept = {
	value = 0
	add = {
		owner = {
			value = bag_rtk_wed_accept
		}
	}
	add = bag_rtk_wed_t_desire
	add = bag_rtk_wed_t_age
}
