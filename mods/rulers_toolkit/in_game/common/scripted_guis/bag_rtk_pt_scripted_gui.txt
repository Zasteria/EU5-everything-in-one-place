# Сгенерировано tools/gen_peace.py — не править руками.

# Шаблон договора (`tools/gen_peace.py`): кнопка запускает шагомер. Приоритет
# идёт до самого большого из выставленных; на каждом — две фазы (категории).
# Scope: country.
bag_rtk_pt_start = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		bag_rtk_pt_defaults = yes
		set_variable = { name = bag_rtk_pt_max value = 0 }
		if = { limit = { var:bag_rtk_pt_p_rep > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_rep } }
		if = { limit = { var:bag_rtk_pt_p_gold > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_gold } }
		if = { limit = { var:bag_rtk_pt_p_annul > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_annul } }
		if = { limit = { var:bag_rtk_pt_p_access > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_access } }
		if = { limit = { var:bag_rtk_pt_p_food > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_food } }
		if = { limit = { var:bag_rtk_pt_p_piracy > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_piracy } }
		if = { limit = { var:bag_rtk_pt_p_build > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_build } }
		if = { limit = { var:bag_rtk_pt_p_slaves > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_slaves } }
		if = { limit = { var:bag_rtk_pt_p_hum > var:bag_rtk_pt_max } set_variable = { name = bag_rtk_pt_max value = var:bag_rtk_pt_p_hum } }
		set_variable = { name = bag_rtk_pt_step value = 0 }
		set_variable = { name = bag_rtk_pt_sub value = 2 }
		set_variable = { name = bag_rtk_pt_run value = yes }
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: peace template started" }
	}
}

# Следующая фаза; после последней — следующий приоритет; после самого
# большого — стоп. Scope: country.
bag_rtk_pt_next = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { var:bag_rtk_pt_sub < 2 }
			change_variable = { name = bag_rtk_pt_sub add = 1 }
		}
		else = {
			set_variable = { name = bag_rtk_pt_sub value = 1 }
			change_variable = { name = bag_rtk_pt_step add = 1 }
		}
		if = {
			limit = { var:bag_rtk_pt_step > var:bag_rtk_pt_max }
			remove_variable = bag_rtk_pt_run
			if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: peace template done" }
		}
	}
}

# Помощник шаблона нажал пункт (зонд). Scope: country.
bag_rtk_pt_clicked = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = { limit = { has_global_variable = bag_rtk_log } debug_log = "bag_rtk: peace template, a term clicked" }
	}
}
