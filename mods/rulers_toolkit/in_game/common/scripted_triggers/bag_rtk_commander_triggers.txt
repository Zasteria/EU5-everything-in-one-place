# Кто годится в коронные командиры. Scope: character, root — страна.
# Правитель или человек сословия короны, взрослый, не в совете и сейчас ничем
# не командует. Пол, возраст, плен — проверка самой игры
# (`is_eligible_military_leader`).
bag_rtk_crown_commander_candidate = {
	is_alive = yes
	is_adult = yes
	is_general = no
	is_admiral = no
	in_cabinet = no
	is_eligible_military_leader = root
	OR = {
		is_ruler = yes
		has_estate = estate_type:crown_estate
	}
}

# Кем уже командует корона: правитель или человек сословия короны.
# Scope: character.
bag_rtk_is_crown_commander = {
	OR = {
		is_ruler = yes
		has_estate = estate_type:crown_estate
	}
}

# Отряд, который стоит отдать коронному командиру. Scope: unit.
# Наёмников, изгнанные отряды и отряды в бою не трогаем.
bag_rtk_unit_open_to_crown = {
	is_mercenary = no
	is_exiled = no
	in_combat = no
	OR = {
		has_commander = no
		leader = { NOT = { bag_rtk_is_crown_commander = yes } }
	}
}
