# Кто годится в коронные командиры. Scope: character, root — страна.
# Правитель или человек сословия короны, взрослый, не в совете и сейчас ничем
# не командует. Пол, возраст, плен — проверка самой игры
# (`is_eligible_military_leader`).
bag_rtk_crown_commander_candidate = {
	is_alive = yes
	is_adult = yes
	is_general = no
	is_admiral = no
	in_cabinet = no
	is_eligible_military_leader = root
	OR = {
		is_ruler = yes
		has_estate = estate_type:crown_estate
	}
}

# Кем уже командует корона: правитель или человек сословия короны.
# Scope: character.
bag_rtk_is_crown_commander = {
	OR = {
		is_ruler = yes
		has_estate = estate_type:crown_estate
	}
}

# Отряд, который считается в дележе власти от командования. Scope: unit.
# Чисто наёмный отряд (ни своих полков, ни ополчения) и изгнанный — мимо.
# `is_mercenary` — триггер полка, не отряда: в 0.1.0 стоял здесь, игра писала
# «Inconsistent trigger scopes (unit vs. sub_unit)», и ни одна армия не
# проходила (его прогон 10-18).
bag_rtk_unit_counts = {
	is_exiled = no
	OR = {
		has_regulars = yes
		has_levies = yes
	}
}

# Отряд, который стоит отдать коронному командиру. Scope: unit.
bag_rtk_unit_open_to_crown = {
	bag_rtk_unit_counts = yes
	in_combat = no
	OR = {
		has_commander = no
		leader = { NOT = { bag_rtk_is_crown_commander = yes } }
	}
}

# Следующий правитель выгоднее на флоте (scope: country, счётчики
# `bag_rtk_count_units`). Власть от командования армиями и от флотов — два
# отдельных котла по 25 %, каждый делится между командирами своей ветви: один
# правитель даёт 1/N своего котла. Значит, туда, где отрядов меньше; поровну —
# в армию.
bag_rtk_navy_pays_more = {
	has_variable = bag_rtk_cmd_navy_on
	any_navy = { bag_rtk_unit_open_to_crown = yes }
	OR = {
		NOT = { any_army = { bag_rtk_unit_open_to_crown = yes } }
		var:bag_rtk_n_navy < var:bag_rtk_n_army
	}
}
