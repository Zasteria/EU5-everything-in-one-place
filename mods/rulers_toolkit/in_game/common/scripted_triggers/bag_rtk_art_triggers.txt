# Творения на продажу (0.4.0, его просьба 10-18). scope:bag_rtk_seller — наша
# страна.

# Творение можно и нужно продать. Scope: work_of_art. Условия видимости —
# из ванильного `country_interactions/sell_work_of_art.txt`; ступени качества —
# из `artists_l_russian.yml` (0–11, 12–24, … 87–100), по галочке на ступень.
bag_rtk_art_for_sale = {
	owner ?= scope:bag_rtk_seller
	art_is_priceless = no
	is_art_destroyed = no
	work_of_art_is_imperial_regalia = no
	OR = {
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q1 }
			art_quality < 12
		}
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q2 }
			art_quality >= 12 art_quality < 25
		}
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q3 }
			art_quality >= 25 art_quality < 37
		}
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q4 }
			art_quality >= 37 art_quality < 50
		}
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q5 }
			art_quality >= 50 art_quality < 62
		}
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q6 }
			art_quality >= 62 art_quality < 75
		}
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q7 }
			art_quality >= 75 art_quality < 87
		}
		AND {
			scope:bag_rtk_seller = { has_variable = bag_rtk_art_q8 }
			art_quality >= 87
		}
	}
}

# Держава купит творение scope:bag_rtk_art. Scope: country. Деньги — как в
# игре (кому не хватает, того нет в списке); согласие — ванильная формула
# `accept` + `diplo_chance`, нужно не меньше 1 (его скрин 10-18: «на 1 больше
# положительных»).
bag_rtk_art_buyer = {
	is_real_country = yes
	NOT = { this = scope:bag_rtk_seller }
	NOT = { country_type = building }
	NOT = { is_enemy_of = scope:bag_rtk_seller }
	gold >= scope:bag_rtk_art.art_price
	bag_rtk_art_accept >= 1
}
