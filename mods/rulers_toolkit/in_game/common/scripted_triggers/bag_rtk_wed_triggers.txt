# Женитьба династии (`bag_rtk_wed.txt`). Наш — правитель или его близкий
# родственник, годен для брака; пара — член короны другой страны, одна группа
# вер, не близкие родственники друг другу.

# Наш член семьи без пары. Scope: character.
bag_rtk_wed_single = {
	age_in_years >= 16
	is_married = no
	is_betrothed = no
	character_can_marry_trigger = yes
	is_eligible_for_royal_marriage = yes
	bag_rtk_wed_royal = yes
}

# Правитель своей страны или его близкий родственник. Scope: character.
bag_rtk_wed_royal = {
	OR = {
		is_ruler = yes
		AND = {
			exists = owner.ruler
			is_close_relative = owner.ruler
		}
	}
}

# Страна невесты, как в окне игры: не наша, монархия (республики и купеческие
# семьи вроде Фуггеров династических браков не делают), в дипломатической
# дистанции. Scope: country.
bag_rtk_wed_land_ok = {
	NOT = { this = scope:bag_rtk_wed_home }
	government_type = government_type:monarchy
	within_diplomatic_range = scope:bag_rtk_wed_home
}

# Пара для scope:bag_rtk_wed_one (он же scope:recipient для проверки игры
# `character_can_marry_recipient_trigger`) — как `auto_marry_royal_partner`:
# правитель или его близкая родственница («closely related enough to their
# ruler» в окне игры), член короны, годен для династического брака, противоположный пол, одна группа
# вер (правило игры для династического брака), не близкий родственник. Scope: character.
bag_rtk_wed_match = {
	age_in_years >= 16
	is_married = no
	is_betrothed = no
	has_estate = estate_type:crown_estate
	character_can_marry_trigger = yes
	is_eligible_for_royal_marriage = yes
	modifier:blocked_from_character_interactions = no
	OR = {
		AND = {
			is_female = yes
			scope:bag_rtk_wed_one = { is_female = no }
		}
		AND = {
			is_female = no
			scope:bag_rtk_wed_one = { is_female = yes }
		}
	}
	religion.group = scope:bag_rtk_wed_one.religion.group
	bag_rtk_wed_royal = yes
	NOT = { is_close_relative = scope:bag_rtk_wed_one }
	character_can_marry_recipient_trigger = yes
}
