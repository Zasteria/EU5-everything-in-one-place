# The settings: show every line, filtered or not; and the test of conditions on hover.
#
# Everything else here is generated. This is not, because it is the mod's whole
# interface with the Community Mod Framework and it has no data behind it.
#
# The switch is deliberately per country rather than global: it is a reading
# preference of the person looking at the tooltip, and CMF's global variants
# store one value for the whole session, which is for rules rather than for
# what a player wants to see.
#
# Scope: country
svx_register = {
	cmf_register_mod = { mod_id = svx }

	cmm_register_bool_setting = {
		mod_id = svx
		setting_id = show_all
		tab_id = main
		group_id = lists
		default_value = 0
	}

	# 10-04: the «not yet» age brackets as links to the object's conditions,
	# behind a checkbox while it is unproven (his ask: «тестовая функция»).
	cmm_register_bool_setting = {
		mod_id = svx
		setting_id = test_conditions
		tab_id = main
		group_id = lists
		default_value = 0
	}
}
