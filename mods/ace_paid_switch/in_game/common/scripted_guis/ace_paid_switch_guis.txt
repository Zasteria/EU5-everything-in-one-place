# ace_paid_switch: the expensive in-depth education on or off as a whole. Switching it back on
# educates the waiting children at once instead of on the next monthly pass.
ace_toggle_paid_off = { scope = country effect = {
 cea_initialize = yes
 if = { limit = { has_variable = cea_paid_off }
  remove_variable = cea_paid_off
  cea_for_each_tracked_child = { effect = cea_auto_educate }
  cea_refresh_list = yes
 }
 else = { set_variable = { name = cea_paid_off value = yes } }
 } }
