# Два крючка CMF: регистрация страницы и обратный звонок с её кнопки.

cmf_on_mod_registration = {
	on_actions = {
		bag_cmfs_on_register
	}
}

# Scope: country
bag_cmfs_on_register = {
	effect = {
		bag_cmfs_register = yes
	}
}

cmf_on_callback = {
	on_actions = {
		bag_cmfs_on_callback
	}
}

# Scope: country
bag_cmfs_on_callback = {
	effect = {
		bag_cmfs_handle_callback = yes
	}
}
