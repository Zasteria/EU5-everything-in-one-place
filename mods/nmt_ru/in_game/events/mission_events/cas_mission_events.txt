# =============================================================================
# Event 20: グラナダ陥落——レコンキスタの完成
# =============================================================================

cas_events.20 = {
	type = country_event
	category = mission_event
	title = cas_events.20.t
	desc  = cas_events.20.d
	fire_only_once = yes
	illustration_tags = { 10 = interior 10 = regular }
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = cas_events.20.a
		add_prestige = 30
		add_country_modifier = { modifier = cas_reconquista_done_mod years = 100 mode = add }
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
	}
}
