# Автоженитьба правящей семьи (наше, не автора). Члены сословия правителей,
# годные для династического брака, женятся на знати своего двора так же, как
# знать женится между собой: раз в месяц, по тем же ограничениям по культуре.

# character: член правящей семьи, которого пора женить
auto_marry_royal_candidate = {
	is_married = no
	has_estate = estate_type:crown_estate
	is_eligible_for_royal_marriage = yes
	character_can_marry_trigger = yes
	modifier:blocked_from_character_interactions = no
	age_in_years > 18
	age_in_years < 40
}

# character: пара для scope:unmarried_royal из знати двора (тот же круг, что у
# браков знати: сословие аристократов и не годные для династического брака
# правители)
auto_marry_royal_suitor = {
	is_married = no
	character_can_marry_trigger = yes
	modifier:blocked_from_character_interactions = no
	noble_auto_marry_culture_check = yes
	OR = {
		has_estate = estate_type:nobles_estate
		AND = {
			has_estate = estate_type:crown_estate
			is_eligible_for_royal_marriage = no
		}
	}
	age_in_years > 18
	age_in_years < 35
	NOT = { is_close_relative = scope:unmarried_royal }
	trigger_if = {
		limit = { scope:unmarried_royal = { is_female = yes } }
		is_female = no
	}
	trigger_else = {
		is_female = yes
	}
	trigger_if = {
		limit = { owner ?= { modifier:country_celtic_marriage_banned = yes } }
		OR = {
			AND = {
				culture ?= { has_culture_group = culture_group:celtic_group }
				scope:unmarried_royal.culture ?= { has_culture_group = culture_group:celtic_group }
			}
			AND = {
				NOT = { culture ?= { has_culture_group = culture_group:celtic_group } }
				NOT = { scope:unmarried_royal.culture ?= { has_culture_group = culture_group:celtic_group } }
			}
		}
	}
}
