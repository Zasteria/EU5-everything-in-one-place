namespace = auto_marry_royal

# Раз в месяц (on_action auto_marry_royal_on_action): один член правящей семьи
# без пары женится на знатном придворном. Скрытое событие подбирает пару и
# женит; если события о свадьбах не скрыты, показывает .2.
auto_marry_royal.1 = {
	type = country_event
	hidden = yes

	trigger = {
		is_ai = no
		NOT = { has_variable = noble_auto_marry_temp_disable }
		NOT = { has_variable = auto_marry_royal_off }
		NOT = { has_variable = noble_auto_marry_royal_planning_wedding }
		any_character = { auto_marry_royal_candidate = yes }
	}

	immediate = {
		random_character = {
			limit = { auto_marry_royal_candidate = yes }
			save_scope_as = unmarried_royal
		}
		random_character = {
			limit = { auto_marry_royal_suitor = yes }
			save_scope_as = target_suitor
		}
		if = {
			limit = {
				exists = scope:unmarried_royal
				exists = scope:target_suitor
			}
			scope:unmarried_royal = { marry_character = scope:target_suitor }
			if = {
				limit = { NOT = { has_variable = hide_marriage_events } }
				trigger_event_non_silently = auto_marry_royal.2
			}
		}
	}

	option = {
		name = noble_auto_marry.3.a
	}
}

auto_marry_royal.2 = {
	type = country_event
	title = noble_auto_marry.3.t
	desc = noble_auto_marry.3.d

	option = {
		name = noble_auto_marry.3.a
	}
	option = {
		name = noble_auto_marry.1.b
		custom_tooltip = { text = hide_marriage_events.tt }
		set_variable = hide_marriage_events
	}
}
