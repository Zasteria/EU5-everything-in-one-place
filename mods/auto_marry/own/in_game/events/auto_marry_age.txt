namespace = auto_marry_age

# Меню брачного возраста (из меню настроек, noble_auto_marry.63). Открытие
# заводит недостающие переменные со значением 19, чтобы описание показывало числа.
auto_marry_age.1 = {
	type = country_event
	title = auto_marry_age.1.t
	desc = auto_marry_age.1.d

	immediate = {
		if = {
			limit = { NOT = { has_variable = auto_marry_age_royal } }
			set_variable = { name = auto_marry_age_royal value = 19 }
		}
		if = {
			limit = { NOT = { has_variable = auto_marry_age_nobles } }
			set_variable = { name = auto_marry_age_nobles value = 19 }
		}
		if = {
			limit = { NOT = { has_variable = auto_marry_age_burghers } }
			set_variable = { name = auto_marry_age_burghers value = 19 }
		}
		if = {
			limit = { NOT = { has_variable = auto_marry_age_clergy } }
			set_variable = { name = auto_marry_age_clergy value = 19 }
		}
		if = {
			limit = { NOT = { has_variable = auto_marry_age_peasants } }
			set_variable = { name = auto_marry_age_peasants value = 19 }
		}
		if = {
			limit = { NOT = { has_variable = auto_marry_age_dhimmi } }
			set_variable = { name = auto_marry_age_dhimmi value = 19 }
		}
		if = {
			limit = { NOT = { has_variable = auto_marry_age_tribes } }
			set_variable = { name = auto_marry_age_tribes value = 19 }
		}
		if = {
			limit = { NOT = { has_variable = auto_marry_age_cossacks } }
			set_variable = { name = auto_marry_age_cossacks value = 19 }
		}
	}

	option = {
		name = auto_marry_age.1.royal
		trigger_event_silently = { id = auto_marry_age.10 }
	}

	option = {
		name = auto_marry_age.1.nobles
		trigger_event_silently = { id = auto_marry_age.11 }
	}

	option = {
		name = auto_marry_age.1.burghers
		trigger_event_silently = { id = auto_marry_age.12 }
	}

	option = {
		name = auto_marry_age.1.clergy
		trigger_event_silently = { id = auto_marry_age.13 }
	}

	option = {
		name = auto_marry_age.1.peasants
		trigger_event_silently = { id = auto_marry_age.14 }
	}

	option = {
		name = auto_marry_age.1.dhimmi
		trigger_event_silently = { id = auto_marry_age.15 }
	}

	option = {
		name = auto_marry_age.1.tribes
		trigger_event_silently = { id = auto_marry_age.16 }
	}

	option = {
		name = auto_marry_age.1.cossacks
		trigger_event_silently = { id = auto_marry_age.17 }
	}

	option = { # back to the settings menu
		name = auto_marry_age.back
		trigger_event_silently = { id = noble_auto_marry.63 }
	}
}
