get_auto_marry_royal_setting = {
	type = country
	text = {
		trigger = { has_variable = auto_marry_royal_off }
		localization_key = auto_marry_royal_off_text
	}
	text = {
		trigger = { NOT = { has_variable = auto_marry_royal_off } }
		localization_key = auto_marry_royal_on_text
	}
}
