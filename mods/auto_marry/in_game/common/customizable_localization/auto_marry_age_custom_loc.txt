	type = country
	text = {
		trigger = { has_variable = auto_marry_age_royal }
		localization_key = auto_marry_age_value_royal
	}
	text = {
		trigger = { NOT = { has_variable = auto_marry_age_royal } }
		localization_key = auto_marry_age_default
	}
}
