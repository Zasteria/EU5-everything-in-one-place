bag_wpr_init = {
	if = { limit = { NOT = { exists = var:bag_wpr_secs } } set_variable = { name = bag_wpr_secs value = 0 } }
	if = { limit = { NOT = { exists = var:bag_wpr_m1 } } set_variable = { name = bag_wpr_m1 value = 0 } }
	if = { limit = { NOT = { exists = var:bag_wpr_m2 } } set_variable = { name = bag_wpr_m2 value = 0 } }
	if = { limit = { NOT = { exists = var:bag_wpr_m3 } } set_variable = { name = bag_wpr_m3 value = 0 } }
	if = { limit = { NOT = { exists = var:bag_wpr_m4 } } set_variable = { name = bag_wpr_m4 value = 0 } }
	if = { limit = { NOT = { exists = var:bag_wpr_m5 } } set_variable = { name = bag_wpr_m5 value = 0 } }
	if = { limit = { NOT = { exists = var:bag_wpr_m6 } } set_variable = { name = bag_wpr_m6 value = 0 } }
}
